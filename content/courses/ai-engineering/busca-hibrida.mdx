Busca vetorial é ótima para paráfrases e péssima para coisas exatas: o código de erro `E-4031`, o nome de uma função, um número de contrato, uma sigla interna. Busca por palavra-chave é o oposto. Em vez de escolher, sistemas de recuperação maduros combinam as duas: é a **busca híbrida**. Para entender por que ela funciona, você precisa conhecer o algoritmo lexical que ainda é o padrão da indústria: o **BM25**.

## BM25 em uma frase

O BM25 dá pontos a um documento para cada termo da consulta que ele contém, com três ajustes:

- **Raridade (IDF):** termos raros na coleção valem mais. "kubernetes" diz muito mais que "de".
- **Saturação da frequência:** aparecer 2 vezes é melhor que 1, mas aparecer 50 vezes não é 50 vezes melhor. O parâmetro `k1` controla essa curva.
- **Normalização por tamanho:** documentos longos contêm mais palavras por acaso, então são penalizados em relação ao tamanho médio. O parâmetro `b` controla a força disso.

```text
score(D, Q) = soma sobre termos t de Q:
    IDF(t) · f(t,D) · (k1 + 1) / ( f(t,D) + k1 · (1 - b + b · |D| / tamanho_medio) )
```

Valores comuns são `k1` entre 1,2 e 2 e `b` perto de 0,75. Uma implementação completa cabe em poucas linhas:

```python
import math
from collections import Counter

docs = {
    "d1": "erro E-4031 ao emitir nota fiscal no sistema",
    "d2": "como emitir nota fiscal eletronica passo a passo",
    "d3": "o sistema caiu e nenhuma nota foi emitida hoje",
    "d4": "politica de reembolso para clientes do plano anual",
}

def tokens(texto):
    return texto.lower().split()

class BM25:
    def __init__(self, docs, k1=1.5, b=0.75):
        self.k1, self.b = k1, b
        self.tf = {d: Counter(tokens(t)) for d, t in docs.items()}
        self.tam = {d: sum(c.values()) for d, c in self.tf.items()}
        self.media = sum(self.tam.values()) / len(docs)
        n = len(docs)
        df = Counter(t for c in self.tf.values() for t in c)
        self.idf = {t: math.log(1 + (n - f + 0.5) / (f + 0.5)) for t, f in df.items()}

    def score(self, consulta, d):
        s = 0.0
        for t in tokens(consulta):
            f = self.tf[d].get(t, 0)
            if f == 0:
                continue
            norm = self.k1 * (1 - self.b + self.b * self.tam[d] / self.media)
            s += self.idf[t] * f * (self.k1 + 1) / (f + norm)
        return s

    def buscar(self, consulta, k=3):
        scores = {d: self.score(consulta, d) for d in self.tf}
        return sorted(((round(s, 3), d) for d, s in scores.items() if s > 0), reverse=True)[:k]

bm25 = BM25(docs)
print(bm25.buscar("E-4031"))
print(bm25.buscar("emitir nota fiscal"))
```

```javascript
const docs = {

  d2: "como emitir nota fiscal eletronica passo a passo",
  d3: "o sistema caiu e nenhuma nota foi emitida hoje",

};

function tokens(texto) {
  return texto.toLowerCase().split(/\s+/).filter(Boolean);
}

function contar(lista) {
  const c = new Map();
  for (const x of lista) c.set(x, (c.get(x) || 0) + 1);
  return c;
}

// round() do Python: o empate exato vai para o par (toFixed arredondaria para cima)
function arred(x, casas) {
  const m = x * 10 ** casas, r = Math.round(m);
  const n = Math.abs(m % 1) === 0.5 && r % 2 !== 0 ? r - 1 : r;
  return n / 10 ** casas;
}

class BM25 {
  constructor(docs, k1 = 1.5, b = 0.75) {
    this.k1 = k1;
    this.b = b;
    this.tf = new Map(Object.entries(docs).map(([d, t]) => [d, contar(tokens(t))]));
    this.tam = new Map();
    for (const [d, c] of this.tf) this.tam.set(d, [...c.values()].reduce((s, v) => s + v, 0));
    this.media = [...this.tam.values()].reduce((s, v) => s + v, 0) / this.tf.size;
    const n = this.tf.size;
    const df = new Map();
    for (const c of this.tf.values()) for (const t of c.keys()) df.set(t, (df.get(t) || 0) + 1);
    this.idf = new Map();
    for (const [t, f] of df) this.idf.set(t, Math.log(1 + (n - f + 0.5) / (f + 0.5)));
  }

  score(consulta, d) {
    let s = 0.0;
    for (const t of tokens(consulta)) {
      const f = this.tf.get(d).get(t) || 0;
      if (f === 0) continue;
      const norm = this.k1 * (1 - this.b + this.b * this.tam.get(d) / this.media);
      s += this.idf.get(t) * f * (this.k1 + 1) / (f + norm);
    }
    return s;
  }

  buscar(consulta, k = 3) {
    const achados = [];
    for (const d of this.tf.keys()) {
      const s = this.score(consulta, d);
      if (s > 0) achados.push([arred(s, 3), d]);
    }
    // sorted(..., reverse=True) em tuplas: maior score e, no empate, maior id
    achados.sort((x, y) => y[0] - x[0] || (y[1] > x[1] ? 1 : y[1] < x[1] ? -1 : 0));
    return achados.slice(0, k);
  }
}

const mostrar = (lista) => "[" + lista.map(([s, d]) => `(${s}, '${d}')`).join(", ") + "]";

const bm25 = new BM25(docs);
console.log(mostrar(bm25.buscar("E-4031")));
console.log(mostrar(bm25.buscar("emitir nota fiscal")));
```

```java
import java.math.*;
import java.util.*;

static List<String> tokens(String texto) {
    String t = texto.toLowerCase(Locale.ROOT).trim();
    return t.isEmpty() ? new ArrayList<>() : Arrays.asList(t.split("\\s+"));
}

// round() do Python: arredonda o valor exato do double, com empate para o par
static double arred(double x, int casas) {
    return new BigDecimal(x).setScale(casas, RoundingMode.HALF_EVEN).doubleValue();
}

class Achado {
    final double score;
    final String doc;
    Achado(double score, String doc) { this.score = score; this.doc = doc; }
    public String toString() { return "(" + score + ", '" + doc + "')"; }
}

class BM25 {
    final double k1, b, media;
    final Map<String, Map<String, Integer>> tf = new LinkedHashMap<>();
    final Map<String, Integer> tam = new LinkedHashMap<>();
    final Map<String, Double> idf = new HashMap<>();

    BM25(Map<String, String> docs, double k1, double b) {
        this.k1 = k1;
        this.b = b;
        for (Map.Entry<String, String> e : docs.entrySet()) {
            Map<String, Integer> c = new LinkedHashMap<>();
            for (String t : tokens(e.getValue())) c.merge(t, 1, Integer::sum);
            tf.put(e.getKey(), c);
            int soma = 0;
            for (int v : c.values()) soma += v;
            tam.put(e.getKey(), soma);
        }
        int total = 0;
        for (int v : tam.values()) total += v;
        media = (double) total / docs.size();
        int n = docs.size();
        Map<String, Integer> df = new HashMap<>();
        for (Map<String, Integer> c : tf.values()) for (String t : c.keySet()) df.merge(t, 1, Integer::sum);
        for (Map.Entry<String, Integer> e : df.entrySet()) {
            int f = e.getValue();
            idf.put(e.getKey(), Math.log(1 + (n - f + 0.5) / (f + 0.5)));
        }
    }

    double score(String consulta, String d) {
        double s = 0.0;
        for (String t : tokens(consulta)) {
            int f = tf.get(d).getOrDefault(t, 0);
            if (f == 0) continue;
            double norm = k1 * (1 - b + b * tam.get(d) / media);
            s += idf.get(t) * f * (k1 + 1) / (f + norm);
        }
        return s;
    }

    List<Achado> buscar(String consulta, int k) {
        List<Achado> achados = new ArrayList<>();
        for (String d : tf.keySet()) {
            double s = score(consulta, d);
            if (s > 0) achados.add(new Achado(arred(s, 3), d));
        }
        // sorted(..., reverse=True) em tuplas: maior score e, no empate, maior id
        achados.sort((x, y) -> {
            int c = Double.compare(y.score, x.score);
            return c != 0 ? c : y.doc.compareTo(x.doc);
        });
        return achados.subList(0, Math.min(k, achados.size()));
    }
}

public static void main(String[] args) {
    Map<String, String> docs = new LinkedHashMap<>();
    docs.put("d1", "erro E-4031 ao emitir nota fiscal no sistema");
    docs.put("d2", "como emitir nota fiscal eletronica passo a passo");
```

```cpp
// lower() só em ASCII, para não quebrar bytes UTF-8; split() em qualquer espaço
vector<string> tokens(const string& texto) {
    string minusculo;
    for (unsigned char c : texto) minusculo += (char)(c < 0x80 ? tolower(c) : c);
    vector<string> saida;
    istringstream in(minusculo);
    for (string t; in >> t;) saida.push_back(t);
    return saida;
}

// round() do Python: o printf arredonda o valor exato do double, com empate para o par
double arred(double x, int casas) {
    char texto[64];
    snprintf(texto, sizeof texto, "%.*f", casas, x);
    return strtod(texto, nullptr);
}

// repr de float do Python: 1.221, 1.0
string pyFloat(double x) {
    char texto[64];
    snprintf(texto, sizeof texto, "%.15g", x);
    string s = texto;
    if (s.find_first_of(".e") == string::npos) s += ".0";
    return s;
}

class BM25 {
    double k1, b, media;
    map<string, map<string, int>> tf;
    map<string, int> tam;
    map<string, double> idf;

public:
    BM25(const vector<pair<string, string>>& docs, double k1 = 1.5, double b = 0.75) : k1(k1), b(b) {
        int total = 0;
        map<string, int> df;
        for (const auto& [d, t] : docs) {
            for (const string& x : tokens(t)) {
                tf[d][x]++;
                tam[d]++;
                total++;
            }
            for (const auto& [x, n] : tf[d]) df[x]++;
        }
        media = (double)total / docs.size();
        int n = docs.size();
        for (const auto& [t, f] : df) idf[t] = log(1 + (n - f + 0.5) / (f + 0.5));
    }

    double score(const string& consulta, const string& d) {
        double s = 0.0;
        for (const string& t : tokens(consulta)) {
            auto it = tf[d].find(t);
            if (it == tf[d].end()) continue;
            int f = it->second;
            double norm = k1 * (1 - b + b * tam[d] / media);
            s += idf[t] * f * (k1 + 1) / (f + norm);
        }
        return s;
    }

    vector<pair<double, string>> buscar(const string& consulta, size_t k = 3) {
        vector<pair<double, string>> achados;
        for (const auto& [d, c] : tf) {
            double s = score(consulta, d);
            if (s > 0) achados.push_back({arred(s, 3), d});
        }
        // sorted(..., reverse=True) em tuplas: maior score e, no empate, maior id
        sort(achados.begin(), achados.end(), greater<pair<double, string>>());
        if (achados.size() > k) achados.resize(k);
        return achados;
    }
};

void mostrar(const vector<pair<double, string>>& lista) {
    string s = "[";
    for (size_t i = 0; i < lista.size(); i++) {
        if (i) s += ", ";
        s += "(" + pyFloat(lista[i].first) + ", '" + lista[i].second + "')";
    }
    printf("%s]\n", s.c_str());
}

    BM25 bm25(docs);
    mostrar(bm25.buscar("E-4031"));
    mostrar(bm25.buscar("emitir nota fiscal"));
}
```

```csharp
using System;
using System.Collections.Generic;
using System.Globalization;
using System.Linq;

static readonly CultureInfo Inv = CultureInfo.InvariantCulture;

static string[] Tokens(string texto) {
    return texto.ToLowerInvariant().Split((char[])null, StringSplitOptions.RemoveEmptyEntries);
}

// round() do Python: parte da forma binária exata (G17) e arredonda com empate para o par
static double Arred(double x, int casas) {
    decimal d = decimal.Parse(x.ToString("G17", Inv), NumberStyles.Float, Inv);
    return (double)Math.Round(d, casas, MidpointRounding.ToEven);
}

class Achado {
    public double Score;
    public string Doc;
    public override string ToString() {
        return "(" + Score.ToString("R", Inv) + ", '" + Doc + "')";
    }
}

class BM25 {
    double k1, b, media;
    Dictionary<string, Dictionary<string, int>> tf = new Dictionary<string, Dictionary<string, int>>();
    Dictionary<string, int> tam = new Dictionary<string, int>();
    Dictionary<string, double> idf = new Dictionary<string, double>();

    public BM25(Dictionary<string, string> docs, double k1 = 1.5, double b = 0.75) {
        this.k1 = k1;
        this.b = b;
        foreach (var par in docs) {
            tf[par.Key] = Tokens(par.Value).GroupBy(t => t).ToDictionary(g => g.Key, g => g.Count());
            tam[par.Key] = tf[par.Key].Values.Sum();
        }
        media = (double)tam.Values.Sum() / docs.Count;
        int n = docs.Count;
        var df = new Dictionary<string, int>();
        foreach (var c in tf.Values) {
            foreach (string t in c.Keys) {
                df.TryGetValue(t, out int atual);
                df[t] = atual + 1;
            }
        }
        foreach (var par in df) idf[par.Key] = Math.Log(1 + (n - par.Value + 0.5) / (par.Value + 0.5));
    }

    public double Score(string consulta, string d) {
        double s = 0.0;
        foreach (string t in Tokens(consulta)) {
            tf[d].TryGetValue(t, out int f);
            if (f == 0) continue;
            double norm = k1 * (1 - b + b * tam[d] / media);
            s += idf[t] * f * (k1 + 1) / (f + norm);
        }
        return s;
    }

    public List<Achado> Buscar(string consulta, int k = 3) {
        var achados = new List<Achado>();
        foreach (string d in tf.Keys) {
            double s = Score(consulta, d);
            if (s > 0) achados.Add(new Achado { Score = Arred(s, 3), Doc = d });
        }
        // sorted(..., reverse=True) em tuplas: maior score e, no empate, maior id
        achados.Sort((x, y) => {
            int c = y.Score.CompareTo(x.Score);
            return c != 0 ? c : string.CompareOrdinal(y.Doc, x.Doc);
        });
        return achados.Take(k).ToList();
    }
}

}
```

```go
import (
	"fmt"
	"math"
	"sort"
	"strconv"
	"strings"
)

var docs = map[string]string{
	"d1": "erro E-4031 ao emitir nota fiscal no sistema",
	"d2": "como emitir nota fiscal eletronica passo a passo",
	"d3": "o sistema caiu e nenhuma nota foi emitida hoje",
	"d4": "politica de reembolso para clientes do plano anual",
}

func tokens(texto string) []string {
	return strings.Fields(strings.ToLower(texto))
}

// round() do Python: o strconv arredonda o valor exato do double, com empate para o par
func arred(x float64, casas int) float64 {
	r, _ := strconv.ParseFloat(strconv.FormatFloat(x, 'f', casas, 64), 64)
	return r
}

type achado struct {
	score float64
	doc   string
}

type BM25 struct {
	k1, b, media float64
	tf           map[string]map[string]int
	tam          map[string]int
	idf          map[string]float64
}

func novoBM25(docs map[string]string, k1, b float64) *BM25 {
	m := &BM25{k1: k1, b: b, tf: map[string]map[string]int{}, tam: map[string]int{}, idf: map[string]float64{}}
	total := 0
	df := map[string]int{}
	for d, texto := range docs {
		c := map[string]int{}
		for _, t := range tokens(texto) {
			c[t]++
			total++
		}
		m.tf[d] = c
		m.tam[d] = len(tokens(texto))
		for t := range c {
			df[t]++
		}
	}
	m.media = float64(total) / float64(len(docs))
	n := float64(len(docs))
	for t, f := range df {
		fr := float64(f)
		m.idf[t] = math.Log(1 + (n-fr+0.5)/(fr+0.5))
	}
	return m
}

func (m *BM25) score(consulta, d string) float64 {
	s := 0.0
	for _, t := range tokens(consulta) {
		f := float64(m.tf[d][t])
		if f == 0 {
			continue
		}
		norm := m.k1 * (1 - m.b + m.b*float64(m.tam[d])/m.media)
		s += m.idf[t] * f * (m.k1 + 1) / (f + norm)
	}
	return s
}

func (m *BM25) buscar(consulta string, k int) []achado {
	achados := []achado{}
	for d := range m.tf {
		if s := m.score(consulta, d); s > 0 {
			achados = append(achados, achado{arred(s, 3), d})
		}
	}
	// sorted(..., reverse=True) em tuplas: maior score e, no empate, maior id
	sort.Slice(achados, func(i, j int) bool {
		if achados[i].score != achados[j].score {
			return achados[i].score > achados[j].score
		}
		return achados[i].doc > achados[j].doc
	})
	if len(achados) > k {
		achados = achados[:k]
	}
	return achados
}

func mostrar(lista []achado) string {
	partes := []string{}
	for _, a := range lista {
		partes = append(partes, fmt.Sprintf("(%s, '%s')", strconv.FormatFloat(a.score, 'f', -1, 64), a.doc))
	}
	return "[" + strings.Join(partes, ", ") + "]"
}

func main() {
	bm25 := novoBM25(docs, 1.5, 0.75)
	fmt.Println(mostrar(bm25.buscar("E-4031", 3)))
	fmt.Println(mostrar(bm25.buscar("emitir nota fiscal", 3)))
```

## Combinando os rankings

Você roda as duas buscas em paralelo, cada uma devolvendo, por exemplo, os 50 melhores, e funde as listas. O problema: os scores não estão na mesma escala (BM25 não tem teto; cosseno vai de -1 a 1). Duas soluções comuns:

- **Fusão por ranking recíproco (RRF):** ignora os scores e usa só a posição. Cada documento recebe `1 / (k + posição)` em cada lista, e as contribuições são somadas. Com `k` perto de 60, é robusto e não precisa de calibração.
- **Combinação ponderada:** normaliza os scores (min-max por consulta) e faz `alfa · vetorial + (1 - alfa) · lexical`. Permite ajustar o peso, mas exige calibrar `alfa` com dados.

```cpp
    vector<pair<string, double>> pontos;
    for (const auto& lista : listas) {

            auto it = find_if(pontos.begin(), pontos.end(), [&](const auto& p) { return p.first == lista[i]; });
```

```go
import (
	"fmt"
	"sort"
)

type pontuado struct {
	doc   string
	score float64
}

func rrf(listas [][]string, k int) []pontuado {
	// fatia de pares: mantém a ordem de inserção do dict do Python
	pontos := []pontuado{}
	indice := map[string]int{}
	for _, lista := range listas {
		for i, doc := range lista {
			posicao := i + 1
			j, existe := indice[doc]
			if !existe {
				j = len(pontos)
				indice[doc] = j
				pontos = append(pontos, pontuado{doc, 0.0})
			}
			pontos[j].score += 1.0 / float64(k+posicao)
		}
	}
	// SliceStable mantém a ordem original nos empates, como o sorted do Python
	sort.SliceStable(pontos, func(a, b int) bool { return pontos[a].score > pontos[b].score })
	return pontos
}

func main() {
	// consulta: "sistema fora do ar ao gerar nota E-4031"
	lexical := []string{"d1", "d2", "d3"}        // BM25: acerta o codigo exato
	vetorial := []string{"d3", "d1", "d2", "d4"} // embeddings: acerta o sentido "sistema caiu"

	for _, p := range rrf([][]string{lexical, vetorial}, 60) {
		fmt.Printf("%s: %.4f\n", p.doc, p.score)
	}
}
```

<Callout type="tip" title="Comece pelo baseline lexical">
Antes de montar um pipeline vetorial, meça o BM25 nas suas consultas reais. Em bases técnicas, com muitos códigos e nomes próprios, ele é um baseline surpreendentemente difícil de bater, e a busca híbrida costuma superar as duas isoladas.
</Callout>

## Armadilhas

- Tokenização ruim no BM25 (sem tratar acentos, plural, maiúsculas) derruba o recall lexical em português.
- Normalizar scores com min-max quando uma das listas vem vazia ou com um único item.
- Manter dois índices exige mantê-los sincronizados: inserir, atualizar e apagar nos dois.
