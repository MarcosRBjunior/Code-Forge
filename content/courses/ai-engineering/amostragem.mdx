O modelo não devolve uma resposta: devolve, a cada passo, uma distribuição de probabilidade sobre o vocabulário. Quem escolhe o token é o **decodificador**, e os parâmetros dele (temperatura, top-k, top-p, semente) mudam bastante o comportamento do sistema. Uma extração de dados que às vezes inventa um campo e um gerador de slogans que sempre repete a mesma frase são, muitas vezes, o mesmo problema com sinais trocados: amostragem mal configurada.

## Greedy versus amostragem

A estratégia mais simples é **greedy**: pegar sempre o token mais provável. Ela é previsível, mas tende a respostas repetitivas e pode entrar em laços. A alternativa é **amostrar** da distribuição, o que traz variedade ao custo de, às vezes, escolher um token ruim de baixa probabilidade.

## Temperatura

Antes da softmax, os logits são divididos pela temperatura `T`:

- `T` menor que 1 deixa a distribuição mais "pontuda": o favorito ganha ainda mais peso. Com `T` perto de 0, o resultado se aproxima do greedy.
- `T` igual a 1 mantém a distribuição original.
- `T` maior que 1 achata a distribuição e dá chance a tokens improváveis.

import math

def softmax(logits, t):
    escalados = [x / t for x in logits]
    m = max(escalados)
    exps = [math.exp(x - m) for x in escalados]
    s = sum(exps)
    return [e / s for e in exps]

tokens = ["Paris", "Lyon", "Roma", "banana"]

    linha = "  ".join(f"{tok}={p:.2f}" for tok, p in zip(tokens, probs))

static double[] softmax(double[] logits, double t) {
    double[] escalados = new double[logits.length];
    for (int i = 0; i < logits.length; i++) escalados[i] = logits[i] / t;
    double m = Arrays.stream(escalados).max().getAsDouble();
    double[] exps = new double[escalados.length];

        s += exps[i];
    }
    double[] probs = new double[exps.length];
    for (int i = 0; i < exps.length; i++) probs[i] = exps[i] / s;
    return probs;
}

func softmax(logits []float64, t float64) []float64 {
	escalados := make([]float64, len(logits))
	m := math.Inf(-1)
	for i, x := range logits {
		escalados[i] = x / t
		m = math.Max(m, escalados[i])
	}
	exps := make([]float64, len(escalados))
	s := 0.0
	for i, x := range escalados {
		exps[i] = math.Exp(x - m)
		s += exps[i]
	}
	probs := make([]float64, len(exps))
	for i, e := range exps {
		probs[i] = e / s
	}
	return probs
}

func main() {
	tokens := []string{"Paris", "Lyon", "Roma", "banana"}

## Top-k e top-p

Mesmo com temperatura moderada, a cauda longa do vocabulário soma muita probabilidade. Dois filtros cortam essa cauda antes do sorteio:

- **Top-k:** mantém só os k tokens mais prováveis e renormaliza.
- **Top-p (nucleus):** mantém o menor conjunto de tokens cuja probabilidade acumulada chega a p. Em distribuições confiantes, sobra um ou dois tokens; em distribuições incertas, sobram vários. Por isso se adapta melhor do que um k fixo.

<CodeTabs>

```python
import random

def top_k(probs, k):
    ordem = sorted(probs.items(), key=lambda kv: -kv[1])[:k]
    total = sum(p for _, p in ordem)
    return {tok: p / total for tok, p in ordem}

def top_p(probs, p):
    ordem = sorted(probs.items(), key=lambda kv: -kv[1])
    mantidos, acumulado = [], 0.0
    for tok, prob in ordem:
        mantidos.append((tok, prob))
        acumulado += prob
        if acumulado >= p:
            break
    total = sum(pr for _, pr in mantidos)
    return {tok: pr / total for tok, pr in mantidos}

probs = {"Paris": 0.55, "Lyon": 0.25, "Roma": 0.12, "Nice": 0.05, "banana": 0.03}
print("top-k=2:", {t: round(p, 2) for t, p in top_k(probs, 2).items()})
print("top-p=0.9:", {t: round(p, 2) for t, p in top_p(probs, 0.9).items()})

filtrado = top_p(probs, 0.9)
rng = random.Random(42)  # semente fixa: mesma sequencia em toda execucao
amostras = rng.choices(list(filtrado), weights=list(filtrado.values()), k=10)
print("10 sorteios:", amostras)
```

function arred(x, casas) {
  const m = x * 10 ** casas, r = Math.round(m);
  const n = Math.abs(m % 1) === 0.5 && r % 2 !== 0 ? r - 1 : r;
  return n / 10 ** casas;
}

function topK(probs, k) {
  const ordem = [...probs].sort((a, b) => b[1] - a[1]).slice(0, k); // sort é estável
  const total = ordem.reduce((s, [, p]) => s + p, 0);
  return new Map(ordem.map(([tok, p]) => [tok, p / total]));
}

function topP(probs, p) {
  const ordem = [...probs].sort((a, b) => b[1] - a[1]);
  const mantidos = [];
  let acumulado = 0.0;
  for (const [tok, prob] of ordem) {
    mantidos.push([tok, prob]);
    acumulado += prob;
    if (acumulado >= p) break;
  }
  const total = mantidos.reduce((s, [, pr]) => s + pr, 0);
  return new Map(mantidos.map(([tok, pr]) => [tok, pr / total]));
}

static LinkedHashMap<String, Double> topK(LinkedHashMap<String, Double> probs, int k) {
    List<Map.Entry<String, Double>> ordem = new ArrayList<>(probs.entrySet());
    ordem.sort((a, b) -> Double.compare(b.getValue(), a.getValue())); // List.sort é estável
    ordem = ordem.subList(0, Math.min(k, ordem.size()));
    double total = 0;

    LinkedHashMap<String, Double> r = new LinkedHashMap<>();
    for (Map.Entry<String, Double> e : ordem) r.put(e.getKey(), e.getValue() / total);
    return r;
}

static List<KeyValuePair<string, double>> TopK(List<KeyValuePair<string, double>> probs, int k)
{
    var ordem = probs.OrderBy(kv => -kv.Value).Take(k).ToList(); // OrderBy é estável
    double total = 0;
    foreach (var kv in ordem) total += kv.Value;
    return ordem.Select(kv => new KeyValuePair<string, double>(kv.Key, kv.Value / total)).ToList();
}

static List<KeyValuePair<string, double>> TopP(List<KeyValuePair<string, double>> probs, double p)
{
    var mantidos = new List<KeyValuePair<string, double>>();
    double acumulado = 0.0;
    foreach (var kv in probs.OrderBy(x => -x.Value))
    {
        mantidos.Add(kv);
        acumulado += kv.Value;
        if (acumulado >= p) break;
    }
    double total = 0;
    foreach (var kv in mantidos) total += kv.Value;
    return mantidos.Select(kv => new KeyValuePair<string, double>(kv.Key, kv.Value / total)).ToList();
}

static string Arredondado(List<KeyValuePair<string, double>> m)
{
    // round(x, 2) do Python arredonda o empate para o par
    var partes = m.Select(kv => "'" + kv.Key + "': " +
        Math.Round(kv.Value, 2, MidpointRounding.ToEven).ToString(CultureInfo.InvariantCulture));
    return "{" + string.Join(", ", partes) + "}";
}

// como random.choices: pesos acumulados + busca binária (o gerador difere do Python)
static List<string> Escolhas(Random rng, List<KeyValuePair<string, double>> itens, int k)
{
    var acum = new double[itens.Count];
    double soma = 0;
    for (int i = 0; i < itens.Count; i++) acum[i] = (soma += itens[i].Value);
    var saida = new List<string>();
    for (int i = 0; i < k; i++)
    {
        double x = rng.NextDouble() * soma;
        int lo = 0, hi = acum.Length - 1;
        while (lo < hi)
        {
            int meio = (lo + hi) / 2;
            if (x < acum[meio]) hi = meio;
            else lo = meio + 1;
        }
        saida.Add(itens[lo].Key);
    }
    return saida;
}

static void Main()
{
    var probs = new List<KeyValuePair<string, double>>
    {
        new KeyValuePair<string, double>("Paris", 0.55),
        new KeyValuePair<string, double>("Lyon", 0.25),

import (
	"fmt"
	"math/rand"
	"sort"
	"strconv"
	"strings"
)

// par token/probabilidade; o slice mantém a ordem de inserção, como o dict
type Par struct {
	Tok  string
	Prob float64
}

// round(x, 2) do Python: o fmt arredonda o empate para o par, como ele
func arred(x float64, casas int) float64 {
	v, _ := strconv.ParseFloat(strconv.FormatFloat(x, 'f', casas, 64), 64)
	return v
}

func ordenar(probs []Par) []Par {
	ordem := append([]Par{}, probs...)
	sort.SliceStable(ordem, func(i, j int) bool { return ordem[i].Prob > ordem[j].Prob })
	return ordem
}

func normalizarPares(itens []Par) []Par {
	total := 0.0
	for _, p := range itens {
		total += p.Prob
	}
	r := []Par{}
	for _, p := range itens {
		r = append(r, Par{p.Tok, p.Prob / total})
	}
	return r
}

func topK(probs []Par, k int) []Par {
	ordem := ordenar(probs)
	if k < len(ordem) {
		ordem = ordem[:k]
	}
	return normalizarPares(ordem)
}

func topP(probs []Par, p float64) []Par {
	mantidos := []Par{}
	acumulado := 0.0
	for _, par := range ordenar(probs) {
		mantidos = append(mantidos, par)
		acumulado += par.Prob
		if acumulado >= p {
			break
		}
	}
	return normalizarPares(mantidos)
}

func arredondado(m []Par) string {
	partes := []string{}
	for _, p := range m {
		partes = append(partes, fmt.Sprintf("'%s': %v", p.Tok, arred(p.Prob, 2)))
	}
	return "{" + strings.Join(partes, ", ") + "}"
}

// como random.choices: pesos acumulados + busca binária (o gerador difere do Python)
func escolhas(rng *rand.Rand, itens []Par, k int) []string {
	acum := []float64{}
	soma := 0.0
	for _, p := range itens {
		soma += p.Prob
		acum = append(acum, soma)
	}
	saida := []string{}
	for i := 0; i < k; i++ {
		x := rng.Float64() * soma
		pos := sort.Search(len(acum), func(j int) bool { return acum[j] > x })
		if pos == len(acum) {
			pos--
		}
		saida = append(saida, "'"+itens[pos].Tok+"'")
	}
	return saida
}

func main() {
	probs := []Par{{"Paris", 0.55}, {"Lyon", 0.25}, {"Roma", 0.12}, {"Nice", 0.05}, {"banana", 0.03}}
	fmt.Println("top-k=2:", arredondado(topK(probs, 2)))
	fmt.Println("top-p=0.9:", arredondado(topP(probs, 0.9)))

	filtrado := topP(probs, 0.9)
	rng := rand.New(rand.NewSource(42)) // semente fixa: mesma sequencia em toda execucao
	amostras := escolhas(rng, filtrado, 10)
	fmt.Println("10 sorteios: [" + strings.Join(amostras, ", ") + "]")
}

## Que valores usar

| Tarefa | Configuração típica | Por quê |
| --- | --- | --- |
| Extração, classificação, SQL | Temperatura baixa (0 a 0,3) | Você quer a resposta mais provável, sempre igual |
| Assistente geral, resumo | Temperatura média (perto de 0,7) | Texto natural sem desviar |
| Brainstorm, variações criativas | Temperatura alta, top-p alto | Diversidade importa mais que precisão |
| Gerar N candidatos e escolher | Temperatura média ou alta | Precisa de candidatos diferentes entre si |

Ajuste temperatura **ou** top-p, de preferência não os dois ao mesmo tempo, e meça com avaliação em vez de chutar.

<Callout type="warning" title="Temperatura 0 não garante determinismo">
Mesmo no greedy, provedores podem devolver respostas levemente diferentes para a mesma entrada, por causa de lotes no servidor, paralelismo em ponto flutuante ou atualizações silenciosas do modelo. Se o seu sistema precisa de reprodutibilidade, registre entradas e saídas e use cache, em vez de confiar na API.
</Callout>

## Outras alavancas

- **Penalidades de repetição/frequência:** reduzem a probabilidade de tokens já usados, úteis contra laços.
- **Sequências de parada:** encerram a geração ao ver um marcador.
- **Limite de tokens de saída:** controla custo, mas corta a resposta se for baixo demais (verifique o motivo de término devolvido pela API).
