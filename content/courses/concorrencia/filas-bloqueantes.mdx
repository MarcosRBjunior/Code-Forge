Quase todo sistema concorrente tem um ponto em que uma parte do código produz trabalho e outra o consome. Você pode coordenar isso com lock e variável de condição, mas é fácil errar: esquecer o `while` em volta do `wait`, avisar a condição errada, deixar a fila crescer sem limite. A **fila bloqueante** empacota esse padrão numa estrutura pronta: ela é thread-safe, bloqueia quem tenta tirar de uma fila vazia e, se tiver capacidade máxima, bloqueia quem tenta colocar numa fila cheia.

## Como funciona

Por dentro, uma fila bloqueante é exatamente o que você montaria à mão: um buffer, um lock e duas condições, "não vazia" e "não cheia".

```text
 produtores                         consumidores
   put() ──►  [ t1 | t2 | t3 | __ ]  ──► get()
   bloqueia se cheia                 bloqueia se vazia
              lock + not_full + not_empty
```

- `put(item)`: adquire o lock, espera enquanto a fila estiver cheia, insere e acorda um consumidor.
- `get()`: adquire o lock, espera enquanto estiver vazia, remove e acorda um produtor.

Esse limite de capacidade é o que dá **backpressure**: se os consumidores ficam lentos, os produtores param naturalmente em vez de encher a memória.

Em Python a classe é `queue.Queue` (também existem `LifoQueue` e `PriorityQueue`). Em Java, a interface `BlockingQueue` tem `ArrayBlockingQueue` (limitada, array circular), `LinkedBlockingQueue` (opcionalmente limitada) e `SynchronousQueue` (capacidade zero, cada `put` espera um `take`). Em Go, o papel é do **canal** com buffer: `make(chan T, n)`. Em C++ não há fila bloqueante na biblioteca padrão; você escreve uma com `std::mutex` e `std::condition_variable`.

## Exemplo: pipeline com fila limitada

O programa abaixo tem um produtor e dois consumidores. O produtor envia um **sentinela** (`None`) por consumidor para avisar que acabou. Cada consumidor guarda seus resultados localmente e o programa só imprime depois dos `join`, com os dados ordenados, para a saída não depender do escalonador.

<CodeTabs>

```python
import queue
import threading

fila = queue.Queue(maxsize=3)   # no máximo 3 itens esperando
FIM = None
resultados = []
lock = threading.Lock()

def produtor(n, consumidores):
    for i in range(1, n + 1):
        fila.put(i)              # bloqueia se a fila estiver cheia
    for _ in range(consumidores):
        fila.put(FIM)            # um sentinela por consumidor

def consumidor():
    locais = []
    while True:
        item = fila.get()        # bloqueia se a fila estiver vazia
        if item is FIM:
            break
        locais.append(item * item)
    with lock:
        resultados.extend(locais)

consumidores = [threading.Thread(target=consumidor) for _ in range(2)]
for t in consumidores:
    t.start()
p = threading.Thread(target=produtor, args=(8, len(consumidores)))
p.start()
p.join()
for t in consumidores:
    t.join()

print("quadrados:", sorted(resultados))
print("soma:", sum(resultados))
```

```java
import java.util.*;
import java.util.concurrent.ArrayBlockingQueue;
import java.util.concurrent.BlockingQueue;

static final BlockingQueue<Integer> fila = new ArrayBlockingQueue<>(3);   // no máximo 3 itens esperando
static final int FIM = -1;   // ArrayBlockingQueue não aceita null: a sentinela é um valor que nunca é item
static final List<Integer> resultados = new ArrayList<>();
static final Object lock = new Object();

static void produtor(int n, int consumidores) {
    try {
        for (int i = 1; i <= n; i++) {
            fila.put(i);             // bloqueia se a fila estiver cheia
        }
        for (int i = 0; i < consumidores; i++) {
            fila.put(FIM);           // um sentinela por consumidor
        }
    } catch (InterruptedException e) {
        Thread.currentThread().interrupt();
    }
}

static void consumidor() {
    List<Integer> locais = new ArrayList<>();
    try {
        while (true) {
            int item = fila.take();  // bloqueia se a fila estiver vazia
            if (item == FIM) {
                break;
            }
            locais.add(item * item);
        }
    } catch (InterruptedException e) {
        Thread.currentThread().interrupt();
    }
    synchronized (lock) {
        resultados.addAll(locais);
    }
}

public static void main(String[] args) throws InterruptedException {
    List<Thread> consumidores = new ArrayList<>();
    for (int i = 0; i < 2; i++) {
        consumidores.add(new Thread(() -> consumidor()));
    }
    for (Thread t : consumidores) t.start();
    Thread p = new Thread(() -> produtor(8, consumidores.size()));
    p.start();
    p.join();
    for (Thread t : consumidores) t.join();

    Collections.sort(resultados);
    int soma = 0;
    for (int x : resultados) soma += x;
    System.out.println("quadrados: " + resultados);
    System.out.println("soma: " + soma);
}
```

```cpp
#include <algorithm>
#include <condition_variable>
#include <deque>
#include <iostream>
#include <mutex>
#include <optional>
#include <thread>
#include <vector>
using namespace std;

// C++ não tem fila bloqueante na biblioteca padrão: um buffer, um lock e duas condições
class FilaBloqueante {
    deque<optional<int>> itens;
    size_t capacidade;
    mutex m;
    condition_variable nao_cheia, nao_vazia;

public:
    explicit FilaBloqueante(size_t capacidade) : capacidade(capacidade) {}

    void put(optional<int> item) {
        unique_lock<mutex> trava(m);
        while (itens.size() == capacidade) {
            nao_cheia.wait(trava);
        }
        itens.push_back(item);
        nao_vazia.notify_one();
    }

    optional<int> get() {
        unique_lock<mutex> trava(m);
        while (itens.empty()) {
            nao_vazia.wait(trava);
        }
        optional<int> item = itens.front();
        itens.pop_front();
        nao_cheia.notify_one();
        return item;
    }
};

FilaBloqueante fila(3);   // no máximo 3 itens esperando
const optional<int> FIM = nullopt;
vector<int> resultados;
mutex lock_resultados;

void produtor(int n, int consumidores) {
    for (int i = 1; i <= n; i++) {
        fila.put(i);             // bloqueia se a fila estiver cheia
    }
    for (int i = 0; i < consumidores; i++) {
        fila.put(FIM);           // um sentinela por consumidor
    }
}

void consumidor() {
    vector<int> locais;
    while (true) {
        optional<int> item = fila.get();  // bloqueia se a fila estiver vazia
        if (item == FIM) {
            break;
        }
        locais.push_back(*item * *item);
    }
    lock_guard<mutex> guarda(lock_resultados);
    resultados.insert(resultados.end(), locais.begin(), locais.end());
}

int main() {
    vector<thread> consumidores;
    for (int i = 0; i < 2; i++) {
        consumidores.emplace_back(consumidor);
    }
    thread p(produtor, 8, (int)consumidores.size());
    p.join();
    for (auto& t : consumidores) {
        t.join();
    }

    sort(resultados.begin(), resultados.end());
    int soma = 0;
    cout << "quadrados: [";
    for (size_t i = 0; i < resultados.size(); i++) {
        cout << (i ? ", " : "") << resultados[i];
        soma += resultados[i];
    }
    cout << "]" << endl;
    cout << "soma: " << soma << endl;
}
```

```go
	"fmt"
	"sort"
	"sync"

var fila = make(chan int, 3) // canal com buffer: no máximo 3 itens esperando

// canal de int não tem None: a sentinela é um valor que nunca é item (fechar o canal seria o mais comum em Go)

var lock sync.Mutex

func produtor(n, consumidores int, wg *sync.WaitGroup) {
	defer wg.Done()
	for i := 1; i <= n; i++ {
		fila <- i // bloqueia se o canal estiver cheio
	}
	for i := 0; i < consumidores; i++ {
		fila <- FIM // um sentinela por consumidor
	}

	defer wg.Done()
	var locais []int

		item := <-fila // bloqueia se o canal estiver vazio
		if item == FIM {
			break
		}
		locais = append(locais, item*item)
	}
	lock.Lock()
	resultados = append(resultados, locais...)
	lock.Unlock()
}

		go consumidor(&wg)
	}

	wg.Wait()

	soma := 0
	for _, x := range resultados {
		soma += x
	}
```

## Timeouts, operações não bloqueantes e task_done

Bloquear para sempre raramente é o que você quer em produção. Toda fila bloqueante oferece variantes: esperar com tempo limite ou falhar na hora. Em Python, `put(item, timeout=s)` lança `queue.Full` e `get(timeout=s)` lança `queue.Empty`; `put_nowait` e `get_nowait` não esperam nada. Em Java são `offer`/`poll` com tempo, e em Go você usa `select` com `default` ou com `time.After`.

Outra peça útil é a contagem de tarefas pendentes: cada `get` seguido de `task_done()` marca um item como processado, e `fila.join()` espera até todos os itens colocados terem sido concluídos.

<CodeTabs>

```python
import queue
import threading

fila = queue.Queue(maxsize=2)
fila.put("a")
fila.put("b")
try:
    fila.put("c", timeout=0.05)
except queue.Full:
    print("cheia: 'c' foi recusado")

print("tirou:", fila.get_nowait(), fila.get_nowait())
try:
    fila.get_nowait()
except queue.Empty:
    print("vazia: nada para tirar")

trabalho = queue.Queue()         # fila nova: 'a' e 'b' nunca tiveram task_done
processados = []

def trabalhador():
    while True:
        item = trabalho.get()
        processados.append(item.upper())
        trabalho.task_done()     # avisa que este item terminou

threading.Thread(target=trabalhador, daemon=True).start()
for letra in "xyz":
    trabalho.put(letra)
trabalho.join()                  # espera todos os task_done
print("processados:", processados)
```

```java
import java.util.*;
import java.util.concurrent.ArrayBlockingQueue;
import java.util.concurrent.BlockingQueue;
import java.util.concurrent.CountDownLatch;
import java.util.concurrent.LinkedBlockingQueue;
import java.util.concurrent.TimeUnit;

static final BlockingQueue<String> trabalho = new LinkedBlockingQueue<>();
// BlockingQueue não tem task_done/join: com o total conhecido, um latch faz esse papel
static final CountDownLatch pendentes = new CountDownLatch(3);
static final List<String> processados = new ArrayList<>();

static void trabalhador() {
    try {
        while (true) {
            String item = trabalho.take();
            processados.add(item.toUpperCase());
            pendentes.countDown();     // avisa que este item terminou
        }
    } catch (InterruptedException e) {
        Thread.currentThread().interrupt();
    }
}

public static void main(String[] args) throws InterruptedException {
    BlockingQueue<String> fila = new ArrayBlockingQueue<>(2);
    fila.put("a");
    fila.put("b");
    if (!fila.offer("c", 50, TimeUnit.MILLISECONDS)) {  // devolve false em vez de lançar exceção
        System.out.println("cheia: 'c' foi recusado");
    }

    String primeiro = fila.poll();
    String segundo = fila.poll();
    System.out.println("tirou: " + primeiro + " " + segundo);
    if (fila.poll() == null) {  // poll sem tempo não espera: devolve null se estiver vazia
        System.out.println("vazia: nada para tirar");
    }

    Thread t = new Thread(() -> trabalhador());
    t.setDaemon(true);
    t.start();
    for (String letra : "xyz".split("")) {
        trabalho.put(letra);
    }
    pendentes.await();                 // espera todos os itens terminarem
    System.out.println("processados: " + processados);
}
```

```cpp
#include <cctype>
#include <chrono>
#include <condition_variable>
#include <deque>
#include <iostream>
#include <mutex>
#include <optional>
#include <string>
#include <thread>
#include <vector>
using namespace std;

// C++ não tem fila bloqueante na biblioteca padrão: esta cobre o que a queue.Queue oferece aqui
class FilaBloqueante {
    deque<string> itens;
    size_t capacidade;   // 0 = sem limite, como na queue.Queue
    size_t pendentes = 0;
    mutex m;
    condition_variable nao_cheia, nao_vazia, tudo_feito;

    bool cheia() const { return capacidade > 0 && itens.size() >= capacidade; }

    void inserir(const string& item) {
        itens.push_back(item);
        pendentes += 1;
        nao_vazia.notify_one();
    }

public:
    explicit FilaBloqueante(size_t capacidade = 0) : capacidade(capacidade) {}

    void put(const string& item) {
        unique_lock<mutex> trava(m);
        while (cheia()) {
            nao_cheia.wait(trava);
        }
        inserir(item);
    }

    // devolve false se o prazo acabar com a fila ainda cheia
    bool put(const string& item, chrono::milliseconds timeout) {
        unique_lock<mutex> trava(m);
        if (!nao_cheia.wait_for(trava, timeout, [this] { return !cheia(); })) {
            return false;
        }
        inserir(item);
        return true;
    }

    string get() {
        unique_lock<mutex> trava(m);
        while (itens.empty()) {
            nao_vazia.wait(trava);
        }
        string item = itens.front();
        itens.pop_front();
        nao_cheia.notify_one();
        return item;
    }

    // não espera: devolve vazio se não houver item
    optional<string> get_nowait() {
        lock_guard<mutex> trava(m);
        if (itens.empty()) {
            return nullopt;
        }
        string item = itens.front();
        itens.pop_front();
        nao_cheia.notify_one();
        return item;
    }

    void task_done() {
        lock_guard<mutex> trava(m);
        pendentes -= 1;
        if (pendentes == 0) {
            tudo_feito.notify_all();
        }
    }

    void join() {
        unique_lock<mutex> trava(m);
        while (pendentes > 0) {
            tudo_feito.wait(trava);
        }
    }
};

FilaBloqueante trabalho;         // fila nova: 'a' e 'b' nunca tiveram task_done
vector<string> processados;

void trabalhador() {
    while (true) {
        string item = trabalho.get();
        if (item.empty()) {      // C++ não tem thread daemon: a sentinela encerra o trabalhador
            return;
        }
        for (char& c : item) {
            c = toupper(c);
        }
        processados.push_back(item);
        trabalho.task_done();    // avisa que este item terminou
    }
}

int main() {
    FilaBloqueante fila(2);
    fila.put("a");
    fila.put("b");
    if (!fila.put("c", chrono::milliseconds(50))) {
        cout << "cheia: 'c' foi recusado" << endl;
    }

    string primeiro = *fila.get_nowait();
    string segundo = *fila.get_nowait();
    cout << "tirou: " << primeiro << " " << segundo << endl;
    if (!fila.get_nowait()) {
        cout << "vazia: nada para tirar" << endl;
    }

    thread t(trabalhador);
    for (char letra : string("xyz")) {
        trabalho.put(string(1, letra));
    }
    trabalho.join();             // espera todos os task_done
    cout << "processados: [";
    for (size_t i = 0; i < processados.size(); i++) {
        cout << (i ? ", '" : "'") << processados[i] << "'";
    }
    cout << "]" << endl;

    trabalho.put("");
    t.join();
}
```

```go
import (
	"fmt"
	"strings"
	"sync"
	"time"
)

func main() {
	fila := make(chan string, 2)
	fila <- "a"
	fila <- "b"
	select {
	case fila <- "c":
	case <-time.After(50 * time.Millisecond):
		fmt.Println("cheia: 'c' foi recusado")
	}

	primeiro, segundo := <-fila, <-fila
	fmt.Println("tirou:", primeiro, segundo)
	select { // com default, o select não espera
	case <-fila:
	default:
		fmt.Println("vazia: nada para tirar")
	}

	trabalho := make(chan string, 3) // canal sempre tem capacidade fixa; com 3 lugares, todos os itens cabem
	var pendentes sync.WaitGroup     // canal não tem task_done/join: o WaitGroup faz esse papel
	var processados []string

	go func() { // como a thread daemon, esta goroutine morre quando main termina
		for item := range trabalho {
			processados = append(processados, strings.ToUpper(item))
			pendentes.Done() // avisa que este item terminou
		}
	}()
	for _, letra := range "xyz" {
		pendentes.Add(1)
		trabalho <- string(letra)
	}
	pendentes.Wait() // espera todos os Done
	fmt.Println("processados:", processados)
}
```

## Armadilhas e trade-offs

- **Fila sem limite** esconde o problema: se o consumidor é mais lento, a memória cresce até o processo morrer. Prefira capacidade explícita.
- **Encerramento**: sem sentinela, `close` ou cancelamento, consumidores ficam presos no `get` para sempre. Em Go, feche o canal e use `for x := range ch`; em Python 3.13, `Queue.shutdown()` também resolve.
- **Esquecer `task_done`** faz o `join` travar; chamar a mais lança `ValueError`.
- **Checar e agir**: `if not fila.empty(): fila.get()` é uma condição de corrida, porque outro consumidor pode esvaziar a fila entre as duas chamadas. Use `get_nowait` e trate a exceção.
- **Capacidade** é um trade-off: pequena demais, produtores passam muito tempo bloqueados; grande demais, a latência dos itens enfileirados aumenta.

<Callout type="tip" title="Fila como fronteira">
Ao desenhar uma solução concorrente, pense em filas como fronteiras entre estágios. Cada estágio tem seu próprio número de threads, e o tamanho da fila define quanto trabalho pode se acumular entre eles.
</Callout>
