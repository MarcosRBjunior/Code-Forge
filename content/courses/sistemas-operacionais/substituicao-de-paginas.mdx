A RAM está cheia e um processo acaba de gerar um page fault. Para trazer a página nova, o kernel precisa expulsar alguma que está na memória. Se escolher uma que vai ser usada logo em seguida, terá outro page fault em instantes. A regra de escolha é o **algoritmo de substituição de páginas**, e o mesmo problema reaparece em qualquer cache que você projetar: Redis, CDN, buffer pool de banco ou um `functools.lru_cache`.

## Os algoritmos clássicos

| Algoritmo | Expulsa | Prós | Contras |
| --- | --- | --- | --- |
| FIFO | A página que entrou há mais tempo | Trivial | Ignora uso; sofre anomalia de Belady |
| OPT (ótimo) | A que vai demorar mais para ser usada | Mínimo possível de faltas | Exige conhecer o futuro |
| LRU | A usada há mais tempo | Ótima aproximação na prática | Caro de manter exato em hardware |
| Clock (segunda chance) | A primeira sem bit de acesso, girando um ponteiro | Barato, aproxima LRU | Menos preciso que LRU |

O OPT serve como régua: nenhum algoritmo real pode fazer menos faltas que ele. Os outros são comparados contra esse limite.

from collections import OrderedDict, deque

def fifo(refs, quadros):
    mem, fila, faltas = set(), deque(), 0
    for p in refs:
        if p not in mem:
            faltas += 1
            if len(mem) == quadros:
                mem.remove(fila.popleft())
            mem.add(p)
            fila.append(p)
    return faltas

def lru(refs, quadros):
    mem, faltas = OrderedDict(), 0
    for p in refs:
        if p in mem:
            mem.move_to_end(p)
        else:
            faltas += 1
            if len(mem) == quadros:
                mem.popitem(last=False)
            mem[p] = True
    return faltas

def opt(refs, quadros):
    mem, faltas = [], 0
    for i, p in enumerate(refs):
        if p in mem:
            continue
        faltas += 1
        if len(mem) == quadros:
            futuro = refs[i + 1:]
            vitima = max(mem, key=lambda q: futuro.index(q) if q in futuro else len(futuro) + 1)
            mem.remove(vitima)
        mem.append(p)
    return faltas

def clock(refs, quadros):
    mem, bit, ponteiro, faltas = [None] * quadros, [0] * quadros, 0, 0
    for p in refs:
        if p in mem:
            bit[mem.index(p)] = 1
            continue
        faltas += 1
        while bit[ponteiro] == 1:          # segunda chance: zera e avanca
            bit[ponteiro] = 0
            ponteiro = (ponteiro + 1) % quadros
        mem[ponteiro], bit[ponteiro] = p, 1
        ponteiro = (ponteiro + 1) % quadros
    return faltas

  const mem = new Set();

    if (!mem.has(p)) {
      faltas += 1;
      if (mem.size === quadros) {

      }
      mem.add(p);
      fila.push(p);
    }
  }

}

function lru(refs, quadros) {
  const mem = new Map();   // Map guarda a ordem de inserção: o primeiro é o menos usado
  let faltas = 0;
  for (const p of refs) {
    if (mem.has(p)) {
      mem.delete(p);       // move para o fim
      mem.set(p, true);
    } else {
      faltas += 1;
      if (mem.size === quadros) {
        mem.delete(mem.keys().next().value);
      }
      mem.set(p, true);
    }
  }
  return faltas;
}

function opt(refs, quadros) {
  const mem = [];
  let faltas = 0;
  refs.forEach((p, i) => {
    if (mem.includes(p)) return;
    faltas += 1;
    if (mem.length === quadros) {
      const futuro = refs.slice(i + 1);
      const distancia = (q) => (futuro.includes(q) ? futuro.indexOf(q) : futuro.length + 1);

      mem.splice(mem.indexOf(vitima), 1);

    mem.push(p);

}

import java.util.*;
import java.util.function.BiFunction;

static int fifo(List<Integer> refs, int quadros) {
    Set<Integer> mem = new HashSet<>();
    ArrayDeque<Integer> fila = new ArrayDeque<>();
    int faltas = 0;
    for (int p : refs) {
        if (!mem.contains(p)) {
            faltas += 1;
            if (mem.size() == quadros) {
                mem.remove(fila.pollFirst());
            }
            mem.add(p);
            fila.addLast(p);
        }
    }
    return faltas;
}

static int lru(List<Integer> refs, int quadros) {
    LinkedHashMap<Integer, Boolean> mem = new LinkedHashMap<>();   // guarda a ordem de inserção: o primeiro é o menos usado
    int faltas = 0;
    for (int p : refs) {
        if (mem.containsKey(p)) {
            mem.remove(p);                                         // move para o fim
            mem.put(p, true);
        } else {
            faltas += 1;
            if (mem.size() == quadros) {
                mem.remove(mem.keySet().iterator().next());
            }
            mem.put(p, true);
        }
    }
    return faltas;
}

static int opt(List<Integer> refs, int quadros) {
    List<Integer> mem = new ArrayList<>();
    int faltas = 0;
    for (int i = 0; i < refs.size(); i++) {
        int p = refs.get(i);
        if (mem.contains(p)) continue;
        faltas += 1;
        if (mem.size() == quadros) {
            List<Integer> futuro = refs.subList(i + 1, refs.size());
            int vitima = mem.get(0);
            int maior = -1;
            for (int q : mem) {
                int d = futuro.contains(q) ? futuro.indexOf(q) : futuro.size() + 1;
                if (d > maior) {   // o primeiro máximo, como o max do Python
                    maior = d;
                    vitima = q;
                }
            }
            mem.remove(Integer.valueOf(vitima));
        }
        mem.add(p);
    }
    return faltas;
}

static int clock(List<Integer> refs, int quadros) {
    Integer[] mem = new Integer[quadros];   // null = quadro vazio
    int[] bit = new int[quadros];
    int ponteiro = 0, faltas = 0;
    for (int p : refs) {
        int pos = Arrays.asList(mem).indexOf(p);

        mem[ponteiro] = p;
        bit[ponteiro] = 1;
        ponteiro = (ponteiro + 1) % quadros;
    }

}

int fifo(const vector<int>& refs, size_t quadros) {
    unordered_set<int> mem;
    deque<int> fila;
    int faltas = 0;
    for (int p : refs) {
        if (!mem.count(p)) {
            faltas += 1;
            if (mem.size() == quadros) {
                mem.erase(fila.front());
                fila.pop_front();
            }
            mem.insert(p);
            fila.push_back(p);
        }
    }
    return faltas;
}

int lru(const vector<int>& refs, size_t quadros) {
    list<int> mem;   // o primeiro da lista é o menos usado
    int faltas = 0;
    for (int p : refs) {
        auto it = find(mem.begin(), mem.end(), p);
        if (it != mem.end()) {
            mem.erase(it);       // move para o fim
            mem.push_back(p);
        } else {
            faltas += 1;
            if (mem.size() == quadros) {
                mem.pop_front();
            }
            mem.push_back(p);
        }
    }
    return faltas;
}

int opt(const vector<int>& refs, size_t quadros) {
    vector<int> mem;
    int faltas = 0;
    for (size_t i = 0; i < refs.size(); i++) {
        int p = refs[i];
        if (find(mem.begin(), mem.end(), p) != mem.end()) continue;
        faltas += 1;
        if (mem.size() == quadros) {
            vector<int> futuro(refs.begin() + i + 1, refs.end());
            auto distancia = [&](int q) {
                auto it = find(futuro.begin(), futuro.end(), q);
                return it != futuro.end() ? (long)(it - futuro.begin()) : (long)futuro.size() + 1;
            };
            // max_element devolve o primeiro máximo, como o max do Python
            auto vitima = max_element(mem.begin(), mem.end(), [&](int a, int b) { return distancia(a) < distancia(b); });
            mem.erase(vitima);
        }
        mem.push_back(p);
    }
    return faltas;
}

int clock(const vector<int>& refs, size_t quadros) {
    vector<optional<int>> mem(quadros);   // nullopt = quadro vazio
    vector<int> bit(quadros, 0);
    size_t ponteiro = 0;
    int faltas = 0;
    for (int p : refs) {
        auto it = find(mem.begin(), mem.end(), optional<int>(p));
        if (it != mem.end()) {
            bit[it - mem.begin()] = 1;
            continue;

        faltas += 1;
        while (bit[ponteiro] == 1) {          // segunda chance: zera e avanca
            bit[ponteiro] = 0;
            ponteiro = (ponteiro + 1) % quadros;
        }
        mem[ponteiro] = p;
        bit[ponteiro] = 1;
        ponteiro = (ponteiro + 1) % quadros;
    }
    return faltas;
}

{
    var mem = new HashSet<int>();
    var fila = new Queue<int>();

    foreach (int p in refs)

        if (!mem.Contains(p))
        {

            if (mem.Count == quadros)

{
    var mem = new LinkedList<int>();   // o primeiro da lista é o menos usado

    foreach (int p in refs)
    {
        if (mem.Contains(p))

            mem.Remove(p);              // move para o fim
            mem.AddLast(p);
        }
        else
        {
            faltas += 1;
            if (mem.Count == quadros)

            mem.AddLast(p);
        }
    }
    return faltas;

import "fmt"

func fifo(refs []int, quadros int) int {
	mem := map[int]bool{}
	fila := []int{}
	faltas := 0
	for _, p := range refs {
		if !mem[p] {
			faltas += 1
			if len(mem) == quadros {
				delete(mem, fila[0])
				fila = fila[1:]
			}
			mem[p] = true
			fila = append(fila, p)
		}
	}
	return faltas
}

func indice(lista []int, x int) int {
	for k, v := range lista {
		if v == x {
			return k
		}
	}
	return -1
}

func lru(refs []int, quadros int) int {
	mem := []int{} // o primeiro da lista é o menos usado
	faltas := 0
	for _, p := range refs {
		if k := indice(mem, p); k >= 0 {
			mem = append(mem[:k], mem[k+1:]...) // move para o fim
			mem = append(mem, p)
		} else {
			faltas += 1
			if len(mem) == quadros {
				mem = mem[1:]
			}
			mem = append(mem, p)
		}
	}
	return faltas
}

func opt(refs []int, quadros int) int {
	mem := []int{}
	faltas := 0
	for i, p := range refs {
		if indice(mem, p) >= 0 {
			continue
		}
		faltas += 1
		if len(mem) == quadros {
			futuro := refs[i+1:]
			vitima, maior := 0, -1
			for k, q := range mem {
				d := indice(futuro, q)
				if d < 0 {
					d = len(futuro) + 1
				}
				if d > maior { // o primeiro máximo, como o max do Python
					maior, vitima = d, k
				}
			}
			mem = append(mem[:vitima], mem[vitima+1:]...)
		}
		mem = append(mem, p)
	}
	return faltas
}

func clock(refs []int, quadros int) int {
	mem := make([]int, quadros) // -1 = quadro vazio
	for k := range mem {
		mem[k] = -1
	}
	bit := make([]int, quadros)
	ponteiro, faltas := 0, 0
	for _, p := range refs {
		if k := indice(mem, p); k >= 0 {
			bit[k] = 1
			continue
		}
		faltas += 1
		for bit[ponteiro] == 1 { // segunda chance: zera e avanca
			bit[ponteiro] = 0
			ponteiro = (ponteiro + 1) % quadros
		}
		mem[ponteiro], bit[ponteiro] = p, 1
		ponteiro = (ponteiro + 1) % quadros
	}
	return faltas
}

## Anomalia de Belady

Intuitivamente, mais memória deveria significar menos faltas. Com FIFO, nem sempre: existem sequências em que aumentar o número de quadros **aumenta** as faltas. LRU e OPT são **algoritmos de pilha** (o conteúdo com N quadros está sempre contido no conteúdo com N + 1) e por isso nunca sofrem dessa anomalia.

## Como o Linux faz de verdade

LRU exato exigiria atualizar uma estrutura a cada acesso à memória, algo impossível de fazer em software na velocidade da CPU. O hardware só oferece o **bit de acesso** na entrada da tabela, ligado automaticamente quando a página é usada. Com ele, o kernel aproxima LRU:

- O Linux mantém listas de páginas **ativas** e **inativas**, separadas para páginas de arquivo e anônimas. Páginas novas entram na inativa e só sobem para a ativa se forem acessadas de novo, o que protege contra uma varredura única expulsar o working set (o mesmo problema que um `SELECT *` gigante causa num buffer pool com LRU ingênuo).
- Desde o kernel 6.1, o **MGLRU** usa várias gerações em vez de duas listas, com melhor precisão sob pressão de memória.
- Páginas **sujas** (modificadas) custam mais para expulsar, porque precisam ser escritas antes. Por isso o kernel prefere expulsar páginas limpas e grava as sujas em segundo plano.

<Callout type="tip" title="O mesmo trade-off nos seus caches">
LRU puro sofre com varreduras sequenciais; LFU sofre com itens que foram populares e deixaram de ser. Algoritmos como LRU-K, 2Q, ARC e W-TinyLFU (usado no Caffeine, em Java) combinam recência e frequência. Se o entrevistador perguntar como você escolheria a política de um cache, fale do padrão de acesso antes de citar um nome.
</Callout>
