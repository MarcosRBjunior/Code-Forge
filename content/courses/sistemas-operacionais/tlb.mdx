Com tabelas de páginas de quatro níveis, cada leitura de memória precisaria de outras quatro só para descobrir o endereço físico. O programa ficaria várias vezes mais lento. O que salva o desempenho é a **TLB** (Translation Lookaside Buffer): um cache pequeno e muito rápido, dentro da MMU, que guarda traduções recentes de página virtual para quadro físico. Quando a tradução está lá (um **hit**), ela sai em cerca de um ciclo. Quando não está (um **miss**), o hardware percorre as tabelas (**page walk**) e guarda o resultado.

## Por que funciona

A TLB tem poucas entradas: algo como 64 no primeiro nível e de 1.000 a 2.000 no segundo. Mesmo assim, as taxas de acerto passam de 99% em programas comuns, graças à **localidade**:

- **Temporal:** a página que você acabou de usar provavelmente será usada de novo (variáveis do laço, pilha).
- **Espacial:** os próximos acessos tendem a cair na mesma página (percorrer um array, executar instruções seguidas).

Uma entrada cobre 4 KB inteiros. Percorrer sequencialmente um array de 4 KB custa um miss e milhares de hits.

```text
 endereço virtual
       │
       ▼
   ┌────────┐  hit (~1 ciclo)
   │  TLB   │ ─────────────────────────► endereço físico
   └────────┘
       │ miss
       ▼
  page walk: PML4 → PDPT → PD → PT   (até 4 acessos à memória)
       │
       ├─ guarda na TLB (expulsando outra entrada)
       ▼
  endereço físico
```

## Simulando uma TLB

A simulação usa uma TLB de 4 entradas com substituição LRU e compara dois padrões de acesso a uma matriz de 64 × 1024 inteiros de 4 bytes (cada linha ocupa exatamente uma página): percorrer por linhas e percorrer por colunas.

<CodeTabs>

```python
from collections import OrderedDict

class TLB:
    def __init__(self, entradas):
        self.cache = OrderedDict()
        self.entradas = entradas
        self.hits = self.misses = 0

    def acessar(self, endereco):
        vpn = endereco // 4096
        if vpn in self.cache:
            self.cache.move_to_end(vpn)
            self.hits += 1
        else:
            self.misses += 1
            if len(self.cache) >= self.entradas:
                self.cache.popitem(last=False)
            self.cache[vpn] = True

LINHAS, COLUNAS, TAM = 64, 1024, 4
for nome in ["por linhas", "por colunas"]:
    tlb = TLB(entradas=4)
    if nome == "por linhas":
        ordem = ((i, j) for i in range(LINHAS) for j in range(COLUNAS))
    else:
        ordem = ((i, j) for j in range(COLUNAS) for i in range(LINHAS))
    for i, j in ordem:
        tlb.acessar((i * COLUNAS + j) * TAM)
    total = tlb.hits + tlb.misses
    print(f"{nome:<12} acessos={total} misses={tlb.misses:>5} taxa_de_acerto={tlb.hits / total:.2%}")
```

```javascript
class TLB {
  constructor(entradas) {
    this.cache = new Map();   // Map guarda a ordem de inserção: o primeiro é o menos usado
    this.entradas = entradas;
    this.hits = this.misses = 0;
  }

  acessar(endereco) {
    const vpn = Math.floor(endereco / 4096);
    if (this.cache.has(vpn)) {
      this.cache.delete(vpn);   // move para o fim
      this.cache.set(vpn, true);
      this.hits += 1;
    } else {
      this.misses += 1;
      if (this.cache.size >= this.entradas) {
        this.cache.delete(this.cache.keys().next().value);
      }
      this.cache.set(vpn, true);
    }
  }
}

const LINHAS = 64, COLUNAS = 1024, TAM = 4;
for (const nome of ["por linhas", "por colunas"]) {
  const tlb = new TLB(4);
  if (nome === "por linhas") {
    for (let i = 0; i < LINHAS; i++) for (let j = 0; j < COLUNAS; j++) tlb.acessar((i * COLUNAS + j) * TAM);
  } else {
    for (let j = 0; j < COLUNAS; j++) for (let i = 0; i < LINHAS; i++) tlb.acessar((i * COLUNAS + j) * TAM);
  }
  const total = tlb.hits + tlb.misses;
  console.log(
    `${nome.padEnd(12)} acessos=${total} misses=${String(tlb.misses).padStart(5)} taxa_de_acerto=${((tlb.hits / total) * 100).toFixed(2)}%`
  );
}
```

```java
import java.util.*;

static class TLB {
    LinkedHashMap<Integer, Boolean> cache = new LinkedHashMap<>();   // guarda a ordem de inserção: o primeiro é o menos usado
    int entradas;
    int hits = 0, misses = 0;

    TLB(int entradas) {
        this.entradas = entradas;
    }

    void acessar(int endereco) {
        int vpn = endereco / 4096;
        if (cache.containsKey(vpn)) {
            cache.remove(vpn);   // move para o fim
            cache.put(vpn, true);
            hits += 1;
        } else {
            misses += 1;
            if (cache.size() >= entradas) {
                cache.remove(cache.keySet().iterator().next());
            }
            cache.put(vpn, true);
        }
    }
}

public static void main(String[] args) {
    final int LINHAS = 64, COLUNAS = 1024, TAM = 4;
    for (String nome : new String[]{"por linhas", "por colunas"}) {
        TLB tlb = new TLB(4);
        if (nome.equals("por linhas")) {
            for (int i = 0; i < LINHAS; i++) for (int j = 0; j < COLUNAS; j++) tlb.acessar((i * COLUNAS + j) * TAM);
        } else {
            for (int j = 0; j < COLUNAS; j++) for (int i = 0; i < LINHAS; i++) tlb.acessar((i * COLUNAS + j) * TAM);
        }
        int total = tlb.hits + tlb.misses;
        System.out.println(String.format(Locale.US, "%-12s acessos=%d misses=%5d taxa_de_acerto=%.2f%%",
                nome, total, tlb.misses, (double) tlb.hits / total * 100));
    }
}
```

```cpp
#include <algorithm>
#include <cstdio>
#include <list>
#include <string>
using namespace std;

class TLB {
    list<int> cache;   // o primeiro da lista é o menos usado
    size_t entradas;

public:
    int hits = 0, misses = 0;

    explicit TLB(size_t entradas) : entradas(entradas) {}

    void acessar(int endereco) {
        int vpn = endereco / 4096;
        auto it = find(cache.begin(), cache.end(), vpn);
        if (it != cache.end()) {
            cache.erase(it);   // move para o fim
            cache.push_back(vpn);
            hits += 1;
        } else {
            misses += 1;
            if (cache.size() >= entradas) {
                cache.pop_front();
            }
            cache.push_back(vpn);
        }
    }
};

int main() {
    const int LINHAS = 64, COLUNAS = 1024, TAM = 4;
    for (string nome : {"por linhas", "por colunas"}) {
        TLB tlb(4);
        if (nome == "por linhas") {
            for (int i = 0; i < LINHAS; i++) for (int j = 0; j < COLUNAS; j++) tlb.acessar((i * COLUNAS + j) * TAM);
```

```csharp
using System;
using System.Collections.Generic;
using System.Globalization;

class TLB
{
    private readonly LinkedList<int> cache = new LinkedList<int>();   // o primeiro da lista é o menos usado
    private readonly int entradas;
    public int Hits = 0, Misses = 0;

    public TLB(int entradas)
    {
        this.entradas = entradas;
    }

    public void Acessar(int endereco)
    {
        int vpn = endereco / 4096;
        if (cache.Contains(vpn))
        {
            cache.Remove(vpn);   // move para o fim
            cache.AddLast(vpn);
            Hits += 1;
        }
        else
        {
            Misses += 1;
            if (cache.Count >= entradas)
                cache.RemoveFirst();
            cache.AddLast(vpn);
        }
    }
}

static void Main()
{
    const int LINHAS = 64, COLUNAS = 1024, TAM = 4;
    foreach (string nome in new[] { "por linhas", "por colunas" })
    {
        var tlb = new TLB(4);
        if (nome == "por linhas")
        {
            for (int i = 0; i < LINHAS; i++) for (int j = 0; j < COLUNAS; j++) tlb.Acessar((i * COLUNAS + j) * TAM);
        }
        else
        {
            for (int j = 0; j < COLUNAS; j++) for (int i = 0; i < LINHAS; i++) tlb.Acessar((i * COLUNAS + j) * TAM);
        }
        int total = tlb.Hits + tlb.Misses;
        Console.WriteLine(string.Format(CultureInfo.InvariantCulture, "{0,-12} acessos={1} misses={2,5} taxa_de_acerto={3:F2}%",
            nome, total, tlb.Misses, (double)tlb.Hits / total * 100));
    }
}
```

```go
import "fmt"

type TLB struct {
	cache    []int // o primeiro da lista é o menos usado
	entradas int
	hits     int
	misses   int
}

func (t *TLB) acessar(endereco int) {
	vpn := endereco / 4096
	for k, v := range t.cache {
		if v == vpn {
			t.cache = append(t.cache[:k], t.cache[k+1:]...) // move para o fim
			t.cache = append(t.cache, vpn)
			t.hits += 1
			return
		}
	}
	t.misses += 1
	if len(t.cache) >= t.entradas {
		t.cache = t.cache[1:]
	}
	t.cache = append(t.cache, vpn)
}

func main() {
	const LINHAS, COLUNAS, TAM = 64, 1024, 4
	for _, nome := range []string{"por linhas", "por colunas"} {
		tlb := &TLB{entradas: 4}
		if nome == "por linhas" {
			for i := 0; i < LINHAS; i++ {
				for j := 0; j < COLUNAS; j++ {
					tlb.acessar((i*COLUNAS + j) * TAM)
				}
			}
		} else {
			for j := 0; j < COLUNAS; j++ {
				for i := 0; i < LINHAS; i++ {
					tlb.acessar((i*COLUNAS + j) * TAM)
				}
			}
		}
		total := tlb.hits + tlb.misses
		fmt.Printf("%-12s acessos=%d misses=%5d taxa_de_acerto=%.2f%%\n", nome, total, tlb.misses, float64(tlb.hits)/float64(total)*100)
	}
}
```

## Tempo efetivo de acesso

O tempo médio de acesso à memória com TLB é uma média ponderada pela taxa de acerto. Supondo 1 ns para consultar a TLB, 100 ns por acesso à memória e page walk de 4 níveis:

<CodeTabs>

```python
TLB_NS, MEM_NS, NIVEIS = 1, 100, 4

print("taxa_acerto  tempo_efetivo   vs_sem_paginacao")
for acerto in [0.80, 0.95, 0.99, 0.999]:
    hit = TLB_NS + MEM_NS
    miss = TLB_NS + NIVEIS * MEM_NS + MEM_NS
    efetivo = acerto * hit + (1 - acerto) * miss
    print(f"{acerto:>11.1%}  {efetivo:>10.1f} ns   {efetivo / MEM_NS:>8.2f}x")
```

## TLB e troca de contexto

As entradas da TLB valem para um espaço de endereçamento. Ao trocar de processo, há duas opções:

| Abordagem | Como funciona | Custo |
| --- | --- | --- |
| Esvaziar a TLB (flush) | Descarta tudo a cada troca | Processo que entra começa com TLB fria |
| Etiquetas ASID/PCID | Cada entrada leva o id do espaço de endereçamento | Entradas de vários processos convivem |

Quando o kernel muda uma entrada da tabela de páginas (por exemplo, ao liberar memória), precisa invalidar a TLB de todos os núcleos que podem ter aquela tradução. Isso é o **TLB shootdown**: uma interrupção entre processadores que custa microssegundos e aparece em workloads que fazem muito `munmap` ou `mprotect`.

<Callout type="tip" title="Huge pages aumentam o alcance da TLB">
Com 1.536 entradas de 4 KB, a TLB cobre cerca de 6 MB de memória. Com páginas de 2 MB, as mesmas entradas cobrem 3 GB. Para um banco de dados com buffer pool de dezenas de gigabytes, essa diferença reduz drasticamente os page walks.
</Callout>

```bash
perf stat -e dTLB-loads,dTLB-load-misses ./programa
grep -i huge /proc/meminfo
```
