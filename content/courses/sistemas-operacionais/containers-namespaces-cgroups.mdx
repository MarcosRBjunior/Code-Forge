Um container não é uma VM pequena. Não existe "kernel do container": ele é um **processo comum** do host, rodando no mesmo kernel que todos os outros, só que com uma visão restrita do sistema e limites de consumo. Duas funcionalidades do kernel Linux fazem esse truque: **namespaces** controlam o que o processo **enxerga**, e **cgroups** controlam quanto ele pode **usar**. Somando um sistema de arquivos em camadas e algumas restrições de segurança, você tem o Docker.

## Namespaces: o que o processo vê

Cada namespace isola uma categoria de recurso. Processos no mesmo namespace compartilham a visão; processos em namespaces diferentes têm visões independentes.

| Namespace | Isola | Efeito no container |
| --- | --- | --- |
| PID | Árvore de processos | Seu app é o PID 1 e não vê processos do host |
| Mount | Pontos de montagem | Tem o próprio sistema de arquivos raiz |
| Network | Interfaces, rotas, portas | Tem o próprio `eth0` e `localhost` |
| UTS | Hostname | `hostname` devolve o id do container |
| IPC | Memória compartilhada System V, filas | Não enxerga IPC do host |
| User | UIDs e GIDs | root dentro pode ser usuário comum fora |
| Cgroup | Visão da hierarquia de cgroups | Vê só o próprio cgroup |

O mesmo processo tem um PID em cada nível. A simulação abaixo modela um namespace de PID aninhado no do host: o processo do container se chama 1 lá dentro, mas é 48213 para o host, e o container não consegue nem nomear processos de fora.

```python
class NamespacePID:
    def __init__(self, nome, pai=None):
        self.nome, self.pai = nome, pai
        self.proximo = 1
        self.tabela = {}          # pid local -> processo

    def registrar(self, proc):
        pid = self.proximo
        self.proximo += 1
        self.tabela[pid] = proc
        return pid

def criar_processo(nome, ns, inicio_host=None):
    proc = {"nome": nome, "pids": {}}
    atual = ns
    while atual:                      # recebe um PID em cada namespace ancestral
        if atual.pai is None and inicio_host is not None:
            atual.proximo = max(atual.proximo, inicio_host)
        proc["pids"][atual.nome] = atual.registrar(proc)
        atual = atual.pai
    return proc

host = NamespacePID("host")
for nome in ["systemd", "sshd", "dockerd"]:
    criar_processo(nome, host)
ctr = NamespacePID("container", pai=host)
app = criar_processo("python app.py", ctr, inicio_host=48213)
worker = criar_processo("gunicorn worker", ctr)

for p in [app, worker]:
    print(f"{p['nome']:<16} pids={p['pids']}")
print("container ve:", sorted((pid, p["nome"]) for pid, p in ctr.tabela.items()))
print("host ve:     ", sorted((pid, p["nome"]) for pid, p in host.tabela.items()))
```

```java
import java.util.*;

class Processo {
    final String nome;
    final Map<String, Integer> pids = new LinkedHashMap<>();

    Processo(String nome) {
        this.nome = nome;
    }
}

class NamespacePID {
    final String nome;
    final NamespacePID pai;
    int proximo = 1;
    final TreeMap<Integer, Processo> tabela = new TreeMap<>(); // pid local -> processo

    NamespacePID(String nome, NamespacePID pai) {
        this.nome = nome;
        this.pai = pai;
    }

    int registrar(Processo proc) {
        int pid = proximo;
        proximo += 1;
        tabela.put(pid, proc);
        return pid;
    }

    String visao() {
        List<String> itens = new ArrayList<>();
        for (Map.Entry<Integer, Processo> e : tabela.entrySet()) itens.add("(" + e.getKey() + ", '" + e.getValue().nome + "')");
        return "[" + String.join(", ", itens) + "]";
    }
}

static Processo criarProcesso(String nome, NamespacePID ns, Integer inicioHost) {
    Processo proc = new Processo(nome);
    NamespacePID atual = ns;
    while (atual != null) { // recebe um PID em cada namespace: o próprio e os ancestrais
        if (atual.pai == null && inicioHost != null) {
            atual.proximo = Math.max(atual.proximo, inicioHost);
        }
        proc.pids.put(atual.nome, atual.registrar(proc));
        atual = atual.pai;
    }
    return proc;
}

public static void main(String[] args) {
    NamespacePID host = new NamespacePID("host", null);
    for (String nome : new String[]{"systemd", "sshd", "dockerd"}) {
        criarProcesso(nome, host, null);
    }
    NamespacePID ctr = new NamespacePID("container", host);
    Processo app = criarProcesso("python app.py", ctr, 48213);
    Processo worker = criarProcesso("gunicorn worker", ctr, null);

    for (Processo p : new Processo[]{app, worker}) {
        System.out.println(String.format("%-16s pids=%s", p.nome, p.pids));
    }

    System.out.println("host ve:      " + host.visao());
```

```go
import (
	"fmt"
	"sort"
	"strings"

type Processo struct {
	nome   string
	nsPids []string // nomes dos namespaces, na ordem em que deram o PID
	pids   map[string]int
}

type NamespacePID struct {
	nome    string
	pai     *NamespacePID
	proximo int
	tabela  map[int]*Processo // pid local -> processo
}

func NovoNamespace(nome string, pai *NamespacePID) *NamespacePID {
	return &NamespacePID{nome: nome, pai: pai, proximo: 1, tabela: map[int]*Processo{}}
}

func (ns *NamespacePID) registrar(proc *Processo) int {
	pid := ns.proximo
	ns.proximo++
	ns.tabela[pid] = proc
	return pid
}

func (ns *NamespacePID) visao() string {
	pids := []int{}
	for pid := range ns.tabela {
		pids = append(pids, pid)
	}
	sort.Ints(pids)
	itens := []string{}
	for _, pid := range pids {
		itens = append(itens, fmt.Sprintf("(%d, '%s')", pid, ns.tabela[pid].nome))
	}
	return "[" + strings.Join(itens, ", ") + "]"
}

func criarProcesso(nome string, ns *NamespacePID, inicioHost int) *Processo { // inicioHost = 0: sem pedido
	proc := &Processo{nome: nome, pids: map[string]int{}}
	for atual := ns; atual != nil; atual = atual.pai { // recebe um PID em cada namespace: o próprio e os ancestrais
		if atual.pai == nil && inicioHost != 0 && inicioHost > atual.proximo {
			atual.proximo = inicioHost
		}
		proc.nsPids = append(proc.nsPids, atual.nome)
		proc.pids[atual.nome] = atual.registrar(proc)
	}
	return proc
}

	for _, nome := range []string{"systemd", "sshd", "dockerd"} {
		criarProcesso(nome, host, 0)

	app := criarProcesso("python app.py", ctr, 48213)

	for _, p := range []*Processo{app, worker} {
		itens := []string{}
		for _, ns := range p.nsPids {
			itens = append(itens, fmt.Sprintf("'%s': %d", ns, p.pids[ns]))
		}
		fmt.Printf("%-16s pids={%s}\n", p.nome, strings.Join(itens, ", "))
	}
	fmt.Println("container ve:", ctr.visao())
	fmt.Println("host ve:     ", host.visao())
}
```

## cgroups: quanto o processo usa

Os **control groups** organizam processos numa hierarquia e aplicam limites e contabilidade a cada grupo. No cgroup v2, os mais importantes para quem roda serviços são:

- **`memory.max`:** limite de memória. Passou, o OOM killer age **dentro do cgroup** e o container morre com código 137.
- **`cpu.max`:** cota de CPU no formato "quota período", por exemplo `50000 100000` = meio núcleo.
- **`cpu.weight`:** peso relativo quando há disputa (equivale aos `requests` do Kubernetes).
- **`io.max`** e **`pids.max`:** limitam E/S e número de processos (protege contra fork bomb).

## O throttling que derruba a latência

A cota de CPU é medida por período (100 ms por padrão), somando o uso de **todas** as threads. Um serviço com limite de 2 núcleos e 8 threads ocupadas consome os 200 ms de cota em 25 ms de relógio, e fica **congelado** pelos 75 ms restantes do período, mesmo com a máquina ociosa. A simulação mostra uma requisição que precisa de 300 ms de CPU, paralelizada em várias threads:

<CodeTabs>

```python
PERIODO_MS = 100

def latencia(trabalho_ms, threads, limite_nucleos):
    quota = limite_nucleos * PERIODO_MS
    relogio, restante, throttled = 0.0, float(trabalho_ms), 0
    while restante > 0:
        capacidade = min(quota, threads * PERIODO_MS)   # cota ou paralelismo
        uso = min(capacidade, restante)
        restante -= uso
        if restante > 0:
            if quota < threads * PERIODO_MS:
                throttled += 1                       # cota acabou antes do periodo
            relogio += PERIODO_MS
        else:
            relogio += uso / threads
    return relogio, throttled

for threads, limite in [(1, 2), (4, 2), (8, 2), (8, 8)]:
    ms, n = latencia(300, threads, limite)
    print(f"threads={threads} limite={limite} nucleos -> latencia={ms:6.1f} ms, periodos throttled={n}")
```

```go
import (
	"fmt"
	"math"
)

const PERIODO_MS = 100

func latencia(trabalhoMs, threads, limiteNucleos int) (float64, int) {
	quota := float64(limiteNucleos * PERIODO_MS)
	relogio, restante, throttled := 0.0, float64(trabalhoMs), 0
	for restante > 0 {
		capacidade := math.Min(quota, float64(threads*PERIODO_MS)) // cota ou paralelismo
		uso := math.Min(capacidade, restante)
		restante -= uso
		if restante > 0 {
			if quota < float64(threads*PERIODO_MS) {
				throttled++ // cota acabou antes do periodo
			}
			relogio += PERIODO_MS
		} else {
			relogio += uso / float64(threads)
		}
	}
	return relogio, throttled
}

func main() {
	for _, c := range [][2]int{{1, 2}, {4, 2}, {8, 2}, {8, 8}} {
		threads, limite := c[0], c[1]
		ms, n := latencia(300, threads, limite)
		fmt.Printf("threads=%d limite=%d nucleos -> latencia=%6.1f ms, periodos throttled=%d\n", threads, limite, ms, n)
	}
}
```

Com 8 threads e limite de 2 núcleos, a requisição gasta a cota inteira em 25 ms, fica congelada por 75 ms e só termina no período seguinte: 112,5 ms para um trabalho que, com 8 núcleos, levaria 37,5 ms. É o motivo por que muitos times removem `limits` de CPU no Kubernetes (mantendo `requests`) ou ajustam o número de threads do runtime ao limite. A métrica `nr_throttled` em `cpu.stat` denuncia o problema.

<Callout type="warning" title="Runtimes que não enxergam o cgroup">
Versões antigas da JVM, do Go e do Node liam o número de núcleos e a memória do **host**, criavam threads e heaps grandes demais e eram mortas ou estranguladas. Hoje a JVM respeita o cgroup (`-XX:+UseContainerSupport`), e no Go você pode usar `GOMAXPROCS` alinhado ao limite. Confira sempre o que o seu runtime enxerga dentro do container.
</Callout>

## Os outros ingredientes

- **Sistema de arquivos em camadas (overlayfs):** a imagem é um conjunto de camadas somente leitura com uma camada gravável por cima. Dez containers da mesma imagem compartilham as camadas em disco e no page cache.
- **Capabilities, seccomp e LSMs (AppArmor, SELinux):** reduzem o que o root do container pode fazer e quais syscalls ele pode chamar.

## Container ou VM?

| Aspecto | Container | VM |
| --- | --- | --- |
| Kernel | Compartilhado com o host | Próprio |
| Inicialização | Milissegundos | Segundos (microVM: cerca de 100 ms) |
| Densidade | Centenas por host | Dezenas |
| Isolamento | Bom, mas um bug no kernel afeta todos | Forte, fronteira de hardware |
| Uso típico | Microsserviços do mesmo dono | Multi-tenant hostil, SOs diferentes |

Para código não confiável de vários clientes, provedores combinam os dois: gVisor (kernel em espaço de usuário), Kata Containers e Firecracker colocam cada container dentro de uma microVM.
