O carrinho de compras é simples até surgirem descontos, cupons e estoque. O que se avalia é como você mantém o cálculo do total extensível sem encher uma classe de condicionais.

## Requisitos

- Adicionar, remover e alterar a quantidade de itens.
- Calcular subtotal, descontos e total.
- Cupons (percentual ou valor fixo) e promoções por item.
- Validar estoque ao adicionar.

## Entidades

| Entidade | Responsabilidade |
| --- | --- |
| `Produto` | SKU, nome, preço em centavos e estoque. |
| `ItemCarrinho` | Produto e quantidade. |
| `Desconto` | Strategy: dado o subtotal e os itens, devolve o desconto. |
| `Carrinho` | Itens, cupons e cálculo do total. |

from dataclasses import dataclass
from typing import Dict, List


@dataclass
class Produto:
    sku: str
    nome: str
    preco: int          # centavos
    estoque: int


class Desconto:
    def calcular(self, itens: Dict[str, "ItemCarrinho"], subtotal: int) -> int:
        raise NotImplementedError


class Percentual(Desconto):
    def __init__(self, pct: int):
        self.pct = pct

    def calcular(self, itens, subtotal):
        return subtotal * self.pct // 100


class ValorFixo(Desconto):
    def __init__(self, valor: int):
        self.valor = valor

    def calcular(self, itens, subtotal):
        return min(self.valor, subtotal)


class Leve3Pague2(Desconto):
    def __init__(self, sku: str):
        self.sku = sku

    def calcular(self, itens, subtotal):
        item = itens.get(self.sku)
        if not item:
            return 0
        return (item.quantidade // 3) * item.produto.preco


@dataclass
class ItemCarrinho:
    produto: Produto
    quantidade: int


class Carrinho:
    def __init__(self):
        self.itens: Dict[str, ItemCarrinho] = {}
        self.descontos: List[Desconto] = []

    def adicionar(self, produto: Produto, qtd: int = 1):
        if qtd <= 0:
            raise ValueError("quantidade inválida")
        atual = self.itens[produto.sku].quantidade if produto.sku in self.itens else 0
        if atual + qtd > produto.estoque:
            raise ValueError("estoque insuficiente")
        self.itens[produto.sku] = ItemCarrinho(produto, atual + qtd)

    def remover(self, sku: str):
        self.itens.pop(sku, None)

    def alterar(self, sku: str, qtd: int):
        if qtd == 0:
            return self.remover(sku)
        item = self.itens[sku]
        if qtd > item.produto.estoque:
            raise ValueError("estoque insuficiente")
        item.quantidade = qtd

    def subtotal(self) -> int:
        return sum(i.produto.preco * i.quantidade for i in self.itens.values())

    def total(self) -> int:
        sub = self.subtotal()
        desc = sum(d.calcular(self.itens, sub) for d in self.descontos)
        return max(0, sub - desc)

caderno = Produto("CAD", "Caderno", 2000, 5)
c = Carrinho()
c.adicionar(caneta, 3)
c.adicionar(caderno)

c.descontos += [Leve3Pague2("CAN"), Percentual(10)]

c.alterar("CAN", 0)
print(c.subtotal(), c.total())
try:
    c.adicionar(caderno, 9)
except ValueError as e:
    print("erro:", e)

      throw new RangeError("estoque insuficiente");

import (
	"errors"
	"fmt"
)

type Produto struct {
	sku, nome string
	preco     int // centavos
	estoque   int
}

type ItemCarrinho struct {
	produto    *Produto
	quantidade int
}

type Desconto interface {
	calcular(itens map[string]*ItemCarrinho, subtotal int) int
}

type Percentual struct{ pct int }

func (p Percentual) calcular(itens map[string]*ItemCarrinho, subtotal int) int {
	return subtotal * p.pct / 100
}

type ValorFixo struct{ valor int }

func (v ValorFixo) calcular(itens map[string]*ItemCarrinho, subtotal int) int {
	if v.valor < subtotal {
		return v.valor
	}
	return subtotal
}

type Leve3Pague2 struct{ sku string }

func (l Leve3Pague2) calcular(itens map[string]*ItemCarrinho, subtotal int) int {
	item, ok := itens[l.sku]
	if !ok {
		return 0
	}
	return (item.quantidade / 3) * item.produto.preco
}

type Carrinho struct {
	itens     map[string]*ItemCarrinho
	descontos []Desconto
}

func NovoCarrinho() *Carrinho {
	return &Carrinho{itens: map[string]*ItemCarrinho{}}
}

func (c *Carrinho) adicionar(produto *Produto, qtd int) error {
	if qtd <= 0 {
		return errors.New("quantidade inválida")
	}
	atual := 0
	if item, ok := c.itens[produto.sku]; ok {
		atual = item.quantidade
	}
	if atual+qtd > produto.estoque {
		return errors.New("estoque insuficiente")
	}
	c.itens[produto.sku] = &ItemCarrinho{produto, atual + qtd}
	return nil
}

func (c *Carrinho) remover(sku string) { delete(c.itens, sku) }

func (c *Carrinho) alterar(sku string, qtd int) error {
	if qtd == 0 {
		c.remover(sku)
		return nil
	}
	item := c.itens[sku]
	if qtd > item.produto.estoque {
		return errors.New("estoque insuficiente")
	}
	item.quantidade = qtd
	return nil
}

func (c *Carrinho) subtotal() int {
	soma := 0
	for _, i := range c.itens {
		soma += i.produto.preco * i.quantidade
	}
	return soma
}

func (c *Carrinho) total() int {
	sub := c.subtotal()
	desc := 0
	for _, d := range c.descontos {
		desc += d.calcular(c.itens, sub)
	}
	if sub-desc < 0 {
		return 0
	}
	return sub - desc
}

func main() {
	caneta := &Produto{"CAN", "Caneta", 500, 10}
	caderno := &Produto{"CAD", "Caderno", 2000, 5}
	c := NovoCarrinho()
	c.adicionar(caneta, 3)
	c.adicionar(caderno, 1)
	fmt.Println(c.subtotal()) // 3500
	c.descontos = append(c.descontos, Leve3Pague2{"CAN"}, Percentual{10})
	fmt.Println(c.total()) // 3500 - 500 - 350 = 2650
	c.alterar("CAN", 0)
	fmt.Println(c.subtotal(), c.total())
	if err := c.adicionar(caderno, 9); err != nil {
		fmt.Println("erro:", err)
	}
}

Descontos independentes entre si são somados sobre o mesmo subtotal. Se a ordem importar (primeiro promoção, depois cupom), o cálculo precisa de uma cadeia explícita; combine isso com o entrevistador.

## Carrinho versus pedido

O carrinho é mutável e provisório. Ao fechar a compra, gere um pedido imutável com o preço do momento, para que mudanças futuras no catálogo não alterem pedidos antigos. Guardar o preço no item do pedido é uma decisão que vale mencionar.

## Concorrência e extensões

Estoque é o ponto crítico: a checagem ao adicionar é só uma dica; a reserva real acontece no checkout, de forma atômica. Outras extensões: carrinho persistente por usuário, frete (Strategy), limite de cupom por cliente e itens com variações (tamanho, cor).

## Na entrevista

- **"Como somar descontos novos?"** Nova subclasse de `Desconto`, sem tocar no carrinho.
- **"Preço no carrinho muda quando o catálogo muda?"** Depende do produto; no pedido, nunca.
