Uma carteira digital guarda saldo e movimenta dinheiro entre contas. Aqui o que importa é correção: nenhum centavo pode aparecer ou sumir. Isso leva a livro-razão (ledger), operações atômicas e idempotência.

## Requisitos

- Carteiras com saldo em centavos.
- Depósito, saque e transferência entre carteiras.
- Saldo nunca negativo.
- Extrato com todas as movimentações.
- Repetir a mesma requisição (por falha de rede) não pode cobrar duas vezes.

## Entidades

| Entidade | Responsabilidade |
| --- | --- |
| `Carteira` | Dono, saldo e lock. |
| `Lancamento` | Registro imutável de uma movimentação. |
| `ServicoCarteira` | Operações, extrato e idempotência. |

Cada alteração de saldo gera um lançamento. O saldo é a soma dos lançamentos, e guardá-lo à parte serve só de cache que deve concordar com o livro-razão.

import threading
from dataclasses import dataclass
from typing import Dict, List


@dataclass(frozen=True)
class Lancamento:
    carteira: str
    valor: int          # positivo credita, negativo debita
    descricao: str


class Carteira:
    def __init__(self, dono: str):
        self.dono = dono
        self.saldo = 0
        self.lock = threading.Lock()
        self.extrato: List[Lancamento] = []


class ServicoCarteira:
    def __init__(self):
        self.carteiras: Dict[str, Carteira] = {}
        self.processadas: Dict[str, str] = {}   # id da requisição -> resultado
        self._global = threading.Lock()

    def criar(self, dono: str) -> Carteira:
        c = Carteira(dono)
        self.carteiras[dono] = c
        return c

    def depositar(self, dono: str, valor: int, req_id: str) -> str:
        return self._idempotente(req_id, lambda: self._creditar(dono, valor, "depósito"))

    def sacar(self, dono: str, valor: int, req_id: str) -> str:
        def op():
            c = self.carteiras[dono]
            with c.lock:
                self._debitar(c, valor, "saque")
            return "ok"
        return self._idempotente(req_id, op)

    def transferir(self, de: str, para: str, valor: int, req_id: str) -> str:
        def op():
            a, b = self.carteiras[de], self.carteiras[para]
            primeiro, segundo = sorted((a, b), key=lambda c: c.dono)  # ordem fixa evita deadlock
            with primeiro.lock, segundo.lock:
                self._debitar(a, valor, f"transferência para {para}")
                b.saldo += valor
                b.extrato.append(Lancamento(para, valor, f"transferência de {de}"))
            return "ok"
        return self._idempotente(req_id, op)

    def _creditar(self, dono, valor, desc):
        if valor <= 0:
            raise ValueError("valor deve ser positivo")
        c = self.carteiras[dono]
        with c.lock:
            c.saldo += valor
            c.extrato.append(Lancamento(dono, valor, desc))
        return "ok"

    def _debitar(self, c: Carteira, valor: int, desc: str):
        if valor <= 0:
            raise ValueError("valor deve ser positivo")
        if c.saldo < valor:
            raise ValueError("saldo insuficiente")
        c.saldo -= valor
        c.extrato.append(Lancamento(c.dono, -valor, desc))

    def _idempotente(self, req_id: str, op) -> str:
        with self._global:
            if req_id in self.processadas:
                return self.processadas[req_id]
        resultado = op()
        with self._global:
            self.processadas[req_id] = resultado
        return resultado

s.criar("ana"); s.criar("bia")
s.depositar("ana", 10000, "r1")
s.depositar("ana", 10000, "r1")        # repetida: ignorada

print(total == 10000)

  constructor(carteira, valor, descricao) {
    this.carteira = carteira;

  }

    this.dono = dono;

    this.extrato = [];
  }
}

    this.processadas = new Map(); // id da requisição -> resultado
  }

    this.carteiras.set(dono, c);
    return c;
  }

    return this._idempotente(reqId, () => this._creditar(dono, valor, "depósito"));

      b.saldo += valor;
      b.extrato.push(new Lancamento(para, valor, `transferência de ${de}`));
      return "ok";

    if (valor <= 0) {
      throw new RangeError("valor deve ser positivo");
    }
    const c = this.carteiras.get(dono);
    c.saldo += valor;
    c.extrato.push(new Lancamento(dono, valor, desc));

    if (valor <= 0) {
      throw new RangeError("valor deve ser positivo");

      throw new RangeError("saldo insuficiente");

    c.saldo -= valor;
    c.extrato.push(new Lancamento(c.dono, -valor, desc));

    if (this.processadas.has(reqId)) {
      return this.processadas.get(reqId);
    }

    this.processadas.set(reqId, resultado);

import java.util.*;
import java.util.concurrent.locks.ReentrantLock;
import java.util.function.Supplier;

class Lancamento {
    final String carteira;
    final int valor; // positivo credita, negativo debita
    final String descricao;

    Lancamento(String carteira, int valor, String descricao) {
        this.carteira = carteira;
        this.valor = valor;
        this.descricao = descricao;
    }
}

class Carteira {
    final String dono;
    int saldo = 0;
    final ReentrantLock lock = new ReentrantLock();
    final List<Lancamento> extrato = new ArrayList<>();

    Carteira(String dono) {
        this.dono = dono;
    }
}

class ServicoCarteira {
    final Map<String, Carteira> carteiras = new LinkedHashMap<>();
    private final Map<String, String> processadas = new HashMap<>(); // id da requisição -> resultado
    private final Object global = new Object();

    Carteira criar(String dono) {
        Carteira c = new Carteira(dono);
        carteiras.put(dono, c);
        return c;
    }

    String depositar(String dono, int valor, String reqId) {
        return idempotente(reqId, () -> creditar(dono, valor, "depósito"));
    }

    String sacar(String dono, int valor, String reqId) {
        return idempotente(reqId, () -> {
            Carteira c = carteiras.get(dono);
            c.lock.lock();
            try {
                debitar(c, valor, "saque");
            } finally {
                c.lock.unlock();
            }
            return "ok";
        });
    }

    String transferir(String de, String para, int valor, String reqId) {
        return idempotente(reqId, () -> {
            Carteira a = carteiras.get(de), b = carteiras.get(para);
            // ordem fixa evita deadlock
            Carteira primeiro = a.dono.compareTo(b.dono) <= 0 ? a : b;
            Carteira segundo = primeiro == a ? b : a;
            primeiro.lock.lock();
            segundo.lock.lock();
            try {
                debitar(a, valor, "transferência para " + para);
                b.saldo += valor;
                b.extrato.add(new Lancamento(para, valor, "transferência de " + de));
            } finally {
                segundo.lock.unlock();
                primeiro.lock.unlock();
            }
            return "ok";
        });
    }

    private String creditar(String dono, int valor, String desc) {
        if (valor <= 0) {
            throw new IllegalArgumentException("valor deve ser positivo");
        }
        Carteira c = carteiras.get(dono);
        c.lock.lock();
        try {
            c.saldo += valor;
            c.extrato.add(new Lancamento(dono, valor, desc));
        } finally {
            c.lock.unlock();
        }
        return "ok";
    }

    private void debitar(Carteira c, int valor, String desc) {
        if (valor <= 0) {
            throw new IllegalArgumentException("valor deve ser positivo");
        }
        if (c.saldo < valor) {
            throw new IllegalArgumentException("saldo insuficiente");
        }
        c.saldo -= valor;
        c.extrato.add(new Lancamento(c.dono, -valor, desc));
    }

    private String idempotente(String reqId, Supplier<String> op) {
        synchronized (global) {
            if (processadas.containsKey(reqId)) {
                return processadas.get(reqId);
            }
        }
        String resultado = op.get();
        synchronized (global) {
            processadas.put(reqId, resultado);
        }
        return resultado;
    }
}

    ServicoCarteira s = new ServicoCarteira();
    s.criar("ana");
    s.criar("bia");
    s.depositar("ana", 10000, "r1");

}

            if (it != processadas.end()) return it->second;

        string resultado = op();
        lock_guard<mutex> g(global);
        processadas[reqId] = resultado;
        return resultado;

        c.saldo -= valor;
        c.extrato.push_back({c.dono, -valor, desc});
    }

        {
            c.Saldo += valor;

	return c
}

func (s *ServicoCarteira) depositar(dono string, valor int, reqId string) (string, error) {
	return s.idempotente(reqId, func() (string, error) { return s.creditar(dono, valor, "depósito") })
}

func (s *ServicoCarteira) sacar(dono string, valor int, reqId string) (string, error) {
	return s.idempotente(reqId, func() (string, error) {
		c := s.carteiras[dono]
		c.lock.Lock()
		defer c.lock.Unlock()
		if err := debitar(c, valor, "saque"); err != nil {
			return "", err
		}
		return "ok", nil
	})
}

func (s *ServicoCarteira) transferir(de, para string, valor int, reqId string) (string, error) {
	return s.idempotente(reqId, func() (string, error) {
		a, b := s.carteiras[de], s.carteiras[para]
		primeiro, segundo := a, b // ordem fixa evita deadlock
		if a.dono > b.dono {
			primeiro, segundo = b, a
		}
		primeiro.lock.Lock()
		defer primeiro.lock.Unlock()
		segundo.lock.Lock()
		defer segundo.lock.Unlock()
		if err := debitar(a, valor, "transferência para "+para); err != nil {
			return "", err
		}
		b.saldo += valor
		b.extrato = append(b.extrato, Lancamento{para, valor, "transferência de " + de})
		return "ok", nil
	})
}

func (s *ServicoCarteira) creditar(dono string, valor int, desc string) (string, error) {
	if valor <= 0 {
		return "", errors.New("valor deve ser positivo")
	}
	c := s.carteiras[dono]
	c.lock.Lock()
	defer c.lock.Unlock()
	c.saldo += valor
	c.extrato = append(c.extrato, Lancamento{dono, valor, desc})
	return "ok", nil
}

func debitar(c *Carteira, valor int, desc string) error {
	if valor <= 0 {
		return errors.New("valor deve ser positivo")
	}
	if c.saldo < valor {
		return errors.New("saldo insuficiente")
	}
	c.saldo -= valor
	c.extrato = append(c.extrato, Lancamento{c.dono, -valor, desc})
	return nil
}

func (s *ServicoCarteira) idempotente(reqId string, op func() (string, error)) (string, error) {
	s.global.Lock()
	if anterior, ok := s.processadas[reqId]; ok {
		s.global.Unlock()
		return anterior, nil
	}
	s.global.Unlock()
	resultado, err := op()
	if err != nil {
		return "", err
	}
	s.global.Lock()
	s.processadas[reqId] = resultado
	s.global.Unlock()
	return resultado, nil
}

func main() {
	s := NovoServico()
	s.criar("ana")
	s.criar("bia")
	s.depositar("ana", 10000, "r1")
	s.depositar("ana", 10000, "r1") // repetida: ignorada
	s.transferir("ana", "bia", 4000, "r2")
	if _, err := s.sacar("bia", 9000, "r3"); err != nil {
		fmt.Println("erro:", err)
	}
	fmt.Println(s.carteiras["ana"].saldo, s.carteiras["bia"].saldo) // 6000 4000
	total := 0
	for _, c := range s.carteiras {
		for _, l := range c.extrato {
			total += l.valor
		}
	}
	fmt.Println(total == 10000)
}

Duas coisas para destacar: a ordem fixa na aquisição dos locks (por nome) evita deadlock quando A transfere para B e B para A ao mesmo tempo, e o id da requisição torna as operações idempotentes. Em produção a marcação de "processada" e a alteração de saldo têm de ocorrer na mesma transação.

## Extensões

Limites diários (política injetável), estorno (lançamento inverso, nunca apagar), múltiplas moedas, taxas e transferências agendadas. Em um banco de dados, o mesmo desenho vira transações com `SELECT ... FOR UPDATE` e uma tabela de lançamentos só de inserção.
