O cache LRU (menos recentemente usado) é o clássico que testa estruturas de dados dentro do design: leitura e escrita em tempo constante exigem um dicionário combinado com uma lista duplamente ligada. Depois vêm política de remoção intercambiável, expiração e concorrência.

## Requisitos

- `get(chave)` e `put(chave, valor)` em tempo O(1).
- Capacidade fixa; ao encher, remove o item menos recentemente usado.
- Tanto ler quanto escrever contam como uso.
- Políticas de remoção trocáveis e expiração opcional por tempo.

## Como funciona

O dicionário dá acesso direto ao nó pela chave. A lista duplamente ligada mantém a ordem de uso: o início é o mais recente, o fim é o próximo a sair. Mover um nó para o início custa O(1) porque conhecemos seus vizinhos.

```text
cabeça <-> [c] <-> [a] <-> [b] <-> cauda
 (mais recente)                  (próximo a sair)
```

```python
import threading
from typing import Callable, Dict, Optional


class No:
    __slots__ = ("chave", "valor", "expira", "ant", "prox")

    def __init__(self, chave=None, valor=None, expira=None):
        self.chave, self.valor, self.expira = chave, valor, expira
        self.ant = self.prox = None


class CacheLRU:
    def __init__(self, capacidade: int, relogio: Callable[[], float] = None):
        if capacidade <= 0:
            raise ValueError("capacidade deve ser positiva")
        self.capacidade = capacidade
        self.relogio = relogio or (lambda: 0.0)
        self.mapa: Dict[object, No] = {}
        self.cabeca, self.cauda = No(), No()      # sentinelas
        self.cabeca.prox, self.cauda.ant = self.cauda, self.cabeca
        self._lock = threading.Lock()

    def _remover(self, no: No):
        no.ant.prox, no.prox.ant = no.prox, no.ant

    def _inserir_inicio(self, no: No):
        no.prox, no.ant = self.cabeca.prox, self.cabeca
        self.cabeca.prox.ant = no
        self.cabeca.prox = no

    def get(self, chave) -> Optional[object]:
        with self._lock:
            no = self.mapa.get(chave)
            if no is None:
                return None
            if no.expira is not None and no.expira <= self.relogio():
                self._remover(no)
                del self.mapa[chave]
                return None
            self._remover(no)
            self._inserir_inicio(no)
            return no.valor

    def put(self, chave, valor, ttl: float = None):
        with self._lock:
            expira = self.relogio() + ttl if ttl is not None else None
            no = self.mapa.get(chave)
            if no:
                no.valor, no.expira = valor, expira
                self._remover(no)
                self._inserir_inicio(no)
                return
            if len(self.mapa) >= self.capacidade:
                velho = self.cauda.ant
                self._remover(velho)
                del self.mapa[velho.chave]
            no = No(chave, valor, expira)
            self.mapa[chave] = no
            self._inserir_inicio(no)

    def __len__(self):
        return len(self.mapa)

c.put("a", 1)
```

```javascript
class No {
  constructor(chave = null, valor = null, expira = null) {
    this.chave = chave;
    this.valor = valor;
    this.expira = expira;
    this.ant = null;
    this.prox = null;
  }
}

// Sem lock: o JavaScript executa um trecho de cada vez. Nas outras linguagens o cache usa um lock.
class CacheLRU {
  constructor(capacidade, relogio = () => 0.0) {
    if (capacidade <= 0) {
      throw new RangeError("capacidade deve ser positiva");
    }
    this.capacidade = capacidade;
    this.relogio = relogio;
    this.mapa = new Map();
    this.cabeca = new No();
    this.cauda = new No(); // sentinelas
    this.cabeca.prox = this.cauda;
    this.cauda.ant = this.cabeca;
  }

  _remover(no) {
    no.ant.prox = no.prox;
    no.prox.ant = no.ant;
  }

  _inserirInicio(no) {
    no.prox = this.cabeca.prox;
    no.ant = this.cabeca;
    this.cabeca.prox.ant = no;
    this.cabeca.prox = no;
  }

  get(chave) {
    const no = this.mapa.get(chave);
    if (no === undefined) {
      return null;
    }
    if (no.expira !== null && no.expira <= this.relogio()) {
      this._remover(no);
      this.mapa.delete(chave);
      return null;
    }
    this._remover(no);
    this._inserirInicio(no);
    return no.valor;
  }

  put(chave, valor, ttl = null) {
    const expira = ttl !== null ? this.relogio() + ttl : null;
    let no = this.mapa.get(chave);
    if (no) {
      no.valor = valor;
      no.expira = expira;
      this._remover(no);
      this._inserirInicio(no);
      return;
    }
    if (this.mapa.size >= this.capacidade) {
      const velho = this.cauda.ant;
      this._remover(velho);
      this.mapa.delete(velho.chave);
    }
    no = new No(chave, valor, expira);
    this.mapa.set(chave, no);
    this._inserirInicio(no);
  }

  get length() {
    return this.mapa.size;
  }
}
```

```java
import java.util.*;
import java.util.function.DoubleSupplier;

class No<K, V> {
    K chave;
    V valor;
    Double expira;
    No<K, V> ant, prox;

    No() {}

    No(K chave, V valor, Double expira) {
        this.chave = chave;
        this.valor = valor;
        this.expira = expira;
    }
}

class CacheLRU<K, V> {
    private final int capacidade;
    private final DoubleSupplier relogio;
    private final Map<K, No<K, V>> mapa = new HashMap<>();
    private final No<K, V> cabeca = new No<>(), cauda = new No<>(); // sentinelas

    CacheLRU(int capacidade, DoubleSupplier relogio) {
        if (capacidade <= 0) {
            throw new IllegalArgumentException("capacidade deve ser positiva");
        }
        this.capacidade = capacidade;
        this.relogio = relogio;
        cabeca.prox = cauda;
        cauda.ant = cabeca;
    }

    CacheLRU(int capacidade) {
        this(capacidade, () -> 0.0);
    }

    private void remover(No<K, V> no) {
        no.ant.prox = no.prox;
        no.prox.ant = no.ant;
    }

    private void inserirInicio(No<K, V> no) {
        no.prox = cabeca.prox;
        no.ant = cabeca;
        cabeca.prox.ant = no;
        cabeca.prox = no;
    }

    synchronized V get(K chave) {
        No<K, V> no = mapa.get(chave);
        if (no == null) {
            return null;
        }
        if (no.expira != null && no.expira <= relogio.getAsDouble()) {
            remover(no);
            mapa.remove(chave);
            return null;
        }
        remover(no);
        inserirInicio(no);
        return no.valor;
    }

    synchronized void put(K chave, V valor, Double ttl) {
        Double expira = ttl != null ? relogio.getAsDouble() + ttl : null;
        No<K, V> no = mapa.get(chave);
        if (no != null) {
            no.valor = valor;
            no.expira = expira;
            remover(no);
            inserirInicio(no);
            return;
        }
        if (mapa.size() >= capacidade) {
            No<K, V> velho = cauda.ant;
            remover(velho);
            mapa.remove(velho.chave);
        }
        no = new No<>(chave, valor, expira);
        mapa.put(chave, no);
        inserirInicio(no);
    }

    void put(K chave, V valor) {
        put(chave, valor, null);
    }

    synchronized int tamanho() {
        return mapa.size();
    }
}

public static void main(String[] args) {
    CacheLRU<String, Integer> c = new CacheLRU<>(2);
    c.put("a", 1);
    c.put("b", 2);
    c.get("a"); // a vira o mais recente
    c.put("c", 3); // remove b
    System.out.println(c.get("b") + " " + c.get("a") + " " + c.get("c")); // null 1 3

    double[] t = {0.0};
    CacheLRU<String, Integer> c2 = new CacheLRU<>(2, () -> t[0]);
    c2.put("x", 9, 5.0);
    t[0] = 6;
    System.out.println(c2.get("x") + " " + c2.tamanho()); // null 0
}
```

```cpp
#include <functional>
#include <iostream>
#include <map>
#include <memory>
#include <mutex>
#include <optional>
#include <stdexcept>
#include <string>
using namespace std;

template <typename K, typename V>
struct No {
    K chave{};
    V valor{};
    optional<double> expira;
    No* ant = nullptr;
    No* prox = nullptr;
};

template <typename K, typename V>
class CacheLRU {
    int capacidade;
    function<double()> relogio;
    map<K, unique_ptr<No<K, V>>> mapa;
    No<K, V> cabeca, cauda; // sentinelas
    mutex trava;

    void remover(No<K, V>* no) {
        no->ant->prox = no->prox;
        no->prox->ant = no->ant;
    }

    void inserirInicio(No<K, V>* no) {
        no->prox = cabeca.prox;
        no->ant = &cabeca;
        cabeca.prox->ant = no;
        cabeca.prox = no;
    }

public:
    CacheLRU(int capacidade, function<double()> relogio = [] { return 0.0; })
        : capacidade(capacidade), relogio(relogio) {
        if (capacidade <= 0) {
            throw invalid_argument("capacidade deve ser positiva");
        }
        cabeca.prox = &cauda;
        cauda.ant = &cabeca;
    }

    optional<V> get(const K& chave) {
        lock_guard<mutex> g(trava);
        auto it = mapa.find(chave);
        if (it == mapa.end()) {
            return nullopt;
        }
        No<K, V>* no = it->second.get();
        if (no->expira && *no->expira <= relogio()) {
            remover(no);
            mapa.erase(it);
            return nullopt;
        }
        remover(no);
        inserirInicio(no);
        return no->valor;
    }

    void put(const K& chave, const V& valor, optional<double> ttl = nullopt) {
        lock_guard<mutex> g(trava);
        optional<double> expira;
        if (ttl) expira = relogio() + *ttl;
        auto it = mapa.find(chave);
        if (it != mapa.end()) {
            No<K, V>* no = it->second.get();
            no->valor = valor;
            no->expira = expira;
            remover(no);
            inserirInicio(no);
            return;
        }
        if ((int)mapa.size() >= capacidade) {
            No<K, V>* velho = cauda.ant;
            remover(velho);
            mapa.erase(velho->chave);
        }
        auto no = make_unique<No<K, V>>();
        no->chave = chave;
        no->valor = valor;
        no->expira = expira;
        No<K, V>* bruto = no.get();
        mapa[chave] = move(no);
        inserirInicio(bruto);
    }

    size_t tamanho() {
        lock_guard<mutex> g(trava);
        return mapa.size();
    }
};

template <typename T>
string mostrar(const optional<T>& v) {
    return v ? to_string(*v) : "nullopt";
}

int main() {
    CacheLRU<string, int> c(2);
    c.put("a", 1);
    c.put("b", 2);
    c.get("a"); // a vira o mais recente
    c.put("c", 3); // remove b
    auto b = c.get("b");
    auto a = c.get("a");
    auto cc = c.get("c");
    cout << mostrar(b) << " " << mostrar(a) << " " << mostrar(cc) << endl; // null 1 3

    double t = 0.0;
    CacheLRU<string, int> c2(2, [&] { return t; });
    c2.put("x", 9, 5.0);
    t = 6;
    auto x = c2.get("x");
    cout << mostrar(x) << " " << c2.tamanho() << endl; // null 0
}
```

```csharp
using System;
using System.Collections.Generic;

class No<K, V>
{
    public K Chave;
    public V Valor;
    public double? Expira;
    public No<K, V> Ant, Prox;

    public No() { }

    public No(K chave, V valor, double? expira)
    {
        Chave = chave;
        Valor = valor;
        Expira = expira;
    }
}

class CacheLRU<K, V> where V : class
{
    private readonly int capacidade;
    private readonly Func<double> relogio;
    private readonly Dictionary<K, No<K, V>> mapa = new Dictionary<K, No<K, V>>();
    private readonly No<K, V> cabeca = new No<K, V>(), cauda = new No<K, V>(); // sentinelas
    private readonly object trava = new object();

    public CacheLRU(int capacidade, Func<double> relogio = null)
    {
        if (capacidade <= 0)
        {
            throw new ArgumentException("capacidade deve ser positiva");
        }
        this.capacidade = capacidade;
        this.relogio = relogio ?? (() => 0.0);
        cabeca.Prox = cauda;
        cauda.Ant = cabeca;
    }

    private void Remover(No<K, V> no)
    {
        no.Ant.Prox = no.Prox;
        no.Prox.Ant = no.Ant;
    }

    private void InserirInicio(No<K, V> no)
    {
        no.Prox = cabeca.Prox;
        no.Ant = cabeca;
        cabeca.Prox.Ant = no;
        cabeca.Prox = no;
    }

    public V Get(K chave)
    {
        lock (trava)
        {
            No<K, V> no;
            if (!mapa.TryGetValue(chave, out no))
            {
                return null;
            }
            if (no.Expira != null && no.Expira.Value <= relogio())
            {
                Remover(no);
                mapa.Remove(chave);
                return null;
            }
            Remover(no);
            InserirInicio(no);
            return no.Valor;
        }
    }

    public void Put(K chave, V valor, double? ttl = null)
    {
        lock (trava)
        {
            double? expira = ttl != null ? relogio() + ttl.Value : (double?)null;
            No<K, V> no;
            if (mapa.TryGetValue(chave, out no))
            {
                no.Valor = valor;
                no.Expira = expira;
                Remover(no);
                InserirInicio(no);
                return;
            }
            if (mapa.Count >= capacidade)
            {
                var velho = cauda.Ant;
                Remover(velho);
                mapa.Remove(velho.Chave);
            }
            no = new No<K, V>(chave, valor, expira);
            mapa[chave] = no;
            InserirInicio(no);
        }
    }

    public int Tamanho
    {
        get { lock (trava) { return mapa.Count; } }
    }
}

static string Mostrar(object v)
{
    return v == null ? "null" : v.ToString();
}

static void Main()
{
    var c = new CacheLRU<string, string>(2);
    c.Put("a", "1");
    c.Put("b", "2");
    c.Get("a"); // a vira o mais recente
    c.Put("c", "3"); // remove b
    Console.WriteLine(Mostrar(c.Get("b")) + " " + Mostrar(c.Get("a")) + " " + Mostrar(c.Get("c"))); // null 1 3

    double t = 0.0;
    var c2 = new CacheLRU<string, string>(2, () => t);
    c2.Put("x", "9", 5);
    t = 6;
    Console.WriteLine(Mostrar(c2.Get("x")) + " " + c2.Tamanho); // null 0
}
```

```go
import (
	"errors"
	"fmt"
	"sync"
)

type No struct {
	chave    string
	valor    int
	expira   *float64
	ant, prox *No
}

type CacheLRU struct {
	capacidade   int
	relogio      func() float64
	mapa         map[string]*No
	cabeca, cauda *No // sentinelas
	trava        sync.Mutex
}

func NovoCache(capacidade int, relogio func() float64) (*CacheLRU, error) {
	if capacidade <= 0 {
		return nil, errors.New("capacidade deve ser positiva")
	}
	if relogio == nil {
		relogio = func() float64 { return 0.0 }
	}
	c := &CacheLRU{capacidade: capacidade, relogio: relogio, mapa: map[string]*No{}, cabeca: &No{}, cauda: &No{}}
	c.cabeca.prox, c.cauda.ant = c.cauda, c.cabeca
	return c, nil
}

func (c *CacheLRU) remover(no *No) {
	no.ant.prox, no.prox.ant = no.prox, no.ant
}

func (c *CacheLRU) inserirInicio(no *No) {
	no.prox, no.ant = c.cabeca.prox, c.cabeca
	c.cabeca.prox.ant = no
	c.cabeca.prox = no
}

func (c *CacheLRU) get(chave string) (int, bool) {
	c.trava.Lock()
	defer c.trava.Unlock()
	no, existe := c.mapa[chave]
	if !existe {
		return 0, false
	}
	if no.expira != nil && *no.expira <= c.relogio() {
		c.remover(no)
		delete(c.mapa, chave)
		return 0, false
	}
	c.remover(no)
	c.inserirInicio(no)
	return no.valor, true
}

func (c *CacheLRU) put(chave string, valor int, ttl *float64) {
	c.trava.Lock()
	defer c.trava.Unlock()
	var expira *float64
	if ttl != nil {
		e := c.relogio() + *ttl
		expira = &e
	}
	if no, existe := c.mapa[chave]; existe {
		no.valor, no.expira = valor, expira
		c.remover(no)
		c.inserirInicio(no)
		return
	}
	if len(c.mapa) >= c.capacidade {
		velho := c.cauda.ant
		c.remover(velho)
		delete(c.mapa, velho.chave)
	}
	no := &No{chave: chave, valor: valor, expira: expira}
	c.mapa[chave] = no
	c.inserirInicio(no)
}

func (c *CacheLRU) tamanho() int {
	c.trava.Lock()
	defer c.trava.Unlock()
	return len(c.mapa)
}

func mostrar(valor int, ok bool) string {
	if !ok {
		return "nil"
	}
	return fmt.Sprint(valor)
}

func main() {
	c, _ := NovoCache(2, nil)
	c.put("a", 1, nil)
	c.put("b", 2, nil)
	c.get("a")      // a vira o mais recente
	c.put("c", 3, nil) // remove b
	vb, okb := c.get("b")
	va, oka := c.get("a")
	vc, okc := c.get("c")
	fmt.Println(mostrar(vb, okb), mostrar(va, oka), mostrar(vc, okc)) // nil 1 3

	t := 0.0
	c2, _ := NovoCache(2, func() float64 { return t })
	cinco := 5.0
	c2.put("x", 9, &cinco)
	t = 6
	vx, okx := c2.get("x")
	fmt.Println(mostrar(vx, okx), c2.tamanho()) // nil 0
}
```

As sentinelas na cabeça e na cauda eliminam todos os casos especiais de lista vazia ou de um elemento só. A expiração é preguiçosa: o item vencido é descartado quando alguém o lê.

## Políticas de remoção

| Política | Remove | Boa para |
| --- | --- | --- |
| LRU | O usado há mais tempo | Acesso com localidade temporal |
| LFU | O menos acessado | Itens populares estáveis |
| FIFO | O mais antigo inserido | Simplicidade |

No design, extraia uma interface `PoliticaRemocao` com ganchos como `ao_acessar` e `escolher_vitima`. O cache fica genérico e cada política mantém sua estrutura.

## Concorrência

O lock global torna tudo correto, mas é um gargalo. Alternativas: fatiar o cache em várias partes por hash da chave, cada uma com seu lock e sua lista. Observe que até o `get` altera a lista, então leituras também precisam de exclusão (ou aproximação como um LRU de relógio).

## Na entrevista

- **"Por que lista duplamente ligada?"** Para remover um nó do meio em O(1) conhecendo seus vizinhos; lista simples exigiria busca.
- **"Pode usar OrderedDict?"** Em Python sim, via `move_to_end`, mas o entrevistador geralmente quer ver a estrutura por baixo.
