**MVC** (Model-View-Controller) separa uma aplicação em três papéis para que a interface possa mudar sem tocar nas regras, e as regras possam ser testadas sem interface. É um dos padrões arquiteturais mais antigos e aparece, com variações, em quase todo framework web e de GUI.

## Os três papéis

| Papel | Responsabilidade | Não deve |
| --- | --- | --- |
| Model | Estado e regras de negócio | Saber como é exibido |
| View | Apresentar o modelo ao usuário | Conter regras de negócio |
| Controller | Receber entradas, chamar o modelo, escolher a view | Guardar estado do domínio |

```text
Usuário --entrada--> Controller --atualiza--> Model
                         |                      |
                         +----escolhe---> View <-+ (lê o estado)
```

O fluxo típico: o usuário age, o controller interpreta, altera o model, e a view reflete o novo estado. Em variações com Observer, o model notifica a view diretamente quando muda.

```python
class ModeloTarefas:
    def __init__(self) -> None:
        self._tarefas: list[tuple[str, bool]] = []

    def adicionar(self, titulo: str) -> None:
        if not titulo.strip():
            raise ValueError("título vazio")
        self._tarefas.append((titulo, False))

    def concluir(self, indice: int) -> None:
        titulo, _ = self._tarefas[indice]
        self._tarefas[indice] = (titulo, True)

    def todas(self) -> list[tuple[str, bool]]:
        return list(self._tarefas)


class VisaoTexto:
    def renderizar(self, tarefas: list[tuple[str, bool]]) -> str:
        linhas = [f"[{'x' if feita else ' '}] {t}" for t, feita in tarefas]
        return "\n".join(linhas) or "(nenhuma tarefa)"


class ControladorTarefas:
    def __init__(self, modelo: ModeloTarefas, visao: VisaoTexto) -> None:
        self.modelo = modelo
        self.visao = visao

    def comando(self, texto: str) -> str:
        acao, _, arg = texto.partition(" ")
        if acao == "add":
            self.modelo.adicionar(arg)
        elif acao == "done":
            self.modelo.concluir(int(arg))
        return self.visao.renderizar(self.modelo.todas())


c = ControladorTarefas(ModeloTarefas(), VisaoTexto())
c.comando("add estudar LLD")
c.comando("add revisar SOLID")
saida = c.comando("done 0")
assert saida == "[x] estudar LLD\n[ ] revisar SOLID"
print(saida)
```

```javascript
class ModeloTarefas {
  #tarefas = [];   // pares [titulo, feita]

  adicionar(titulo) {
    if (!titulo.trim()) throw new Error("título vazio");
    this.#tarefas.push([titulo, false]);
  }

  concluir(indice) {
    const [titulo] = this.#tarefas[indice];
    this.#tarefas[indice] = [titulo, true];
  }

  todas() { return this.#tarefas.map((t) => [...t]); }
}

  renderizar(tarefas) {
    const linhas = tarefas.map(([t, feita]) => `[${feita ? "x" : " "}] ${t}`);
    return linhas.join("\n") || "(nenhuma tarefa)";
  }
}

class ControladorTarefas {
  constructor(modelo, visao) {
    this.modelo = modelo;
    this.visao = visao;
  }

  comando(texto) {
    const espaco = texto.indexOf(" ");
    const acao = espaco === -1 ? texto : texto.slice(0, espaco);
    const arg = espaco === -1 ? "" : texto.slice(espaco + 1);
    if (acao === "add") {
      this.modelo.adicionar(arg);
    } else if (acao === "done") {
      this.modelo.concluir(parseInt(arg, 10));
    }
    return this.visao.renderizar(this.modelo.todas());
  }
}
```

```java
import java.util.*;

class Tarefa {
    final String titulo;
    final boolean feita;

    Tarefa(String titulo, boolean feita) {
        this.titulo = titulo;
        this.feita = feita;
    }
}

class ModeloTarefas {
    private final List<Tarefa> tarefas = new ArrayList<>();

    void adicionar(String titulo) {
        if (titulo.isBlank()) throw new IllegalArgumentException("título vazio");
        tarefas.add(new Tarefa(titulo, false));
    }

    void concluir(int indice) {
        tarefas.set(indice, new Tarefa(tarefas.get(indice).titulo, true));
    }

    List<Tarefa> todas() { return new ArrayList<>(tarefas); }
}

class VisaoTexto {
    String renderizar(List<Tarefa> tarefas) {
        List<String> linhas = new ArrayList<>();
        for (Tarefa t : tarefas) linhas.add("[" + (t.feita ? "x" : " ") + "] " + t.titulo);
        return linhas.isEmpty() ? "(nenhuma tarefa)" : String.join("\n", linhas);
    }
}

class ControladorTarefas {
    final ModeloTarefas modelo;
    final VisaoTexto visao;

    ControladorTarefas(ModeloTarefas modelo, VisaoTexto visao) {
        this.modelo = modelo;
        this.visao = visao;
    }

    String comando(String texto) {
        int espaco = texto.indexOf(' ');
        String acao = espaco == -1 ? texto : texto.substring(0, espaco);
        String arg = espaco == -1 ? "" : texto.substring(espaco + 1);
        if (acao.equals("add")) {
            modelo.adicionar(arg);
        } else if (acao.equals("done")) {
            modelo.concluir(Integer.parseInt(arg));
        }
        return visao.renderizar(modelo.todas());
    }
}

public static void main(String[] args) {
    ControladorTarefas c = new ControladorTarefas(new ModeloTarefas(), new VisaoTexto());
    c.comando("add estudar LLD");
    c.comando("add revisar SOLID");
    String saida = c.comando("done 0");
    if (!saida.equals("[x] estudar LLD\n[ ] revisar SOLID")) throw new AssertionError();
    System.out.println(saida);
}
```

```cpp
using Tarefa = pair<string, bool>;

class ModeloTarefas {

    string renderizar(const vector<Tarefa>& tarefas) const {
        string saida;
        for (size_t i = 0; i < tarefas.size(); i++) {
            saida += string(i ? "\n" : "") + "[" + (tarefas[i].second ? "x" : " ") + "] " + tarefas[i].first;
        }
        return saida.empty() ? "(nenhuma tarefa)" : saida;
    }
};

class ControladorTarefas {
    ModeloTarefas& modelo;
    VisaoTexto& visao;

    string comando(const string& texto) {
        size_t espaco = texto.find(' ');
        string acao = espaco == string::npos ? texto : texto.substr(0, espaco);
        string arg = espaco == string::npos ? "" : texto.substr(espaco + 1);
        if (acao == "add") {
            modelo.adicionar(arg);
        } else if (acao == "done") {
            modelo.concluir(stoi(arg));
        }
        return visao.renderizar(modelo.todas());
    }
```

```go
import (
	"fmt"
	"strconv"
	"strings"
)

type Tarefa struct {
	titulo string
	feita  bool
}

type ModeloTarefas struct {
	tarefas []Tarefa
}

func (m *ModeloTarefas) adicionar(titulo string) {
	if strings.TrimSpace(titulo) == "" {
		panic("título vazio")
	}
	m.tarefas = append(m.tarefas, Tarefa{titulo, false})
}

func (m *ModeloTarefas) concluir(indice int) { m.tarefas[indice].feita = true }

func (m *ModeloTarefas) todas() []Tarefa { return append([]Tarefa{}, m.tarefas...) }

type VisaoTexto struct{}

func (VisaoTexto) renderizar(tarefas []Tarefa) string {
	linhas := []string{}
	for _, t := range tarefas {
		marca := " "
		if t.feita {
			marca = "x"
		}
		linhas = append(linhas, fmt.Sprintf("[%s] %s", marca, t.titulo))
	}
	if len(linhas) == 0 {
		return "(nenhuma tarefa)"
	}
	return strings.Join(linhas, "\n")
}

type ControladorTarefas struct {
	modelo *ModeloTarefas
	visao  VisaoTexto
}

func (c *ControladorTarefas) comando(texto string) string {
	acao, arg, _ := strings.Cut(texto, " ")
	switch acao {
	case "add":
		c.modelo.adicionar(arg)
	case "done":
		n, _ := strconv.Atoi(arg)
		c.modelo.concluir(n)
	}
	return c.visao.renderizar(c.modelo.todas())
}

func main() {
	c := &ControladorTarefas{&ModeloTarefas{}, VisaoTexto{}}
	c.comando("add estudar LLD")
	c.comando("add revisar SOLID")
	saida := c.comando("done 0")
	if saida != "[x] estudar LLD\n[ ] revisar SOLID" {
		panic("falhou")
	}
	fmt.Println(saida)
}
```

Trocar `VisaoTexto` por uma visão HTML não exige mexer no modelo nem no controlador. E o `ModeloTarefas` pode ser testado sem nenhuma saída.

## Variações

- **MVP**: o Presenter intermedeia tudo; a view é passiva e fica ainda mais fácil de testar.
- **MVVM**: o ViewModel expõe estado observável e a view se liga a ele automaticamente (comum em front-ends reativos).
- **MVC web**: o controller recebe a requisição HTTP, o model acessa dados e a view é um template.

## Armadilhas

- **Controller gordo**: regras de negócio escorrem para o controller. Mantenha-o fino e leve a lógica para o model.
- **View que consulta o banco**: quebra a separação e dificulta testes.
- **Dependência circular**: se model, view e controller se conhecem demais, o ganho some. Use interfaces ou eventos.

## Na entrevista

Em problemas com interface (jogo da velha, caixa eletrônico) separe a lógica do jogo da exibição: diga que o model é testável sozinho e que a saída em console é só uma view. Follow-ups: "qual a diferença entre MVC e MVP?", "onde ficaria a validação?", "como a view sabe que o model mudou?".
