Regras de filtragem tendem a crescer de forma desorganizada: "produtos ativos, com estoque, abaixo de 100 reais ou em promoção". Se cada combinação vira um `if` gigante ou um método novo (`buscar_ativos_baratos_em_promocao`), o código explode. O padrão **Specification** encapsula cada critério num objeto pequeno com um método `satisfeito_por(objeto)`, e permite **combinar** critérios com E, OU e NÃO.

## A ideia

Cada regra é uma peça reutilizável. Combinar peças gera regras novas sem escrever lógica nova:

```text
ativo E (barato OU promocao) E NÃO esgotado
```

```python
from __future__ import annotations
from abc import ABC, abstractmethod
from dataclasses import dataclass


@dataclass
class Produto:
    nome: str
    preco: float
    estoque: int
    ativo: bool = True
    promocao: bool = False


class Especificacao(ABC):
    @abstractmethod
    def satisfeita_por(self, p: Produto) -> bool: ...

    def e(self, outra: Especificacao) -> Especificacao:
        return E(self, outra)

    def ou(self, outra: Especificacao) -> Especificacao:
        return Ou(self, outra)

    def nao(self) -> Especificacao:
        return Nao(self)


class E(Especificacao):
    def __init__(self, a: Especificacao, b: Especificacao) -> None:
        self.a, self.b = a, b

    def satisfeita_por(self, p: Produto) -> bool:
        return self.a.satisfeita_por(p) and self.b.satisfeita_por(p)


class Ou(Especificacao):
    def __init__(self, a: Especificacao, b: Especificacao) -> None:
        self.a, self.b = a, b

    def satisfeita_por(self, p: Produto) -> bool:
        return self.a.satisfeita_por(p) or self.b.satisfeita_por(p)


class Nao(Especificacao):
    def __init__(self, a: Especificacao) -> None:
        self.a = a

    def satisfeita_por(self, p: Produto) -> bool:
        return not self.a.satisfeita_por(p)


class Ativo(Especificacao):
    def satisfeita_por(self, p: Produto) -> bool:
        return p.ativo


class PrecoAte(Especificacao):
    def __init__(self, limite: float) -> None:
        self.limite = limite

    def satisfeita_por(self, p: Produto) -> bool:
        return p.preco <= self.limite


class EmPromocao(Especificacao):
    def satisfeita_por(self, p: Produto) -> bool:
        return p.promocao


class Esgotado(Especificacao):
    def satisfeita_por(self, p: Produto) -> bool:
        return p.estoque == 0


catalogo = [
    Produto("Mouse", 50, 10),
    Produto("Teclado", 250, 5, promocao=True),
    Produto("Monitor", 900, 0),
    Produto("Cabo", 20, 0),
    Produto("Webcam", 300, 3, ativo=False, promocao=True),
]

regra = Ativo().e(PrecoAte(100).ou(EmPromocao())).e(Esgotado().nao())
nomes = [p.nome for p in catalogo if regra.satisfeita_por(p)]
assert nomes == ["Mouse", "Teclado"]
print(nomes)
```

```cpp
#include <iostream>
#include <memory>
#include <stdexcept>
#include <string>
#include <vector>
using namespace std;

struct Produto {
    string nome;
    double preco;
    int estoque;
    bool ativo = true;
    bool promocao = false;
};

class Especificacao;
using Espec = shared_ptr<Especificacao>;

class Especificacao : public enable_shared_from_this<Especificacao> {
public:
    virtual ~Especificacao() = default;
    virtual bool satisfeitaPor(const Produto& p) const = 0;

    Espec e(Espec outra);
    Espec ou(Espec outra);
    Espec nao();
};

class E : public Especificacao {
    Espec a, b;

public:
    E(Espec a, Espec b) : a(a), b(b) {}
    bool satisfeitaPor(const Produto& p) const override { return a->satisfeitaPor(p) && b->satisfeitaPor(p); }
};

class Ou : public Especificacao {
    Espec a, b;

public:
    Ou(Espec a, Espec b) : a(a), b(b) {}
    bool satisfeitaPor(const Produto& p) const override { return a->satisfeitaPor(p) || b->satisfeitaPor(p); }
};

class Nao : public Especificacao {
    Espec a;

public:
    Nao(Espec a) : a(a) {}
    bool satisfeitaPor(const Produto& p) const override { return !a->satisfeitaPor(p); }
};

Espec Especificacao::e(Espec outra) { return make_shared<E>(shared_from_this(), outra); }
Espec Especificacao::ou(Espec outra) { return make_shared<Ou>(shared_from_this(), outra); }
Espec Especificacao::nao() { return make_shared<Nao>(shared_from_this()); }

class Ativo : public Especificacao {
public:
    bool satisfeitaPor(const Produto& p) const override { return p.ativo; }
};

class PrecoAte : public Especificacao {
    double limite;

public:
    PrecoAte(double limite) : limite(limite) {}
    bool satisfeitaPor(const Produto& p) const override { return p.preco <= limite; }
};

class EmPromocao : public Especificacao {
public:
    bool satisfeitaPor(const Produto& p) const override { return p.promocao; }
};

class Esgotado : public Especificacao {
public:
    bool satisfeitaPor(const Produto& p) const override { return p.estoque == 0; }
};

int main() {
    vector<Produto> catalogo = {
        {"Mouse", 50, 10},
        {"Teclado", 250, 5, true, true},
        {"Monitor", 900, 0},
        {"Cabo", 20, 0},
        {"Webcam", 300, 3, false, true},
    };

    Espec regra = make_shared<Ativo>()
                      ->e(make_shared<PrecoAte>(100)->ou(make_shared<EmPromocao>()))
                      ->e(make_shared<Esgotado>()->nao());
    vector<string> nomes;
    for (auto& p : catalogo) if (regra->satisfeitaPor(p)) nomes.push_back(p.nome);
    if (nomes != vector<string>{"Mouse", "Teclado"}) throw runtime_error("falhou");
    cout << "[";
    for (size_t i = 0; i < nomes.size(); i++) cout << (i ? ", '" : "'") << nomes[i] << "'";
    cout << "]" << endl;
}
```

```go
import (
	"fmt"
	"strings"
)

type Produto struct {
	nome     string
	preco    float64
	estoque  int
	ativo    bool
	promocao bool
}

// Especificacao é uma função: compor regras é compor funções.
type Especificacao func(p Produto) bool

func (a Especificacao) e(b Especificacao) Especificacao {
	return func(p Produto) bool { return a(p) && b(p) }
}

func (a Especificacao) ou(b Especificacao) Especificacao {
	return func(p Produto) bool { return a(p) || b(p) }
}

func (a Especificacao) nao() Especificacao {
	return func(p Produto) bool { return !a(p) }
}

func Ativo() Especificacao { return func(p Produto) bool { return p.ativo } }

func PrecoAte(limite float64) Especificacao {
	return func(p Produto) bool { return p.preco <= limite }
}

func EmPromocao() Especificacao { return func(p Produto) bool { return p.promocao } }

func Esgotado() Especificacao { return func(p Produto) bool { return p.estoque == 0 } }

func main() {
	catalogo := []Produto{
		{"Mouse", 50, 10, true, false},
		{"Teclado", 250, 5, true, true},
		{"Monitor", 900, 0, true, false},
		{"Cabo", 20, 0, true, false},
		{"Webcam", 300, 3, false, true},
	}

	regra := Ativo().e(PrecoAte(100).ou(EmPromocao())).e(Esgotado().nao())
	nomes := []string{}
	for _, p := range catalogo {
		if regra(p) {
			nomes = append(nomes, "'"+p.nome+"'")
		}
	}
	if strings.Join(nomes, ",") != "'Mouse','Teclado'" {
		panic("falhou")
	}
	fmt.Println("[" + strings.Join(nomes, ", ") + "]")
}
```

A regra final lê quase como a frase original, e cada peça tem um teste simples e isolado.

## Quando vale a pena

| Situação | Specification? |
| --- | --- |
| Filtros combináveis em tempo de execução (busca avançada) | Sim |
| Regras de elegibilidade reutilizadas em vários lugares | Sim |
| Um único filtro fixo | Não, uma função basta |
| Consultas pesadas em banco | Cuidado: traduza para a consulta, não filtre tudo em memória |

Em Python, funções de ordem superior (`lambda`) resolvem casos simples. A classe vale quando a regra tem nome de negócio, parâmetros e será combinada ou testada em separado.

<Callout type="tip" title="Parentes próximos">
Specification lembra Strategy (um critério intercambiável) e Composite (E e OU compõem outras especificações como uma árvore). A combinação dos dois é a essência do padrão.
</Callout>

## Armadilhas

- **Explosão de classes** para regras triviais: use funções quando bastar.
- **Filtrar em memória** o que o banco faria melhor: com muitos dados, converta a especificação em consulta.
- **Estado escondido**: especificações devem ser puras, sem efeitos colaterais.

## Na entrevista

Aparece em sistemas de busca e filtro: estacionamento por tipo de vaga, catálogo de produtos, regras de promoção. Follow-ups: "como adicionar um novo filtro sem mudar o código existente?" (resposta: nova classe, princípio aberto/fechado) e "como combinaria regras?".
