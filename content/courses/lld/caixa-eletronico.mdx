O caixa eletrônico (ATM) junta várias preocupações em um problema só: um **fluxo por estados** (cartão, PIN, menu, operação), **regras de dinheiro** (saldo, limite, notas disponíveis) e **segurança** (tentativas de PIN). O entrevistador avalia se você separa a máquina física da conta bancária e se trata a entrega de notas como um problema próprio.

## Requisitos

- **Autenticação**: inserir cartão e PIN; após 3 erros, o cartão é bloqueado.
- **Operações**: consultar saldo, sacar, depositar.
- **Saque**: não pode exceder o saldo, e o caixa precisa ter notas para compor o valor.
- **Notas**: o caixa guarda quantidades por denominação (por exemplo 100, 50, 20) e entrega com o menor número de notas possível.
- Fora de escopo: rede com o banco, recibo impresso e depósitos em envelope.

## Entidades

| Entidade | Responsabilidade |
| --- | --- |
| `Conta` | Saldo e operações de débito e crédito, sem saber de cartões |
| `Cartao` | Número, PIN, tentativas erradas e bloqueio |
| `Banco` | Mapeia cartões para contas e valida o PIN |
| `Dispensador` | Estoque de notas e cálculo de como pagar um valor |
| `CaixaEletronico` | Orquestra a sessão e o fluxo de estados |

O ponto central: `Dispensador` e `Conta` são independentes. A conta pode ter saldo e o caixa não ter notas, e o inverso. Um saque só acontece quando **ambos** concordam, e o débito acontece depois de garantir que as notas existem.

class Conta:
    def __init__(self, saldo: int) -> None:
        self.saldo = saldo

    def debitar(self, valor: int) -> None:
        if valor > self.saldo:
            raise ValueError("saldo insuficiente")
        self.saldo -= valor

    def creditar(self, valor: int) -> None:
        self.saldo += valor


class Cartao:
    MAX_TENTATIVAS = 3

    def __init__(self, numero: str, pin: str, conta: Conta) -> None:
        self.numero, self._pin, self.conta = numero, pin, conta
        self.erros = 0
        self.bloqueado = False

    def validar(self, pin: str) -> bool:
        if self.bloqueado:
            return False
        if pin == self._pin:
            self.erros = 0
            return True
        self.erros += 1
        self.bloqueado = self.erros >= self.MAX_TENTATIVAS
        return False


class Dispensador:
    def __init__(self, notas: dict[int, int]) -> None:
        self.notas = dict(notas)  # denominação -> quantidade

    def plano(self, valor: int) -> dict[int, int] | None:
        """Guloso: serve para denominações em que cada uma divide a maior."""
        restante, resultado = valor, {}
        for nota in sorted(self.notas, reverse=True):
            qtd = min(restante // nota, self.notas[nota])
            if qtd:
                resultado[nota] = qtd
                restante -= qtd * nota
        return resultado if restante == 0 else None

    def entregar(self, plano: dict[int, int]) -> None:
        for nota, qtd in plano.items():
            self.notas[nota] -= qtd


class CaixaEletronico:
    def __init__(self, dispensador: Dispensador) -> None:
        self.dispensador = dispensador
        self.cartao: Cartao | None = None
        self.autenticado = False

    def inserir_cartao(self, cartao: Cartao) -> None:
        self.cartao, self.autenticado = cartao, False

    def digitar_pin(self, pin: str) -> bool:
        assert self.cartao is not None
        self.autenticado = self.cartao.validar(pin)
        return self.autenticado

    def saldo(self) -> int:
        self._exigir_sessao()
        return self.cartao.conta.saldo

    def sacar(self, valor: int) -> dict[int, int]:
        self._exigir_sessao()
        plano = self.dispensador.plano(valor)
        if plano is None:
            raise ValueError("caixa sem notas para este valor")
        self.cartao.conta.debitar(valor)  # só debita depois de saber que dá para pagar
        self.dispensador.entregar(plano)
        return plano

    def depositar(self, valor: int) -> None:
        self._exigir_sessao()
        self.cartao.conta.creditar(valor)

    def ejetar(self) -> None:
        self.cartao, self.autenticado = None, False

    def _exigir_sessao(self) -> None:
        if not (self.cartao and self.autenticado):
            raise PermissionError("faça login")

cartao = Cartao("1111", "4321", conta)

for _ in range(3):
    cartao.validar("9999")
assert cartao.bloqueado
print("atm ok")

class Conta {
  constructor(saldo) {
    this.saldo = saldo;
  }

  debitar(valor) {
    if (valor > this.saldo) throw new Error("saldo insuficiente");
    this.saldo -= valor;
  }

  creditar(valor) {
    this.saldo += valor;
  }
}

class Cartao {
  static MAX_TENTATIVAS = 3;
  #pin;

  constructor(numero, pin, conta) {
    this.numero = numero;
    this.#pin = pin;
    this.conta = conta;
    this.erros = 0;
    this.bloqueado = false;
  }

  validar(pin) {
    if (this.bloqueado) return false;
    if (pin === this.#pin) {
      this.erros = 0;
      return true;
    }
    this.erros += 1;
    this.bloqueado = this.erros >= Cartao.MAX_TENTATIVAS;
    return false;
  }
}

class Dispensador {
  constructor(notas) {
    this.notas = new Map(Object.entries(notas).map(([k, v]) => [Number(k), v])); // denominação -> quantidade
  }

  // Guloso: serve para denominações em que cada uma divide a maior.
  plano(valor) {
    let restante = valor;
    const resultado = new Map();
    for (const nota of [...this.notas.keys()].sort((a, b) => b - a)) {
      const qtd = Math.min(Math.floor(restante / nota), this.notas.get(nota));
      if (qtd) {
        resultado.set(nota, qtd);
        restante -= qtd * nota;
      }
    }
    return restante === 0 ? resultado : null;
  }

  entregar(plano) {
    for (const [nota, qtd] of plano) {
      this.notas.set(nota, this.notas.get(nota) - qtd);
    }
  }
}

import java.util.*;

class Conta {
    int saldo;

    Conta(int saldo) {
        this.saldo = saldo;
    }

    void debitar(int valor) {
        if (valor > saldo) throw new IllegalArgumentException("saldo insuficiente");
        saldo -= valor;
    }

    void creditar(int valor) {
        saldo += valor;
    }
}

class Cartao {
    static final int MAX_TENTATIVAS = 3;

    final String numero;
    private final String pin;
    final Conta conta;
    int erros = 0;
    boolean bloqueado = false;

    Cartao(String numero, String pin, Conta conta) {
        this.numero = numero;
        this.pin = pin;
        this.conta = conta;
    }

    boolean validar(String tentativa) {
        if (bloqueado) return false;
        if (tentativa.equals(pin)) {
            erros = 0;
            return true;
        }
        erros += 1;
        bloqueado = erros >= MAX_TENTATIVAS;
        return false;
    }
}

class Dispensador {
    private final Map<Integer, Integer> notas = new TreeMap<>(Comparator.reverseOrder()); // denominação -> quantidade

    Dispensador(Map<Integer, Integer> notas) {
        this.notas.putAll(notas);
    }

    // Guloso: serve para denominações em que cada uma divide a maior.
    Map<Integer, Integer> plano(int valor) {
        int restante = valor;
        Map<Integer, Integer> resultado = new LinkedHashMap<>();
        for (Map.Entry<Integer, Integer> e : notas.entrySet()) {
            int nota = e.getKey();
            int qtd = Math.min(restante / nota, e.getValue());
            if (qtd > 0) {
                resultado.put(nota, qtd);
                restante -= qtd * nota;
            }
        }
        return restante == 0 ? resultado : null;
    }

    void entregar(Map<Integer, Integer> plano) {
        for (Map.Entry<Integer, Integer> e : plano.entrySet()) {
            notas.put(e.getKey(), notas.get(e.getKey()) - e.getValue());
        }
    }
}

class CaixaEletronico {
    private final Dispensador dispensador;
    private Cartao cartao = null;
    private boolean autenticado = false;

    CaixaEletronico(Dispensador dispensador) {
        this.dispensador = dispensador;
    }

        this.cartao = cartao;
        this.autenticado = false;

        autenticado = cartao.validar(pin);
        return autenticado;
    }

    int saldo() {
        exigirSessao();
        return cartao.conta.saldo;
    }

        exigirSessao();
        Map<Integer, Integer> plano = dispensador.plano(valor);
        if (plano == null) {
            throw new IllegalArgumentException("caixa sem notas para este valor");
        }
        cartao.conta.debitar(valor); // só debita depois de saber que dá para pagar

        return plano;

        exigirSessao();
        cartao.conta.creditar(valor);
    }

    void ejetar() {
        cartao = null;
        autenticado = false;
    }

    private void exigirSessao() {
        if (!(cartao != null && autenticado)) {
            throw new SecurityException("faça login");
        }
    }
}

using System.Linq;

class Conta
{
    public int Saldo;

    public Conta(int saldo)
    {
        Saldo = saldo;
    }

    public void Debitar(int valor)
    {
        if (valor > Saldo) throw new ArgumentException("saldo insuficiente");

    }

    public void Creditar(int valor)
    {
        Saldo += valor;
    }

class Cartao
{
    const int MAX_TENTATIVAS = 3;

    public readonly string Numero;
    private readonly string pin;
    public readonly Conta Conta;
    public int Erros = 0;
    public bool Bloqueado = false;

    public Cartao(string numero, string pin, Conta conta)
    {
        Numero = numero;
        this.pin = pin;
        Conta = conta;
    }

    public bool Validar(string tentativa)
    {
        if (Bloqueado) return false;
        if (tentativa == pin)
        {
            Erros = 0;
            return true;
        }
        Erros += 1;
        Bloqueado = Erros >= MAX_TENTATIVAS;
        return false;
    }
}

class Dispensador
{
    private readonly Dictionary<int, int> notas; // denominação -> quantidade

    public Dispensador(Dictionary<int, int> notas)
    {
        this.notas = new Dictionary<int, int>(notas);
    }

    // Guloso: serve para denominações em que cada uma divide a maior.
    public Dictionary<int, int> Plano(int valor)
    {
        int restante = valor;
        var resultado = new Dictionary<int, int>();
        foreach (int nota in notas.Keys.OrderByDescending(n => n))
        {
            int qtd = Math.Min(restante / nota, notas[nota]);
            if (qtd > 0)
            {
                resultado[nota] = qtd;
                restante -= qtd * nota;
            }
        }
        return restante == 0 ? resultado : null;
    }

    public void Entregar(Dictionary<int, int> plano)
    {
        foreach (var par in plano) notas[par.Key] -= par.Value;
    }
}

        ExigirSessao();
        return cartao.Conta.Saldo;

        var plano = dispensador.Plano(valor);
        if (plano == null)
        {
            throw new ArgumentException("caixa sem notas para este valor");
        }
        cartao.Conta.Debitar(valor); // só debita depois de saber que dá para pagar
        dispensador.Entregar(plano);
        return plano;
    }

    public void Depositar(int valor)
    {
        ExigirSessao();
        cartao.Conta.Creditar(valor);
    }

    public void Ejetar()
    {
        cartao = null;
        autenticado = false;
    }

    private void ExigirSessao()
    {
        if (!(cartao != null && autenticado))
        {
            throw new UnauthorizedAccessException("faça login");
        }
    }

import (

	"fmt"
	"reflect"
	"sort"

type Conta struct {
	saldo int
}

func (c *Conta) debitar(valor int) error {
	if valor > c.saldo {
		return errors.New("saldo insuficiente")
	}
	c.saldo -= valor
	return nil
}

func (c *Conta) creditar(valor int) { c.saldo += valor }

const maxTentativas = 3

type Cartao struct {
	numero     string
	pin        string
	conta      *Conta
	erros      int
	bloqueado  bool
}

func (c *Cartao) validar(tentativa string) bool {
	if c.bloqueado {
		return false
	}
	if tentativa == c.pin {
		c.erros = 0
		return true
	}
	c.erros++
	c.bloqueado = c.erros >= maxTentativas
	return false
}

type Dispensador struct {
	notas map[int]int // denominação -> quantidade
}

// Guloso: serve para denominações em que cada uma divide a maior.
func (d *Dispensador) plano(valor int) map[int]int {
	restante := valor
	resultado := map[int]int{}
	denominacoes := []int{}
	for nota := range d.notas {
		denominacoes = append(denominacoes, nota)
	}
	sort.Sort(sort.Reverse(sort.IntSlice(denominacoes)))
	for _, nota := range denominacoes {
		qtd := restante / nota
		if d.notas[nota] < qtd {
			qtd = d.notas[nota]
		}
		if qtd > 0 {
			resultado[nota] = qtd
			restante -= qtd * nota
		}
	}
	if restante == 0 {
		return resultado
	}
	return nil
}

func (d *Dispensador) entregar(plano map[int]int) {
	for nota, qtd := range plano {
		d.notas[nota] -= qtd
	}
}

type CaixaEletronico struct {
	dispensador *Dispensador
	cartao      *Cartao
	autenticado bool
}

func (c *CaixaEletronico) inserirCartao(cartao *Cartao) {
	c.cartao, c.autenticado = cartao, false
}

func (c *CaixaEletronico) digitarPin(pin string) bool {
	c.autenticado = c.cartao.validar(pin)
	return c.autenticado
}

func (c *CaixaEletronico) exigirSessao() error {
	if !(c.cartao != nil && c.autenticado) {
		return errors.New("faça login")
	}
	return nil
}

func (c *CaixaEletronico) saldo() (int, error) {
	if err := c.exigirSessao(); err != nil {
		return 0, err
	}
	return c.cartao.conta.saldo, nil
}

func (c *CaixaEletronico) sacar(valor int) (map[int]int, error) {
	if err := c.exigirSessao(); err != nil {
		return nil, err
	}
	plano := c.dispensador.plano(valor)
	if plano == nil {
		return nil, errors.New("caixa sem notas para este valor")
	}
	if err := c.cartao.conta.debitar(valor); err != nil { // só debita depois de saber que dá para pagar
		return nil, err
	}
	c.dispensador.entregar(plano)
	return plano, nil
}

func (c *CaixaEletronico) depositar(valor int) error {
	if err := c.exigirSessao(); err != nil {
		return err
	}
	c.cartao.conta.creditar(valor)
	return nil
}

func (c *CaixaEletronico) ejetar() { c.cartao, c.autenticado = nil, false }

func verificar(condicao bool) {
	if !condicao {
		panic("verificação falhou")
	}
}

func main() {
	conta := &Conta{1000}
	cartao := &Cartao{numero: "1111", pin: "4321", conta: conta}
	atm := &CaixaEletronico{dispensador: &Dispensador{map[int]int{100: 5, 50: 5, 20: 5}}}
	atm.inserirCartao(cartao)
	verificar(atm.digitarPin("0000") == false)
	verificar(atm.digitarPin("4321") == true)

	saldo, _ := atm.saldo()
	verificar(saldo == 730)
	atm.sacar(15) // sem notas para 15: devolve erro e não debita
	verificar(conta.saldo == 730)
	for i := 0; i < 3; i++ {
		cartao.validar("9999")
	}
	verificar(cartao.bloqueado)

}

## Decisões que merecem fala

- **Dinheiro em inteiros**, nunca em ponto flutuante.
- **Ordem das operações no saque**: planejar notas, depois debitar e entregar. Em um sistema real isso seria uma transação com compensação se a entrega mecânica falhasse.
- **Algoritmo de notas**: o guloso funciona para denominações "canônicas" como 100, 50 e 20 combinadas com um limite de estoque razoável, mas pode falhar em outras combinações; nesses casos use programação dinâmica.
- **Estado da sessão**: aqui é um flag; se o fluxo crescer (menu, troca de PIN, timeouts), modele com State.

<Callout type="warning" title="Concorrência e consistência">
Duas máquinas sacando da mesma conta ao mesmo tempo exigem uma operação atômica de débito no banco. Cite isso sem implementar, a menos que o entrevistador peça.
</Callout>
