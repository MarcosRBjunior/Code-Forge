Um site de perguntas e respostas no estilo Stack Overflow mistura conteúdo, votos, reputação e busca por tags. O desafio de modelagem é manter a reputação coerente com os votos sem acoplar tudo.

## Requisitos

- Usuários publicam perguntas (com tags) e respostas.
- Votos positivos e negativos em perguntas e respostas, um voto por usuário por item.
- O autor da pergunta aceita uma resposta.
- Reputação do autor muda com votos e aceitação.
- Busca de perguntas por tag.

## Entidades

| Entidade | Responsabilidade |
| --- | --- |
| `Usuario` | Nome e reputação. |
| `Postagem` | Base comum: autor, texto, votos. |
| `Pergunta` | Postagem com título, tags, respostas e resposta aceita. |
| `Resposta` | Postagem ligada a uma pergunta. |
| `Site` | Publicar, votar, aceitar, buscar por tag. |

Pergunta e resposta compartilham votos, então herdam de `Postagem`. Os votos ficam num dicionário de usuário para valor (+1 ou -1), o que garante um voto por usuário e permite trocar de opinião.

from collections import defaultdict
from typing import Dict, List, Optional, Set

PONTOS_VOTO_POSITIVO = 10
PONTOS_VOTO_NEGATIVO = -2
PONTOS_ACEITA = 15


class Usuario:
    def __init__(self, nome: str):
        self.nome, self.reputacao = nome, 0


class Postagem:
    def __init__(self, autor: Usuario, texto: str):
        self.autor, self.texto = autor, texto
        self.votos: Dict[Usuario, int] = {}

    @property
    def pontuacao(self) -> int:
        return sum(self.votos.values())

    def votar(self, usuario: Usuario, valor: int):
        if usuario is self.autor:
            raise ValueError("não pode votar no próprio conteúdo")
        anterior = self.votos.get(usuario, 0)
        if anterior == valor:
            return
        self.autor.reputacao -= self._pontos(anterior)
        self.votos[usuario] = valor
        self.autor.reputacao += self._pontos(valor)

    @staticmethod
    def _pontos(valor: int) -> int:
        return {1: PONTOS_VOTO_POSITIVO, -1: PONTOS_VOTO_NEGATIVO, 0: 0}[valor]


class Resposta(Postagem):
    pass


class Pergunta(Postagem):
    def __init__(self, autor, titulo: str, texto: str, tags: Set[str]):
        super().__init__(autor, texto)
        self.titulo, self.tags = titulo, set(tags)
        self.respostas: List[Resposta] = []
        self.aceita: Optional[Resposta] = None


class Site:
    def __init__(self):
        self.por_tag: Dict[str, List[Pergunta]] = defaultdict(list)

    def perguntar(self, autor, titulo, texto, tags) -> Pergunta:
        p = Pergunta(autor, titulo, texto, tags)
        for t in p.tags:
            self.por_tag[t].append(p)
        return p

    def responder(self, pergunta: Pergunta, autor, texto) -> Resposta:
        r = Resposta(autor, texto)
        pergunta.respostas.append(r)
        return r

    def aceitar(self, pergunta: Pergunta, resposta: Resposta):
        if resposta not in pergunta.respostas:
            raise ValueError("resposta de outra pergunta")
        if pergunta.aceita is not None:
            pergunta.aceita.autor.reputacao -= PONTOS_ACEITA
        pergunta.aceita = resposta
        resposta.autor.reputacao += PONTOS_ACEITA

    def buscar(self, tag: str) -> List[Pergunta]:
        return sorted(self.por_tag[tag], key=lambda p: -p.pontuacao)

const PONTOS_VOTO_NEGATIVO = -2;
const PONTOS_ACEITA = 15;

class Usuario {
  constructor(nome) {
    this.nome = nome;
    this.reputacao = 0;
  }
}

class Postagem {
  constructor(autor, texto) {
    this.autor = autor;
    this.texto = texto;
    this.votos = new Map();
  }

  get pontuacao() {
    let soma = 0;
    for (const v of this.votos.values()) soma += v;
    return soma;
  }

  votar(usuario, valor) {
    if (usuario === this.autor) {
      throw new Error("não pode votar no próprio conteúdo");
    }
    const anterior = this.votos.get(usuario) ?? 0;
    if (anterior === valor) {
      return;
    }
    this.autor.reputacao -= Postagem.#pontos(anterior);
    this.votos.set(usuario, valor);
    this.autor.reputacao += Postagem.#pontos(valor);
  }

  static #pontos(valor) {
    return { 1: PONTOS_VOTO_POSITIVO, [-1]: PONTOS_VOTO_NEGATIVO, 0: 0 }[valor];
  }
}

  constructor(autor, titulo, texto, tags) {

    this.titulo = titulo;

    this.aceita = null;

}

    for (const t of p.tags) {

    pergunta.respostas.push(r);
    return r;

    pergunta.aceita = resposta;
    resposta.autor.reputacao += PONTOS_ACEITA;

import java.util.*;

class Usuario {
    final String nome;
    int reputacao = 0;

    Usuario(String nome) {
        this.nome = nome;
    }
}

class Postagem {
    static final int PONTOS_VOTO_POSITIVO = 10;
    static final int PONTOS_VOTO_NEGATIVO = -2;
    static final int PONTOS_ACEITA = 15;

    final Usuario autor;
    final String texto;
    final Map<Usuario, Integer> votos = new HashMap<>();

    Postagem(Usuario autor, String texto) {
        this.autor = autor;
        this.texto = texto;
    }

    int pontuacao() {
        int soma = 0;
        for (int v : votos.values()) soma += v;
        return soma;
    }

    void votar(Usuario usuario, int valor) {
        if (usuario == autor) {
            throw new IllegalArgumentException("não pode votar no próprio conteúdo");
        }
        int anterior = votos.getOrDefault(usuario, 0);
        if (anterior == valor) {
            return;
        }
        autor.reputacao -= pontos(anterior);
        votos.put(usuario, valor);
        autor.reputacao += pontos(valor);
    }

    private static int pontos(int valor) {
        switch (valor) {
            case 1: return PONTOS_VOTO_POSITIVO;
            case -1: return PONTOS_VOTO_NEGATIVO;
            default: return 0;
        }
    }
}

class Resposta extends Postagem {
    Resposta(Usuario autor, String texto) {
        super(autor, texto);
    }
}

class Pergunta extends Postagem {
    final String titulo;
    final Set<String> tags;
    final List<Resposta> respostas = new ArrayList<>();
    Resposta aceita = null;

    Pergunta(Usuario autor, String titulo, String texto, Set<String> tags) {
        super(autor, texto);
        this.titulo = titulo;
        this.tags = new LinkedHashSet<>(tags);
    }
}

class Site {
    final Map<String, List<Pergunta>> porTag = new HashMap<>();

    Pergunta perguntar(Usuario autor, String titulo, String texto, Set<String> tags) {
        Pergunta p = new Pergunta(autor, titulo, texto, tags);
        for (String t : p.tags) porTag.computeIfAbsent(t, k -> new ArrayList<>()).add(p);
        return p;
    }

    Resposta responder(Pergunta pergunta, Usuario autor, String texto) {
        Resposta r = new Resposta(autor, texto);
        pergunta.respostas.add(r);
        return r;
    }

    void aceitar(Pergunta pergunta, Resposta resposta) {
        if (!pergunta.respostas.contains(resposta)) {
            throw new IllegalArgumentException("resposta de outra pergunta");
        }
        if (pergunta.aceita != null) {
            pergunta.aceita.autor.reputacao -= Postagem.PONTOS_ACEITA;
        }
        pergunta.aceita = resposta;
        resposta.autor.reputacao += Postagem.PONTOS_ACEITA;
    }

    List<Pergunta> buscar(String tag) {
        List<Pergunta> r = new ArrayList<>(porTag.getOrDefault(tag, new ArrayList<>()));
        r.sort((a, b) -> b.pontuacao() - a.pontuacao());
        return r;
    }
}

public static void main(String[] args) {
    Usuario ana = new Usuario("Ana"), bia = new Usuario("Bia"), caio = new Usuario("Caio");
    Site site = new Site();
    Pergunta p = site.perguntar(ana, "O que é um mutex?", "Explique.", Set.of("concorrencia"));
    Resposta r = site.responder(p, bia, "Trava de exclusão mútua.");
    r.votar(caio, 1);
    r.votar(caio, -1); // Caio muda de ideia
    site.aceitar(p, r);
    System.out.println(bia.reputacao + " " + r.pontuacao()); // 13 -1
    List<String> titulos = new ArrayList<>();
    for (Pergunta q : site.buscar("concorrencia")) titulos.add("'" + q.titulo + "'");
    System.out.println("[" + String.join(", ", titulos) + "]");
}

const int PONTOS_VOTO_NEGATIVO = -2;
const int PONTOS_ACEITA = 15;

struct Usuario {
    string nome;
    int reputacao = 0;
};

class Postagem {
    static int pontos(int valor) {
        switch (valor) {
            case 1: return PONTOS_VOTO_POSITIVO;
            case -1: return PONTOS_VOTO_NEGATIVO;
            default: return 0;
        }
    }

public:
    Usuario* autor;
    string texto;
    map<Usuario*, int> votos;

    Postagem(Usuario* autor, string texto) : autor(autor), texto(texto) {}
    virtual ~Postagem() = default;

    int pontuacao() const {
        int soma = 0;
        for (auto& [u, v] : votos) soma += v;
        return soma;
    }

    void votar(Usuario* usuario, int valor) {
        if (usuario == autor) {
            throw invalid_argument("não pode votar no próprio conteúdo");
        }
        int anterior = votos.count(usuario) ? votos[usuario] : 0;
        if (anterior == valor) {
            return;
        }
        autor->reputacao -= pontos(anterior);
        votos[usuario] = valor;
        autor->reputacao += pontos(valor);
    }
};

class Resposta : public Postagem {
public:
    using Postagem::Postagem;
};

class Pergunta : public Postagem {
public:
    string titulo;
    set<string> tags;
    vector<Resposta*> respostas;
    Resposta* aceita = nullptr;

    Pergunta(Usuario* autor, string titulo, string texto, set<string> tags)
        : Postagem(autor, texto), titulo(titulo), tags(tags) {}
};

class Site {
    vector<unique_ptr<Pergunta>> perguntas;
    vector<unique_ptr<Resposta>> todasRespostas;

public:
    map<string, vector<Pergunta*>> porTag;

    Pergunta* perguntar(Usuario* autor, const string& titulo, const string& texto, const set<string>& tags) {
        perguntas.push_back(make_unique<Pergunta>(autor, titulo, texto, tags));
        Pergunta* p = perguntas.back().get();
        for (auto& t : p->tags) porTag[t].push_back(p);
        return p;
    }

    Resposta* responder(Pergunta* pergunta, Usuario* autor, const string& texto) {
        todasRespostas.push_back(make_unique<Resposta>(autor, texto));
        Resposta* r = todasRespostas.back().get();
        pergunta->respostas.push_back(r);
        return r;
    }

    void aceitar(Pergunta* pergunta, Resposta* resposta) {
        if (find(pergunta->respostas.begin(), pergunta->respostas.end(), resposta) == pergunta->respostas.end()) {
            throw invalid_argument("resposta de outra pergunta");
        }
        if (pergunta->aceita != nullptr) {
            pergunta->aceita->autor->reputacao -= PONTOS_ACEITA;
        }
        pergunta->aceita = resposta;
        resposta->autor->reputacao += PONTOS_ACEITA;
    }

    vector<Pergunta*> buscar(const string& tag) {
        vector<Pergunta*> r = porTag[tag];
        stable_sort(r.begin(), r.end(), [](Pergunta* a, Pergunta* b) { return a->pontuacao() > b->pontuacao(); });
        return r;
    }
};

}

using System;
using System.Collections.Generic;
using System.Linq;

class Usuario
{
    public readonly string Nome;
    public int Reputacao = 0;

    public Usuario(string nome)
    {
        Nome = nome;
    }
}

class Postagem
{
    public const int PONTOS_VOTO_POSITIVO = 10;
    public const int PONTOS_VOTO_NEGATIVO = -2;
    public const int PONTOS_ACEITA = 15;

    public readonly Usuario Autor;
    public readonly string Texto;
    public readonly Dictionary<Usuario, int> Votos = new Dictionary<Usuario, int>();

    public Postagem(Usuario autor, string texto)
    {
        Autor = autor;
        Texto = texto;
    }

    public int Pontuacao
    {
        get { return Votos.Values.Sum(); }
    }

    public void Votar(Usuario usuario, int valor)
    {
        if (usuario == Autor)
        {
            throw new ArgumentException("não pode votar no próprio conteúdo");
        }
        int anterior;
        Votos.TryGetValue(usuario, out anterior);
        if (anterior == valor)
        {
            return;
        }
        Autor.Reputacao -= Pontos(anterior);
        Votos[usuario] = valor;
        Autor.Reputacao += Pontos(valor);
    }

    private static int Pontos(int valor)
    {
        switch (valor)
        {
            case 1: return PONTOS_VOTO_POSITIVO;
            case -1: return PONTOS_VOTO_NEGATIVO;
            default: return 0;
        }
    }
}

class Resposta : Postagem
{
    public Resposta(Usuario autor, string texto) : base(autor, texto) { }
}

class Pergunta : Postagem
{
    public readonly string Titulo;
    public readonly HashSet<string> Tags;
    public readonly List<Resposta> Respostas = new List<Resposta>();
    public Resposta Aceita = null;

    public Pergunta(Usuario autor, string titulo, string texto, IEnumerable<string> tags) : base(autor, texto)
    {
        Titulo = titulo;
        Tags = new HashSet<string>(tags);
    }
}

class Site
{
    public readonly Dictionary<string, List<Pergunta>> PorTag = new Dictionary<string, List<Pergunta>>();

    public Pergunta Perguntar(Usuario autor, string titulo, string texto, IEnumerable<string> tags)
    {
        var p = new Pergunta(autor, titulo, texto, tags);
        foreach (var t in p.Tags)
        {
            if (!PorTag.ContainsKey(t)) PorTag[t] = new List<Pergunta>();
            PorTag[t].Add(p);
        }
        return p;
    }

    public Resposta Responder(Pergunta pergunta, Usuario autor, string texto)
    {
        var r = new Resposta(autor, texto);
        pergunta.Respostas.Add(r);
        return r;
    }

    public void Aceitar(Pergunta pergunta, Resposta resposta)
    {
        if (!pergunta.Respostas.Contains(resposta))
        {
            throw new ArgumentException("resposta de outra pergunta");
        }
        if (pergunta.Aceita != null)
        {
            pergunta.Aceita.Autor.Reputacao -= Postagem.PONTOS_ACEITA;
        }
        pergunta.Aceita = resposta;
        resposta.Autor.Reputacao += Postagem.PONTOS_ACEITA;
    }

    public List<Pergunta> Buscar(string tag)
    {
        List<Pergunta> lista;
        if (!PorTag.TryGetValue(tag, out lista)) lista = new List<Pergunta>();
        return lista.OrderByDescending(p => p.Pontuacao).ToList();
    }
}

}

	pontosVotoPositivo = 10
	pontosVotoNegativo = -2
	pontosAceita       = 15
)

type Usuario struct {
	nome      string
	reputacao int
}

type Postagem struct {
	autor *Usuario
	texto string
	votos map[*Usuario]int
}

func NovaPostagem(autor *Usuario, texto string) Postagem {
	return Postagem{autor: autor, texto: texto, votos: map[*Usuario]int{}}
}

func (p *Postagem) pontuacao() int {
	soma := 0
	for _, v := range p.votos {
		soma += v
	}
	return soma
}

func pontos(valor int) int {
	switch valor {
	case 1:
		return pontosVotoPositivo
	case -1:
		return pontosVotoNegativo
	}
	return 0
}

func (p *Postagem) votar(usuario *Usuario, valor int) error {
	if usuario == p.autor {
		return errors.New("não pode votar no próprio conteúdo")
	}
	anterior := p.votos[usuario]
	if anterior == valor {
		return nil
	}
	p.autor.reputacao -= pontos(anterior)
	p.votos[usuario] = valor
	p.autor.reputacao += pontos(valor)
	return nil
}

type Resposta struct {
	Postagem
}

type Pergunta struct {
	Postagem
	titulo    string
	tags      []string
	respostas []*Resposta
	aceita    *Resposta
}

type Site struct {
	porTag map[string][]*Pergunta
}

func (s *Site) perguntar(autor *Usuario, titulo, texto string, tags []string) *Pergunta {
	p := &Pergunta{Postagem: NovaPostagem(autor, texto), titulo: titulo, tags: tags}
	for _, t := range tags {
		s.porTag[t] = append(s.porTag[t], p)
	}
	return p
}

func (s *Site) responder(pergunta *Pergunta, autor *Usuario, texto string) *Resposta {
	r := &Resposta{NovaPostagem(autor, texto)}
	pergunta.respostas = append(pergunta.respostas, r)
	return r
}

func (s *Site) aceitar(pergunta *Pergunta, resposta *Resposta) error {
	encontrada := false
	for _, r := range pergunta.respostas {
		if r == resposta {
			encontrada = true
		}
	}
	if !encontrada {
		return errors.New("resposta de outra pergunta")
	}
	if pergunta.aceita != nil {
		pergunta.aceita.autor.reputacao -= pontosAceita
	}
	pergunta.aceita = resposta
	resposta.autor.reputacao += pontosAceita
	return nil
}

func (s *Site) buscar(tag string) []*Pergunta {
	r := append([]*Pergunta{}, s.porTag[tag]...)
	sort.SliceStable(r, func(i, j int) bool { return r[i].pontuacao() > r[j].pontuacao() })
	return r
}

func main() {
	ana, bia, caio := &Usuario{nome: "Ana"}, &Usuario{nome: "Bia"}, &Usuario{nome: "Caio"}
	site := &Site{porTag: map[string][]*Pergunta{}}
	p := site.perguntar(ana, "O que é um mutex?", "Explique.", []string{"concorrencia"})
	r := site.responder(p, bia, "Trava de exclusão mútua.")
	r.votar(caio, 1)
	r.votar(caio, -1) // Caio muda de ideia
	site.aceitar(p, r)
	fmt.Println(bia.reputacao, r.pontuacao()) // 13 -1
	titulos := []string{}
	for _, q := range site.buscar("concorrencia") {
		titulos = append(titulos, "'"+q.titulo+"'")
	}
	fmt.Println("[" + strings.Join(titulos, ", ") + "]")
}

Ao trocar o voto, a reputação anterior é desfeita antes de aplicar a nova, o que evita acúmulo indevido.

## Concorrência e extensões

Votos simultâneos no mesmo item exigem lock por postagem (ou contadores atômicos). Extensões: comentários (Composite leve), moderação e privilégios por reputação, histórico de edições, e busca textual com índice invertido. Reputação pode ser derivada de eventos de voto, o que facilita recalcular.
