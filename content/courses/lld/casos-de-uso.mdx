Antes de decidir classes, você precisa saber **o que o sistema oferece e para quem**. O diagrama de casos de uso responde exatamente isso, em alto nível e sem detalhes de implementação. É a ferramenta ideal para a fase de requisitos de uma entrevista de LLD: ele organiza as funcionalidades e evita que você modele coisas que ninguém pediu.

## Elementos básicos

- **Ator**: quem interage com o sistema de fora. Pode ser uma pessoa (cliente, administrador) ou outro sistema (gateway de pagamento). Desenhado como um boneco.
- **Caso de uso**: um objetivo que o ator quer alcançar, escrito como verbo + objeto ("Sacar dinheiro", "Reservar sala"). Desenhado como elipse.
- **Fronteira do sistema**: retângulo que separa o que está dentro do sistema do que está fora.
- **Associação**: linha ligando ator ao caso de uso.

```text
 +----------------- Caixa Eletrônico -----------------+
 |                                                    |
 |   (Consultar saldo)                                |
 |        |                                           |
 |   (Sacar dinheiro) ---- inclui ---> (Autenticar)   |
 |        |                                ^          |
 |   (Depositar) ------- inclui -----------+          |
 |                                                    |
 +----------------------------------------------------+
   Cliente (ator)  --- liga-se aos três casos de uso
   Banco (ator externo) --- liga-se a "Sacar dinheiro"
```

## Include, extend e generalização

| Relação | Quando usar | Exemplo |
| --- | --- | --- |
| include | comportamento obrigatório reaproveitado | Sacar inclui Autenticar |
| extend | comportamento opcional, em condição especial | Pagar pode ser estendido por Aplicar cupom |
| generalização | variação de ator ou caso | Administrador é um tipo de Usuário |

A dica prática: `include` é "sempre acontece", `extend` é "às vezes acontece". Misturar os dois é o erro mais comum.

## Do caso de uso ao código

Cada caso de uso costuma virar um método de serviço (ou um comando). Veja uma tradução direta:

```python
class Autenticacao:
    def __init__(self, senhas: dict[str, str]) -> None:
        self._senhas = senhas

    def validar(self, cliente: str, senha: str) -> bool:
        return self._senhas.get(cliente) == senha


class CaixaEletronico:
    def __init__(self, auth: Autenticacao, saldos: dict[str, float]) -> None:
        self._auth = auth
        self._saldos = saldos

    def sacar(self, cliente: str, senha: str, valor: float) -> bool:
        if not self._auth.validar(cliente, senha):  # include: Autenticar
            return False
        if self._saldos[cliente] < valor:
            return False
        self._saldos[cliente] -= valor
        return True

    def consultar_saldo(self, cliente: str, senha: str) -> float | None:
        if not self._auth.validar(cliente, senha):
            return None
        return self._saldos[cliente]


caixa = CaixaEletronico(Autenticacao({"ana": "123"}), {"ana": 200.0})
assert caixa.sacar("ana", "123", 50) is True
assert caixa.sacar("ana", "errada", 10) is False
print(caixa.consultar_saldo("ana", "123"))
```

```java
import java.util.*;

class Autenticacao {
    private final Map<String, String> senhas;

    Autenticacao(Map<String, String> senhas) { this.senhas = senhas; }

    boolean validar(String cliente, String senha) { return senha.equals(senhas.get(cliente)); }
}

class CaixaEletronico {
    private final Autenticacao auth;
    private final Map<String, Double> saldos;

    CaixaEletronico(Autenticacao auth, Map<String, Double> saldos) {
        this.auth = auth;
        this.saldos = saldos;
    }

    boolean sacar(String cliente, String senha, double valor) {
        if (!auth.validar(cliente, senha)) return false;   // include: Autenticar
        if (saldos.get(cliente) < valor) return false;
        saldos.put(cliente, saldos.get(cliente) - valor);
        return true;
    }

    Double consultarSaldo(String cliente, String senha) {
        if (!auth.validar(cliente, senha)) return null;
        return saldos.get(cliente);
    }
}

public static void main(String[] args) {
    CaixaEletronico caixa = new CaixaEletronico(new Autenticacao(Map.of("ana", "123")), new HashMap<>(Map.of("ana", 200.0)));
    if (!caixa.sacar("ana", "123", 50)) throw new AssertionError();
    if (caixa.sacar("ana", "errada", 10)) throw new AssertionError();
    System.out.println(caixa.consultarSaldo("ana", "123"));
}
```

```cpp
#include <cassert>
#include <iostream>
#include <map>
#include <optional>
#include <string>
using namespace std;

class Autenticacao {
    map<string, string> senhas;

public:
    Autenticacao(map<string, string> senhas) : senhas(senhas) {}

    bool validar(const string& cliente, const string& senha) const {
        auto it = senhas.find(cliente);
        return it != senhas.end() && it->second == senha;
    }
};

class CaixaEletronico {
    Autenticacao auth;
    map<string, double> saldos;

public:
    CaixaEletronico(Autenticacao auth, map<string, double> saldos) : auth(auth), saldos(saldos) {}

    bool sacar(const string& cliente, const string& senha, double valor) {
        if (!auth.validar(cliente, senha)) return false;   // include: Autenticar
        if (saldos[cliente] < valor) return false;
        saldos[cliente] -= valor;
        return true;
    }

    optional<double> consultarSaldo(const string& cliente, const string& senha) {
        if (!auth.validar(cliente, senha)) return nullopt;
        return saldos[cliente];
    }
};

int main() {
    CaixaEletronico caixa(Autenticacao(map<string, string>{{"ana", "123"}}), map<string, double>{{"ana", 200.0}});
    assert(caixa.sacar("ana", "123", 50) == true);
    assert(caixa.sacar("ana", "errada", 10) == false);
    cout << *caixa.consultarSaldo("ana", "123") << endl;
}
```

```go
import "fmt"

type Autenticacao struct {
	senhas map[string]string
}

func (a Autenticacao) validar(cliente, senha string) bool {
	guardada, ok := a.senhas[cliente]
	return ok && guardada == senha
}

type CaixaEletronico struct {
	auth   Autenticacao
	saldos map[string]float64
}

func (c *CaixaEletronico) sacar(cliente, senha string, valor float64) bool {
	if !c.auth.validar(cliente, senha) { // include: Autenticar
		return false
	}
	if c.saldos[cliente] < valor {
		return false
	}
	c.saldos[cliente] -= valor
	return true
}

func (c *CaixaEletronico) consultarSaldo(cliente, senha string) (float64, bool) {
	if !c.auth.validar(cliente, senha) {
		return 0, false
	}
	return c.saldos[cliente], true
}

func main() {
	caixa := &CaixaEletronico{Autenticacao{map[string]string{"ana": "123"}}, map[string]float64{"ana": 200.0}}
	if !caixa.sacar("ana", "123", 50) || caixa.sacar("ana", "errada", 10) {
		panic("falhou")
	}
	saldo, _ := caixa.consultarSaldo("ana", "123")
	fmt.Println(saldo)
}
```

A relação `include` virou uma chamada compartilhada a `Autenticacao`, o que evita repetir a regra em cada operação.

## Boas práticas

- Nomeie casos de uso do ponto de vista do ator, não do sistema ("Comprar ingresso", não "Inserir registro de venda").
- Mantenha poucos casos: de 5 a 10 principais bastam numa entrevista.
- Não desenhe passos internos. Se precisar detalhar o fluxo, use o diagrama de atividades ou de sequência.

<Callout type="note" title="Texto também vale">
Em quadro branco, uma lista de atores e uma lista de casos de uso, com um traço ligando cada um, comunica o mesmo e economiza tempo.
</Callout>

## Na entrevista

Use o diagrama ou uma lista equivalente logo após as perguntas de esclarecimento, para confirmar o escopo: "então o sistema tem estes atores e estes casos; algo fora disso?". Follow-ups comuns: "qual a diferença entre include e extend?", "esse ator é humano ou sistema externo?".
