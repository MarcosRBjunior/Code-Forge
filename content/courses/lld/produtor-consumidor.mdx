Duas partes do sistema trabalham em ritmos diferentes: uma gera trabalho (pedidos chegando, linhas lidas de um arquivo) e outra o processa (cobrança, gravação). Se a geradora chamasse a processadora diretamente, qualquer lentidão de uma travaria a outra. **Produtor-consumidor** coloca entre elas um **buffer compartilhado**: produtores colocam itens, consumidores retiram, e os dois lados evoluem de forma independente.

## O problema de coordenar

Um buffer compartilhado exige regras:

- Se o buffer está **cheio**, o produtor precisa esperar.
- Se está **vazio**, o consumidor precisa esperar.
- Acessos simultâneos não podem corromper a estrutura.

Implementar isso com travas e variáveis de condição é clássico e fácil de errar. Em Python, `queue.Queue` entrega tudo isso pronto: é thread-safe e bloqueia sozinha quando cheia ou vazia.

## Exemplo

```python
import queue
import threading

SENTINELA = None  # sinaliza fim da produção

buffer: queue.Queue = queue.Queue(maxsize=3)  # limitado: contrapressão
resultados: list[int] = []
trava = threading.Lock()


def produtor(n: int) -> None:
    for i in range(n):
        buffer.put(i)  # bloqueia se o buffer estiver cheio
    buffer.put(SENTINELA)


def consumidor() -> None:
    while True:
        item = buffer.get()  # bloqueia se estiver vazio
        if item is SENTINELA:
            break
        with trava:
            resultados.append(item * 10)


t1 = threading.Thread(target=produtor, args=(8,))
t2 = threading.Thread(target=consumidor)
t1.start()
t2.start()
t1.join()
t2.join()

assert resultados == [0, 10, 20, 30, 40, 50, 60, 70]
print(resultados)
```

```javascript
const SENTINELA = null;   // sinaliza fim da produção

// Fila limitada para JavaScript: put() espera quando está cheia, get() espera quando está vazia
class FilaLimitada {
  #itens = [];
  #capacidade;
  #esperandoEspaco = [];
  #esperandoItem = [];

  constructor(capacidade) { this.#capacidade = capacidade; }

  async put(item) {
    while (this.#itens.length >= this.#capacidade) {
      await new Promise((r) => this.#esperandoEspaco.push(r));
    }
    this.#itens.push(item);
    this.#esperandoItem.shift()?.();
  }

  async get() {
    while (this.#itens.length === 0) {
      await new Promise((r) => this.#esperandoItem.push(r));
    }
    const item = this.#itens.shift();
    this.#esperandoEspaco.shift()?.();
    return item;
  }
}

const buffer = new FilaLimitada(3);   // limitado: contrapressão
const resultados = [];

async function produtor(n) {
  for (let i = 0; i < n; i++) {
    await buffer.put(i);   // espera se o buffer estiver cheio
  }
  await buffer.put(SENTINELA);
}

async function consumidor() {
  while (true) {
    const item = await buffer.get();   // espera se estiver vazio
    if (item === SENTINELA) break;
    resultados.push(item * 10);
  }
}

(async () => {
  await Promise.all([produtor(8), consumidor()]);
  console.assert(JSON.stringify(resultados) === JSON.stringify([0, 10, 20, 30, 40, 50, 60, 70]));
  console.log(resultados);
})();
```

```typescript
const SENTINELA = null;   // sinaliza fim da produção

// Fila limitada para TypeScript: put() espera quando está cheia, get() espera quando está vazia
class FilaLimitada<T> {
  private itens: T[] = [];
  private esperandoEspaco: (() => void)[] = [];
  private esperandoItem: (() => void)[] = [];

  constructor(private capacidade: number) {}

  async put(item: T): Promise<void> {
    while (this.itens.length >= this.capacidade) {
      await new Promise<void>((r) => this.esperandoEspaco.push(r));
    }
    this.itens.push(item);
    const acordar = this.esperandoItem.shift();
    if (acordar) acordar();
  }

  async get(): Promise<T> {
    while (this.itens.length === 0) {
      await new Promise<void>((r) => this.esperandoItem.push(r));
    }
    const item = this.itens.shift() as T;
    const acordar = this.esperandoEspaco.shift();
    if (acordar) acordar();
    return item;
  }
}
```

```java
import java.util.*;
import java.util.concurrent.*;

static final Integer SENTINELA = Integer.MIN_VALUE;   // sinaliza fim da produção (um valor que nunca é produzido)

static final BlockingQueue<Integer> buffer = new ArrayBlockingQueue<>(3);   // limitado: contrapressão
static final List<Integer> resultados = new ArrayList<>();
static final Object trava = new Object();

static void produtor(int n) throws InterruptedException {
    for (int i = 0; i < n; i++) {
        buffer.put(i);   // bloqueia se o buffer estiver cheio
    }
    buffer.put(SENTINELA);
}

static void consumidor() throws InterruptedException {
    while (true) {
        Integer item = buffer.take();   // bloqueia se estiver vazio
        if (item.equals(SENTINELA)) break;
        synchronized (trava) {
            resultados.add(item * 10);
        }
    }
}

public static void main(String[] args) throws Exception {
    Thread t1 = new Thread(() -> {
        try { produtor(8); } catch (InterruptedException e) { Thread.currentThread().interrupt(); }
    });
    Thread t2 = new Thread(() -> {
        try { consumidor(); } catch (InterruptedException e) { Thread.currentThread().interrupt(); }
    });
    t1.start();
    t2.start();
    t1.join();
    t2.join();

    if (!resultados.equals(List.of(0, 10, 20, 30, 40, 50, 60, 70))) throw new AssertionError();
    System.out.println(resultados);
}
```

```cpp
#include <cassert>
#include <condition_variable>
#include <iostream>
#include <mutex>
#include <queue>
#include <thread>
#include <vector>
using namespace std;

const int SENTINELA = -1;   // sinaliza fim da produção (um valor que nunca é produzido)

// Fila limitada: put() bloqueia quando está cheia, get() bloqueia quando está vazia
class FilaLimitada {
    queue<int> itens;
    size_t capacidade;
    mutex m;
    condition_variable temEspaco, temItem;

public:
    FilaLimitada(size_t capacidade) : capacidade(capacidade) {}

    void put(int item) {
        unique_lock<mutex> lk(m);
        temEspaco.wait(lk, [&] { return itens.size() < capacidade; });
        itens.push(item);
        temItem.notify_one();
    }

    int get() {
        unique_lock<mutex> lk(m);
        temItem.wait(lk, [&] { return !itens.empty(); });
        int item = itens.front();
        itens.pop();
        temEspaco.notify_one();
        return item;
    }
};

FilaLimitada buffer(3);   // limitado: contrapressão
vector<int> resultados;
mutex trava;

void produtor(int n) {
    for (int i = 0; i < n; i++) {
        buffer.put(i);   // bloqueia se o buffer estiver cheio
    }
    buffer.put(SENTINELA);
}

void consumidor() {
    while (true) {
        int item = buffer.get();   // bloqueia se estiver vazio
        if (item == SENTINELA) break;
        lock_guard<mutex> guarda(trava);
        resultados.push_back(item * 10);
    }
}

int main() {
    thread t1(produtor, 8);
    thread t2(consumidor);
    t1.join();
    t2.join();

    assert((resultados == vector<int>{0, 10, 20, 30, 40, 50, 60, 70}));
    cout << "[";
    for (size_t i = 0; i < resultados.size(); i++) cout << (i ? ", " : "") << resultados[i];
    cout << "]" << endl;
}
```

```csharp
using System;
using System.Collections.Concurrent;
using System.Collections.Generic;
using System.Linq;
using System.Threading;

const int SENTINELA = -1;   // sinaliza fim da produção (um valor que nunca é produzido)

static BlockingCollection<int> buffer = new BlockingCollection<int>(3);   // limitado: contrapressão
static List<int> resultados = new List<int>();
static object trava = new object();

static void Produtor(int n)
{
    for (int i = 0; i < n; i++)
    {
        buffer.Add(i);   // bloqueia se o buffer estiver cheio
    }
    buffer.Add(SENTINELA);
}

static void Consumidor()
{
    while (true)
    {
        int item = buffer.Take();   // bloqueia se estiver vazio
        if (item == SENTINELA) break;
        lock (trava)
        {
            resultados.Add(item * 10);
        }
    }
}

static void Main()
{
    var t1 = new Thread(() => Produtor(8));
    var t2 = new Thread(Consumidor);
    t1.Start();
    t2.Start();
    t1.Join();
    t2.Join();

    if (!resultados.SequenceEqual(new[] { 0, 10, 20, 30, 40, 50, 60, 70 })) throw new Exception("falhou");
    Console.WriteLine("[" + string.Join(", ", resultados) + "]");
}
```

```go
import (
	"fmt"
	"sync"
)

func main() {
	buffer := make(chan int, 3) // limitado: contrapressão (a capacidade do canal)
	resultados := []int{}
	var trava sync.Mutex
	var wg sync.WaitGroup

	produtor := func(n int) {
		defer wg.Done()
		for i := 0; i < n; i++ {
			buffer <- i // bloqueia se o buffer estiver cheio
		}
		close(buffer) // fechar o canal faz o papel da sentinela
	}

	consumidor := func() {
		defer wg.Done()
		for item := range buffer { // bloqueia se estiver vazio; termina quando o canal fecha
			trava.Lock()
			resultados = append(resultados, item*10)
			trava.Unlock()
		}
	}

	wg.Add(2)
	go produtor(8)
	go consumidor()
	wg.Wait()

	fmt.Println(resultados)
}
```

</CodeTabs>

Com um único produtor e um consumidor a ordem é preservada. Com vários consumidores, você coloca uma sentinela por consumidor, e a ordem de saída deixa de ser garantida.

## Variações

| Cenário | Ajuste |
| --- | --- |
| Vários produtores | Contar quantos terminaram antes de enviar as sentinelas |
| Vários consumidores | Uma sentinela para cada consumidor |
| Itens com prioridade | `queue.PriorityQueue` |
| Buffer ilimitado | `Queue()` sem `maxsize`: cuidado com memória |
| Processos em vez de threads | `multiprocessing.Queue` |

<Callout type="tip" title="Contrapressão">
Um buffer limitado é uma ferramenta, não um incômodo. Quando o consumidor é mais lento, o produtor desacelera em vez de encher a memória. Em sistemas distribuídos, o equivalente é uma fila de mensagens com limite.
</Callout>

## Onde aparece

- **Thread pool**: a fila de tarefas é o buffer; quem chama `submit` é o produtor e os workers são os consumidores.
- **Logging assíncrono**: a aplicação produz mensagens e uma thread de log as grava.
- **Pipelines**: a saída de um estágio é a entrada do próximo, encadeando buffers.
- **Sistemas de mensagens**: o broker é o buffer entre serviços.

## Armadilhas

- **Deadlock por término**: consumidores que ficam esperando para sempre porque ninguém enviou a sentinela.
- **Perda de itens** ao encerrar abruptamente: decida se descarta ou esvazia a fila.
- **Contenção**: um único buffer disputado por muitas threads vira gargalo.
- **Esquecer a trava** ao atualizar estado compartilhado fora da fila (como a lista de resultados acima).
