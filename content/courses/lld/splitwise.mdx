O problema do Splitwise: um grupo divide despesas e o sistema mostra quem deve quanto a quem. Dois pontos pesam na avaliação: estratégias de divisão (igual, exata, percentual) e a simplificação das dívidas.

## Requisitos

- Usuários registram despesas pagas por um deles.
- Divisão igual, por valores exatos ou por percentuais.
- Consultar o saldo de cada pessoa.
- Simplificar: reduzir o número de transferências para quitar tudo.

## Entidades

| Entidade | Responsabilidade |
| --- | --- |
| `Divisao` | Strategy que gera a parte de cada participante. |
| `Despesa` | Pagador, valor e partes. |
| `Grupo` | Registra despesas e mantém saldos. |
| `simplificar` | Função que gera as transferências mínimas. |

Guardamos o saldo líquido de cada pessoa: positivo significa que tem a receber, negativo que deve. Dinheiro em centavos inteiros, para a soma das partes sempre bater.

```python
from collections import defaultdict
from typing import Dict, List, Tuple


class Divisao:
    def partes(self, total: int, participantes: List[str]) -> Dict[str, int]:
        raise NotImplementedError


class Igual(Divisao):
    def partes(self, total, participantes):
        base, resto = divmod(total, len(participantes))
        return {p: base + (1 if i < resto else 0) for i, p in enumerate(participantes)}


class Exata(Divisao):
    def __init__(self, valores: Dict[str, int]):
        self.valores = valores

    def partes(self, total, participantes):
        if sum(self.valores.values()) != total:
            raise ValueError("valores não somam o total")
        return dict(self.valores)


class Percentual(Divisao):
    def __init__(self, percentuais: Dict[str, int]):
        if sum(percentuais.values()) != 100:
            raise ValueError("percentuais devem somar 100")
        self.percentuais = percentuais

    def partes(self, total, participantes):
        partes = {p: total * pct // 100 for p, pct in self.percentuais.items()}
        sobra = total - sum(partes.values())
        for p in list(partes)[:sobra]:
            partes[p] += 1
        return partes


class Grupo:
    def __init__(self):
        self.saldo: Dict[str, int] = defaultdict(int)

    def despesa(self, pagador: str, total: int, participantes: List[str], divisao: Divisao):
        partes = divisao.partes(total, participantes)
        self.saldo[pagador] += total
        for pessoa, valor in partes.items():
            self.saldo[pessoa] -= valor

    def simplificar(self) -> List[Tuple[str, str, int]]:
        credores = sorted(((v, p) for p, v in self.saldo.items() if v > 0), reverse=True)
        devedores = sorted((-v, p) for p, v in self.saldo.items() if v < 0)[::-1]
        transf = []
        i = j = 0
        while i < len(devedores) and j < len(credores):
            dv, dp = devedores[i]
            cv, cp = credores[j]
            valor = min(dv, cv)
            transf.append((dp, cp, valor))
            devedores[i], credores[j] = (dv - valor, dp), (cv - valor, cp)
            if devedores[i][0] == 0:
                i += 1
            if credores[j][0] == 0:
                j += 1
        return transf


g = Grupo()
g.despesa("ana", 9000, ["ana", "bia", "caio"], Igual())                 # 30,00 cada
g.despesa("bia", 5000, ["ana", "bia"], Percentual({"ana": 60, "bia": 40}))
g.despesa("caio", 3000, ["ana", "caio"], Exata({"ana": 1000, "caio": 2000}))
print(dict(g.saldo))
print(g.simplificar())
assert sum(g.saldo.values()) == 0
```

```javascript
  constructor(valores) {
    super();
    this.valores = valores;
  }

    }
    return new Map(this.valores);
  }
}

  constructor(percentuais) {
    super();

    }
    this.percentuais = percentuais;

    for (const p of [...partes.keys()].slice(0, sobra)) {
      partes.set(p, partes.get(p) + 1);
    }
    return partes;
  }
}
```

```java
import java.util.*;

abstract class Divisao {
    abstract Map<String, Integer> partes(int total, List<String> participantes);
}

class Igual extends Divisao {
    Map<String, Integer> partes(int total, List<String> participantes) {
        int base = total / participantes.size();
        int resto = total % participantes.size();
        Map<String, Integer> r = new LinkedHashMap<>();
        for (int i = 0; i < participantes.size(); i++) r.put(participantes.get(i), base + (i < resto ? 1 : 0));
        return r;
    }
}

class Exata extends Divisao {
    private final Map<String, Integer> valores;

    Exata(Map<String, Integer> valores) {
        this.valores = valores;
    }

    Map<String, Integer> partes(int total, List<String> participantes) {
        int soma = 0;
        for (int v : valores.values()) soma += v;
        if (soma != total) {
            throw new IllegalArgumentException("valores não somam o total");
        }
        return new LinkedHashMap<>(valores);
    }
}

class Percentual extends Divisao {
    private final Map<String, Integer> percentuais;

    Percentual(Map<String, Integer> percentuais) {
        int soma = 0;
        for (int v : percentuais.values()) soma += v;
        if (soma != 100) {
            throw new IllegalArgumentException("percentuais devem somar 100");
        }
        this.percentuais = percentuais;
    }

    Map<String, Integer> partes(int total, List<String> participantes) {
        Map<String, Integer> partes = new LinkedHashMap<>();
        int usado = 0;
        for (Map.Entry<String, Integer> e : percentuais.entrySet()) {
            int v = total * e.getValue() / 100;
            partes.put(e.getKey(), v);
            usado += v;
        }
        int sobra = total - usado;
        for (String p : new ArrayList<>(partes.keySet()).subList(0, sobra)) {
            partes.put(p, partes.get(p) + 1);
        }
        return partes;
    }
}

class Grupo {
    final Map<String, Integer> saldo = new LinkedHashMap<>();

    void despesa(String pagador, int total, List<String> participantes, Divisao divisao) {
        Map<String, Integer> partes = divisao.partes(total, participantes);
        saldo.merge(pagador, total, Integer::sum);
        for (Map.Entry<String, Integer> e : partes.entrySet()) {
            saldo.merge(e.getKey(), -e.getValue(), Integer::sum);
        }
    }

    // cada transferência: {devedor, credor, valor} em strings já formatadas
    List<String[]> simplificar() {
        List<Object[]> credores = new ArrayList<>(), devedores = new ArrayList<>();
        for (Map.Entry<String, Integer> e : saldo.entrySet()) {
            if (e.getValue() > 0) credores.add(new Object[] {e.getValue(), e.getKey()});
            if (e.getValue() < 0) devedores.add(new Object[] {-e.getValue(), e.getKey()});
        }
        Comparator<Object[]> porValorENome = (a, b) -> {
            int c = Integer.compare((Integer) a[0], (Integer) b[0]);
            return c != 0 ? c : ((String) a[1]).compareTo((String) b[1]);
        };
        credores.sort(porValorENome.reversed());
        devedores.sort(porValorENome);
        Collections.reverse(devedores);
        List<String[]> transf = new ArrayList<>();
        int i = 0, j = 0;
        while (i < devedores.size() && j < credores.size()) {
            int dv = (Integer) devedores.get(i)[0], cv = (Integer) credores.get(j)[0];
            String dp = (String) devedores.get(i)[1], cp = (String) credores.get(j)[1];
            int valor = Math.min(dv, cv);
            transf.add(new String[] {dp, cp, String.valueOf(valor)});
            devedores.set(i, new Object[] {dv - valor, dp});
            credores.set(j, new Object[] {cv - valor, cp});
            if ((Integer) devedores.get(i)[0] == 0) i += 1;
            if ((Integer) credores.get(j)[0] == 0) j += 1;
        }
        return transf;
    }
}

public static void main(String[] args) {
    Grupo g = new Grupo();
    g.despesa("ana", 9000, List.of("ana", "bia", "caio"), new Igual()); // 30,00 cada
    Map<String, Integer> pct = new LinkedHashMap<>();
    pct.put("ana", 60);
    pct.put("bia", 40);
    g.despesa("bia", 5000, List.of("ana", "bia"), new Percentual(pct));
    Map<String, Integer> exatos = new LinkedHashMap<>();
    exatos.put("ana", 1000);
    exatos.put("caio", 2000);
    g.despesa("caio", 3000, List.of("ana", "caio"), new Exata(exatos));

    List<String> pares = new ArrayList<>();
    for (Map.Entry<String, Integer> e : g.saldo.entrySet()) pares.add("'" + e.getKey() + "': " + e.getValue());
    System.out.println("{" + String.join(", ", pares) + "}");
    List<String> ts = new ArrayList<>();
    for (String[] t : g.simplificar()) ts.add("('" + t[0] + "', '" + t[1] + "', " + t[2] + ")");
    System.out.println("[" + String.join(", ", ts) + "]");
    int soma = 0;
    for (int v : g.saldo.values()) soma += v;
    if (soma != 0) throw new AssertionError("falhou");
}
```

```cpp
        return valores;

            r.push_back({p, v});
            usado += v;

        for (int i = 0; i < sobra; i++) r[i].second += 1;
        return r;
    }
};

    void ajustar(const string& pessoa, int delta) {
        for (auto& par : saldo) {
            if (par.first == pessoa) {
                par.second += delta;
                return;
            }
        }
        saldo.push_back({pessoa, delta});

        ajustar(pagador, total);
        for (auto& [pessoa, valor] : partes) ajustar(pessoa, -valor);

        vector<pair<int, string>> credores, devedores;
        for (auto& [p, v] : saldo) {
            if (v > 0) credores.push_back({v, p});
            if (v < 0) devedores.push_back({-v, p});

        sort(credores.begin(), credores.end(), greater<pair<int, string>>());
        sort(devedores.begin(), devedores.end());

            credores[j] = {cv - valor, cp};
            if (devedores[i].first == 0) i += 1;
            if (credores[j].first == 0) j += 1;
        }
        return transf;
    }
};
```

```csharp
using System;
using System.Collections.Generic;
using System.Linq;

abstract class Divisao
{
    public abstract List<KeyValuePair<string, int>> Partes(int total, List<string> participantes);
}

class Igual : Divisao
{
    public override List<KeyValuePair<string, int>> Partes(int total, List<string> participantes)
    {
        int baseValor = total / participantes.Count;
        int resto = total % participantes.Count;
        return participantes.Select((p, i) => new KeyValuePair<string, int>(p, baseValor + (i < resto ? 1 : 0))).ToList();
    }
}

class Exata : Divisao
{
    private readonly List<KeyValuePair<string, int>> valores;

    public Exata(List<KeyValuePair<string, int>> valores)
    {
        this.valores = valores;
    }

    public override List<KeyValuePair<string, int>> Partes(int total, List<string> participantes)
    {
        if (valores.Sum(v => v.Value) != total)
        {
            throw new ArgumentException("valores não somam o total");
        }
        return new List<KeyValuePair<string, int>>(valores);
    }
}

class Percentual : Divisao
{
    private readonly List<KeyValuePair<string, int>> percentuais;

    public Percentual(List<KeyValuePair<string, int>> percentuais)
    {
        if (percentuais.Sum(v => v.Value) != 100)
        {
            throw new ArgumentException("percentuais devem somar 100");
        }
        this.percentuais = percentuais;
    }

    public override List<KeyValuePair<string, int>> Partes(int total, List<string> participantes)
    {
        var r = percentuais.Select(par => new KeyValuePair<string, int>(par.Key, total * par.Value / 100)).ToList();
        int sobra = total - r.Sum(v => v.Value);
        for (int i = 0; i < sobra; i++) r[i] = new KeyValuePair<string, int>(r[i].Key, r[i].Value + 1);
        return r;
    }
}

class Grupo
{
    public readonly List<KeyValuePair<string, int>> Saldo = new List<KeyValuePair<string, int>>(); // mantém a ordem de inserção

    private void Ajustar(string pessoa, int delta)
    {
        int i = Saldo.FindIndex(par => par.Key == pessoa);
        if (i < 0) Saldo.Add(new KeyValuePair<string, int>(pessoa, delta));
        else Saldo[i] = new KeyValuePair<string, int>(pessoa, Saldo[i].Value + delta);
    }

    public void Despesa(string pagador, int total, List<string> participantes, Divisao divisao)
    {
        var partes = divisao.Partes(total, participantes);
        Ajustar(pagador, total);
        foreach (var par in partes) Ajustar(par.Key, -par.Value);
    }

    public List<Tuple<string, string, int>> Simplificar()
    {
        var credores = Saldo.Where(p => p.Value > 0).Select(p => Tuple.Create(p.Value, p.Key))
            .OrderByDescending(t => t.Item1).ThenByDescending(t => t.Item2, StringComparer.Ordinal).ToList();
        var devedores = Saldo.Where(p => p.Value < 0).Select(p => Tuple.Create(-p.Value, p.Key))
            .OrderBy(t => t.Item1).ThenBy(t => t.Item2, StringComparer.Ordinal).Reverse().ToList();
        var transf = new List<Tuple<string, string, int>>();
        int i = 0, j = 0;
        while (i < devedores.Count && j < credores.Count)
        {
            int dv = devedores[i].Item1, cv = credores[j].Item1;
            string dp = devedores[i].Item2, cp = credores[j].Item2;
            int valor = Math.Min(dv, cv);
            transf.Add(Tuple.Create(dp, cp, valor));
            devedores[i] = Tuple.Create(dv - valor, dp);
            credores[j] = Tuple.Create(cv - valor, cp);
            if (devedores[i].Item1 == 0) i += 1;
            if (credores[j].Item1 == 0) j += 1;
        }
        return transf;
    }
}

static void Main()
{
    var g = new Grupo();
    g.Despesa("ana", 9000, new List<string> { "ana", "bia", "caio" }, new Igual()); // 30,00 cada
    g.Despesa("bia", 5000, new List<string> { "ana", "bia" },
        new Percentual(new List<KeyValuePair<string, int>> { new KeyValuePair<string, int>("ana", 60), new KeyValuePair<string, int>("bia", 40) }));
    g.Despesa("caio", 3000, new List<string> { "ana", "caio" },
        new Exata(new List<KeyValuePair<string, int>> { new KeyValuePair<string, int>("ana", 1000), new KeyValuePair<string, int>("caio", 2000) }));

    Console.WriteLine("{" + string.Join(", ", g.Saldo.Select(p => "'" + p.Key + "': " + p.Value)) + "}");
    Console.WriteLine("[" + string.Join(", ", g.Simplificar().Select(t => "('" + t.Item1 + "', '" + t.Item2 + "', " + t.Item3 + ")")) + "]");
    if (g.Saldo.Sum(p => p.Value) != 0) throw new Exception("falhou");
}
```

```go
import (
	"errors"
	"fmt"
	"sort"
	"strings"
)

// Fatias de pares mantêm a ordem de inserção, como o dict do Python.
type Parte struct {
	pessoa string
	valor  int
}

type Divisao interface {
	partes(total int, participantes []string) ([]Parte, error)
}

type Igual struct{}

func (Igual) partes(total int, participantes []string) ([]Parte, error) {
	base, resto := total/len(participantes), total%len(participantes)
	r := []Parte{}
	for i, p := range participantes {
		extra := 0
		if i < resto {
			extra = 1
		}
		r = append(r, Parte{p, base + extra})
	}
	return r, nil
}

type Exata struct {
	valores []Parte
}

func (e Exata) partes(total int, participantes []string) ([]Parte, error) {
	soma := 0
	for _, p := range e.valores {
		soma += p.valor
	}
	if soma != total {
		return nil, errors.New("valores não somam o total")
	}
	return append([]Parte{}, e.valores...), nil
}

type Percentual struct {
	percentuais []Parte
}

func NovoPercentual(percentuais []Parte) (Percentual, error) {
	soma := 0
	for _, p := range percentuais {
		soma += p.valor
	}
	if soma != 100 {
		return Percentual{}, errors.New("percentuais devem somar 100")
	}
	return Percentual{percentuais}, nil
}

func (p Percentual) partes(total int, participantes []string) ([]Parte, error) {
	r := []Parte{}
	usado := 0
	for _, par := range p.percentuais {
		v := total * par.valor / 100
		r = append(r, Parte{par.pessoa, v})
		usado += v
	}
	sobra := total - usado
	for i := 0; i < sobra; i++ {
		r[i].valor++
	}
	return r, nil
}

type Grupo struct {
	saldo []Parte
}

func (g *Grupo) ajustar(pessoa string, delta int) {
	for i := range g.saldo {
		if g.saldo[i].pessoa == pessoa {
			g.saldo[i].valor += delta
			return
		}
	}
	g.saldo = append(g.saldo, Parte{pessoa, delta})
}

func (g *Grupo) despesa(pagador string, total int, participantes []string, divisao Divisao) error {
	partes, err := divisao.partes(total, participantes)
	if err != nil {
		return err
	}
	g.ajustar(pagador, total)
	for _, p := range partes {
		g.ajustar(p.pessoa, -p.valor)
	}
	return nil
}

type Transferencia struct {
	devedor, credor string
	valor           int
}

func (g *Grupo) simplificar() []Transferencia {
	credores, devedores := []Parte{}, []Parte{}
	for _, p := range g.saldo {
		if p.valor > 0 {
			credores = append(credores, p)
		}
		if p.valor < 0 {
			devedores = append(devedores, Parte{p.pessoa, -p.valor})
		}
	}
	sort.Slice(credores, func(a, b int) bool {
		if credores[a].valor != credores[b].valor {
			return credores[a].valor > credores[b].valor
		}
		return credores[a].pessoa > credores[b].pessoa
	})
	sort.Slice(devedores, func(a, b int) bool {
		if devedores[a].valor != devedores[b].valor {
			return devedores[a].valor > devedores[b].valor
		}
		return devedores[a].pessoa > devedores[b].pessoa
	})
	transf := []Transferencia{}
	i, j := 0, 0
	for i < len(devedores) && j < len(credores) {
		dv, cv := devedores[i].valor, credores[j].valor
		dp, cp := devedores[i].pessoa, credores[j].pessoa
		valor := dv
		if cv < valor {
			valor = cv
		}
		transf = append(transf, Transferencia{dp, cp, valor})
		devedores[i] = Parte{dp, dv - valor}
		credores[j] = Parte{cp, cv - valor}
		if devedores[i].valor == 0 {
			i++
		}
		if credores[j].valor == 0 {
			j++
		}
	}
	return transf
}

func main() {
	g := &Grupo{}
	g.despesa("ana", 9000, []string{"ana", "bia", "caio"}, Igual{}) // 30,00 cada
	pct, _ := NovoPercentual([]Parte{{"ana", 60}, {"bia", 40}})
	g.despesa("bia", 5000, []string{"ana", "bia"}, pct)
	g.despesa("caio", 3000, []string{"ana", "caio"}, Exata{[]Parte{{"ana", 1000}, {"caio", 2000}}})

	pares := []string{}
	soma := 0
	for _, p := range g.saldo {
		pares = append(pares, fmt.Sprintf("'%s': %d", p.pessoa, p.valor))
		soma += p.valor
	}
	fmt.Println("{" + strings.Join(pares, ", ") + "}")
	ts := []string{}
	for _, t := range g.simplificar() {
		ts = append(ts, fmt.Sprintf("('%s', '%s', %d)", t.devedor, t.credor, t.valor))
	}
	fmt.Println("[" + strings.Join(ts, ", ") + "]")
	if soma != 0 {
		panic("falhou")
	}
}
```

A soma de todos os saldos é sempre zero, uma invariante ótima para testes. A simplificação é gulosa: casa o maior devedor com o maior credor repetidamente. Ela produz no máximo N-1 transferências, e encontrar o mínimo absoluto é um problema difícil (NP-difícil), o que vale mencionar na entrevista.

## Extensões

Despesas em moedas diferentes, quitação parcial (registrar um pagamento como despesa inversa), histórico auditável de despesas em vez de só saldos e grupos aninhados.

## Concorrência

Várias pessoas lançando despesas no mesmo grupo: atualizações dos saldos precisam ser atômicas (lock por grupo) ou derivadas do log de despesas, que é só anexado.

## Na entrevista

- **"Por que Strategy na divisão?"** Novas formas (por ação, por peso) entram sem alterar `Grupo`.
- **"Por que centavos?"** Ponto flutuante acumula erro e a invariante de soma zero quebraria.
