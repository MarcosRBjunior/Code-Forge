O estacionamento é o clássico de entrevistas de LLD. Parece simples, mas esconde decisões de modelagem: tipos de veículo e de vaga, estratégia de alocação, cálculo de tarifa e concorrência na hora de ocupar a última vaga livre.

## Requisitos

- Vários andares, cada um com vagas de tamanhos diferentes (moto, carro, grande).
- Na entrada, o sistema escolhe uma vaga compatível e emite um ticket.
- Na saída, calcula o valor pelo tempo de permanência e libera a vaga.
- Se não houver vaga, recusa a entrada.

## Entidades

| Entidade | Responsabilidade |
| --- | --- |
| `Vaga` | Tem tamanho e guarda o ticket ocupante, se houver. |
| `Andar` | Agrupa vagas e encontra a menor livre compatível. |
| `Ticket` | Placa, vaga, horário de entrada. |
| `Tarifa` | Estratégia que converte duração em valor. |
| `Estacionamento` | Fachada: entrar, sair, consultar lotação. |

Um veículo cabe em vagas do seu tamanho ou maiores. Modele isso com um enum ordenado, e a regra de compatibilidade vira uma comparação.

```python
from enum import IntEnum
from dataclasses import dataclass
from threading import Lock
from typing import List, Optional


class Tamanho(IntEnum):
    MOTO = 1
    CARRO = 2
    GRANDE = 3


class Vaga:
    def __init__(self, id: str, tamanho: Tamanho):
        self.id, self.tamanho, self.ticket = id, tamanho, None

    @property
    def livre(self):
        return self.ticket is None

    def comporta(self, tam: Tamanho):
        return self.livre and self.tamanho >= tam


class Andar:
    def __init__(self, vagas: List[Vaga]):
        self.vagas = vagas

    def buscar(self, tam: Tamanho) -> Optional[Vaga]:
        candidatas = [v for v in self.vagas if v.comporta(tam)]
        return min(candidatas, key=lambda v: v.tamanho, default=None)


@dataclass
class Ticket:
    placa: str
    vaga: Vaga
    entrada: float


class TarifaPorHora:
    def __init__(self, valor_hora: float):
        self.valor_hora = valor_hora

    def calcular(self, segundos: float) -> float:
        horas = max(1, -(-int(segundos) // 3600))  # arredonda para cima
        return horas * self.valor_hora


class Estacionamento:
    def __init__(self, andares: List[Andar], tarifa, relogio):
        self.andares, self.tarifa, self.relogio = andares, tarifa, relogio
        self._lock = Lock()

    def entrar(self, placa: str, tam: Tamanho) -> Ticket:
        with self._lock:
            for andar in self.andares:
                vaga = andar.buscar(tam)
                if vaga:
                    t = Ticket(placa, vaga, self.relogio())
                    vaga.ticket = t
                    return t
        raise RuntimeError("sem vaga")

    def sair(self, ticket: Ticket) -> float:
        with self._lock:
            valor = self.tarifa.calcular(self.relogio() - ticket.entrada)
            ticket.vaga.ticket = None
            return valor


agora = [0.0]
est = Estacionamento(
    [Andar([Vaga("A1", Tamanho.MOTO), Vaga("A2", Tamanho.CARRO)])],
    TarifaPorHora(8.0),
    lambda: agora[0],
)
t1 = est.entrar("ABC-1234", Tamanho.MOTO)
print(t1.vaga.id)            # A1 (menor vaga que comporta)
t2 = est.entrar("XYZ-9999", Tamanho.MOTO)
print(t2.vaga.id)            # A2 (moto em vaga de carro)
agora[0] = 3700
print(est.sair(t1))          # 16.0 (2 horas)
```

```javascript
    this.tamanho = tamanho;
    this.ticket = null;
```

```java
import java.util.*;
import java.util.function.DoubleSupplier;

enum Tamanho {
    MOTO(1), CARRO(2), GRANDE(3);

    final int nivel;

    Tamanho(int nivel) {
        this.nivel = nivel;
    }
}

class Vaga {
    final String id;
    final Tamanho tamanho;
    Ticket ticket = null;

    Vaga(String id, Tamanho tamanho) {
        this.id = id;
        this.tamanho = tamanho;
    }

    boolean livre() {
        return ticket == null;
    }

    boolean comporta(Tamanho tam) {
        return livre() && tamanho.nivel >= tam.nivel;
    }
}

class Andar {
    private final List<Vaga> vagas;

    Andar(List<Vaga> vagas) {
        this.vagas = vagas;
    }

    Vaga buscar(Tamanho tam) {
        Vaga melhor = null;
        for (Vaga v : vagas) {
            if (v.comporta(tam) && (melhor == null || v.tamanho.nivel < melhor.tamanho.nivel)) melhor = v;
        }
        return melhor;
    }
}

class Ticket {
    final String placa;
    final Vaga vaga;
    final double entrada;

    Ticket(String placa, Vaga vaga, double entrada) {
        this.placa = placa;
        this.vaga = vaga;
        this.entrada = entrada;
    }
}

class TarifaPorHora {
    private final double valorHora;

    TarifaPorHora(double valorHora) {
        this.valorHora = valorHora;
    }

    double calcular(double segundos) {
        long horas = Math.max(1, (long) Math.ceil((long) segundos / 3600.0)); // arredonda para cima
        return horas * valorHora;
    }
}

class Estacionamento {
    private final List<Andar> andares;
    private final TarifaPorHora tarifa;
    private final DoubleSupplier relogio;

    Estacionamento(List<Andar> andares, TarifaPorHora tarifa, DoubleSupplier relogio) {
        this.andares = andares;
        this.tarifa = tarifa;
        this.relogio = relogio;
    }

    synchronized Ticket entrar(String placa, Tamanho tam) {
        for (Andar andar : andares) {
            Vaga vaga = andar.buscar(tam);
            if (vaga != null) {
                Ticket t = new Ticket(placa, vaga, relogio.getAsDouble());
                vaga.ticket = t;
                return t;
            }
        }
        throw new IllegalStateException("sem vaga");
    }

    synchronized double sair(Ticket ticket) {
        double valor = tarifa.calcular(relogio.getAsDouble() - ticket.entrada);
        ticket.vaga.ticket = null;
        return valor;
    }
}

public static void main(String[] args) {
    double[] agora = {0.0};
    Estacionamento est = new Estacionamento(
        List.of(new Andar(List.of(new Vaga("A1", Tamanho.MOTO), new Vaga("A2", Tamanho.CARRO)))),
        new TarifaPorHora(8.0),
        () -> agora[0]);
    Ticket t1 = est.entrar("ABC-1234", Tamanho.MOTO);
    System.out.println(t1.vaga.id); // A1 (menor vaga que comporta)
    Ticket t2 = est.entrar("XYZ-9999", Tamanho.MOTO);
    System.out.println(t2.vaga.id); // A2 (moto em vaga de carro)
    agora[0] = 3700;
    System.out.println(est.sair(t1)); // 16.0 (2 horas)
}
```

```cpp
#include <cmath>
#include <functional>
#include <iostream>
#include <memory>
#include <mutex>
#include <stdexcept>
#include <string>
#include <vector>
using namespace std;

enum class Tamanho { MOTO = 1, CARRO = 2, GRANDE = 3 };

struct Ticket;

struct Vaga {
    string id;
    Tamanho tamanho;
    Ticket* ticket = nullptr;

    bool livre() const { return ticket == nullptr; }
    bool comporta(Tamanho tam) const { return livre() && (int)tamanho >= (int)tam; }
};

class Andar {
public:
    vector<Vaga> vagas;

    Vaga* buscar(Tamanho tam) {
        Vaga* melhor = nullptr;
        for (auto& v : vagas) {
            if (v.comporta(tam) && (!melhor || (int)v.tamanho < (int)melhor->tamanho)) melhor = &v;
        }
        return melhor;
    }
};

struct Ticket {
    string placa;
    Vaga* vaga;
    double entrada;
};

class TarifaPorHora {
    double valorHora;

public:
    TarifaPorHora(double valorHora) : valorHora(valorHora) {}

    double calcular(double segundos) const {
        long horas = max(1L, (long)ceil((long)segundos / 3600.0)); // arredonda para cima
        return horas * valorHora;
    }
};

class Estacionamento {
    vector<Andar>& andares;
    TarifaPorHora tarifa;
    function<double()> relogio;
    mutex trava;
    vector<unique_ptr<Ticket>> emitidos;

public:
    Estacionamento(vector<Andar>& andares, TarifaPorHora tarifa, function<double()> relogio)
        : andares(andares), tarifa(tarifa), relogio(relogio) {}

    Ticket* entrar(const string& placa, Tamanho tam) {
        lock_guard<mutex> g(trava);
        for (auto& andar : andares) {
            Vaga* vaga = andar.buscar(tam);
            if (vaga) {
                emitidos.push_back(make_unique<Ticket>(Ticket{placa, vaga, relogio()}));
                Ticket* t = emitidos.back().get();
                vaga->ticket = t;
                return t;
            }
        }
        throw runtime_error("sem vaga");
    }

    double sair(Ticket* ticket) {
        lock_guard<mutex> g(trava);
        double valor = tarifa.calcular(relogio() - ticket->entrada);
        ticket->vaga->ticket = nullptr;
        return valor;
    }
};

int main() {
    double agora = 0.0;
    vector<Andar> andares = {Andar{{Vaga{"A1", Tamanho::MOTO}, Vaga{"A2", Tamanho::CARRO}}}};
    Estacionamento est(andares, TarifaPorHora(8.0), [&] { return agora; });
    Ticket* t1 = est.entrar("ABC-1234", Tamanho::MOTO);
    cout << t1->vaga->id << endl; // A1 (menor vaga que comporta)
    Ticket* t2 = est.entrar("XYZ-9999", Tamanho::MOTO);
    cout << t2->vaga->id << endl; // A2 (moto em vaga de carro)
    agora = 3700;
    cout << est.sair(t1) << endl; // 16.0 (2 horas)
}
```

```csharp
using System.Collections.Generic;
using System.Linq;

enum Tamanho { Moto = 1, Carro = 2, Grande = 3 }

class Vaga
{
    public readonly string Id;
    public readonly Tamanho Tamanho;
    public Ticket Ticket = null;

    public Vaga(string id, Tamanho tamanho)
    {
        Id = id;
        Tamanho = tamanho;
    }

    public bool Livre
    {
        get { return Ticket == null; }

    public bool Comporta(Tamanho tam)
    {
        return Livre && Tamanho >= tam;

{

    public Andar(List<Vaga> vagas)
    {
        this.vagas = vagas;

    public Vaga Buscar(Tamanho tam)
    {
        return vagas.Where(v => v.Comporta(tam)).OrderBy(v => v.Tamanho).FirstOrDefault();
    }
}

{
    public readonly string Placa;
    public readonly Vaga Vaga;
    public readonly double Entrada;

    public Ticket(string placa, Vaga vaga, double entrada)

        Placa = placa;
        Vaga = vaga;
        Entrada = entrada;
    }
}

{
    private readonly double valorHora;

    public TarifaPorHora(double valorHora)
    {
        this.valorHora = valorHora;
    }

    public double Calcular(double segundos)
    {
        long horas = Math.Max(1L, (long)Math.Ceiling((long)segundos / 3600.0)); // arredonda para cima
        return horas * valorHora;
    }
}

    private readonly TarifaPorHora tarifa;
    private readonly Func<double> relogio;
    private readonly object trava = new object();

        this.andares = andares;
        this.tarifa = tarifa;
        this.relogio = relogio;

    {
        lock (trava)
        {
            foreach (var andar in andares)
            {
                var vaga = andar.Buscar(tam);
                if (vaga != null)
                {
                    var t = new Ticket(placa, vaga, relogio());
                    vaga.Ticket = t;
                    return t;
                }
            }
        }
        throw new InvalidOperationException("sem vaga");

    {
        lock (trava)
        {
            double valor = tarifa.Calcular(relogio() - ticket.Entrada);
            ticket.Vaga.Ticket = null;
            return valor;
        }
    }
```

```go
import (
	"errors"
	"fmt"
	"math"
	"sync"
)

type Tamanho int

const (
	Moto   Tamanho = 1
	Carro  Tamanho = 2
	Grande Tamanho = 3
)

type Vaga struct {
	id      string
	tamanho Tamanho
	ticket  *Ticket
}

func (v *Vaga) livre() bool { return v.ticket == nil }

func (v *Vaga) comporta(tam Tamanho) bool { return v.livre() && v.tamanho >= tam }

type Andar struct {
	vagas []*Vaga
}

func (a *Andar) buscar(tam Tamanho) *Vaga {
	var melhor *Vaga
	for _, v := range a.vagas {
		if v.comporta(tam) && (melhor == nil || v.tamanho < melhor.tamanho) {
			melhor = v
		}
	}
	return melhor
}

type Ticket struct {
	placa   string
	vaga    *Vaga
	entrada float64
}

type TarifaPorHora struct {
	valorHora float64
}

func (t TarifaPorHora) calcular(segundos float64) float64 {
	horas := math.Max(1, math.Ceil(math.Trunc(segundos)/3600)) // arredonda para cima
	return horas * t.valorHora
}

type Estacionamento struct {
	andares []*Andar
	tarifa  TarifaPorHora
	relogio func() float64
	trava   sync.Mutex
}

func (e *Estacionamento) entrar(placa string, tam Tamanho) (*Ticket, error) {
	e.trava.Lock()
	defer e.trava.Unlock()
	for _, andar := range e.andares {
		if vaga := andar.buscar(tam); vaga != nil {
			t := &Ticket{placa, vaga, e.relogio()}
			vaga.ticket = t
			return t, nil
		}
	}
	return nil, errors.New("sem vaga")
}

func (e *Estacionamento) sair(ticket *Ticket) float64 {
	e.trava.Lock()
	defer e.trava.Unlock()
	valor := e.tarifa.calcular(e.relogio() - ticket.entrada)
	ticket.vaga.ticket = nil
	return valor
}

func main() {
	agora := 0.0
	est := &Estacionamento{
		andares: []*Andar{{vagas: []*Vaga{{id: "A1", tamanho: Moto}, {id: "A2", tamanho: Carro}}}},
		tarifa:  TarifaPorHora{8.0},
		relogio: func() float64 { return agora },
	}
	t1, _ := est.entrar("ABC-1234", Moto)
	fmt.Println(t1.vaga.id) // A1 (menor vaga que comporta)
	t2, _ := est.entrar("XYZ-9999", Moto)
	fmt.Println(t2.vaga.id) // A2 (moto em vaga de carro)
	agora = 3700
	fmt.Println(est.sair(t1)) // 16.0 (2 horas)
}
```

O relógio é injetado como função, o que permite testar a cobrança sem esperar horas de verdade. Escolher a menor vaga compatível evita que uma moto ocupe a única vaga grande.

## Concorrência e extensões

Dois carros chegando juntos podem disputar a mesma vaga. O lock global resolve, mas serializa todas as entradas. Em escala maior, use um lock por andar ou uma fila de vagas livres por tamanho (cada retirada é atômica). Extensões comuns: tarifa por faixa (Strategy), vagas reservadas ou para deficientes (novo atributo e filtro), pagamento em máquina separada do portão e displays que observam a lotação (Observer).

## Na entrevista

- **"Como trocar a regra de escolha da vaga?"** Extraia uma `EstrategiaAlocacao` (mais próxima da entrada, menor andar, etc.).
- **"E se o ticket for perdido?"** Busque pela placa ou cobre a diária máxima.
- **"Como saber a lotação rápido?"** Mantenha contadores por tamanho atualizados na entrada e saída.
