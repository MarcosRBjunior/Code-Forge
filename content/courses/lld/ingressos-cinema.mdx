Reservar ingressos de cinema é o exemplo clássico de recurso disputado: dois clientes querem a mesma poltrona ao mesmo tempo. O design precisa de um estado transitório de retenção, com prazo, antes da compra confirmada.

## Requisitos

- Cinema com várias salas; cada sala tem poltronas.
- Sessões ligam um filme, uma sala e um horário.
- Cliente escolhe poltronas livres; elas ficam retidas por alguns minutos.
- Confirmar o pagamento transforma a retenção em reserva; sem pagar, a retenção expira.

## Entidades

| Entidade | Responsabilidade |
| --- | --- |
| `Sessao` | Filme, horário e mapa de poltronas. |
| `Poltrona` | Estado: livre, retida ou vendida. |
| `Retencao` | Cliente, poltronas e instante de expiração. |
| `Cinema` | Retém, confirma e libera. |

Estados por poltrona e por sessão: livre, retida (com dono e prazo) e vendida. Uma poltrona retida cujo prazo passou conta como livre.

```python
from dataclasses import dataclass
from enum import Enum
from threading import Lock
from typing import Callable, Dict, List, Optional


class Estado(Enum):
    LIVRE = 1
    RETIDA = 2
    VENDIDA = 3


@dataclass
class Poltrona:
    estado: Estado = Estado.LIVRE
    dono: Optional[str] = None
    expira: float = 0.0


class Sessao:
    def __init__(self, filme: str, poltronas: List[str], relogio: Callable[[], float],
                 ttl: float = 300):
        self.filme, self.relogio, self.ttl = filme, relogio, ttl
        self.mapa: Dict[str, Poltrona] = {p: Poltrona() for p in poltronas}
        self._lock = Lock()

    def _disponivel(self, p: Poltrona, cliente: str) -> bool:
        if p.estado is Estado.LIVRE:
            return True
        return p.estado is Estado.RETIDA and (p.expira <= self.relogio() or p.dono == cliente)

    def reter(self, cliente: str, assentos: List[str]) -> None:
        with self._lock:                       # tudo ou nada
            for a in assentos:
                if a not in self.mapa or not self._disponivel(self.mapa[a], cliente):
                    raise ValueError(f"poltrona {a} indisponível")
            for a in assentos:
                self.mapa[a] = Poltrona(Estado.RETIDA, cliente, self.relogio() + self.ttl)

    def confirmar(self, cliente: str, assentos: List[str]) -> None:
        with self._lock:
            for a in assentos:
                p = self.mapa[a]
                if p.estado is not Estado.RETIDA or p.dono != cliente or p.expira <= self.relogio():
                    raise ValueError(f"retenção de {a} inválida ou expirada")
            for a in assentos:
                self.mapa[a].estado = Estado.VENDIDA

    def livres(self) -> List[str]:
        with self._lock:
            return sorted(a for a, p in self.mapa.items() if self._disponivel(p, ""))
```

```java
import java.util.*;
import java.util.function.DoubleSupplier;

enum Estado { LIVRE, RETIDA, VENDIDA }

class Poltrona {
    Estado estado = Estado.LIVRE;
    String dono = null;
    double expira = 0.0;

    Poltrona() {}

    Poltrona(Estado estado, String dono, double expira) {
        this.estado = estado;
        this.dono = dono;
        this.expira = expira;
    }
}

class Sessao {
    final String filme;
    private final DoubleSupplier relogio;
    private final double ttl;
    private final Map<String, Poltrona> mapa = new LinkedHashMap<>();

    Sessao(String filme, List<String> poltronas, DoubleSupplier relogio, double ttl) {
        this.filme = filme;
        this.relogio = relogio;
        this.ttl = ttl;
        for (String p : poltronas) mapa.put(p, new Poltrona());
    }

    private boolean disponivel(Poltrona p, String cliente) {
        if (p.estado == Estado.LIVRE) {
            return true;
        }
        return p.estado == Estado.RETIDA && (p.expira <= relogio.getAsDouble() || cliente.equals(p.dono));
    }

    synchronized void reter(String cliente, List<String> assentos) { // tudo ou nada
        for (String a : assentos) {
            if (!mapa.containsKey(a) || !disponivel(mapa.get(a), cliente)) {
                throw new IllegalArgumentException("poltrona " + a + " indisponível");
            }
        }
        for (String a : assentos) {
            mapa.put(a, new Poltrona(Estado.RETIDA, cliente, relogio.getAsDouble() + ttl));
        }
    }

    synchronized void confirmar(String cliente, List<String> assentos) {
        for (String a : assentos) {
            Poltrona p = mapa.get(a);
            if (p.estado != Estado.RETIDA || !cliente.equals(p.dono) || p.expira <= relogio.getAsDouble()) {
                throw new IllegalArgumentException("retenção de " + a + " inválida ou expirada");
            }
        }
        for (String a : assentos) {
            mapa.get(a).estado = Estado.VENDIDA;
        }
    }

    synchronized List<String> livres() {
        List<String> r = new ArrayList<>();
        for (Map.Entry<String, Poltrona> e : mapa.entrySet()) if (disponivel(e.getValue(), "")) r.add(e.getKey());
        Collections.sort(r);
        return r;
    }
}

public static void main(String[] args) {
    double[] t = {0.0};
    Sessao s = new Sessao("Filme X", List.of("A1", "A2", "A3"), () -> t[0], 300);
    s.reter("ana", List.of("A1", "A2"));
    try {
        s.reter("bia", List.of("A2", "A3")); // A2 retida: nada é retido para Bia
    } catch (IllegalArgumentException e) {
        System.out.println("erro: " + e.getMessage());
    }
    System.out.println(s.livres()); // ['A3']
    s.confirmar("ana", List.of("A1"));
    t[0] = 301; // retenção de A2 expira
    System.out.println(s.livres()); // ['A2', 'A3']
    s.reter("bia", List.of("A2", "A3"));
}
```

```cpp
#include <algorithm>
#include <functional>
#include <iostream>
#include <map>
#include <mutex>
#include <stdexcept>
#include <string>
#include <vector>
using namespace std;

enum class Estado { LIVRE = 1, RETIDA = 2, VENDIDA = 3 };

struct Poltrona {
    Estado estado = Estado::LIVRE;
    string dono;
    double expira = 0.0;
};

class Sessao {
    string filme;
    function<double()> relogio;
    double ttl;
    map<string, Poltrona> mapa;
    mutex trava;

    bool disponivel(const Poltrona& p, const string& cliente) const {
        if (p.estado == Estado::LIVRE) {
            return true;
        }
        return p.estado == Estado::RETIDA && (p.expira <= relogio() || p.dono == cliente);
    }

public:
    Sessao(string filme, const vector<string>& poltronas, function<double()> relogio, double ttl = 300)
        : filme(filme), relogio(relogio), ttl(ttl) {
        for (auto& p : poltronas) mapa[p] = Poltrona();
    }

    void reter(const string& cliente, const vector<string>& assentos) { // tudo ou nada
        lock_guard<mutex> g(trava);
        for (auto& a : assentos) {
            auto it = mapa.find(a);
            if (it == mapa.end() || !disponivel(it->second, cliente)) {
                throw invalid_argument("poltrona " + a + " indisponível");
            }
        }
        for (auto& a : assentos) {
            mapa[a] = Poltrona{Estado::RETIDA, cliente, relogio() + ttl};
        }
    }

    void confirmar(const string& cliente, const vector<string>& assentos) {
        lock_guard<mutex> g(trava);
        for (auto& a : assentos) {
            const Poltrona& p = mapa.at(a);
            if (p.estado != Estado::RETIDA || p.dono != cliente || p.expira <= relogio()) {
                throw invalid_argument("retenção de " + a + " inválida ou expirada");
            }
        }
        for (auto& a : assentos) {
            mapa.at(a).estado = Estado::VENDIDA;
        }
    }

    vector<string> livres() {
        lock_guard<mutex> g(trava);
        vector<string> r;
        for (auto& [a, p] : mapa) if (disponivel(p, "")) r.push_back(a);
        sort(r.begin(), r.end());
        return r;
    }
};

void mostrar(const vector<string>& v) {
    cout << "[";
    for (size_t i = 0; i < v.size(); i++) cout << (i ? ", '" : "'") << v[i] << "'";
    cout << "]" << endl;
}

int main() {
    double t = 0.0;
    Sessao s("Filme X", {"A1", "A2", "A3"}, [&] { return t; }, 300);
    s.reter("ana", {"A1", "A2"});
    try {
        s.reter("bia", {"A2", "A3"}); // A2 retida: nada é retido para Bia
    } catch (const invalid_argument& e) {
        cout << "erro: " << e.what() << endl;
    }
    mostrar(s.livres()); // ['A3']
    s.confirmar("ana", {"A1"});
    t = 301; // retenção de A2 expira
    mostrar(s.livres()); // ['A2', 'A3']
    s.reter("bia", {"A2", "A3"});
}
```

```csharp
using System;
using System.Collections.Generic;
using System.Linq;

enum Estado { Livre = 1, Retida = 2, Vendida = 3 }

class Poltrona
{
    public Estado Estado = Estado.Livre;
    public string Dono = null;
    public double Expira = 0.0;

    public Poltrona() { }

    public Poltrona(Estado estado, string dono, double expira)
    {
        Estado = estado;
        Dono = dono;
        Expira = expira;
    }
}

class Sessao
{
    public readonly string Filme;
    private readonly Func<double> relogio;
    private readonly double ttl;
    private readonly Dictionary<string, Poltrona> mapa = new Dictionary<string, Poltrona>();
    private readonly object trava = new object();

    public Sessao(string filme, List<string> poltronas, Func<double> relogio, double ttl = 300)
    {
        Filme = filme;
        this.relogio = relogio;
        this.ttl = ttl;
        foreach (var p in poltronas) mapa[p] = new Poltrona();
    }

    private bool Disponivel(Poltrona p, string cliente)
    {
        if (p.Estado == Estado.Livre)
        {
            return true;
        }
        return p.Estado == Estado.Retida && (p.Expira <= relogio() || p.Dono == cliente);
    }

    public void Reter(string cliente, List<string> assentos)
    {
        lock (trava) // tudo ou nada
        {
            foreach (var a in assentos)
            {
                Poltrona p;
                if (!mapa.TryGetValue(a, out p) || !Disponivel(p, cliente))
                {
                    throw new ArgumentException("poltrona " + a + " indisponível");
                }
            }
            foreach (var a in assentos)
            {
                mapa[a] = new Poltrona(Estado.Retida, cliente, relogio() + ttl);
            }
        }
    }

    public void Confirmar(string cliente, List<string> assentos)
    {
        lock (trava)
        {
            foreach (var a in assentos)
            {
                var p = mapa[a];
                if (p.Estado != Estado.Retida || p.Dono != cliente || p.Expira <= relogio())
                {
                    throw new ArgumentException("retenção de " + a + " inválida ou expirada");
                }
            }
            foreach (var a in assentos)
            {
                mapa[a].Estado = Estado.Vendida;
            }
        }
    }

    public List<string> Livres()
    {
        lock (trava)
        {
            return mapa.Where(par => Disponivel(par.Value, "")).Select(par => par.Key).OrderBy(a => a, StringComparer.Ordinal).ToList();
        }
    }
}

static string Mostrar(List<string> v)
{
    return "[" + string.Join(", ", v.Select(x => "'" + x + "'")) + "]";
}

static void Main()
{
    double t = 0.0;
    var s = new Sessao("Filme X", new List<string> { "A1", "A2", "A3" }, () => t, 300);
    s.Reter("ana", new List<string> { "A1", "A2" });
    try
    {
        s.Reter("bia", new List<string> { "A2", "A3" }); // A2 retida: nada é retido para Bia
    }
    catch (ArgumentException e)
    {
        Console.WriteLine("erro: " + e.Message);
    }
    Console.WriteLine(Mostrar(s.Livres())); // ['A3']
    s.Confirmar("ana", new List<string> { "A1" });
    t = 301; // retenção de A2 expira
    Console.WriteLine(Mostrar(s.Livres())); // ['A2', 'A3']
    s.Reter("bia", new List<string> { "A2", "A3" });
}
```

```go
import (
	"fmt"
	"sort"
	"sync"
)

type Estado int

const (
	Livre   Estado = 1
	Retida  Estado = 2
	Vendida Estado = 3
)

type Poltrona struct {
	estado Estado
	dono   string
	expira float64
}

type Sessao struct {
	filme   string
	relogio func() float64
	ttl     float64
	mapa    map[string]*Poltrona
	trava   sync.Mutex
}

func NovaSessao(filme string, poltronas []string, relogio func() float64, ttl float64) *Sessao {
	s := &Sessao{filme: filme, relogio: relogio, ttl: ttl, mapa: map[string]*Poltrona{}}
	for _, p := range poltronas {
		s.mapa[p] = &Poltrona{estado: Livre}
	}
	return s
}

func (s *Sessao) disponivel(p *Poltrona, cliente string) bool {
	if p.estado == Livre {
		return true
	}
	return p.estado == Retida && (p.expira <= s.relogio() || p.dono == cliente)
}

func (s *Sessao) reter(cliente string, assentos []string) error {
	s.trava.Lock() // tudo ou nada
	defer s.trava.Unlock()
	for _, a := range assentos {
		p, existe := s.mapa[a]
		if !existe || !s.disponivel(p, cliente) {
			return fmt.Errorf("poltrona %s indisponível", a)
		}
	}
	for _, a := range assentos {
		s.mapa[a] = &Poltrona{Retida, cliente, s.relogio() + s.ttl}
	}
	return nil
}

func (s *Sessao) confirmar(cliente string, assentos []string) error {
	s.trava.Lock()
	defer s.trava.Unlock()
	for _, a := range assentos {
		p := s.mapa[a]
		if p.estado != Retida || p.dono != cliente || p.expira <= s.relogio() {
			return fmt.Errorf("retenção de %s inválida ou expirada", a)
		}
	}
	for _, a := range assentos {
		s.mapa[a].estado = Vendida
	}
	return nil
}

func (s *Sessao) livres() []string {
	s.trava.Lock()
	defer s.trava.Unlock()
	r := []string{}
	for a, p := range s.mapa {
		if s.disponivel(p, "") {
			r = append(r, a)
		}
	}
	sort.Strings(r)
	return r
}

func main() {
	t := 0.0
	s := NovaSessao("Filme X", []string{"A1", "A2", "A3"}, func() float64 { return t }, 300)
	s.reter("ana", []string{"A1", "A2"})
	if err := s.reter("bia", []string{"A2", "A3"}); err != nil { // A2 retida: nada é retido para Bia
		fmt.Println("erro:", err)
	}
	fmt.Println(s.livres()) // [A3]
	s.confirmar("ana", []string{"A1"})
	t = 301 // retenção de A2 expira
	fmt.Println(s.livres()) // [A2 A3]
	s.reter("bia", []string{"A2", "A3"})
}
```

A retenção é tudo ou nada: se uma poltrona falhar, nenhuma é segurada. A expiração é preguiçosa (avaliada na consulta), o que dispensa um processo de limpeza para a correção; ele só é útil para liberar memória.

## Concorrência

O lock por sessão serializa tentativas sobre o mesmo mapa e evita venda dupla. Em banco de dados, o equivalente é uma restrição de unicidade em (sessão, poltrona) ou atualização condicional que só vence se o estado ainda for livre.

## Extensões

Preços por tipo de poltrona (Strategy), cancelamento com reembolso, lista de espera, e um Observer que atualiza o mapa na tela de todos os clientes quando algo muda.

## Na entrevista

- **"Por que reter antes de pagar?"** O pagamento leva tempo; sem retenção, o cliente pagaria por uma poltrona já vendida.
- **"Como expirar?"** Prazo gravado na retenção, avaliado na leitura, com limpeza opcional em segundo plano.
