O coração de um sistema hoteleiro é a disponibilidade por intervalo de datas. Quarto livre não é um booleano: é a ausência de reservas sobrepostas. Entender isso separa um bom design de um ingênuo.

## Requisitos

- Quartos de tipos diferentes (solteiro, casal, suíte) com diárias distintas.
- Buscar quartos livres para um período.
- Reservar, cancelar e fazer check-in e check-out.
- Calcular o valor da estadia.

## Entidades

| Entidade | Responsabilidade |
| --- | --- |
| `Quarto` | Número, tipo, diária e suas reservas. |
| `Reserva` | Hóspede, quarto, check-in, check-out, status. |
| `Hotel` | Busca, reserva e cancelamento. |

Intervalos são meio abertos: o dia do check-out fica livre para o próximo hóspede. Dois intervalos se sobrepõem quando `a.entrada < b.saida` e `b.entrada < a.saida`.

```python
from dataclasses import dataclass
from enum import Enum
from typing import List


class Status(Enum):
    CONFIRMADA = "confirmada"
    CANCELADA = "cancelada"


@dataclass
class Reserva:
    hospede: str
    entrada: int   # dia (inteiro) de check-in
    saida: int     # dia de check-out (exclusivo)
    status: Status = Status.CONFIRMADA

    def sobrepoe(self, entrada: int, saida: int) -> bool:
        return self.status is Status.CONFIRMADA and self.entrada < saida and entrada < self.saida


class Quarto:
    def __init__(self, numero: int, tipo: str, diaria: float):
        self.numero, self.tipo, self.diaria = numero, tipo, diaria
        self.reservas: List[Reserva] = []

    def livre(self, entrada: int, saida: int) -> bool:
        return not any(r.sobrepoe(entrada, saida) for r in self.reservas)


class Hotel:
    def __init__(self, quartos: List[Quarto]):
        self.quartos = quartos

    def buscar(self, tipo: str, entrada: int, saida: int) -> List[Quarto]:
        if saida <= entrada:
            raise ValueError("período inválido")
        return [q for q in self.quartos if q.tipo == tipo and q.livre(entrada, saida)]

    def reservar(self, hospede: str, tipo: str, entrada: int, saida: int) -> Reserva:
        disponiveis = self.buscar(tipo, entrada, saida)
        if not disponiveis:
            raise RuntimeError("sem disponibilidade")
        quarto = min(disponiveis, key=lambda q: q.numero)
        r = Reserva(hospede, entrada, saida)
        r.quarto = quarto
        quarto.reservas.append(r)
        return r

    def cancelar(self, reserva: Reserva):
        reserva.status = Status.CANCELADA

    def valor(self, reserva: Reserva) -> float:
        return (reserva.saida - reserva.entrada) * reserva.quarto.diaria


h = Hotel([Quarto(101, "casal", 300.0), Quarto(102, "casal", 320.0)])
a = h.reservar("Ana", "casal", 1, 4)

c2 = h.reservar("Duda", "casal", 3, 6)  # entrada no dia do check-out de Caio
print(c2.quarto.numero)
```

```javascript
const Status = Object.freeze({ CONFIRMADA: "confirmada", CANCELADA: "cancelada" });

class Reserva {
  constructor(hospede, entrada, saida) {
    this.hospede = hospede;
    this.entrada = entrada; // dia (inteiro) de check-in
    this.saida = saida;     // dia de check-out (exclusivo)
    this.status = Status.CONFIRMADA;
    this.quarto = null;
  }

  sobrepoe(entrada, saida) {
    return this.status === Status.CONFIRMADA && this.entrada < saida && entrada < this.saida;
  }
}

  constructor(numero, tipo, diaria) {
    this.numero = numero;
    this.tipo = tipo;
    this.diaria = diaria;
    this.reservas = [];
  }

  livre(entrada, saida) {
    return !this.reservas.some((r) => r.sobrepoe(entrada, saida));
  }
}

  constructor(quartos) {
    this.quartos = quartos;
  }

  buscar(tipo, entrada, saida) {
    if (saida <= entrada) {
      throw new RangeError("período inválido");

    return this.quartos.filter((q) => q.tipo === tipo && q.livre(entrada, saida));

    r.quarto = quarto;
    quarto.reservas.push(r);
    return r;
```

```java
import java.util.*;

enum Status { CONFIRMADA, CANCELADA }

class Reserva {
    final String hospede;
    final int entrada; // dia (inteiro) de check-in
    final int saida;   // dia de check-out (exclusivo)
    Status status = Status.CONFIRMADA;
    Quarto quarto;

    Reserva(String hospede, int entrada, int saida) {
        this.hospede = hospede;
        this.entrada = entrada;
        this.saida = saida;
    }

    boolean sobrepoe(int entrada, int saida) {
        return status == Status.CONFIRMADA && this.entrada < saida && entrada < this.saida;
    }
}

class Quarto {
    final int numero;
    final String tipo;
    final double diaria;
    final List<Reserva> reservas = new ArrayList<>();

    Quarto(int numero, String tipo, double diaria) {
        this.numero = numero;
        this.tipo = tipo;
        this.diaria = diaria;
    }

    boolean livre(int entrada, int saida) {
        for (Reserva r : reservas) if (r.sobrepoe(entrada, saida)) return false;
        return true;
    }
}

class Hotel {
    private final List<Quarto> quartos;

    Hotel(List<Quarto> quartos) {
        this.quartos = quartos;
    }

    List<Quarto> buscar(String tipo, int entrada, int saida) {
        if (saida <= entrada) {
            throw new IllegalArgumentException("período inválido");
        }
        List<Quarto> r = new ArrayList<>();
        for (Quarto q : quartos) if (q.tipo.equals(tipo) && q.livre(entrada, saida)) r.add(q);
        return r;
    }

    Reserva reservar(String hospede, String tipo, int entrada, int saida) {
        List<Quarto> disponiveis = buscar(tipo, entrada, saida);
        if (disponiveis.isEmpty()) {
            throw new IllegalStateException("sem disponibilidade");
        }
        Quarto quarto = Collections.min(disponiveis, Comparator.comparingInt(q -> q.numero));
        Reserva r = new Reserva(hospede, entrada, saida);
        r.quarto = quarto;
        quarto.reservas.add(r);
        return r;
    }

    void cancelar(Reserva reserva) {
        reserva.status = Status.CANCELADA;
    }

    double valor(Reserva reserva) {
        return (reserva.saida - reserva.entrada) * reserva.quarto.diaria;
    }
}

public static void main(String[] args) {
    Hotel h = new Hotel(List.of(new Quarto(101, "casal", 300.0), new Quarto(102, "casal", 320.0)));
    Reserva a = h.reservar("Ana", "casal", 1, 4);
    Reserva b = h.reservar("Bia", "casal", 3, 5); // 101 ocupado em 3, vai para 102
    System.out.println(a.quarto.numero + " " + b.quarto.numero + " " + h.valor(a));
    h.cancelar(a);
    Reserva c = h.reservar("Caio", "casal", 1, 3);
    System.out.println(c.quarto.numero); // 101 voltou a ficar livre
    Reserva c2 = h.reservar("Duda", "casal", 3, 6); // entrada no dia do check-out de Caio
    System.out.println(c2.quarto.numero);
}
```

```cpp
#include <algorithm>
#include <iostream>
#include <memory>
#include <stdexcept>
#include <string>
#include <vector>
using namespace std;

enum class Status { CONFIRMADA, CANCELADA };

struct Quarto;

struct Reserva {
    string hospede;
    int entrada; // dia (inteiro) de check-in
    int saida;   // dia de check-out (exclusivo)
    Status status = Status::CONFIRMADA;
    Quarto* quarto = nullptr;

    bool sobrepoe(int e, int s) const { return status == Status::CONFIRMADA && entrada < s && e < saida; }
};

struct Quarto {
    int numero;
    string tipo;
    double diaria;
    vector<Reserva*> reservas;

    bool livre(int entrada, int saida) const {
        return none_of(reservas.begin(), reservas.end(), [&](Reserva* r) { return r->sobrepoe(entrada, saida); });
    }
};

class Hotel {
    vector<Quarto>& quartos;
    vector<unique_ptr<Reserva>> todas;

public:
    Hotel(vector<Quarto>& quartos) : quartos(quartos) {}

    vector<Quarto*> buscar(const string& tipo, int entrada, int saida) {
        if (saida <= entrada) {
            throw invalid_argument("período inválido");
        }
        vector<Quarto*> r;
        for (auto& q : quartos) if (q.tipo == tipo && q.livre(entrada, saida)) r.push_back(&q);
        return r;
    }

    Reserva* reservar(const string& hospede, const string& tipo, int entrada, int saida) {
        auto disponiveis = buscar(tipo, entrada, saida);
        if (disponiveis.empty()) {
            throw runtime_error("sem disponibilidade");
        }
        Quarto* quarto = *min_element(disponiveis.begin(), disponiveis.end(),
                                      [](Quarto* a, Quarto* b) { return a->numero < b->numero; });
        todas.push_back(make_unique<Reserva>(Reserva{hospede, entrada, saida}));
        Reserva* r = todas.back().get();
        r->quarto = quarto;
        quarto->reservas.push_back(r);
        return r;
    }

    void cancelar(Reserva* reserva) { reserva->status = Status::CANCELADA; }

    double valor(const Reserva* reserva) const { return (reserva->saida - reserva->entrada) * reserva->quarto->diaria; }
};

int main() {
    vector<Quarto> quartos = {{101, "casal", 300.0, {}}, {102, "casal", 320.0, {}}};
    Hotel h(quartos);
    Reserva* a = h.reservar("Ana", "casal", 1, 4);
    Reserva* b = h.reservar("Bia", "casal", 3, 5); // 101 ocupado em 3, vai para 102
    cout << a->quarto->numero << " " << b->quarto->numero << " " << h.valor(a) << endl;
    h.cancelar(a);
    Reserva* c = h.reservar("Caio", "casal", 1, 3);
    cout << c->quarto->numero << endl; // 101 voltou a ficar livre
    Reserva* c2 = h.reservar("Duda", "casal", 3, 6); // entrada no dia do check-out de Caio
    cout << c2->quarto->numero << endl;
}
```

```csharp
using System.Linq;

enum Status { Confirmada, Cancelada }

class Reserva
{
    public readonly string Hospede;
    public readonly int Entrada; // dia (inteiro) de check-in
    public readonly int Saida;   // dia de check-out (exclusivo)

    {
        Hospede = hospede;
        Entrada = entrada;
        Saida = saida;

    }
}

{
    public readonly int Numero;
    public readonly string Tipo;
    public readonly double Diaria;
    public readonly List<Reserva> Reservas = new List<Reserva>();

    public Quarto(int numero, string tipo, double diaria)
    {
        Numero = numero;
        Tipo = tipo;
        Diaria = diaria;
    }

    public bool Livre(int entrada, int saida)
    {
        return !Reservas.Any(r => r.Sobrepoe(entrada, saida));
    }
}

class Hotel
{
    private readonly List<Quarto> quartos;

    public Hotel(List<Quarto> quartos)
    {
        this.quartos = quartos;
    }

    public List<Quarto> Buscar(string tipo, int entrada, int saida)
    {
        if (saida <= entrada)
        {
            throw new ArgumentException("período inválido");
        }
        return quartos.Where(q => q.Tipo == tipo && q.Livre(entrada, saida)).ToList();
    }

    {
        var disponiveis = Buscar(tipo, entrada, saida);
        if (disponiveis.Count == 0)
        {
            throw new InvalidOperationException("sem disponibilidade");
        }
        var quarto = disponiveis.OrderBy(q => q.Numero).First();
        var r = new Reserva(hospede, entrada, saida);
        r.Quarto = quarto;
        quarto.Reservas.Add(r);
        return r;

        reserva.Status = Status.Cancelada;
    }

    public double Valor(Reserva reserva)
    {
        return (reserva.Saida - reserva.Entrada) * reserva.Quarto.Diaria;
    }
```

```go
import (
	"errors"
	"fmt"
)

type Status int

const (
	Confirmada Status = iota
	Cancelada
)

type Reserva struct {
	hospede string
	entrada int // dia (inteiro) de check-in
	saida   int // dia de check-out (exclusivo)
	status  Status
	quarto  *Quarto
}

func (r *Reserva) sobrepoe(entrada, saida int) bool {
	return r.status == Confirmada && r.entrada < saida && entrada < r.saida
}

type Quarto struct {
	numero   int
	tipo     string
	diaria   float64
	reservas []*Reserva
}

func (q *Quarto) livre(entrada, saida int) bool {
	for _, r := range q.reservas {
		if r.sobrepoe(entrada, saida) {
			return false
		}
	}
	return true
}

type Hotel struct {
	quartos []*Quarto
}

func (h *Hotel) buscar(tipo string, entrada, saida int) ([]*Quarto, error) {
	if saida <= entrada {
		return nil, errors.New("período inválido")
	}
	r := []*Quarto{}
	for _, q := range h.quartos {
		if q.tipo == tipo && q.livre(entrada, saida) {
			r = append(r, q)
		}
	}
	return r, nil
}

func (h *Hotel) reservar(hospede, tipo string, entrada, saida int) (*Reserva, error) {
	disponiveis, err := h.buscar(tipo, entrada, saida)
	if err != nil {
		return nil, err
	}
	if len(disponiveis) == 0 {
		return nil, errors.New("sem disponibilidade")
	}
	quarto := disponiveis[0]
	for _, q := range disponiveis {
		if q.numero < quarto.numero {
			quarto = q
		}
	}
	r := &Reserva{hospede: hospede, entrada: entrada, saida: saida, status: Confirmada, quarto: quarto}
	quarto.reservas = append(quarto.reservas, r)
	return r, nil
}

func (h *Hotel) cancelar(r *Reserva) { r.status = Cancelada }

func (h *Hotel) valor(r *Reserva) float64 {
	return float64(r.saida-r.entrada) * r.quarto.diaria
}

func main() {
	h := &Hotel{[]*Quarto{{numero: 101, tipo: "casal", diaria: 300.0}, {numero: 102, tipo: "casal", diaria: 320.0}}}
	a, _ := h.reservar("Ana", "casal", 1, 4)

	h.cancelar(a)
	c, _ := h.reservar("Caio", "casal", 1, 3)

	c2, _ := h.reservar("Duda", "casal", 3, 6) // entrada no dia do check-out de Caio
```

## Concorrência

Dois recepcionistas reservando o último quarto no mesmo período é o caso crítico. A verificação de sobreposição e a inserção devem ser atômicas: lock por quarto, ou no banco uma restrição de exclusão por intervalo. Sem isso você terá overbooking.

## Extensões

Preço dinâmico por temporada (Strategy de tarifa), múltiplos quartos por reserva, serviços extras na conta (Decorator no valor), políticas de cancelamento com multa e overbooking intencional controlado, que alguns hotéis usam de propósito.
