Na rodada de **machine coding** você recebe um problema (por exemplo, um sistema de estacionamento ou um jogo) e tem de 60 a 120 minutos para entregar um programa **funcionando**, executável, com código limpo. Ao contrário do quadro branco, aqui o entrevistador roda seu código, lê a estrutura e muitas vezes pede uma alteração na hora para ver se o design aguenta.

## O que é avaliado

| Critério | O que olham |
| --- | --- |
| Funcionamento | O fluxo principal roda e os casos de borda não quebram |
| Modelagem | Classes com responsabilidades claras, sem classe-deus |
| Extensibilidade | Adicionar um tipo novo exige pouco código novo e pouca edição |
| Legibilidade | Nomes bons, métodos curtos, sem código morto |
| Testes / demo | Um `main` ou testes que mostram o sistema funcionando |

## Estratégia de tempo

1. **10 min**: ler com calma, listar requisitos e decidir o que fica de fora. Anote suposições.
2. **10 min**: desenhar classes e interfaces no papel.
3. **60 a 70 min**: implementar do centro para as bordas. Primeiro o fluxo principal de ponta a ponta, depois os extras.
4. **15 min**: rodar, corrigir, escrever uma demonstração e limpar o código.

Prefira um fluxo simples **completo** a muitas funcionalidades pela metade. Se sobrar tempo, adicione extras na ordem de valor.

## Esqueleto que rende

Mantenha camadas simples: modelos de dados, serviço com as regras, e um ponto de entrada. Armazene tudo em memória (dicionários e listas); ninguém espera banco de dados, a menos que seja pedido.

```python
from dataclasses import dataclass, field


@dataclass
class Tarefa:
    id: int
    titulo: str
    feita: bool = False


class GerenciadorTarefas:
    def __init__(self) -> None:
        self._tarefas: dict[int, Tarefa] = {}
        self._proximo_id = 1

    def criar(self, titulo: str) -> Tarefa:
        if not titulo.strip():
            raise ValueError("título obrigatório")
        t = Tarefa(self._proximo_id, titulo)
        self._tarefas[t.id] = t
        self._proximo_id += 1
        return t

    def concluir(self, id_: int) -> None:
        if id_ not in self._tarefas:
            raise KeyError(f"tarefa {id_} não existe")
        self._tarefas[id_].feita = True

    def pendentes(self) -> list[Tarefa]:
        return [t for t in self._tarefas.values() if not t.feita]


def main() -> None:
    g = GerenciadorTarefas()
    a = g.criar("estudar")
    g.criar("treinar")
    g.concluir(a.id)
    assert [t.titulo for t in g.pendentes()] == ["treinar"]
    try:
        g.concluir(99)
    except KeyError as erro:
        print("erro tratado:", erro)
    print("demo ok")


main()
```

```java
import java.util.*;
import java.util.stream.Collectors;

class Tarefa {
    final int id;
    final String titulo;
    boolean feita = false;

    Tarefa(int id, String titulo) {
        this.id = id;
        this.titulo = titulo;
    }
}

class GerenciadorTarefas {
    private final Map<Integer, Tarefa> tarefas = new LinkedHashMap<>();
    private int proximoId = 1;

    Tarefa criar(String titulo) {
        if (titulo.isBlank()) throw new IllegalArgumentException("título obrigatório");
        Tarefa t = new Tarefa(proximoId, titulo);
        tarefas.put(t.id, t);
        proximoId += 1;
        return t;
    }

    void concluir(int id) {
        if (!tarefas.containsKey(id)) throw new NoSuchElementException("tarefa " + id + " não existe");
        tarefas.get(id).feita = true;
    }

    List<Tarefa> pendentes() {
        return tarefas.values().stream().filter(t -> !t.feita).collect(Collectors.toList());
    }
}

public static void main(String[] args) {
    GerenciadorTarefas g = new GerenciadorTarefas();
    Tarefa a = g.criar("estudar");
    g.criar("treinar");
    g.concluir(a.id);
    List<String> titulos = g.pendentes().stream().map(t -> t.titulo).collect(Collectors.toList());
    if (!titulos.equals(List.of("treinar"))) throw new AssertionError();
    try {
        g.concluir(99);
    } catch (NoSuchElementException erro) {
        System.out.println("erro tratado: " + erro.getMessage());
    }
    System.out.println("demo ok");
}
```

```cpp
#include <cassert>
#include <iostream>
#include <map>
#include <stdexcept>
#include <string>
#include <vector>
using namespace std;

struct Tarefa {
    int id;
    string titulo;
    bool feita = false;
};

class GerenciadorTarefas {
    map<int, Tarefa> tarefas;
    int proximoId = 1;

public:
    Tarefa criar(const string& titulo) {
        if (titulo.find_first_not_of(" \t\n") == string::npos) throw invalid_argument("título obrigatório");
        Tarefa t{proximoId, titulo};
        tarefas[t.id] = t;
        proximoId += 1;
        return t;
    }

    void concluir(int id) {
        auto it = tarefas.find(id);
        if (it == tarefas.end()) throw out_of_range("tarefa " + to_string(id) + " não existe");
        it->second.feita = true;
    }

    vector<Tarefa> pendentes() const {
        vector<Tarefa> saida;
        for (const auto& [id, t] : tarefas) if (!t.feita) saida.push_back(t);
        return saida;
    }
};

void principal() {
    GerenciadorTarefas g;
    Tarefa a = g.criar("estudar");
    g.criar("treinar");
    g.concluir(a.id);
    auto pend = g.pendentes();
    assert(pend.size() == 1 && pend[0].titulo == "treinar");
    try {
        g.concluir(99);
    } catch (const out_of_range& erro) {
        cout << "erro tratado: " << erro.what() << endl;
    }
    cout << "demo ok" << endl;
}

int main() {
    principal();
}
```

```csharp
using System;

class Tarefa
{
    public readonly int Id;
    public readonly string Titulo;
    public bool Feita = false;

    public Tarefa(int id, string titulo)
    {
        Id = id;
        Titulo = titulo;

class GerenciadorTarefas

    private readonly Dictionary<int, Tarefa> tarefas = new Dictionary<int, Tarefa>();
    private int proximoId = 1;

    {
        if (string.IsNullOrWhiteSpace(titulo)) throw new ArgumentException("título obrigatório");
        var t = new Tarefa(proximoId, titulo);
        tarefas[t.Id] = t;
        proximoId += 1;
        return t;

    {
        if (!tarefas.ContainsKey(id)) throw new KeyNotFoundException("tarefa " + id + " não existe");
        tarefas[id].Feita = true;
    }

    public List<Tarefa> Pendentes()
    {
        return tarefas.Values.Where(t => !t.Feita).ToList();
```

```go
import (
	"fmt"
	"sort"
	"strings"
)

type Tarefa struct {
	id     int
	titulo string
	feita  bool
}

type GerenciadorTarefas struct {
	tarefas   map[int]*Tarefa
	proximoID int
}

func novoGerenciador() *GerenciadorTarefas {
	return &GerenciadorTarefas{tarefas: map[int]*Tarefa{}, proximoID: 1}
}

func (g *GerenciadorTarefas) criar(titulo string) (*Tarefa, error) {
	if strings.TrimSpace(titulo) == "" {
		return nil, fmt.Errorf("título obrigatório")
	}
	t := &Tarefa{id: g.proximoID, titulo: titulo}
	g.tarefas[t.id] = t
	g.proximoID++
	return t, nil
}

func (g *GerenciadorTarefas) concluir(id int) error {
	t, ok := g.tarefas[id]
	if !ok {
		return fmt.Errorf("tarefa %d não existe", id)
	}
	t.feita = true
	return nil
}

func (g *GerenciadorTarefas) pendentes() []*Tarefa {
	saida := []*Tarefa{}
	for _, t := range g.tarefas {
		if !t.feita {
			saida = append(saida, t)
		}
	}
	sort.Slice(saida, func(i, j int) bool { return saida[i].id < saida[j].id })
	return saida
}

func main() {
	g := novoGerenciador()
	a, _ := g.criar("estudar")
	g.criar("treinar")
	g.concluir(a.id)
	pend := g.pendentes()
	if len(pend) != 1 || pend[0].titulo != "treinar" {
		panic("falhou")
	}
	if erro := g.concluir(99); erro != nil {
		fmt.Println("erro tratado:", erro)
	}
	fmt.Println("demo ok")
}
```

<Callout type="warning" title="Cuidado com o excesso">
Não crie cinco interfaces e três fábricas antes de ter o fluxo principal rodando. Abstraia onde o enunciado já indica variação (tipos de veículo, formas de pagamento) e nada além disso.
</Callout>

## Dicas práticas

- Faça commits (ou salvamentos) frequentes de versões que funcionam.
- Mantenha uma lista de pendências no topo do arquivo para mostrar o que ficaria para depois.
- Use enums para estados e tipos fixos, no lugar de strings soltas.
- Trate entradas inválidas com exceções claras, não com `print` silencioso.
- Evite singleton e estado global; passe dependências pelo construtor.

## Na entrevista

O entrevistador costuma pedir uma mudança depois ("agora adicione um novo tipo de desconto"). Se o seu código tem uma interface no ponto certo, a mudança é adicionar uma classe. Esse é o momento em que você demonstra o valor do design. Fale também sobre o que faria com mais tempo: persistência, concorrência, testes unitários.
