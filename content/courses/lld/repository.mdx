Quando a regra de negócio chama SQL diretamente, ela passa a depender do banco, do formato das tabelas e do driver. Trocar de banco, ou apenas testar sem um banco, vira dor de cabeça. O padrão **Repository** coloca entre o domínio e o armazenamento uma interface com cara de **coleção de objetos**: o domínio pede `salvar`, `buscar`, `listar`, sem saber onde ou como os dados vivem.

## A ideia

```text
Serviço de domínio  --->  [ Repositório (interface) ]
                                  ^            ^
                    RepositorioMemoria    RepositorioSQL
```

O serviço depende apenas da abstração (Dependency Inversion). As implementações concretas ficam nas bordas do sistema.

```python
from abc import ABC, abstractmethod
from dataclasses import dataclass


@dataclass
class Usuario:
    id: int
    nome: str
    ativo: bool = True


class RepositorioUsuarios(ABC):
    @abstractmethod
    def salvar(self, usuario: Usuario) -> None: ...

    @abstractmethod
    def buscar(self, id_: int) -> Usuario | None: ...

    @abstractmethod
    def listar_ativos(self) -> list[Usuario]: ...


class RepositorioUsuariosMemoria(RepositorioUsuarios):
    def __init__(self) -> None:
        self._dados: dict[int, Usuario] = {}

    def salvar(self, usuario: Usuario) -> None:
        self._dados[usuario.id] = usuario

    def buscar(self, id_: int) -> Usuario | None:
        return self._dados.get(id_)

    def listar_ativos(self) -> list[Usuario]:
        return [u for u in self._dados.values() if u.ativo]


class ServicoUsuarios:
    def __init__(self, repo: RepositorioUsuarios) -> None:
        self.repo = repo

    def cadastrar(self, id_: int, nome: str) -> Usuario:
        if self.repo.buscar(id_) is not None:
            raise ValueError("id já existe")
        usuario = Usuario(id_, nome)
        self.repo.salvar(usuario)
        return usuario

    def desativar(self, id_: int) -> None:
        usuario = self.repo.buscar(id_)
        if usuario is None:
            raise KeyError(id_)
        usuario.ativo = False
        self.repo.salvar(usuario)


servico = ServicoUsuarios(RepositorioUsuariosMemoria())
servico.cadastrar(1, "Ana")
servico.cadastrar(2, "Bia")
servico.desativar(2)
assert [u.nome for u in servico.repo.listar_ativos()] == ["Ana"]
print("repositório ok")
```

```javascript
class Usuario {
  constructor(id, nome, ativo = true) {
    this.id = id;
    this.nome = nome;
    this.ativo = ativo;
  }
}

// Contrato do repositório (por convenção): salvar(usuario), buscar(id), listarAtivos()
class RepositorioUsuariosMemoria {
  #dados = new Map();

  salvar(usuario) {
    this.#dados.set(usuario.id, usuario);
  }

  buscar(id) {
    return this.#dados.get(id) ?? null;
  }

  listarAtivos() {
    return [...this.#dados.values()].filter((u) => u.ativo);
  }
}

class ServicoUsuarios {
  constructor(repo) {
    this.repo = repo;
  }

  cadastrar(id, nome) {
    if (this.repo.buscar(id) !== null) {
      throw new Error("id já existe");
    }
    const usuario = new Usuario(id, nome);
    this.repo.salvar(usuario);
    return usuario;
  }

  desativar(id) {
    const usuario = this.repo.buscar(id);
    if (usuario === null) {
      throw new RangeError(String(id));
    }
    usuario.ativo = false;
    this.repo.salvar(usuario);
  }
}

servico.cadastrar(2, "Bia");
```

```java
import java.util.*;

class Usuario {
    final int id;
    final String nome;
    boolean ativo = true;

    Usuario(int id, String nome) {
        this.id = id;
        this.nome = nome;
    }
}

interface RepositorioUsuarios {
    void salvar(Usuario usuario);
    Usuario buscar(int id); // null se não existir
    List<Usuario> listarAtivos();
}

class RepositorioUsuariosMemoria implements RepositorioUsuarios {
    private final Map<Integer, Usuario> dados = new LinkedHashMap<>();

    public void salvar(Usuario usuario) {
        dados.put(usuario.id, usuario);
    }

    public Usuario buscar(int id) {
        return dados.get(id);
    }

    public List<Usuario> listarAtivos() {
        List<Usuario> r = new ArrayList<>();
        for (Usuario u : dados.values()) if (u.ativo) r.add(u);
        return r;
    }
}

class ServicoUsuarios {
    final RepositorioUsuarios repo;

    ServicoUsuarios(RepositorioUsuarios repo) {
        this.repo = repo;
    }

    Usuario cadastrar(int id, String nome) {
        if (repo.buscar(id) != null) {
            throw new IllegalArgumentException("id já existe");
        }
        Usuario usuario = new Usuario(id, nome);
        repo.salvar(usuario);
        return usuario;
    }

    void desativar(int id) {
        Usuario usuario = repo.buscar(id);
        if (usuario == null) {
            throw new NoSuchElementException(String.valueOf(id));
        }
        usuario.ativo = false;
        repo.salvar(usuario);
    }
}

public static void main(String[] args) {
    ServicoUsuarios servico = new ServicoUsuarios(new RepositorioUsuariosMemoria());
    servico.cadastrar(1, "Ana");
    servico.cadastrar(2, "Bia");
    servico.desativar(2);
    List<Usuario> ativos = servico.repo.listarAtivos();
    if (ativos.size() != 1 || !ativos.get(0).nome.equals("Ana")) throw new AssertionError("falhou");
    System.out.println("repositório ok");
}
```

```cpp
#include <iostream>
#include <map>
#include <memory>
#include <stdexcept>
#include <string>
#include <vector>
using namespace std;

struct Usuario {
    int id;
    string nome;
    bool ativo = true;
};

class RepositorioUsuarios {
public:
    virtual ~RepositorioUsuarios() = default;
    virtual void salvar(const Usuario& usuario) = 0;
    virtual Usuario* buscar(int id) = 0; // nullptr se não existir
    virtual vector<Usuario> listarAtivos() const = 0;
};

class RepositorioUsuariosMemoria : public RepositorioUsuarios {
    map<int, Usuario> dados;

public:
    void salvar(const Usuario& usuario) override { dados[usuario.id] = usuario; }

    Usuario* buscar(int id) override {
        auto it = dados.find(id);
        return it == dados.end() ? nullptr : &it->second;
    }

    vector<Usuario> listarAtivos() const override {
        vector<Usuario> r;
        for (auto& [id, u] : dados) if (u.ativo) r.push_back(u);
        return r;
    }
};

class ServicoUsuarios {
public:
    RepositorioUsuarios& repo;

    ServicoUsuarios(RepositorioUsuarios& repo) : repo(repo) {}

    Usuario cadastrar(int id, const string& nome) {
        if (repo.buscar(id) != nullptr) {
            throw invalid_argument("id já existe");
        }
        Usuario usuario{id, nome};
        repo.salvar(usuario);
        return usuario;
    }

    void desativar(int id) {
        Usuario* usuario = repo.buscar(id);
        if (usuario == nullptr) {
            throw out_of_range(to_string(id));
        }
        usuario->ativo = false;
        repo.salvar(*usuario);
    }
};

int main() {
    RepositorioUsuariosMemoria memoria;
    ServicoUsuarios servico(memoria);
    servico.cadastrar(1, "Ana");
    servico.cadastrar(2, "Bia");
    servico.desativar(2);
    auto ativos = servico.repo.listarAtivos();
    if (ativos.size() != 1 || ativos[0].nome != "Ana") throw runtime_error("falhou");
    cout << "repositório ok" << endl;
}
```

```csharp
using System;
using System.Collections.Generic;
using System.Linq;

class Usuario
{
    public readonly int Id;
    public readonly string Nome;
    public bool Ativo = true;

    public Usuario(int id, string nome)
    {
        Id = id;
        Nome = nome;
    }
}

interface IRepositorioUsuarios
{
    void Salvar(Usuario usuario);
    Usuario Buscar(int id); // null se não existir
    List<Usuario> ListarAtivos();
}

class RepositorioUsuariosMemoria : IRepositorioUsuarios
{
    private readonly Dictionary<int, Usuario> dados = new Dictionary<int, Usuario>();

    public void Salvar(Usuario usuario)
    {
        dados[usuario.Id] = usuario;
    }

    public Usuario Buscar(int id)
    {
        Usuario u;
        return dados.TryGetValue(id, out u) ? u : null;
    }

    public List<Usuario> ListarAtivos()
    {
        return dados.Values.Where(u => u.Ativo).ToList();
    }
}

class ServicoUsuarios
{
    public readonly IRepositorioUsuarios Repo;

    public ServicoUsuarios(IRepositorioUsuarios repo)
    {
        Repo = repo;
    }

    public Usuario Cadastrar(int id, string nome)
    {
        if (Repo.Buscar(id) != null)
        {
            throw new ArgumentException("id já existe");
        }
        var usuario = new Usuario(id, nome);
        Repo.Salvar(usuario);
        return usuario;
    }

    public void Desativar(int id)
    {
        var usuario = Repo.Buscar(id);
        if (usuario == null)
        {
            throw new KeyNotFoundException(id.ToString());
        }
        usuario.Ativo = false;
        Repo.Salvar(usuario);
    }
}

static void Main()
{
    var servico = new ServicoUsuarios(new RepositorioUsuariosMemoria());
    servico.Cadastrar(1, "Ana");
    servico.Cadastrar(2, "Bia");
    servico.Desativar(2);
    var ativos = servico.Repo.ListarAtivos();
    if (ativos.Count != 1 || ativos[0].Nome != "Ana") throw new Exception("falhou");
    Console.WriteLine("repositório ok");
}
```

```go
import (
	"errors"
	"fmt"
)

type Usuario struct {
	id    int
	nome  string
	ativo bool
}

type RepositorioUsuarios interface {
	salvar(usuario *Usuario)
	buscar(id int) *Usuario // nil se não existir
	listarAtivos() []*Usuario
}

type RepositorioUsuariosMemoria struct {
	dados map[int]*Usuario
	ordem []int
}

func NovoRepositorioMemoria() *RepositorioUsuariosMemoria {
	return &RepositorioUsuariosMemoria{dados: map[int]*Usuario{}}
}

func (r *RepositorioUsuariosMemoria) salvar(u *Usuario) {
	if _, existe := r.dados[u.id]; !existe {
		r.ordem = append(r.ordem, u.id)
	}
	r.dados[u.id] = u
}

func (r *RepositorioUsuariosMemoria) buscar(id int) *Usuario { return r.dados[id] }

func (r *RepositorioUsuariosMemoria) listarAtivos() []*Usuario {
	ativos := []*Usuario{}
	for _, id := range r.ordem {
		if r.dados[id].ativo {
			ativos = append(ativos, r.dados[id])
		}
	}
	return ativos
}

type ServicoUsuarios struct {
	repo RepositorioUsuarios
}

func (s *ServicoUsuarios) cadastrar(id int, nome string) (*Usuario, error) {
	if s.repo.buscar(id) != nil {
		return nil, errors.New("id já existe")
	}
	usuario := &Usuario{id, nome, true}
	s.repo.salvar(usuario)
	return usuario, nil
}

func (s *ServicoUsuarios) desativar(id int) error {
	usuario := s.repo.buscar(id)
	if usuario == nil {
		return fmt.Errorf("usuário %d não encontrado", id)
	}
	usuario.ativo = false
	s.repo.salvar(usuario)
	return nil
}

func main() {
	servico := &ServicoUsuarios{NovoRepositorioMemoria()}
	servico.cadastrar(1, "Ana")
	servico.cadastrar(2, "Bia")
	servico.desativar(2)
	ativos := servico.repo.listarAtivos()
	if len(ativos) != 1 || ativos[0].nome != "Ana" {
		panic("falhou")
	}
	fmt.Println("repositório ok")
}
```

Uma implementação `RepositorioUsuariosSQL` teria os mesmos métodos, usando uma conexão de banco por dentro. O `ServicoUsuarios` não mudaria uma linha.

## Benefícios

- **Testabilidade**: nos testes, a versão em memória substitui o banco.
- **Troca de tecnologia**: de SQL para documento ou para API externa, mexendo só na implementação.
- **Consultas com nome de domínio**: `listar_ativos()` diz mais do que uma query espalhada.

## Armadilhas

- **Repositório genérico demais**: um `buscar(filtro_sql)` vaza a tecnologia para o domínio. Prefira métodos com nome de negócio.
- **Um repositório por tabela**: o repositório é por **agregado** (por exemplo, `Pedido` junto com seus itens), não por tabela.
- **Lógica de negócio dentro dele**: ele só guarda e recupera. Regras ficam no domínio.
- **Confundir com DAO**: o DAO é mais próximo do banco (uma tabela, CRUD); o Repository fala a língua do domínio e abstrai uma coleção.

<Callout type="tip" title="Unidade de trabalho">
Quando várias mudanças devem ser salvas atomicamente, o Repository costuma vir acompanhado do padrão Unit of Work, que agrupa as operações numa transação.
</Callout>

## Na entrevista
