Pub/sub desacopla quem produz mensagens de quem as consome. Em LLD, é o padrão Observer promovido a sistema: tópicos, assinantes, entrega e o que acontece quando um assinante é lento ou falha.

## Requisitos

- Publicadores enviam mensagens para tópicos.
- Assinantes se inscrevem e cancelam inscrição em tópicos.
- Cada assinante recebe todas as mensagens publicadas após assinar, em ordem.
- Um assinante lento ou com erro não pode travar os outros.

## Entidades

| Entidade | Responsabilidade |
| --- | --- |
| `Mensagem` | Tópico, conteúdo e número de sequência. |
| `Topico` | Lista de assinantes e histórico de sequência. |
| `Assinante` | Interface `receber(msg)`; mantém sua própria fila. |
| `Broker` | Cria tópicos, publica e gerencia inscrições. |

A decisão-chave é dar uma fila a cada assinante. O publicador apenas enfileira (rápido) e cada assinante consome no seu ritmo.

```python
import queue
import threading
from dataclasses import dataclass
from typing import Callable, Dict, List


@dataclass(frozen=True)
class Mensagem:
    topico: str
    conteudo: str
    seq: int


class Assinante:
    def __init__(self, nome: str, tratador: Callable[[Mensagem], None]):
        self.nome, self.tratador = nome, tratador
        self.fila: "queue.Queue" = queue.Queue()
        self.falhas: List[Mensagem] = []
        self._thread = threading.Thread(target=self._laco, daemon=True)
        self._thread.start()

    def _laco(self):
        while True:
            msg = self.fila.get()
            if msg is None:
                self.fila.task_done()
                return
            try:
                self.tratador(msg)
            except Exception:
                self.falhas.append(msg)   # isolada: não afeta os demais
            self.fila.task_done()

    def aguardar(self):
        self.fila.join()

    def parar(self):
        self.fila.put(None)
        self._thread.join()


class Broker:
    def __init__(self):
        self.topicos: Dict[str, List[Assinante]] = {}
        self.seq: Dict[str, int] = {}
        self._lock = threading.Lock()

    def assinar(self, topico: str, assinante: Assinante):
        with self._lock:
            self.topicos.setdefault(topico, []).append(assinante)
            self.seq.setdefault(topico, 0)

    def cancelar(self, topico: str, assinante: Assinante):
        with self._lock:
            if assinante in self.topicos.get(topico, []):
                self.topicos[topico].remove(assinante)

    def publicar(self, topico: str, conteudo: str) -> int:
        with self._lock:
            self.seq[topico] = self.seq.get(topico, 0) + 1
            msg = Mensagem(topico, conteudo, self.seq[topico])
            destinos = list(self.topicos.get(topico, []))
        for a in destinos:
            a.fila.put(msg)
        return len(destinos)

def bom(m):
    recebidas.append(("bom", m.seq, m.conteudo))

def ruim(m):
    raise RuntimeError("falhei")
```

```java
import java.util.*;
import java.util.concurrent.*;
import java.util.function.*;

class Mensagem {
    final String topico, conteudo;
    final int seq;

    Mensagem(String topico, String conteudo, int seq) {
        this.topico = topico;
        this.conteudo = conteudo;
        this.seq = seq;
    }
}

class Assinante {
    private static final Mensagem PARAR = new Mensagem("", "", 0); // sentinela (o None do Python)

    final String nome;
    private final Consumer<Mensagem> tratador;
    private final BlockingQueue<Mensagem> fila = new LinkedBlockingQueue<>();
    final List<Mensagem> falhas = Collections.synchronizedList(new ArrayList<>());
    private int pendentes = 0;
    private final Thread thread;

    Assinante(String nome, Consumer<Mensagem> tratador) {
        this.nome = nome;
        this.tratador = tratador;
        this.thread = new Thread(this::laco);
        this.thread.setDaemon(true);
        this.thread.start();
    }

    void entregar(Mensagem msg) {
        synchronized (this) {
            pendentes++;
        }
        fila.add(msg);
    }

    private void laco() {
        try {
            while (true) {
                Mensagem msg = fila.take();
                if (msg == PARAR) {
                    concluido();
                    return;
                }
                try {
                    tratador.accept(msg);
                } catch (Exception e) {
                    falhas.add(msg); // isolada: não afeta os demais
                }
                concluido();
            }
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
        }
    }

    private synchronized void concluido() {
        pendentes--;
        notifyAll();
    }

    synchronized void aguardar() throws InterruptedException {
        while (pendentes > 0) wait();
    }

    void parar() throws InterruptedException {
        entregar(PARAR);
        thread.join();
    }
}

class Broker {
    final Map<String, List<Assinante>> topicos = new HashMap<>();
    final Map<String, Integer> seq = new HashMap<>();

    synchronized void assinar(String topico, Assinante assinante) {
        topicos.computeIfAbsent(topico, t -> new ArrayList<>()).add(assinante);
        seq.putIfAbsent(topico, 0);
    }

    synchronized void cancelar(String topico, Assinante assinante) {
        List<Assinante> lista = topicos.get(topico);
        if (lista != null) lista.remove(assinante);
    }

    int publicar(String topico, String conteudo) {
        Mensagem msg;
        List<Assinante> destinos;
        synchronized (this) {
            seq.merge(topico, 1, Integer::sum);
            msg = new Mensagem(topico, conteudo, seq.get(topico));
            destinos = new ArrayList<>(topicos.getOrDefault(topico, new ArrayList<>()));
        }
        for (Assinante a : destinos) a.entregar(msg);
        return destinos.size();
    }
}

public static void main(String[] args) throws Exception {
    List<String> recebidas = Collections.synchronizedList(new ArrayList<>());

    Consumer<Mensagem> bom = m -> recebidas.add("('bom', " + m.seq + ", '" + m.conteudo + "')");
    Consumer<Mensagem> ruim = m -> { throw new RuntimeException("falhei"); };

    Broker broker = new Broker();
    Assinante a = new Assinante("a", bom), b = new Assinante("b", ruim);
    broker.assinar("pedidos", a);
    broker.assinar("pedidos", b);
    System.out.println(broker.publicar("pedidos", "novo pedido")); // 2
    System.out.println(broker.publicar("outro", "ninguém ouve"));  // 0
    broker.publicar("pedidos", "pedido 2");
    a.aguardar();
    b.aguardar();
    System.out.println(recebidas + " " + b.falhas.size());
    a.parar();
    b.parar();
}
```

```cpp
#include <condition_variable>
#include <functional>
#include <iostream>
#include <map>
#include <memory>
#include <mutex>
#include <queue>
#include <string>
#include <thread>
#include <vector>
using namespace std;

struct Mensagem {
    string topico, conteudo;
    int seq;
};

class Assinante {
    function<void(const Mensagem&)> tratador;
    queue<pair<bool, Mensagem>> fila; // bool = é a sentinela de parada (o None do Python)
    mutex m;
    condition_variable temItem, vazio;
    int pendentes = 0;
    thread t;

    void laco() {
        while (true) {
            unique_lock<mutex> lk(m);
            temItem.wait(lk, [&] { return !fila.empty(); });
            auto [parar, msg] = fila.front();
            fila.pop();
            lk.unlock();
            if (!parar) {
                try {
                    tratador(msg);
                } catch (...) {
                    lock_guard<mutex> g(m);
                    falhas.push_back(msg); // isolada: não afeta os demais
                }
            }
            lock_guard<mutex> g(m);
            if (--pendentes == 0) vazio.notify_all();
            if (parar) return;
        }
    }

public:
    string nome;
    vector<Mensagem> falhas;

    Assinante(string nome, function<void(const Mensagem&)> tratador) : tratador(tratador), nome(nome) {
        t = thread(&Assinante::laco, this);
    }

    void entregar(const Mensagem& msg, bool parar = false) {
        lock_guard<mutex> g(m);
        pendentes++;
        fila.push({parar, msg});
        temItem.notify_one();
    }

    void aguardar() {
        unique_lock<mutex> lk(m);
        vazio.wait(lk, [&] { return pendentes == 0; });
    }

    void parar() {
        entregar(Mensagem{}, true);
        t.join();
    }

    size_t totalFalhas() {
        lock_guard<mutex> g(m);
        return falhas.size();
    }
};

class Broker {
    map<string, vector<Assinante*>> topicos;
    map<string, int> seq;
    mutex lk;

public:
    void assinar(const string& topico, Assinante* assinante) {
        lock_guard<mutex> g(lk);
        topicos[topico].push_back(assinante);
        seq.emplace(topico, 0);
    }

    void cancelar(const string& topico, Assinante* assinante) {
        lock_guard<mutex> g(lk);
        auto& lista = topicos[topico];
        for (auto it = lista.begin(); it != lista.end(); ++it) {
            if (*it == assinante) {
                lista.erase(it);
                break;
            }
        }
    }

    int publicar(const string& topico, const string& conteudo) {
        Mensagem msg;
        vector<Assinante*> destinos;
        {
            lock_guard<mutex> g(lk);
            seq[topico] += 1;
            msg = Mensagem{topico, conteudo, seq[topico]};
            destinos = topicos[topico];
        }
        for (auto* a : destinos) a->entregar(msg);
        return (int)destinos.size();
    }
};

int main() {
    vector<string> recebidas;
    mutex recebidasMutex;

    auto bom = [&](const Mensagem& m) {
        lock_guard<mutex> g(recebidasMutex);
        recebidas.push_back("('bom', " + to_string(m.seq) + ", '" + m.conteudo + "')");
    };
    auto ruim = [](const Mensagem&) { throw runtime_error("falhei"); };

    Broker broker;
    Assinante a("a", bom), b("b", ruim);
    broker.assinar("pedidos", &a);
    broker.assinar("pedidos", &b);
    cout << broker.publicar("pedidos", "novo pedido") << endl; // 2
    cout << broker.publicar("outro", "ninguém ouve") << endl;  // 0
    broker.publicar("pedidos", "pedido 2");
    a.aguardar();
    b.aguardar();
    cout << "[";
    for (size_t i = 0; i < recebidas.size(); i++) cout << (i ? ", " : "") << recebidas[i];
    cout << "] " << b.totalFalhas() << endl;
    a.parar();
    b.parar();
}
```

```csharp
using System;
using System.Collections.Generic;
using System.Linq;
using System.Threading;

class Mensagem
{
    public readonly string Topico, Conteudo;
    public readonly int Seq;

    public Mensagem(string topico, string conteudo, int seq)
    {
        Topico = topico;
        Conteudo = conteudo;
        Seq = seq;
    }
}

class Assinante
{
    private static readonly Mensagem PARAR = new Mensagem("", "", 0); // sentinela (o None do Python)

    public readonly string Nome;
    private readonly Action<Mensagem> tratador;
    private readonly Queue<Mensagem> fila = new Queue<Mensagem>();
    private readonly object trava = new object();
    public readonly List<Mensagem> Falhas = new List<Mensagem>();
    private int pendentes = 0;
    private readonly Thread thread;

    public Assinante(string nome, Action<Mensagem> tratador)
    {
        Nome = nome;
        this.tratador = tratador;
        thread = new Thread(Laco);
        thread.IsBackground = true;
        thread.Start();
    }

    public void Entregar(Mensagem msg)
    {
        lock (trava)
        {
            pendentes++;
            fila.Enqueue(msg);
            Monitor.PulseAll(trava);
        }
    }

    private void Laco()
    {
        while (true)
        {
            Mensagem msg;
            lock (trava)
            {
                while (fila.Count == 0) Monitor.Wait(trava);
                msg = fila.Dequeue();
            }
            if (msg != PARAR)
            {
                try
                {
                    tratador(msg);
                }
                catch (Exception)
                {
                    lock (trava) { Falhas.Add(msg); } // isolada: não afeta os demais
                }
            }
            lock (trava)
            {
                pendentes--;
                Monitor.PulseAll(trava);
            }
            if (msg == PARAR) return;
        }
    }

    public void Aguardar()
    {
        lock (trava)
        {
            while (pendentes > 0) Monitor.Wait(trava);
        }
    }

    public void Parar()
    {
        Entregar(PARAR);
        thread.Join();
    }

    public int TotalFalhas()
    {
        lock (trava) { return Falhas.Count; }
    }
}

class Broker
{
    private readonly Dictionary<string, List<Assinante>> topicos = new Dictionary<string, List<Assinante>>();
    private readonly Dictionary<string, int> seq = new Dictionary<string, int>();
    private readonly object trava = new object();

    public void Assinar(string topico, Assinante assinante)
    {
        lock (trava)
        {
            if (!topicos.ContainsKey(topico)) topicos[topico] = new List<Assinante>();
            topicos[topico].Add(assinante);
            if (!seq.ContainsKey(topico)) seq[topico] = 0;
        }
    }

    public void Cancelar(string topico, Assinante assinante)
    {
        lock (trava)
        {
            List<Assinante> lista;
            if (topicos.TryGetValue(topico, out lista)) lista.Remove(assinante);
        }
    }

    public int Publicar(string topico, string conteudo)
    {
        Mensagem msg;
        List<Assinante> destinos;
        lock (trava)
        {
            int atual;
            seq.TryGetValue(topico, out atual);
            seq[topico] = atual + 1;
            msg = new Mensagem(topico, conteudo, seq[topico]);
            List<Assinante> lista;
            destinos = topicos.TryGetValue(topico, out lista) ? new List<Assinante>(lista) : new List<Assinante>();
        }
        foreach (var a in destinos) a.Entregar(msg);
        return destinos.Count;
    }
}

    b.Aguardar();
    Console.WriteLine("[" + string.Join(", ", recebidas) + "] " + b.TotalFalhas());
    a.Parar();
    b.Parar();
}
```

```go
import (
	"fmt"
	"strings"
	"sync"
)

type Mensagem struct {
	topico, conteudo string
	seq              int
}

type Assinante struct {
	nome     string
	tratador func(Mensagem) error
	fila     chan *Mensagem // nil = sentinela de parada (o None do Python)
	espera   sync.WaitGroup // conta mensagens ainda não tratadas (queue.join)
	feito    chan struct{}
	mu       sync.Mutex
	falhas   []Mensagem
}

func NovoAssinante(nome string, tratador func(Mensagem) error) *Assinante {
	a := &Assinante{nome: nome, tratador: tratador, fila: make(chan *Mensagem, 100), feito: make(chan struct{})}
	go a.laco()
	return a
}

func (a *Assinante) entregar(msg *Mensagem) {
	a.espera.Add(1)
	a.fila <- msg
}

func (a *Assinante) laco() {
	for msg := range a.fila {
		if msg == nil {
			a.espera.Done()
			close(a.feito)
			return
		}
		if err := a.tratador(*msg); err != nil {
			a.mu.Lock()
			a.falhas = append(a.falhas, *msg) // isolada: não afeta os demais
			a.mu.Unlock()
		}
		a.espera.Done()
	}
}

func (a *Assinante) aguardar() { a.espera.Wait() }

func (a *Assinante) parar() {
	a.entregar(nil)
	<-a.feito
}

type Broker struct {
	mu      sync.Mutex
	topicos map[string][]*Assinante
	seq     map[string]int
}

func NovoBroker() *Broker {
	return &Broker{topicos: map[string][]*Assinante{}, seq: map[string]int{}}
}

func (b *Broker) assinar(topico string, a *Assinante) {
	b.mu.Lock()
	defer b.mu.Unlock()
	b.topicos[topico] = append(b.topicos[topico], a)
	if _, ok := b.seq[topico]; !ok {
		b.seq[topico] = 0
	}
}

func (b *Broker) cancelar(topico string, a *Assinante) {
	b.mu.Lock()
	defer b.mu.Unlock()
	lista := b.topicos[topico]
	for i, x := range lista {
		if x == a {
			b.topicos[topico] = append(lista[:i], lista[i+1:]...)
			return
		}
	}
}

func (b *Broker) publicar(topico, conteudo string) int {
	b.mu.Lock()
	b.seq[topico]++
	msg := &Mensagem{topico, conteudo, b.seq[topico]}
	destinos := append([]*Assinante{}, b.topicos[topico]...)
	b.mu.Unlock()
	for _, a := range destinos {
		a.entregar(msg)
	}
	return len(destinos)
}

func main() {
	var recebidas []string
	var mu sync.Mutex

	bom := func(m Mensagem) error {
		mu.Lock()
		defer mu.Unlock()
		recebidas = append(recebidas, fmt.Sprintf("('bom', %d, '%s')", m.seq, m.conteudo))
		return nil
	}
	ruim := func(m Mensagem) error { return fmt.Errorf("falhei") }

	broker := NovoBroker()
	a, b := NovoAssinante("a", bom), NovoAssinante("b", ruim)
	broker.assinar("pedidos", a)
	broker.assinar("pedidos", b)
	fmt.Println(broker.publicar("pedidos", "novo pedido")) // 2
	fmt.Println(broker.publicar("outro", "ninguém ouve"))  // 0
	broker.publicar("pedidos", "pedido 2")
	a.aguardar()
	b.aguardar()
	b.mu.Lock()
	falhas := len(b.falhas)
	b.mu.Unlock()
	fmt.Println("["+strings.Join(recebidas, ", ")+"]", falhas)
	a.parar()
	b.parar()
}
```

O número de sequência é atribuído dentro do lock, e a entrega é feita fora dele, para que um assinante nunca bloqueie a publicação.

## Semânticas de entrega

| Garantia | Significado | Custo |
| --- | --- | --- |
| No máximo uma vez | Pode perder, nunca duplica | Simples |
| Pelo menos uma vez | Reenvia até confirmar; pode duplicar | Assinante idempotente |
| Exatamente uma vez | Na prática, pelo menos uma vez mais deduplicação | Alto |

## Extensões

Mensagens de falha em uma fila morta (dead letter), filtros por atributo na assinatura, grupos de consumidores onde só um membro recebe cada mensagem (competição), retenção do histórico para reprocessar a partir de um deslocamento, e fila limitada com política de descarte quando o assinante não acompanha (backpressure).

## Na entrevista

- **"Pub/sub é igual a Observer?"** Parecido, mas pub/sub tem um intermediário (broker) e os lados não se conhecem.
- **"E se a fila do assinante encher?"** Limite o tamanho e escolha: bloquear, descartar o mais antigo ou desconectar.
