O aluguel de carros combina disponibilidade por período (como no hotel) com cálculo de preço por categoria, extras e devolução com possíveis taxas. Foque em separar o veículo do contrato de aluguel.

## Requisitos

- Frota com categorias (econômico, SUV, luxo), cada uma com diária própria.
- Buscar veículos livres de uma categoria em um período.
- Reservar, retirar e devolver.
- Preço final: diárias, extras (cadeirinha, seguro) e multa por atraso.

## Entidades

| Entidade | Responsabilidade |
| --- | --- |
| `Veiculo` | Placa, categoria e reservas. |
| `Categoria` | Diária base. |
| `Extra` | Item opcional com preço por dia. |
| `Aluguel` | Veículo, cliente, período, extras e status. |
| `Locadora` | Busca, reserva e devolução. |

```python
from dataclasses import dataclass, field
from enum import Enum
from typing import List


class Categoria(Enum):
    ECONOMICO = 100
    SUV = 180
    LUXO = 350


@dataclass(frozen=True)
class Extra:
    nome: str
    por_dia: int


class Veiculo:
    def __init__(self, placa: str, categoria: Categoria):
        self.placa, self.categoria = placa, categoria
        self.alugueis: List["Aluguel"] = []

    def livre(self, inicio: int, fim: int) -> bool:
        return all(a.cancelado or a.fim <= inicio or fim <= a.inicio for a in self.alugueis)


@dataclass
class Aluguel:
    cliente: str
    veiculo: Veiculo
    inicio: int
    fim: int
    extras: List[Extra] = field(default_factory=list)
    cancelado: bool = False
    devolvido_em: int = None

    def dias_previstos(self) -> int:
        return self.fim - self.inicio


class Locadora:
    MULTA_ATRASO = 1.5      # diária cobrada a 150% por dia de atraso

    def __init__(self, frota: List[Veiculo]):
        self.frota = frota

    def buscar(self, categoria: Categoria, inicio: int, fim: int) -> List[Veiculo]:
        if fim <= inicio:
            raise ValueError("período inválido")
        return [v for v in self.frota if v.categoria is categoria and v.livre(inicio, fim)]

    def reservar(self, cliente, categoria, inicio, fim, extras=()) -> Aluguel:
        livres = self.buscar(categoria, inicio, fim)
        if not livres:
            raise RuntimeError("sem veículo disponível")
        a = Aluguel(cliente, livres[0], inicio, fim, list(extras))
        livres[0].alugueis.append(a)
        return a

    def devolver(self, aluguel: Aluguel, dia: int) -> float:
        aluguel.devolvido_em = dia
        diaria = aluguel.veiculo.categoria.value
        extras = sum(e.por_dia for e in aluguel.extras)
        base = aluguel.dias_previstos() * (diaria + extras)
        atraso = max(0, dia - aluguel.fim)
        return base + atraso * diaria * self.MULTA_ATRASO

loc = Locadora(frota)
cadeirinha = Extra("cadeirinha", 15)
a1 = loc.reservar("Ana", Categoria.ECONOMICO, 1, 4, [cadeirinha])

print(a1.veiculo.placa, a2.veiculo.placa)       # carros diferentes

    loc.reservar("Caio", Categoria.ECONOMICO, 3, 4)
except RuntimeError as e:
    print("erro:", e)
print(loc.devolver(a1, 5))     # 3*(100+15) + 1*100*1.5 = 495.0
```

```javascript
class Extra {
  constructor(nome, porDia) {
    this.nome = nome;
    this.porDia = porDia;
    Object.freeze(this);
  }
}

    this.categoria = categoria;

  }

    return this.alugueis.every((a) => a.cancelado || a.fim <= inicio || fim <= a.inicio);
  }
}

  constructor(cliente, veiculo, inicio, fim, extras = []) {
    this.cliente = cliente;
    this.veiculo = veiculo;
    this.inicio = inicio;
    this.fim = fim;
    this.extras = extras;
    this.cancelado = false;
    this.devolvidoEm = null;
  }

  diasPrevistos() {
    return this.fim - this.inicio;
  }
}

  constructor(frota) {
    this.frota = frota;
  }

  buscar(categoria, inicio, fim) {
    if (fim <= inicio) {
      throw new RangeError("período inválido");

    return this.frota.filter((v) => v.categoria === categoria && v.livre(inicio, fim));
  }

    const livres = this.buscar(categoria, inicio, fim);
    if (livres.length === 0) {
      throw new Error("sem veículo disponível");

    const a = new Aluguel(cliente, livres[0], inicio, fim, [...extras]);
    livres[0].alugueis.push(a);
    return a;
```

```java
import java.util.*;

enum Categoria {
    ECONOMICO(100), SUV(180), LUXO(350);

    final int diaria;

    Categoria(int diaria) {
        this.diaria = diaria;
    }
}

class Extra {
    final String nome;
    final int porDia;

    Extra(String nome, int porDia) {
        this.nome = nome;
        this.porDia = porDia;
    }
}

class Veiculo {
    final String placa;
    final Categoria categoria;
    final List<Aluguel> alugueis = new ArrayList<>();

    Veiculo(String placa, Categoria categoria) {
        this.placa = placa;
        this.categoria = categoria;
    }

    boolean livre(int inicio, int fim) {
        for (Aluguel a : alugueis) {
            if (!(a.cancelado || a.fim <= inicio || fim <= a.inicio)) return false;
        }
        return true;
    }
}

class Aluguel {
    final String cliente;
    final Veiculo veiculo;
    final int inicio, fim;
    final List<Extra> extras;
    boolean cancelado = false;
    Integer devolvidoEm = null;

    Aluguel(String cliente, Veiculo veiculo, int inicio, int fim, List<Extra> extras) {
        this.cliente = cliente;
        this.veiculo = veiculo;
        this.inicio = inicio;
        this.fim = fim;
        this.extras = extras;
    }

    int diasPrevistos() {
        return fim - inicio;
    }
}

class Locadora {
    static final double MULTA_ATRASO = 1.5; // diária cobrada a 150% por dia de atraso
    private final List<Veiculo> frota;

    Locadora(List<Veiculo> frota) {
        this.frota = frota;
    }

    List<Veiculo> buscar(Categoria categoria, int inicio, int fim) {
        if (fim <= inicio) {
            throw new IllegalArgumentException("período inválido");
        }
        List<Veiculo> r = new ArrayList<>();
        for (Veiculo v : frota) if (v.categoria == categoria && v.livre(inicio, fim)) r.add(v);
        return r;
    }

    Aluguel reservar(String cliente, Categoria categoria, int inicio, int fim, List<Extra> extras) {
        List<Veiculo> livres = buscar(categoria, inicio, fim);
        if (livres.isEmpty()) {
            throw new IllegalStateException("sem veículo disponível");
        }
        Aluguel a = new Aluguel(cliente, livres.get(0), inicio, fim, new ArrayList<>(extras));
        livres.get(0).alugueis.add(a);
        return a;
    }

    Aluguel reservar(String cliente, Categoria categoria, int inicio, int fim) {
        return reservar(cliente, categoria, inicio, fim, new ArrayList<>());
    }

        aluguel.devolvidoEm = dia;
        int diaria = aluguel.veiculo.categoria.diaria;

        for (Extra e : aluguel.extras) extras += e.porDia;

        int atraso = Math.max(0, dia - aluguel.fim);
        return base + atraso * diaria * MULTA_ATRASO;
    }
}
```

```go
import (
	"errors"
	"fmt"
)

type Categoria int

const (
	Economico Categoria = 100
	Suv       Categoria = 180
	Luxo      Categoria = 350
)

type Extra struct {
	nome  string
	porDia int
}

type Veiculo struct {
	placa     string
	categoria Categoria
	alugueis  []*Aluguel
}

func (v *Veiculo) livre(inicio, fim int) bool {
	for _, a := range v.alugueis {
		if !(a.cancelado || a.fim <= inicio || fim <= a.inicio) {
			return false
		}
	}
	return true
}

type Aluguel struct {
	cliente    string
	veiculo    *Veiculo
	inicio, fim int
	extras     []Extra
	cancelado  bool
	devolvidoEm int
}

func (a *Aluguel) diasPrevistos() int { return a.fim - a.inicio }

const multaAtraso = 1.5 // diária cobrada a 150% por dia de atraso

type Locadora struct {
	frota []*Veiculo
}

func (l *Locadora) buscar(categoria Categoria, inicio, fim int) ([]*Veiculo, error) {
	if fim <= inicio {
		return nil, errors.New("período inválido")
	}
	r := []*Veiculo{}
	for _, v := range l.frota {
		if v.categoria == categoria && v.livre(inicio, fim) {
			r = append(r, v)
		}
	}
	return r, nil
}

func (l *Locadora) reservar(cliente string, categoria Categoria, inicio, fim int, extras ...Extra) (*Aluguel, error) {
	livres, err := l.buscar(categoria, inicio, fim)
	if err != nil {
		return nil, err
	}
	if len(livres) == 0 {
		return nil, errors.New("sem veículo disponível")
	}
	a := &Aluguel{cliente: cliente, veiculo: livres[0], inicio: inicio, fim: fim, extras: append([]Extra{}, extras...)}
	livres[0].alugueis = append(livres[0].alugueis, a)
	return a, nil
}

func (l *Locadora) devolver(aluguel *Aluguel, dia int) float64 {
	aluguel.devolvidoEm = dia
	diaria := int(aluguel.veiculo.categoria)
	extras := 0
	for _, e := range aluguel.extras {
		extras += e.porDia
	}
	base := aluguel.diasPrevistos() * (diaria + extras)
	atraso := 0
	if dia > aluguel.fim {
		atraso = dia - aluguel.fim
	}
	return float64(base) + float64(atraso*diaria)*multaAtraso
}

func main() {
	frota := []*Veiculo{{placa: "AAA-1111", categoria: Economico}, {placa: "BBB-2222", categoria: Economico}}
```

## Concorrência e extensões

O mesmo cuidado do hotel: verificar sobreposição e inserir o aluguel numa única seção crítica por veículo. Extensões: cálculo de preço por Strategy (fim de semana, temporada), vistoria na retirada e na devolução com registro de danos, quilometragem franqueada e retirada em uma filial com devolução em outra, o que exige modelar filiais e transferência de frota.
