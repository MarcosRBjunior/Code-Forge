No LLD de uma rede social você não projeta o feed distribuído, e sim o modelo de objetos: usuários, amizades, posts, curtidas e a montagem do feed. O ponto que mais pesa é a escolha entre relação simétrica (amizade) ou assimétrica (seguir).

## Requisitos

- Usuários se seguem (relação assimétrica).
- Publicar posts, curtir e comentar.
- Feed: posts de quem o usuário segue, do mais recente para o mais antigo.
- Notificar o autor quando curtem ou comentam.

## Entidades

| Entidade | Responsabilidade |
| --- | --- |
| `Usuario` | Nome, seguidos, posts e notificações. |
| `Post` | Autor, texto, instante, curtidas e comentários. |
| `Comentario` | Autor e texto. |
| `Rede` | Operações de seguir, publicar, curtir e montar o feed. |

import heapq
from dataclasses import dataclass, field
from typing import List, Set


@dataclass
class Comentario:
    autor: "Usuario"
    texto: str


class Post:
    def __init__(self, autor: "Usuario", texto: str, instante: int):
        self.autor, self.texto, self.instante = autor, texto, instante
        self.curtidas: Set["Usuario"] = set()
        self.comentarios: List[Comentario] = []


class Usuario:
    def __init__(self, nome: str):
        self.nome = nome
        self.seguindo: Set["Usuario"] = set()
        self.posts: List[Post] = []
        self.notificacoes: List[str] = []

    def notificar(self, msg: str):
        self.notificacoes.append(msg)


class Rede:
    def __init__(self):
        self._tempo = 0

    def _tick(self) -> int:
        self._tempo += 1
        return self._tempo

    def seguir(self, a: Usuario, b: Usuario):
        if a is b:
            raise ValueError("não pode seguir a si mesmo")
        a.seguindo.add(b)

    def deixar_de_seguir(self, a: Usuario, b: Usuario):
        a.seguindo.discard(b)

    def publicar(self, autor: Usuario, texto: str) -> Post:
        p = Post(autor, texto, self._tick())
        autor.posts.append(p)
        return p

    def curtir(self, usuario: Usuario, post: Post):
        if usuario not in post.curtidas:
            post.curtidas.add(usuario)
            post.autor.notificar(f"{usuario.nome} curtiu seu post")

    def comentar(self, usuario: Usuario, post: Post, texto: str):
        post.comentarios.append(Comentario(usuario, texto))
        post.autor.notificar(f"{usuario.nome} comentou: {texto}")

    def feed(self, usuario: Usuario, limite: int = 10) -> List[Post]:
        # cada lista de posts já está em ordem cronológica; junta as mais recentes
        fontes = [reversed(u.posts) for u in usuario.seguindo]
        mesclado = heapq.merge(*fontes, key=lambda p: -p.instante)
        return [p for _, p in zip(range(limite), mesclado)]

rede.comentar(ana, p2, "bom dia!")
print([p.texto for p in rede.feed(ana)])   # ['tudo bem?', 'bom dia', 'oi']
print(caio.notificacoes)

#include <algorithm>
#include <iostream>
#include <set>
#include <stdexcept>
#include <string>
#include <vector>
using namespace std;

struct Usuario;

struct Comentario {
    Usuario* autor;
    string texto;
};

struct Post {
    Usuario* autor;
    string texto;
    int instante;
    set<Usuario*> curtidas;
    vector<Comentario> comentarios;
};

struct Usuario {
    string nome;
    vector<Usuario*> seguindo;
    vector<Post*> posts;
    vector<string> notificacoes;

    void notificar(const string& msg) { notificacoes.push_back(msg); }
};

class Rede {
    int tempo = 0;
    vector<Post*> todos;

    int tick() { return ++tempo; }

public:
    ~Rede() {
        for (auto* p : todos) delete p;
    }

    void seguir(Usuario& a, Usuario& b) {
        if (&a == &b) {
            throw invalid_argument("não pode seguir a si mesmo");
        }
        if (find(a.seguindo.begin(), a.seguindo.end(), &b) == a.seguindo.end()) a.seguindo.push_back(&b);
    }

    void deixarDeSeguir(Usuario& a, Usuario& b) {
        a.seguindo.erase(remove(a.seguindo.begin(), a.seguindo.end(), &b), a.seguindo.end());
    }

    Post* publicar(Usuario& autor, const string& texto) {
        Post* p = new Post{&autor, texto, tick(), {}, {}};
        todos.push_back(p);
        autor.posts.push_back(p);
        return p;
    }

    void curtir(Usuario& usuario, Post& post) {
        if (!post.curtidas.count(&usuario)) {
            post.curtidas.insert(&usuario);
            post.autor->notificar(usuario.nome + " curtiu seu post");
        }
    }

    void comentar(Usuario& usuario, Post& post, const string& texto) {
        post.comentarios.push_back({&usuario, texto});
        post.autor->notificar(usuario.nome + " comentou: " + texto);
    }

    // Junta os posts de quem o usuário segue, do mais recente para o mais antigo.
    // (Um merge de k listas com heap faria o mesmo sem juntar tudo antes.)
    vector<Post*> feed(const Usuario& usuario, size_t limite = 10) const {
        vector<Post*> r;
        for (auto* u : usuario.seguindo) r.insert(r.end(), u->posts.begin(), u->posts.end());
        sort(r.begin(), r.end(), [](Post* a, Post* b) { return a->instante > b->instante; });
        if (r.size() > limite) r.resize(limite);
        return r;
    }
};

string lista(const vector<string>& xs) {
    string r = "[";
    for (size_t i = 0; i < xs.size(); i++) r += (i ? ", '" : "'") + xs[i] + "'";
    return r + "]";
}

int main() {
    Rede rede;
    Usuario ana{"Ana"}, bia{"Bia"}, caio{"Caio"};
    rede.seguir(ana, bia);
    rede.seguir(ana, caio);

}

import (
	"errors"
	"fmt"
	"sort"
	"strings"
)

type Comentario struct {
	autor *Usuario
	texto string
}

type Post struct {
	autor       *Usuario
	texto       string
	instante    int
	curtidas    map[*Usuario]bool
	comentarios []Comentario
}

type Usuario struct {
	nome         string
	seguindo     []*Usuario
	posts        []*Post
	notificacoes []string
}

func (u *Usuario) notificar(msg string) { u.notificacoes = append(u.notificacoes, msg) }

type Rede struct {
	tempo int
}

func (r *Rede) tick() int {
	r.tempo++
	return r.tempo
}

func (r *Rede) seguir(a, b *Usuario) error {
	if a == b {
		return errors.New("não pode seguir a si mesmo")
	}
	for _, x := range a.seguindo {
		if x == b {
			return nil
		}
	}
	a.seguindo = append(a.seguindo, b)
	return nil
}

func (r *Rede) deixarDeSeguir(a, b *Usuario) {
	for i, x := range a.seguindo {
		if x == b {
			a.seguindo = append(a.seguindo[:i], a.seguindo[i+1:]...)
			return
		}
	}
}

func (r *Rede) publicar(autor *Usuario, texto string) *Post {
	p := &Post{autor: autor, texto: texto, instante: r.tick(), curtidas: map[*Usuario]bool{}}
	autor.posts = append(autor.posts, p)
	return p
}

func (r *Rede) curtir(usuario *Usuario, post *Post) {
	if !post.curtidas[usuario] {
		post.curtidas[usuario] = true
		post.autor.notificar(usuario.nome + " curtiu seu post")
	}
}

func (r *Rede) comentar(usuario *Usuario, post *Post, texto string) {
	post.comentarios = append(post.comentarios, Comentario{usuario, texto})
	post.autor.notificar(usuario.nome + " comentou: " + texto)
}

// Junta os posts de quem o usuário segue, do mais recente para o mais antigo.
// (Um merge de k listas com heap faria o mesmo sem juntar tudo antes.)
func (r *Rede) feed(usuario *Usuario, limite int) []*Post {
	todos := []*Post{}
	for _, u := range usuario.seguindo {
		todos = append(todos, u.posts...)
	}
	sort.Slice(todos, func(i, j int) bool { return todos[i].instante > todos[j].instante })
	if len(todos) > limite {
		todos = todos[:limite]
	}
	return todos
}

func lista(xs []string) string {
	citadas := []string{}
	for _, x := range xs {
		citadas = append(citadas, "'"+x+"'")
	}
	return "[" + strings.Join(citadas, ", ") + "]"
}

O feed usa `heapq.merge` sobre listas já ordenadas, evitando ordenar tudo a cada leitura: o custo é proporcional ao limite pedido, mais o número de seguidos.

## Trade-offs de feed

| Estratégia | Como funciona | Quando serve |
| --- | --- | --- |
| Pull (na leitura) | Mescla posts dos seguidos ao abrir o feed | Poucos seguidos, simples |
| Push (na escrita) | Copia o post no feed de cada seguidor | Muita leitura, poucos seguidores por autor |

No LLD, implemente o pull e cite o push como evolução. O trade-off completo é assunto de System Design.

## Extensões

Curtir comentários, privacidade por post (visibilidade como Strategy), bloqueio de usuários e notificações por canais diferentes com Observer, em vez de a rede chamar `notificar` diretamente.

## Na entrevista

- **"Seguir ou amizade?"** Amizade é uma relação mútua que exige aceite; seguir é um conjunto no lado de quem segue.
- **"Como paginar o feed?"** Use o instante do último post visto como cursor, em vez de deslocamento numérico.
