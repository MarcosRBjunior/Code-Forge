Uma entrevista de design orientado a objetos (OOD) é aberta de propósito: "projete um estacionamento". Não existe resposta única, e o entrevistador avalia **como você pensa**, não se chegou a um desenho idêntico ao dele. Ter um roteiro evita o erro mais comum, que é sair codando nos primeiros dois minutos.

## Roteiro em seis passos

| Passo | O que fazer | Tempo aproximado |
| --- | --- | --- |
| 1. Requisitos | Perguntar, delimitar escopo, listar o que fica de fora | 5 a 8 min |
| 2. Entidades | Substantivos viram classes candidatas; verbos viram métodos | 5 min |
| 3. Relações | Herança, composição, interfaces; desenhar o diagrama | 5 a 8 min |
| 4. Padrões | Aplicar só onde resolvem um problema real | 3 min |
| 5. Código | Implementar o núcleo, classes pequenas e legíveis | 15 a 20 min |
| 6. Extensão e testes | Mostrar como o design aguenta mudanças | 5 min |

## Passo 1: requisitos

Faça perguntas que mudam o design: quantos tipos de veículo? Há vários andares? Preço por hora ou por faixa? Concorrência importa? Ao final, repita o escopo em voz alta: "então vou modelar X e Y, e deixar Z como extensão". Isso protege você de ser cobrado por algo que nunca foi combinado.

## Passo 2 e 3: entidades e relações

Pegue as frases dos requisitos e sublinhe os substantivos. Depois pergunte, para cada candidato: tem estado? tem comportamento? muda de forma independente? Se não, talvez seja só um atributo. Em seguida decida as relações, preferindo composição à herança quando a relação não é um "é um" verdadeiro.

## Passo 5: um esqueleto que roda

Mesmo em quadro branco, escreva código com a cara de quem vai executar. Exemplo do núcleo de um estacionamento, deixando o preço como estratégia:

```python
from abc import ABC, abstractmethod


class Tarifa(ABC):
    @abstractmethod
    def calcular(self, horas: int) -> float: ...


class TarifaHoraria(Tarifa):
    def __init__(self, valor_hora: float) -> None:
        self.valor_hora = valor_hora

    def calcular(self, horas: int) -> float:
        return self.valor_hora * max(1, horas)


class Estacionamento:
    def __init__(self, vagas: int, tarifa: Tarifa) -> None:
        self.livres = vagas
        self.tarifa = tarifa
        self.ocupadas: dict[str, int] = {}

    def entrar(self, placa: str, hora: int) -> bool:
        if self.livres == 0 or placa in self.ocupadas:
            return False
        self.livres -= 1
        self.ocupadas[placa] = hora
        return True

    def sair(self, placa: str, hora: int) -> float:
        entrada = self.ocupadas.pop(placa)
        self.livres += 1
        return self.tarifa.calcular(hora - entrada)


e = Estacionamento(1, TarifaHoraria(5))
assert e.entrar("ABC", 8) is True
assert e.entrar("XYZ", 8) is False
assert e.sair("ABC", 11) == 15
```

```javascript
class Tarifa {
  calcular(horas) {
    throw new Error("método abstrato");
  }
}

class TarifaHoraria extends Tarifa {
  constructor(valorHora) {
    super();
    this.valorHora = valorHora;
  }

  calcular(horas) {
    return this.valorHora * Math.max(1, horas);
  }
}

class Estacionamento {
  constructor(vagas, tarifa) {
    this.livres = vagas;
    this.tarifa = tarifa;
    this.ocupadas = new Map();
  }

  entrar(placa, hora) {
    if (this.livres === 0 || this.ocupadas.has(placa)) {
      return false;
    }
    this.livres -= 1;
    this.ocupadas.set(placa, hora);
    return true;
  }

  sair(placa, hora) {
    const entrada = this.ocupadas.get(placa);
    this.ocupadas.delete(placa);
    this.livres += 1;
    return this.tarifa.calcular(hora - entrada);
  }
}

function verificar(condicao) {
  if (!condicao) throw new Error("verificação falhou");
}

const e = new Estacionamento(1, new TarifaHoraria(5));
verificar(e.entrar("ABC", 8) === true);
verificar(e.entrar("XYZ", 8) === false);
verificar(e.sair("ABC", 11) === 15);
```

```java
import java.util.*;

interface Tarifa {
    double calcular(int horas);
}

class TarifaHoraria implements Tarifa {
    private final double valorHora;

    TarifaHoraria(double valorHora) {
        this.valorHora = valorHora;
    }

    public double calcular(int horas) {
        return valorHora * Math.max(1, horas);
    }
}

class Estacionamento {
    private int livres;
    private final Tarifa tarifa;
    private final Map<String, Integer> ocupadas = new HashMap<>();

    Estacionamento(int vagas, Tarifa tarifa) {
        this.livres = vagas;
        this.tarifa = tarifa;
    }

    boolean entrar(String placa, int hora) {
        if (livres == 0 || ocupadas.containsKey(placa)) {
            return false;
        }
        livres--;
        ocupadas.put(placa, hora);
        return true;
    }

    double sair(String placa, int hora) {
        int entrada = ocupadas.remove(placa);
        livres++;
        return tarifa.calcular(hora - entrada);
    }
}

static void verificar(boolean condicao) {
    if (!condicao) throw new AssertionError("verificação falhou");
}

public static void main(String[] args) {
    Estacionamento e = new Estacionamento(1, new TarifaHoraria(5));
    verificar(e.entrar("ABC", 8));
    verificar(!e.entrar("XYZ", 8));
    verificar(e.sair("ABC", 11) == 15);
}
```

```cpp
#include <algorithm>
#include <map>
#include <stdexcept>
#include <string>
using namespace std;

class Tarifa {
public:
    virtual ~Tarifa() = default;
    virtual double calcular(int horas) const = 0;
};

class TarifaHoraria : public Tarifa {
    double valorHora;

public:
    TarifaHoraria(double valorHora) : valorHora(valorHora) {}

    double calcular(int horas) const override {
        return valorHora * max(1, horas);
    }
};

class Estacionamento {
    int livres;
    const Tarifa& tarifa;
    map<string, int> ocupadas;

public:
    Estacionamento(int vagas, const Tarifa& tarifa) : livres(vagas), tarifa(tarifa) {}

    bool entrar(const string& placa, int hora) {
        if (livres == 0 || ocupadas.count(placa)) {
            return false;
        }
        livres--;
        ocupadas[placa] = hora;
        return true;
    }

    double sair(const string& placa, int hora) {
        int entrada = ocupadas[placa];
        ocupadas.erase(placa);
        livres++;
        return tarifa.calcular(hora - entrada);
    }
};

void verificar(bool condicao) {
    if (!condicao) throw runtime_error("verificação falhou");
}

int main() {
    TarifaHoraria tarifa(5);
    Estacionamento e(1, tarifa);
    verificar(e.entrar("ABC", 8) == true);
    verificar(e.entrar("XYZ", 8) == false);
    verificar(e.sair("ABC", 11) == 15);
}
```

```csharp

using System.Collections.Generic;

interface ITarifa
{
    double Calcular(int horas);
}

class TarifaHoraria : ITarifa
```

```go
import "errors"

type Tarifa interface {
	calcular(horas int) float64
}

type TarifaHoraria struct {
	valorHora float64
}

func (t TarifaHoraria) calcular(horas int) float64 {
	h := horas
	if h < 1 {
		h = 1
	}
	return t.valorHora * float64(h)
}

type Estacionamento struct {
	livres   int
	tarifa   Tarifa
	ocupadas map[string]int
}

func NovoEstacionamento(vagas int, tarifa Tarifa) *Estacionamento {
	return &Estacionamento{livres: vagas, tarifa: tarifa, ocupadas: map[string]int{}}
}

func (e *Estacionamento) entrar(placa string, hora int) bool {
	if _, ocupada := e.ocupadas[placa]; e.livres == 0 || ocupada {
		return false
	}
	e.livres--
	e.ocupadas[placa] = hora
	return true
}

func (e *Estacionamento) sair(placa string, hora int) float64 {
	entrada := e.ocupadas[placa]
	delete(e.ocupadas, placa)
	e.livres++
	return e.tarifa.calcular(hora - entrada)
}

func verificar(condicao bool) {
	if !condicao {
		panic(errors.New("verificação falhou"))
	}
}

func main() {
	e := NovoEstacionamento(1, TarifaHoraria{5})
	verificar(e.entrar("ABC", 8))
	verificar(!e.entrar("XYZ", 8))
	verificar(e.sair("ABC", 11) == 15)
}
```

## Pense em voz alta

O entrevistador só avalia o que escuta. Explique por que escolheu uma interface, por que um método ficou numa classe e não em outra, e nomeie os trade-offs. Quando errar, corrija com naturalidade: isso mostra maturidade.

<Callout type="tip" title="Comece pequeno">
Entregue primeiro uma versão simples e correta, depois evolua. Um design enxuto que funciona vale mais do que dez padrões que não compilam.
</Callout>

## Erros comuns

- Pular perguntas e supor requisitos.
- Criar hierarquias profundas de herança logo de início.
- Aplicar padrões por vaidade, sem problema que os justifique.
- Ignorar casos de borda e erros (vaga cheia, id inexistente).
- Não testar o próprio código com um exemplo simples no final.

## Na entrevista

Se o tempo apertar, priorize: requisitos claros, diagrama, código do fluxo principal. Extensões podem ser descritas em palavras ("para pagar com Pix, adicionaria uma nova implementação da interface").
