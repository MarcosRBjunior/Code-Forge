Um sistema de biblioteca cobre catálogo, empréstimos, devoluções, reservas e multas. A armadilha clássica é misturar o conceito de livro (obra) com o de exemplar físico. Acertar essa separação mostra maturidade de modelagem.

## Requisitos

- Catálogo de livros, cada um com vários exemplares.
- Membros pegam exemplares emprestados com limite de itens e prazo.
- Atraso gera multa por dia.
- Se todos os exemplares estão emprestados, o membro pode reservar.

## Entidades

| Entidade | Responsabilidade |
| --- | --- |
| `Livro` | Dados da obra: título, ISBN. |
| `Exemplar` | Unidade física, com estado disponível ou emprestado. |
| `Membro` | Quem empresta; guarda empréstimos ativos. |
| `Emprestimo` | Liga membro e exemplar, com data de saída e prazo. |
| `Biblioteca` | Orquestra regras: limites, multas, fila de reserva. |

from dataclasses import dataclass
from collections import defaultdict, deque
from typing import Dict, List, Optional

DIAS_PRAZO = 14
LIMITE = 2
MULTA_DIA = 0.5


@dataclass(frozen=True)
class Livro:
    isbn: str
    titulo: str


class Exemplar:
    def __init__(self, codigo: str, livro: Livro):
        self.codigo, self.livro, self.emprestado = codigo, livro, False


@dataclass
class Emprestimo:
    membro: "Membro"
    exemplar: Exemplar
    saida: int

    @property
    def vencimento(self):
        return self.saida + DIAS_PRAZO


class Membro:
    def __init__(self, nome: str):
        self.nome = nome
        self.emprestimos: List[Emprestimo] = []


class Biblioteca:
    def __init__(self):
        self.exemplares: Dict[str, List[Exemplar]] = defaultdict(list)
        self.reservas: Dict[str, deque] = defaultdict(deque)

    def adicionar(self, ex: Exemplar):
        self.exemplares[ex.livro.isbn].append(ex)

    def emprestar(self, membro: Membro, isbn: str, hoje: int) -> Optional[Emprestimo]:
        if len(membro.emprestimos) >= LIMITE:
            raise ValueError("limite de empréstimos atingido")
        fila = self.reservas[isbn]
        if fila and fila[0] is not membro:
            self.reservar(membro, isbn)
            return None
        for ex in self.exemplares[isbn]:
            if not ex.emprestado:
                if fila and fila[0] is membro:
                    fila.popleft()
                ex.emprestado = True
                emp = Emprestimo(membro, ex, hoje)
                membro.emprestimos.append(emp)
                return emp
        self.reservar(membro, isbn)
        return None

    def reservar(self, membro: Membro, isbn: str):
        if membro not in self.reservas[isbn]:
            self.reservas[isbn].append(membro)

    def devolver(self, emp: Emprestimo, hoje: int) -> float:
        emp.exemplar.emprestado = False
        emp.membro.emprestimos.remove(emp)
        atraso = max(0, hoje - emp.vencimento)
        return atraso * MULTA_DIA

const DIAS_PRAZO = 14;
const LIMITE = 2;
const MULTA_DIA = 0.5;

class Livro {
  constructor(isbn, titulo) {
    this.isbn = isbn;
    this.titulo = titulo;
    Object.freeze(this);
  }
}

class Exemplar {
  constructor(codigo, livro) {
    this.codigo = codigo;
    this.livro = livro;
    this.emprestado = false;
  }
}

class Emprestimo {
  constructor(membro, exemplar, saida) {
    this.membro = membro;
    this.exemplar = exemplar;
    this.saida = saida;
  }

  get vencimento() {
    return this.saida + DIAS_PRAZO;
  }
}

class Membro {
  constructor(nome) {
    this.nome = nome;
    this.emprestimos = [];
  }
}

class Biblioteca {
  constructor() {
    this.exemplares = new Map(); // isbn -> lista de exemplares
    this.reservas = new Map(); // isbn -> fila de membros
  }

  #lista(mapa, isbn) {
    if (!mapa.has(isbn)) mapa.set(isbn, []);
    return mapa.get(isbn);
  }

  adicionar(ex) {
    this.#lista(this.exemplares, ex.livro.isbn).push(ex);
  }

  emprestar(membro, isbn, hoje) {
    if (membro.emprestimos.length >= LIMITE) {
      throw new Error("limite de empréstimos atingido");
    }
    const fila = this.#lista(this.reservas, isbn);
    if (fila.length > 0 && fila[0] !== membro) {
      this.reservar(membro, isbn);
      return null;
    }
    for (const ex of this.#lista(this.exemplares, isbn)) {
      if (!ex.emprestado) {
        if (fila.length > 0 && fila[0] === membro) {
          fila.shift();
        }
        ex.emprestado = true;
        const emp = new Emprestimo(membro, ex, hoje);
        membro.emprestimos.push(emp);
        return emp;
      }
    }
    this.reservar(membro, isbn);
    return null;
  }

  reservar(membro, isbn) {
    const fila = this.#lista(this.reservas, isbn);
    if (!fila.includes(membro)) {
      fila.push(membro);
    }
  }

  devolver(emp, hoje) {
    emp.exemplar.emprestado = false;
    emp.membro.emprestimos.splice(emp.membro.emprestimos.indexOf(emp), 1);
    const atraso = Math.max(0, hoje - emp.vencimento);
    return atraso * MULTA_DIA;
  }
}

import java.util.*;

class Livro {
    final String isbn, titulo;

    Livro(String isbn, String titulo) {
        this.isbn = isbn;
        this.titulo = titulo;
    }
}

class Exemplar {
    final String codigo;
    final Livro livro;
    boolean emprestado = false;

    Exemplar(String codigo, Livro livro) {
        this.codigo = codigo;
        this.livro = livro;
    }
}

class Membro {
    final String nome;
    final List<Emprestimo> emprestimos = new ArrayList<>();

    Membro(String nome) {
        this.nome = nome;
    }
}

class Emprestimo {
    static final int DIAS_PRAZO = 14;

    final Membro membro;
    final Exemplar exemplar;
    final int saida;

    Emprestimo(Membro membro, Exemplar exemplar, int saida) {
        this.membro = membro;
        this.exemplar = exemplar;
        this.saida = saida;
    }

    int vencimento() {
        return saida + DIAS_PRAZO;
    }
}

class Biblioteca {
    static final int LIMITE = 2;
    static final double MULTA_DIA = 0.5;

    private final Map<String, List<Exemplar>> exemplares = new HashMap<>();
    private final Map<String, Deque<Membro>> reservas = new HashMap<>();

    void adicionar(Exemplar ex) {
        exemplares.computeIfAbsent(ex.livro.isbn, k -> new ArrayList<>()).add(ex);
    }

    Emprestimo emprestar(Membro membro, String isbn, int hoje) {
        if (membro.emprestimos.size() >= LIMITE) {
            throw new IllegalStateException("limite de empréstimos atingido");
        }
        Deque<Membro> fila = reservas.computeIfAbsent(isbn, k -> new ArrayDeque<>());
        if (!fila.isEmpty() && fila.peekFirst() != membro) {
            reservar(membro, isbn);
            return null;
        }
        for (Exemplar ex : exemplares.getOrDefault(isbn, new ArrayList<>())) {
            if (!ex.emprestado) {
                if (!fila.isEmpty() && fila.peekFirst() == membro) {
                    fila.pollFirst();
                }
                ex.emprestado = true;
                Emprestimo emp = new Emprestimo(membro, ex, hoje);
                membro.emprestimos.add(emp);
                return emp;
            }
        }
        reservar(membro, isbn);
        return null;
    }

    void reservar(Membro membro, String isbn) {
        Deque<Membro> fila = reservas.computeIfAbsent(isbn, k -> new ArrayDeque<>());
        if (!fila.contains(membro)) {
            fila.addLast(membro);
        }
    }

    double devolver(Emprestimo emp, int hoje) {
        emp.exemplar.emprestado = false;
        emp.membro.emprestimos.remove(emp);
        int atraso = Math.max(0, hoje - emp.vencimento());
        return atraso * MULTA_DIA;
    }
}

public static void main(String[] args) {
    Biblioteca b = new Biblioteca();
    Livro livro = new Livro("111", "Código Limpo");
    b.adicionar(new Exemplar("E1", livro));
    Membro ana = new Membro("Ana"), rui = new Membro("Rui");
    Emprestimo e = b.emprestar(ana, "111", 0);
    System.out.println(b.emprestar(rui, "111", 1)); // null: ficou na fila
    System.out.println(b.devolver(e, 20));          // 3.0 (6 dias de atraso)
    System.out.println(b.emprestar(rui, "111", 21).exemplar.codigo);
}

#include <algorithm>
#include <deque>
#include <iostream>
#include <map>
#include <memory>
#include <stdexcept>
#include <string>
#include <vector>
using namespace std;

const int DIAS_PRAZO = 14;
const int LIMITE = 2;
const double MULTA_DIA = 0.5;

struct Livro {
    string isbn, titulo;
};

struct Exemplar {
    string codigo;
    Livro livro;
    bool emprestado = false;
};

struct Emprestimo;

struct Membro {
    string nome;
    vector<Emprestimo*> emprestimos;
};

struct Emprestimo {
    Membro* membro;
    Exemplar* exemplar;
    int saida;

    int vencimento() const { return saida + DIAS_PRAZO; }
};

class Biblioteca {
    map<string, vector<Exemplar*>> exemplares;
    map<string, deque<Membro*>> reservas;
    vector<unique_ptr<Emprestimo>> todos;

public:
    void adicionar(Exemplar* ex) { exemplares[ex->livro.isbn].push_back(ex); }

    Emprestimo* emprestar(Membro* membro, const string& isbn, int hoje) {
        if ((int)membro->emprestimos.size() >= LIMITE) {
            throw runtime_error("limite de empréstimos atingido");
        }
        auto& fila = reservas[isbn];
        if (!fila.empty() && fila.front() != membro) {
            reservar(membro, isbn);
            return nullptr;
        }
        for (Exemplar* ex : exemplares[isbn]) {
            if (!ex->emprestado) {
                if (!fila.empty() && fila.front() == membro) {
                    fila.pop_front();
                }
                ex->emprestado = true;
                todos.push_back(make_unique<Emprestimo>(Emprestimo{membro, ex, hoje}));
                Emprestimo* emp = todos.back().get();
                membro->emprestimos.push_back(emp);
                return emp;
            }
        }
        reservar(membro, isbn);
        return nullptr;
    }

    void reservar(Membro* membro, const string& isbn) {
        auto& fila = reservas[isbn];
        if (find(fila.begin(), fila.end(), membro) == fila.end()) {
            fila.push_back(membro);
        }
    }

    double devolver(Emprestimo* emp, int hoje) {
        emp->exemplar->emprestado = false;
        auto& lista = emp->membro->emprestimos;
        lista.erase(find(lista.begin(), lista.end(), emp));
        int atraso = max(0, hoje - emp->vencimento());
        return atraso * MULTA_DIA;
    }
};

int main() {
    Biblioteca b;
    Livro livro{"111", "Código Limpo"};
    Exemplar e1{"E1", livro};
    b.adicionar(&e1);
    Membro ana{"Ana"}, rui{"Rui"};
    Emprestimo* e = b.emprestar(&ana, "111", 0);
    Emprestimo* espera = b.emprestar(&rui, "111", 1);
    cout << (espera == nullptr ? "null" : "emprestimo") << endl; // null: ficou na fila
    cout << b.devolver(e, 20) << endl;                           // 3.0 (6 dias de atraso)
    cout << b.emprestar(&rui, "111", 21)->exemplar->codigo << endl;
}

using System.Collections.Generic;
using System.Linq;

class Livro
{
    public readonly string Isbn, Titulo;

    public Livro(string isbn, string titulo)
    {
        Isbn = isbn;
        Titulo = titulo;
    }
}

class Exemplar
{
    public readonly string Codigo;
    public readonly Livro Livro;
    public bool Emprestado = false;

    public Exemplar(string codigo, Livro livro)
    {
        Codigo = codigo;
        Livro = livro;
    }
}

class Membro
{
    public readonly string Nome;
    public readonly List<Emprestimo> Emprestimos = new List<Emprestimo>();

    public Membro(string nome)
    {
        Nome = nome;
    }
}

class Emprestimo
{
    public const int DIAS_PRAZO = 14;

    public readonly Membro Membro;
    public readonly Exemplar Exemplar;
    public readonly int Saida;

    public Emprestimo(Membro membro, Exemplar exemplar, int saida)
    {
        Membro = membro;
        Exemplar = exemplar;
        Saida = saida;
    }

    public int Vencimento
    {
        get { return Saida + DIAS_PRAZO; }
    }
}

class Biblioteca
{
    const int LIMITE = 2;
    const double MULTA_DIA = 0.5;

    private readonly Dictionary<string, List<Exemplar>> exemplares = new Dictionary<string, List<Exemplar>>();
    private readonly Dictionary<string, LinkedList<Membro>> reservas = new Dictionary<string, LinkedList<Membro>>();

    private LinkedList<Membro> Fila(string isbn)
    {
        if (!reservas.ContainsKey(isbn)) reservas[isbn] = new LinkedList<Membro>();
        return reservas[isbn];
    }

    public void Adicionar(Exemplar ex)
    {
        if (!exemplares.ContainsKey(ex.Livro.Isbn)) exemplares[ex.Livro.Isbn] = new List<Exemplar>();
        exemplares[ex.Livro.Isbn].Add(ex);
    }

    public Emprestimo Emprestar(Membro membro, string isbn, int hoje)
    {
        if (membro.Emprestimos.Count >= LIMITE)
        {
            throw new InvalidOperationException("limite de empréstimos atingido");
        }
        var fila = Fila(isbn);
        if (fila.Count > 0 && fila.First.Value != membro)
        {
            Reservar(membro, isbn);
            return null;
        }
        List<Exemplar> lista;
        exemplares.TryGetValue(isbn, out lista);
        foreach (var ex in lista ?? new List<Exemplar>())
        {
            if (!ex.Emprestado)
            {
                if (fila.Count > 0 && fila.First.Value == membro)
                {
                    fila.RemoveFirst();
                }
                ex.Emprestado = true;
                var emp = new Emprestimo(membro, ex, hoje);
                membro.Emprestimos.Add(emp);
                return emp;
            }
        }
        Reservar(membro, isbn);
        return null;
    }

    public void Reservar(Membro membro, string isbn)
    {
        var fila = Fila(isbn);
        if (!fila.Contains(membro))
        {
            fila.AddLast(membro);
        }
    }

    public double Devolver(Emprestimo emp, int hoje)
    {
        emp.Exemplar.Emprestado = false;
        emp.Membro.Emprestimos.Remove(emp);
        int atraso = Math.Max(0, hoje - emp.Vencimento);
        return atraso * MULTA_DIA;
    }
}

import (
	"errors"
	"fmt"
)

const (
	diasPrazo = 14
	limite    = 2
	multaDia  = 0.5
)

type Livro struct {
	isbn, titulo string
}

type Exemplar struct {
	codigo     string
	livro      Livro
	emprestado bool
}

type Membro struct {
	nome        string
	emprestimos []*Emprestimo
}

type Emprestimo struct {
	membro   *Membro
	exemplar *Exemplar
	saida    int
}

func (e *Emprestimo) vencimento() int { return e.saida + diasPrazo }

type Biblioteca struct {
	exemplares map[string][]*Exemplar
	reservas   map[string][]*Membro
}

func NovaBiblioteca() *Biblioteca {
	return &Biblioteca{exemplares: map[string][]*Exemplar{}, reservas: map[string][]*Membro{}}
}

func (b *Biblioteca) adicionar(ex *Exemplar) {
	b.exemplares[ex.livro.isbn] = append(b.exemplares[ex.livro.isbn], ex)
}

func (b *Biblioteca) emprestar(membro *Membro, isbn string, hoje int) (*Emprestimo, error) {
	if len(membro.emprestimos) >= limite {
		return nil, errors.New("limite de empréstimos atingido")
	}
	fila := b.reservas[isbn]
	if len(fila) > 0 && fila[0] != membro {
		b.reservar(membro, isbn)
		return nil, nil
	}
	for _, ex := range b.exemplares[isbn] {
		if !ex.emprestado {
			if len(fila) > 0 && fila[0] == membro {
				b.reservas[isbn] = fila[1:]
			}
			ex.emprestado = true
			emp := &Emprestimo{membro, ex, hoje}
			membro.emprestimos = append(membro.emprestimos, emp)
			return emp, nil
		}
	}
	b.reservar(membro, isbn)
	return nil, nil
}

func (b *Biblioteca) reservar(membro *Membro, isbn string) {
	for _, m := range b.reservas[isbn] {
		if m == membro {
			return
		}
	}
	b.reservas[isbn] = append(b.reservas[isbn], membro)
}

func (b *Biblioteca) devolver(emp *Emprestimo, hoje int) float64 {
	emp.exemplar.emprestado = false
	lista := emp.membro.emprestimos
	for i, x := range lista {
		if x == emp {
			emp.membro.emprestimos = append(lista[:i], lista[i+1:]...)
			break
		}
	}
	atraso := 0
	if hoje > emp.vencimento() {
		atraso = hoje - emp.vencimento()
	}
	return float64(atraso) * multaDia
}

func main() {
	b := NovaBiblioteca()
	livro := Livro{"111", "Código Limpo"}
	b.adicionar(&Exemplar{"E1", livro, false})
	ana, rui := &Membro{nome: "Ana"}, &Membro{nome: "Rui"}
	e, _ := b.emprestar(ana, "111", 0)
	espera, _ := b.emprestar(rui, "111", 1)
	if espera == nil {
		fmt.Println("nil") // ficou na fila
	}
	fmt.Println(b.devolver(e, 20)) // 3.0 (6 dias de atraso)
	proximo, _ := b.emprestar(rui, "111", 21)
	fmt.Println(proximo.exemplar.codigo)
}

Quem está na frente da fila tem prioridade no exemplar devolvido; os demais continuam esperando.

## Extensões

Renovação (limitada se houver reserva), tipos de membro com limites diferentes (Strategy de política), notificação ao primeiro da fila quando o exemplar volta (Observer) e itens de mídia diferentes de livros (herança leve ou composição em um `Item` base).

## Concorrência

Dois atendentes podem emprestar o último exemplar ao mesmo tempo. Proteja a verificação e a marcação como emprestado numa única seção crítica, ou use atualização condicional no banco.

## Na entrevista

- **"Onde fica a multa?"** Em uma política separada; o empréstimo só informa o atraso.
- **"Livro versus exemplar?"** Livro é a obra, exemplar é a cópia. Reservas são por livro, empréstimos por exemplar.
