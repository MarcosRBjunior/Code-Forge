Um sistema de arquivos em memória é o caso de manual do padrão Composite: diretórios contêm arquivos e outros diretórios, e muitas operações (tamanho total, busca, listagem) tratam os dois do mesmo jeito.

## Requisitos

- Diretórios e arquivos organizados em árvore a partir da raiz.
- Criar diretório, criar arquivo com conteúdo, ler, remover.
- Resolver caminhos como `/docs/notas.txt`.
- Listar diretório e calcular tamanho total de um diretório.
- Buscar por nome ou extensão.

## Entidades

| Entidade | Responsabilidade |
| --- | --- |
| `No` | Base comum: nome, pai, `tamanho()`. |
| `Arquivo` | Folha com conteúdo. |
| `Diretorio` | Composto: dicionário nome para nó. |
| `SistemaArquivos` | Resolve caminhos e expõe as operações. |

Todo nó conhece o pai, o que permite reconstruir o caminho completo. Os filhos ficam em um dicionário por nome, para busca em O(1) e unicidade de nomes dentro do diretório.

from typing import Dict, List, Optional


class No:
    def __init__(self, nome: str):
        self.nome = nome
        self.pai: Optional["Diretorio"] = None

    def tamanho(self) -> int:
        raise NotImplementedError

    def caminho(self) -> str:
        if self.pai is None:
            return "/"
        base = self.pai.caminho()
        return base + self.nome if base == "/" else base + "/" + self.nome


class Arquivo(No):
    def __init__(self, nome: str, conteudo: str = ""):
        super().__init__(nome)
        self.conteudo = conteudo

    def tamanho(self) -> int:
        return len(self.conteudo.encode("utf-8"))


class Diretorio(No):
    def __init__(self, nome: str):
        super().__init__(nome)
        self.filhos: Dict[str, No] = {}

    def adicionar(self, no: No):
        if no.nome in self.filhos:
            raise FileExistsError(f"{no.nome} já existe")
        no.pai = self
        self.filhos[no.nome] = no

    def tamanho(self) -> int:
        return sum(f.tamanho() for f in self.filhos.values())


class SistemaArquivos:
    def __init__(self):
        self.raiz = Diretorio("")

    def _resolver(self, caminho: str) -> No:
        atual: No = self.raiz
        for parte in [p for p in caminho.split("/") if p]:
            if not isinstance(atual, Diretorio) or parte not in atual.filhos:
                raise FileNotFoundError(caminho)
            atual = atual.filhos[parte]
        return atual

    def _pai_e_nome(self, caminho: str):
        pai_caminho, _, nome = caminho.rstrip("/").rpartition("/")
        pai = self._resolver(pai_caminho or "/")
        if not isinstance(pai, Diretorio):
            raise NotADirectoryError(pai_caminho)
        return pai, nome

    def mkdir(self, caminho: str):
        pai, nome = self._pai_e_nome(caminho)
        pai.adicionar(Diretorio(nome))

    def escrever(self, caminho: str, conteudo: str):
        pai, nome = self._pai_e_nome(caminho)
        existente = pai.filhos.get(nome)
        if isinstance(existente, Arquivo):
            existente.conteudo = conteudo
        else:
            pai.adicionar(Arquivo(nome, conteudo))

    def ler(self, caminho: str) -> str:
        no = self._resolver(caminho)
        if not isinstance(no, Arquivo):
            raise IsADirectoryError(caminho)
        return no.conteudo

    def ls(self, caminho: str = "/") -> List[str]:
        no = self._resolver(caminho)
        return [no.nome] if isinstance(no, Arquivo) else sorted(no.filhos)

    def remover(self, caminho: str):
        no = self._resolver(caminho)
        if no.pai is None:
            raise PermissionError("não é possível remover a raiz")
        del no.pai.filhos[no.nome]

    def tamanho(self, caminho: str = "/") -> int:
        return self._resolver(caminho).tamanho()

    def buscar(self, sufixo: str, a_partir: str = "/") -> List[str]:
        achados: List[str] = []

        def visitar(no: No):
            if isinstance(no, Arquivo) and no.nome.endswith(sufixo):
                achados.append(no.caminho())
            elif isinstance(no, Diretorio):
                for f in no.filhos.values():
                    visitar(f)

        visitar(self._resolver(a_partir))
        return sorted(achados)

class No {
  constructor(nome) {
    this.nome = nome;
    this.pai = null;
  }

  tamanho() {
    throw new Error("não implementado");
  }

  caminho() {
    if (this.pai === null) {
      return "/";
    }
    const base = this.pai.caminho();
    return base === "/" ? base + this.nome : base + "/" + this.nome;
  }
}

class Arquivo extends No {
  constructor(nome, conteudo = "") {
    super(nome);
    this.conteudo = conteudo;
  }

  tamanho() {
    return new TextEncoder().encode(this.conteudo).length; // bytes em UTF-8
  }
}

class Diretorio extends No {
  constructor(nome) {
    super(nome);
    this.filhos = new Map();
  }

  adicionar(no) {
    if (this.filhos.has(no.nome)) {
      throw new Error(`${no.nome} já existe`);
    }
    no.pai = this;
    this.filhos.set(no.nome, no);
  }

  tamanho() {
    let soma = 0;
    for (const f of this.filhos.values()) soma += f.tamanho();
    return soma;
  }
}

class SistemaArquivos {
  constructor() {
    this.raiz = new Diretorio("");
  }

  _resolver(caminho) {
    let atual = this.raiz;
    for (const parte of caminho.split("/").filter((p) => p)) {
      if (!(atual instanceof Diretorio) || !atual.filhos.has(parte)) {
        throw new Error(`não encontrado: ${caminho}`);
      }
      atual = atual.filhos.get(parte);
    }
    return atual;
  }

  _paiENome(caminho) {
    const limpo = caminho.replace(/\/+$/, "");
    const i = limpo.lastIndexOf("/");
    const paiCaminho = limpo.slice(0, i);
    const nome = limpo.slice(i + 1);
    const pai = this._resolver(paiCaminho || "/");
    if (!(pai instanceof Diretorio)) {
      throw new Error(`não é um diretório: ${paiCaminho}`);
    }
    return [pai, nome];
  }

  mkdir(caminho) {
    const [pai, nome] = this._paiENome(caminho);
    pai.adicionar(new Diretorio(nome));
  }

  escrever(caminho, conteudo) {
    const [pai, nome] = this._paiENome(caminho);
    const existente = pai.filhos.get(nome);
    if (existente instanceof Arquivo) {
      existente.conteudo = conteudo;
    } else {
      pai.adicionar(new Arquivo(nome, conteudo));
    }
  }

  ler(caminho) {
    const no = this._resolver(caminho);
    if (!(no instanceof Arquivo)) {
      throw new Error(`é um diretório: ${caminho}`);
    }
    return no.conteudo;
  }

    const no = this._resolver(caminho);
    return no instanceof Arquivo ? [no.nome] : [...no.filhos.keys()].sort();
  }

  remover(caminho) {

    if (no.pai === null) {
      throw new Error("não é possível remover a raiz");
    }
    no.pai.filhos.delete(no.nome);
  }

    return this._resolver(caminho).tamanho();

    return achados.sort();

}

import java.nio.charset.StandardCharsets;
import java.util.*;

abstract class No {
    final String nome;
    Diretorio pai = null;

    No(String nome) {
        this.nome = nome;
    }

    abstract int tamanho();

    String caminho() {
        if (pai == null) {
            return "/";
        }
        String base = pai.caminho();
        return base.equals("/") ? base + nome : base + "/" + nome;
    }
}

class Arquivo extends No {
    String conteudo;

    Arquivo(String nome, String conteudo) {
        super(nome);
        this.conteudo = conteudo;
    }

    int tamanho() {
        return conteudo.getBytes(StandardCharsets.UTF_8).length; // bytes em UTF-8
    }
}

class Diretorio extends No {
    final Map<String, No> filhos = new LinkedHashMap<>();

    Diretorio(String nome) {
        super(nome);
    }

    void adicionar(No no) {
        if (filhos.containsKey(no.nome)) {
            throw new IllegalStateException(no.nome + " já existe");
        }
        no.pai = this;
        filhos.put(no.nome, no);
    }

    int tamanho() {
        int soma = 0;
        for (No f : filhos.values()) soma += f.tamanho();
        return soma;
    }
}

class SistemaArquivos {
    final Diretorio raiz = new Diretorio("");

    private No resolver(String caminho) {
        No atual = raiz;
        for (String parte : caminho.split("/")) {
            if (parte.isEmpty()) continue;
            if (!(atual instanceof Diretorio) || !((Diretorio) atual).filhos.containsKey(parte)) {
                throw new NoSuchElementException("não encontrado: " + caminho);
            }
            atual = ((Diretorio) atual).filhos.get(parte);
        }
        return atual;
    }

    private Object[] paiENome(String caminho) {
        String limpo = caminho.replaceAll("/+$", "");
        int i = limpo.lastIndexOf('/');
        String paiCaminho = limpo.substring(0, i);
        String nome = limpo.substring(i + 1);
        No pai = resolver(paiCaminho.isEmpty() ? "/" : paiCaminho);
        if (!(pai instanceof Diretorio)) {
            throw new IllegalStateException("não é um diretório: " + paiCaminho);
        }
        return new Object[] {pai, nome};
    }

    void mkdir(String caminho) {
        Object[] r = paiENome(caminho);
        ((Diretorio) r[0]).adicionar(new Diretorio((String) r[1]));
    }

    void escrever(String caminho, String conteudo) {
        Object[] r = paiENome(caminho);
        Diretorio pai = (Diretorio) r[0];
        String nome = (String) r[1];
        No existente = pai.filhos.get(nome);
        if (existente instanceof Arquivo) {
            ((Arquivo) existente).conteudo = conteudo;
        } else {
            pai.adicionar(new Arquivo(nome, conteudo));
        }
    }

    String ler(String caminho) {
        No no = resolver(caminho);
        if (!(no instanceof Arquivo)) {
            throw new IllegalStateException("é um diretório: " + caminho);
        }
        return ((Arquivo) no).conteudo;
    }

    List<String> ls(String caminho) {
        No no = resolver(caminho);
        if (no instanceof Arquivo) return List.of(no.nome);
        List<String> nomes = new ArrayList<>(((Diretorio) no).filhos.keySet());
        Collections.sort(nomes);
        return nomes;
    }

    void remover(String caminho) {
        No no = resolver(caminho);
        if (no.pai == null) {
            throw new SecurityException("não é possível remover a raiz");
        }
        no.pai.filhos.remove(no.nome);
    }

    int tamanho(String caminho) {
        return resolver(caminho).tamanho();
    }

    List<String> buscar(String sufixo, String aPartirDe) {
        List<String> achados = new ArrayList<>();
        visitar(resolver(aPartirDe), sufixo, achados);
        Collections.sort(achados);
        return achados;
    }

    private void visitar(No no, String sufixo, List<String> achados) {
        if (no instanceof Arquivo && no.nome.endsWith(sufixo)) {
            achados.add(no.caminho());
        } else if (no instanceof Diretorio) {
            for (No f : ((Diretorio) no).filhos.values()) visitar(f, sufixo, achados);
        }
    }
}

static String lista(List<String> xs) {
    List<String> citadas = new ArrayList<>();
    for (String x : xs) citadas.add("'" + x + "'");
    return "[" + String.join(", ", citadas) + "]";
}

}

#include <algorithm>
#include <iostream>
#include <map>
#include <memory>
#include <stdexcept>
#include <string>
#include <vector>
using namespace std;

class Diretorio;

class No {
public:
    string nome;
    Diretorio* pai = nullptr;

    No(string nome) : nome(nome) {}
    virtual ~No() = default;

    virtual int tamanho() const = 0;
    string caminho() const;
};

class Arquivo : public No {
public:
    string conteudo;

    Arquivo(string nome, string conteudo = "") : No(nome), conteudo(conteudo) {}

    int tamanho() const override { return (int)conteudo.size(); } // bytes em UTF-8
};

class Diretorio : public No {
public:
    map<string, unique_ptr<No>> filhos;

    Diretorio(string nome) : No(nome) {}

    void adicionar(unique_ptr<No> no) {
        if (filhos.count(no->nome)) {
            throw runtime_error(no->nome + " já existe");
        }
        no->pai = this;
        string chave = no->nome;
        filhos[chave] = move(no);
    }

    int tamanho() const override {
        int soma = 0;
        for (auto& [nome, f] : filhos) soma += f->tamanho();
        return soma;
    }
};

string No::caminho() const {
    if (pai == nullptr) {
        return "/";
    }
    string base = pai->caminho();
    return base == "/" ? base + nome : base + "/" + nome;
}

class SistemaArquivos {
    Diretorio raiz{""};

    No* resolver(const string& caminho) {
        No* atual = &raiz;
        size_t i = 0;
        while (i <= caminho.size()) {
            size_t j = caminho.find('/', i);
            if (j == string::npos) j = caminho.size();
            string parte = caminho.substr(i, j - i);
            i = j + 1;
            if (parte.empty()) continue;
            auto* dir = dynamic_cast<Diretorio*>(atual);
            if (!dir || !dir->filhos.count(parte)) {
                throw out_of_range("não encontrado: " + caminho);
            }
            atual = dir->filhos[parte].get();
        }
        return atual;
    }

    pair<Diretorio*, string> paiENome(string caminho) {
        while (!caminho.empty() && caminho.back() == '/') caminho.pop_back();
        size_t i = caminho.rfind('/');
        string paiCaminho = caminho.substr(0, i);
        string nome = caminho.substr(i + 1);
        auto* pai = dynamic_cast<Diretorio*>(resolver(paiCaminho.empty() ? "/" : paiCaminho));
        if (!pai) {
            throw logic_error("não é um diretório: " + paiCaminho);
        }
        return {pai, nome};
    }

    void visitar(No* no, const string& sufixo, vector<string>& achados) {
        if (auto* arq = dynamic_cast<Arquivo*>(no)) {
            if (arq->nome.size() >= sufixo.size() && arq->nome.compare(arq->nome.size() - sufixo.size(), sufixo.size(), sufixo) == 0) {
                achados.push_back(arq->caminho());
            }
        } else if (auto* dir = dynamic_cast<Diretorio*>(no)) {
            for (auto& [nome, f] : dir->filhos) visitar(f.get(), sufixo, achados);
        }
    }

public:
    void mkdir(const string& caminho) {
        auto [pai, nome] = paiENome(caminho);
        pai->adicionar(make_unique<Diretorio>(nome));
    }

    void escrever(const string& caminho, const string& conteudo) {
        auto [pai, nome] = paiENome(caminho);
        auto it = pai->filhos.find(nome);
        if (it != pai->filhos.end() && dynamic_cast<Arquivo*>(it->second.get())) {
            static_cast<Arquivo*>(it->second.get())->conteudo = conteudo;
        } else {
            pai->adicionar(make_unique<Arquivo>(nome, conteudo));
        }
    }

    string ler(const string& caminho) {
        auto* arq = dynamic_cast<Arquivo*>(resolver(caminho));
        if (!arq) {
            throw logic_error("é um diretório: " + caminho);
        }
        return arq->conteudo;
    }

    vector<string> ls(const string& caminho = "/") {
        No* no = resolver(caminho);
        if (dynamic_cast<Arquivo*>(no)) return {no->nome};
        vector<string> nomes;
        for (auto& [nome, f] : static_cast<Diretorio*>(no)->filhos) nomes.push_back(nome);
        sort(nomes.begin(), nomes.end());
        return nomes;
    }

    void remover(const string& caminho) {
        No* no = resolver(caminho);
        if (no->pai == nullptr) {
            throw runtime_error("não é possível remover a raiz");
        }
        no->pai->filhos.erase(no->nome);
    }

    int tamanho(const string& caminho = "/") { return resolver(caminho)->tamanho(); }

    vector<string> buscar(const string& sufixo, const string& aPartirDe = "/") {
        vector<string> achados;
        visitar(resolver(aPartirDe), sufixo, achados);
        sort(achados.begin(), achados.end());
        return achados;
    }
};

string lista(const vector<string>& xs) {
    string r = "[";
    for (size_t i = 0; i < xs.size(); i++) r += (i ? ", '" : "'") + xs[i] + "'";
    return r + "]";
}

    SistemaArquivos fs;

using System;
using System.Collections.Generic;
using System.Linq;
using System.Text;

abstract class No
{
    public readonly string Nome;
    public Diretorio Pai = null;

    protected No(string nome)
    {
        Nome = nome;
    }

    public abstract int Tamanho();

    public string Caminho()
    {
        if (Pai == null)
        {
            return "/";
        }
        string baseCaminho = Pai.Caminho();
        return baseCaminho == "/" ? baseCaminho + Nome : baseCaminho + "/" + Nome;
    }
}

class Arquivo : No
{
    public string Conteudo;

    public Arquivo(string nome, string conteudo = "") : base(nome)
    {
        Conteudo = conteudo;
    }

    public override int Tamanho()
    {
        return Encoding.UTF8.GetByteCount(Conteudo); // bytes em UTF-8
    }
}

class Diretorio : No
{
    public readonly Dictionary<string, No> Filhos = new Dictionary<string, No>();

    public Diretorio(string nome) : base(nome) { }

    public void Adicionar(No no)
    {
        if (Filhos.ContainsKey(no.Nome))
        {
            throw new InvalidOperationException(no.Nome + " já existe");
        }
        no.Pai = this;
        Filhos[no.Nome] = no;
    }

    public override int Tamanho()
    {
        return Filhos.Values.Sum(f => f.Tamanho());
    }
}

class SistemaArquivos
{
    private readonly Diretorio raiz = new Diretorio("");

    private No Resolver(string caminho)
    {
        No atual = raiz;
        foreach (var parte in caminho.Split('/').Where(p => p != ""))
        {
            var dir = atual as Diretorio;
            if (dir == null || !dir.Filhos.ContainsKey(parte))
            {
                throw new KeyNotFoundException("não encontrado: " + caminho);
            }
            atual = dir.Filhos[parte];
        }
        return atual;
    }

    private Tuple<Diretorio, string> PaiENome(string caminho)
    {
        string limpo = caminho.TrimEnd('/');
        int i = limpo.LastIndexOf('/');
        string paiCaminho = limpo.Substring(0, i);
        string nome = limpo.Substring(i + 1);
        var pai = Resolver(paiCaminho == "" ? "/" : paiCaminho) as Diretorio;
        if (pai == null)
        {
            throw new InvalidOperationException("não é um diretório: " + paiCaminho);
        }
        return Tuple.Create(pai, nome);
    }

    public void Mkdir(string caminho)
    {
        var r = PaiENome(caminho);
        r.Item1.Adicionar(new Diretorio(r.Item2));
    }

    public void Escrever(string caminho, string conteudo)
    {
        var r = PaiENome(caminho);
        No existente;
        if (r.Item1.Filhos.TryGetValue(r.Item2, out existente) && existente is Arquivo)
        {
            ((Arquivo)existente).Conteudo = conteudo;
        }
        else
        {
            r.Item1.Adicionar(new Arquivo(r.Item2, conteudo));
        }
    }

    public string Ler(string caminho)
    {
        var arq = Resolver(caminho) as Arquivo;
        if (arq == null)
        {
            throw new InvalidOperationException("é um diretório: " + caminho);
        }
        return arq.Conteudo;
    }

    public List<string> Ls(string caminho = "/")
    {
        var no = Resolver(caminho);
        if (no is Arquivo) return new List<string> { no.Nome };
        return ((Diretorio)no).Filhos.Keys.OrderBy(n => n, StringComparer.Ordinal).ToList();
    }

    public void Remover(string caminho)
    {
        var no = Resolver(caminho);
        if (no.Pai == null)
        {
            throw new UnauthorizedAccessException("não é possível remover a raiz");
        }
        no.Pai.Filhos.Remove(no.Nome);
    }

    public int Tamanho(string caminho = "/")
    {
        return Resolver(caminho).Tamanho();
    }

    public List<string> Buscar(string sufixo, string aPartirDe = "/")
    {
        var achados = new List<string>();
        Visitar(Resolver(aPartirDe), sufixo, achados);
        return achados.OrderBy(c => c, StringComparer.Ordinal).ToList();
    }

    private void Visitar(No no, string sufixo, List<string> achados)
    {
        if (no is Arquivo && no.Nome.EndsWith(sufixo, StringComparison.Ordinal))
        {
            achados.Add(no.Caminho());
        }
        else if (no is Diretorio)
        {
            foreach (var f in ((Diretorio)no).Filhos.Values) Visitar(f, sufixo, achados);
        }
    }
}

static string Lista(IEnumerable<string> xs)
{
    return "[" + string.Join(", ", xs.Select(x => "'" + x + "'")) + "]";
}

static void Main()
{
    var fs = new SistemaArquivos();
    fs.Mkdir("/docs");
    fs.Mkdir("/docs/2024");
    fs.Escrever("/docs/notas.txt", "olá mundo");
    fs.Escrever("/docs/2024/plano.txt", "abc");
    fs.Escrever("/leia.md", "x");

    Console.WriteLine(Lista(fs.Buscar(".txt")));
    Console.WriteLine(fs.Ler("/docs/notas.txt"));
    fs.Remover("/docs/2024");
    Console.WriteLine(fs.Tamanho("/docs"));
}

import (
	"errors"
	"fmt"
	"sort"
	"strings"
)

type No interface {
	nome() string
	pai() *Diretorio
	definirPai(d *Diretorio)
	tamanho() int
}

func caminho(n No) string {
	if n.pai() == nil {
		return "/"
	}
	base := caminho(n.pai())
	if base == "/" {
		return base + n.nome()
	}
	return base + "/" + n.nome()
}

type base struct {
	n string
	p *Diretorio
}

func (b *base) nome() string           { return b.n }
func (b *base) pai() *Diretorio        { return b.p }
func (b *base) definirPai(d *Diretorio) { b.p = d }

type Arquivo struct {
	base
	conteudo string
}

func (a *Arquivo) tamanho() int { return len(a.conteudo) } // bytes em UTF-8

type Diretorio struct {
	base
	filhos map[string]No
}

func NovoDiretorio(nome string) *Diretorio {
	return &Diretorio{base: base{n: nome}, filhos: map[string]No{}}
}

func (d *Diretorio) adicionar(no No) error {
	if _, existe := d.filhos[no.nome()]; existe {
		return fmt.Errorf("%s já existe", no.nome())
	}
	no.definirPai(d)
	d.filhos[no.nome()] = no
	return nil
}

func (d *Diretorio) tamanho() int {
	soma := 0
	for _, f := range d.filhos {
		soma += f.tamanho()
	}
	return soma
}

type SistemaArquivos struct {
	raiz *Diretorio
}

func (s *SistemaArquivos) resolver(cam string) (No, error) {
	var atual No = s.raiz
	for _, parte := range strings.Split(cam, "/") {
		if parte == "" {
			continue
		}
		dir, ok := atual.(*Diretorio)
		if !ok {
			return nil, fmt.Errorf("não encontrado: %s", cam)
		}
		filho, existe := dir.filhos[parte]
		if !existe {
			return nil, fmt.Errorf("não encontrado: %s", cam)
		}
		atual = filho
	}
	return atual, nil
}

func (s *SistemaArquivos) paiENome(cam string) (*Diretorio, string, error) {
	limpo := strings.TrimRight(cam, "/")
	i := strings.LastIndex(limpo, "/")
	paiCaminho, nome := limpo[:i], limpo[i+1:]
	if paiCaminho == "" {
		paiCaminho = "/"
	}
	no, err := s.resolver(paiCaminho)
	if err != nil {
		return nil, "", err
	}
	pai, ok := no.(*Diretorio)
	if !ok {
		return nil, "", fmt.Errorf("não é um diretório: %s", paiCaminho)
	}
	return pai, nome, nil
}

func (s *SistemaArquivos) mkdir(cam string) error {
	pai, nome, err := s.paiENome(cam)
	if err != nil {
		return err
	}
	return pai.adicionar(NovoDiretorio(nome))
}

func (s *SistemaArquivos) escrever(cam, conteudo string) error {
	pai, nome, err := s.paiENome(cam)
	if err != nil {
		return err
	}
	if arq, ok := pai.filhos[nome].(*Arquivo); ok {
		arq.conteudo = conteudo
		return nil
	}
	return pai.adicionar(&Arquivo{base: base{n: nome}, conteudo: conteudo})
}

func (s *SistemaArquivos) ler(cam string) (string, error) {
	no, err := s.resolver(cam)
	if err != nil {
		return "", err
	}
	arq, ok := no.(*Arquivo)
	if !ok {
		return "", fmt.Errorf("é um diretório: %s", cam)
	}
	return arq.conteudo, nil
}

func (s *SistemaArquivos) ls(cam string) []string {
	no, _ := s.resolver(cam)
	if arq, ok := no.(*Arquivo); ok {
		return []string{arq.nome()}
	}
	nomes := []string{}
	for n := range no.(*Diretorio).filhos {
		nomes = append(nomes, n)
	}
	sort.Strings(nomes)
	return nomes
}

func (s *SistemaArquivos) remover(cam string) error {
	no, err := s.resolver(cam)
	if err != nil {
		return err
	}
	if no.pai() == nil {
		return errors.New("não é possível remover a raiz")
	}
	delete(no.pai().filhos, no.nome())
	return nil
}

func (s *SistemaArquivos) tamanho(cam string) int {
	no, _ := s.resolver(cam)
	return no.tamanho()
}

func (s *SistemaArquivos) buscar(sufixo, aPartirDe string) []string {
	achados := []string{}
	var visitar func(no No)
	visitar = func(no No) {
		switch n := no.(type) {
		case *Arquivo:
			if strings.HasSuffix(n.nome(), sufixo) {
				achados = append(achados, caminho(n))
			}
		case *Diretorio:
			for _, f := range n.filhos {
				visitar(f)
			}
		}
	}
	inicio, _ := s.resolver(aPartirDe)
	visitar(inicio)
	sort.Strings(achados)
	return achados
}

func lista(xs []string) string {
	citadas := []string{}
	for _, x := range xs {
		citadas = append(citadas, "'"+x+"'")
	}
	return "[" + strings.Join(citadas, ", ") + "]"
}

	fmt.Println(conteudo)
	fs.remover("/docs/2024")

}

O tamanho de um diretório é recursivo e calculado na hora; se isso for lento, guarde o valor e atualize os ancestrais a cada mudança (cache com invalidação). O cálculo de caminho sobe pelos pais, sem precisar armazenar a string.

## Extensões

Permissões (usuário, grupo, leitura e escrita), links simbólicos (um nó que aponta para outro caminho, com cuidado com ciclos), mover e copiar (copiar é recursivo), cotas de espaço e observadores de mudança. Para operações que visitam a árvore (busca, tamanho, listagem recursiva), considere o padrão Visitor, que separa as operações da estrutura.

## Na entrevista
