O sistema de elevadores combina **máquina de estados** com **escalonamento**: cada elevador tem estados e direção, e um controlador decide qual elevador atende cada chamada. É um problema excelente para o entrevistador ver se você separa o que muda (a política de despacho) do que é estável (o movimento de um elevador).

## Requisitos

- **Quantos elevadores e andares?** Configuráveis.
- **Chamadas externas**: botão no andar, com direção (subir ou descer).
- **Chamadas internas**: botão do andar de destino dentro do elevador.
- **Política de despacho**: escolher o melhor elevador para uma chamada externa.
- **Movimento**: o elevador continua na direção atual atendendo paradas pelo caminho (varredura), em vez de ir e voltar.
- Fora de escopo: capacidade de peso, portas, manutenção e prioridade (extensões).

## Entidades

| Entidade | Responsabilidade |
| --- | --- |
| `Direcao` | Enum: SUBINDO, DESCENDO, PARADO |
| `Elevador` | Andar atual, direção, conjunto de paradas pendentes; método `passo()` que avança um andar |
| `Controlador` | Recebe chamadas, escolhe elevador via estratégia, coordena os passos |
| `EstrategiaDespacho` | Interface para escolher o elevador (Strategy) |

```python
from abc import ABC, abstractmethod
from enum import Enum


class Direcao(Enum):
    SUBINDO = 1
    DESCENDO = -1
    PARADO = 0


class Elevador:
    def __init__(self, id_: int, andar: int = 0) -> None:
        self.id = id_
        self.andar = andar
        self.direcao = Direcao.PARADO
        self.paradas: set[int] = set()

    def adicionar_parada(self, andar: int) -> None:
        if andar != self.andar:
            self.paradas.add(andar)

    def _escolher_direcao(self) -> Direcao:
        if not self.paradas:
            return Direcao.PARADO
        acima = any(p > self.andar for p in self.paradas)
        abaixo = any(p < self.andar for p in self.paradas)
        if self.direcao is Direcao.SUBINDO and acima:
            return Direcao.SUBINDO
        if self.direcao is Direcao.DESCENDO and abaixo:
            return Direcao.DESCENDO
        return Direcao.SUBINDO if acima else Direcao.DESCENDO

    def passo(self) -> None:
        self.direcao = self._escolher_direcao()
        if self.direcao is Direcao.PARADO:
            return
        self.andar += self.direcao.value
        self.paradas.discard(self.andar)  # chegou: atende a parada
        if not self.paradas:
            self.direcao = Direcao.PARADO

    def custo(self, andar: int) -> int:
        """Estimativa simples: distância mais uma penalidade por parada pendente."""
        return abs(self.andar - andar) + 2 * len(self.paradas)


class EstrategiaDespacho(ABC):
    @abstractmethod
    def escolher(self, elevadores: list[Elevador], andar: int) -> Elevador: ...


class MaisProximo(EstrategiaDespacho):
    def escolher(self, elevadores, andar):
        return min(elevadores, key=lambda e: (e.custo(andar), e.id))


class Controlador:
    def __init__(self, elevadores: list[Elevador], estrategia: EstrategiaDespacho) -> None:
        self.elevadores = elevadores
        self.estrategia = estrategia

    def chamar(self, andar: int) -> int:
        elevador = self.estrategia.escolher(self.elevadores, andar)
        elevador.adicionar_parada(andar)
        return elevador.id

    def selecionar_destino(self, id_elevador: int, andar: int) -> None:
        self.elevadores[id_elevador].adicionar_parada(andar)

    def passo(self) -> None:
        for e in self.elevadores:
            e.passo()


ctrl = Controlador([Elevador(0, andar=0), Elevador(1, andar=10)], MaisProximo())
assert ctrl.chamar(8) == 1          # o elevador do 10 está mais perto
assert ctrl.chamar(1) == 0
ctrl.selecionar_destino(1, 3)
for _ in range(10):
    ctrl.passo()
assert ctrl.elevadores[0].andar == 1
assert ctrl.elevadores[1].paradas == set()
print("elevadores ok")
```

```javascript
const Direcao = Object.freeze({ SUBINDO: 1, DESCENDO: -1, PARADO: 0 });

class Elevador {
  constructor(id, andar = 0) {
    this.id = id;
    this.andar = andar;
    this.direcao = Direcao.PARADO;
    this.paradas = new Set();
  }

  adicionarParada(andar) {
    if (andar !== this.andar) {
      this.paradas.add(andar);
    }
  }

  _escolherDirecao() {
    if (this.paradas.size === 0) {
      return Direcao.PARADO;
    }
    const acima = [...this.paradas].some((p) => p > this.andar);
    const abaixo = [...this.paradas].some((p) => p < this.andar);
    if (this.direcao === Direcao.SUBINDO && acima) {
      return Direcao.SUBINDO;
    }
    if (this.direcao === Direcao.DESCENDO && abaixo) {
      return Direcao.DESCENDO;
    }
    return acima ? Direcao.SUBINDO : Direcao.DESCENDO;
  }

  passo() {
    this.direcao = this._escolherDirecao();
    if (this.direcao === Direcao.PARADO) {
      return;
    }
    this.andar += this.direcao;
    this.paradas.delete(this.andar); // chegou: atende a parada
    if (this.paradas.size === 0) {
      this.direcao = Direcao.PARADO;
    }
  }

  // Estimativa simples: distância mais uma penalidade por parada pendente.
  custo(andar) {
    return Math.abs(this.andar - andar) + 2 * this.paradas.size;
  }
}

      const ce = e.custo(andar);
      const cm = melhor.custo(andar);

    });
  }
}

    return elevador.id;

    this.elevadores[idElevador].adicionarParada(andar);
  }

  passo() {
    for (const e of this.elevadores) {
      e.passo();
    }
  }
}
```

```java
import java.util.*;

enum Direcao {
    SUBINDO(1), DESCENDO(-1), PARADO(0);

    final int valor;

    Direcao(int valor) {
        this.valor = valor;
    }
}

class Elevador {
    final int id;
    int andar;
    Direcao direcao = Direcao.PARADO;
    final Set<Integer> paradas = new HashSet<>();

    Elevador(int id, int andar) {
        this.id = id;
        this.andar = andar;
    }

    void adicionarParada(int andar) {
        if (andar != this.andar) {
            paradas.add(andar);
        }
    }

    private Direcao escolherDirecao() {
        if (paradas.isEmpty()) {
            return Direcao.PARADO;
        }
        boolean acima = false, abaixo = false;
        for (int p : paradas) {
            if (p > andar) acima = true;
            if (p < andar) abaixo = true;
        }
        if (direcao == Direcao.SUBINDO && acima) {
            return Direcao.SUBINDO;
        }
        if (direcao == Direcao.DESCENDO && abaixo) {
            return Direcao.DESCENDO;
        }
        return acima ? Direcao.SUBINDO : Direcao.DESCENDO;
    }

    void passo() {
        direcao = escolherDirecao();
        if (direcao == Direcao.PARADO) {
            return;
        }
        andar += direcao.valor;
        paradas.remove(andar); // chegou: atende a parada
        if (paradas.isEmpty()) {
            direcao = Direcao.PARADO;
        }
    }

    // Estimativa simples: distância mais uma penalidade por parada pendente.
    int custo(int andar) {
        return Math.abs(this.andar - andar) + 2 * paradas.size();
    }
}

interface EstrategiaDespacho {
    Elevador escolher(List<Elevador> elevadores, int andar);
}

class MaisProximo implements EstrategiaDespacho {
    public Elevador escolher(List<Elevador> elevadores, int andar) {
        Elevador melhor = elevadores.get(0);
        for (Elevador e : elevadores) {
            int ce = e.custo(andar), cm = melhor.custo(andar);
            if (ce < cm || (ce == cm && e.id < melhor.id)) melhor = e;
        }
        return melhor;
    }
}

    final List<Elevador> elevadores;
    private final EstrategiaDespacho estrategia;

        this.elevadores = elevadores;
        this.estrategia = estrategia;

        Elevador elevador = estrategia.escolher(elevadores, andar);

        return elevador.id;

        elevadores.get(idElevador).adicionarParada(andar);
    }

    void passo() {
        for (Elevador e : elevadores) {
            e.passo();
        }
    }

}
```

```csharp
        {
            return Direcao.Parado;

        if (Direcao == Direcao.Descendo && abaixo)

            return Direcao.Descendo;
        }
        return acima ? Direcao.Subindo : Direcao.Descendo;
    }

    public void Passo()
    {
        Direcao = EscolherDirecao();
        if (Direcao == Direcao.Parado)
        {
            return;
        }
        Andar += (int)Direcao;
        Paradas.Remove(Andar); // chegou: atende a parada
        if (Paradas.Count == 0)
        {
            Direcao = Direcao.Parado;
        }
```

```go
import (
	"fmt"
)

type Direcao int

const (
	Subindo  Direcao = 1
	Descendo Direcao = -1
	Parado   Direcao = 0
)

type Elevador struct {
	id      int
	andar   int
	direcao Direcao
	paradas map[int]bool
}

func NovoElevador(id, andar int) *Elevador {
	return &Elevador{id: id, andar: andar, direcao: Parado, paradas: map[int]bool{}}
}

func (e *Elevador) adicionarParada(andar int) {
	if andar != e.andar {
		e.paradas[andar] = true
	}
}

func (e *Elevador) escolherDirecao() Direcao {
	if len(e.paradas) == 0 {
		return Parado
	}
	acima, abaixo := false, false
	for p := range e.paradas {
		if p > e.andar {
			acima = true
		}
		if p < e.andar {
			abaixo = true
		}
	}
	if e.direcao == Subindo && acima {
		return Subindo
	}
	if e.direcao == Descendo && abaixo {
		return Descendo
	}
	if acima {
		return Subindo
	}
	return Descendo
}

func (e *Elevador) passo() {
	e.direcao = e.escolherDirecao()
	if e.direcao == Parado {
		return
	}
	e.andar += int(e.direcao)
	delete(e.paradas, e.andar) // chegou: atende a parada
	if len(e.paradas) == 0 {
		e.direcao = Parado
	}
}

func abs(x int) int {

		return -x
	}

}

// Estimativa simples: distância mais uma penalidade por parada pendente.
func (e *Elevador) custo(andar int) int {
	return abs(e.andar-andar) + 2*len(e.paradas)
}

type EstrategiaDespacho interface {
	escolher(elevadores []*Elevador, andar int) *Elevador
}

type MaisProximo struct{}

func (MaisProximo) escolher(elevadores []*Elevador, andar int) *Elevador {
	melhor := elevadores[0]
	for _, e := range elevadores {
		ce, cm := e.custo(andar), melhor.custo(andar)
		if ce < cm || (ce == cm && e.id < melhor.id) {
			melhor = e
		}
	}
	return melhor
}

type Controlador struct {
	elevadores []*Elevador
	estrategia EstrategiaDespacho
}

func (c *Controlador) chamar(andar int) int {
	elevador := c.estrategia.escolher(c.elevadores, andar)
	elevador.adicionarParada(andar)
	return elevador.id
}

func (c *Controlador) selecionarDestino(idElevador, andar int) {
	c.elevadores[idElevador].adicionarParada(andar)
}

func (c *Controlador) passo() {
	for _, e := range c.elevadores {
		e.passo()
	}
}

func verificar(condicao bool) {
	if !condicao {
		panic("verificação falhou")
	}
}

func main() {
	ctrl := &Controlador{[]*Elevador{NovoElevador(0, 0), NovoElevador(1, 10)}, MaisProximo{}}
	verificar(ctrl.chamar(8) == 1) // o elevador do 10 está mais perto
	verificar(ctrl.chamar(1) == 0)
	ctrl.selecionarDestino(1, 3)
	for i := 0; i < 10; i++ {
		ctrl.passo()
	}
	verificar(ctrl.elevadores[0].andar == 1)
	verificar(len(ctrl.elevadores[1].paradas) == 0)
	fmt.Println("elevadores ok")
}
```

## Decisões que merecem fala

- **Varredura**: manter a direção enquanto houver paradas à frente reduz viagens; é a ideia do algoritmo SCAN de discos.
- **Strategy no despacho**: trocar por uma política que minimiza o tempo médio de espera não altera `Elevador`.
- **Conjunto de paradas**: evita duplicatas quando várias pessoas apertam o mesmo andar.
- **Controlador separado**: o elevador conhece só seu próprio movimento; quem enxerga o conjunto é o controlador.

<Callout type="tip" title="Estado e direção">
Se pedirem portas e manutenção, estenda os estados (PARADO_PORTA_ABERTA, MANUTENCAO) e use o padrão State em vez de crescer os ifs de `passo`.
</Callout>

## Extensões

Capacidade e peso, andares restritos por cartão, modo bombeiro (cancela chamadas e vai ao térreo) e prioridade. Para escala, vários prédios compartilham o mesmo controlador abstrato. Para concorrência, o controlador precisa de uma trava ou fila de eventos, já que botões chegam de várias threads.
