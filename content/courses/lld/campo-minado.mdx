Campo minado é um problema de LLD que parece de jogo, mas avalia três coisas concretas: separar o **estado interno** do tabuleiro da sua **visão para o jogador**, tratar a regra de abrir células com **propagação** (a abertura em cascata) e garantir que o primeiro clique nunca seja uma mina. O entrevistador também observa se o aleatório é injetável, para o jogo poder ser testado.

## Requisitos

Perguntas de esclarecimento:

- **Tamanho e número de minas?** Configuráveis (linhas, colunas, minas).
- **Ações do jogador?** Abrir uma célula, marcar e desmarcar bandeira.
- **Fim de jogo?** Vitória quando todas as células sem mina estão abertas; derrota ao abrir uma mina.
- **Primeiro clique seguro?** Sim, as minas são sorteadas depois do primeiro clique.
- **Interface?** Apenas a lógica; a exibição fica separada.

## Entidades

| Entidade | Responsabilidade |
| --- | --- |
| `Celula` | Tem mina ou não, está aberta, tem bandeira, guarda o número de minas vizinhas |
| `Tabuleiro` | Grade de células, distribuição das minas, contagem de vizinhas, abertura em cascata |
| `StatusJogo` | Enum: EM_ANDAMENTO, VITORIA, DERROTA |
| `Jogo` | Regras de alto nível: abrir, marcar, atualizar o status |

A decisão central: a `Celula` guarda o fato (tem mina?), e quem decide **o que mostrar** é a camada de exibição. Assim a mesma lógica serve para console, web ou testes.

```python
import random
from enum import Enum


class StatusJogo(Enum):
    EM_ANDAMENTO = "em andamento"
    VITORIA = "vitória"
    DERROTA = "derrota"


class Celula:
    def __init__(self) -> None:
        self.mina = False
        self.aberta = False
        self.bandeira = False
        self.vizinhas = 0


class Jogo:
    def __init__(self, linhas: int, colunas: int, minas: int, rng: random.Random | None = None) -> None:
        if minas >= linhas * colunas:
            raise ValueError("minas demais")
        self.linhas, self.colunas, self.minas = linhas, colunas, minas
        self.rng = rng or random.Random()
        self.grade = [[Celula() for _ in range(colunas)] for _ in range(linhas)]
        self.status = StatusJogo.EM_ANDAMENTO
        self._minas_sorteadas = False

    def _vizinhos(self, r: int, c: int):
        for dr in (-1, 0, 1):
            for dc in (-1, 0, 1):
                nr, nc = r + dr, c + dc
                if (dr or dc) and 0 <= nr < self.linhas and 0 <= nc < self.colunas:
                    yield nr, nc

    def _sortear_minas(self, seguro: tuple[int, int]) -> None:
        livres = [(r, c) for r in range(self.linhas) for c in range(self.colunas) if (r, c) != seguro]
        for r, c in self.rng.sample(livres, self.minas):
            self.grade[r][c].mina = True
        for r in range(self.linhas):
            for c in range(self.colunas):
                self.grade[r][c].vizinhas = sum(self.grade[nr][nc].mina for nr, nc in self._vizinhos(r, c))
        self._minas_sorteadas = True

    def marcar(self, r: int, c: int) -> None:
        cel = self.grade[r][c]
        if not cel.aberta:
            cel.bandeira = not cel.bandeira

    def abrir(self, r: int, c: int) -> StatusJogo:
        if self.status is not StatusJogo.EM_ANDAMENTO:
            return self.status
        if not self._minas_sorteadas:
            self._sortear_minas((r, c))
        cel = self.grade[r][c]
        if cel.bandeira or cel.aberta:
            return self.status
        if cel.mina:
            cel.aberta = True
            self.status = StatusJogo.DERROTA
            return self.status
        self._abrir_cascata(r, c)
        if self._venceu():
            self.status = StatusJogo.VITORIA
        return self.status

    def _abrir_cascata(self, r: int, c: int) -> None:
        pilha = [(r, c)]  # iterativo: evita estouro de recursão em grades grandes
        while pilha:
            cr, cc = pilha.pop()
            cel = self.grade[cr][cc]
            if cel.aberta or cel.bandeira or cel.mina:
                continue
            cel.aberta = True
            if cel.vizinhas == 0:
                pilha.extend(self._vizinhos(cr, cc))

    def _venceu(self) -> bool:
        return all(c.aberta or c.mina for linha in self.grade for c in linha)


jogo = Jogo(5, 5, 3, random.Random(42))
assert jogo.abrir(0, 0) in (StatusJogo.EM_ANDAMENTO, StatusJogo.VITORIA)
assert not jogo.grade[0][0].mina  # primeiro clique sempre seguro
minas = [(r, c) for r in range(5) for c in range(5) if jogo.grade[r][c].mina]
assert len(minas) == 3
r, c = minas[0]
assert jogo.abrir(r, c) is StatusJogo.DERROTA
print("campo minado ok")
```

```javascript
const StatusJogo = Object.freeze({
  EM_ANDAMENTO: "em andamento",
  VITORIA: "vitória",
  DERROTA: "derrota",
});

class Celula {
  constructor() {
    this.mina = false;
    this.aberta = false;
    this.bandeira = false;
    this.vizinhas = 0;
  }
}

class Jogo {
  // rng: função que devolve um número em [0, 1), para poder fixar a semente nos testes
  constructor(linhas, colunas, minas, rng = Math.random) {
    if (minas >= linhas * colunas) {
      throw new RangeError("minas demais");
    }
    this.linhas = linhas;
    this.colunas = colunas;
    this.minas = minas;
    this.rng = rng;
    this.grade = Array.from({ length: linhas }, () => Array.from({ length: colunas }, () => new Celula()));
    this.status = StatusJogo.EM_ANDAMENTO;
    this._minasSorteadas = false;
  }

  *_vizinhos(r, c) {
    for (const dr of [-1, 0, 1]) {
      for (const dc of [-1, 0, 1]) {
        const nr = r + dr;
        const nc = c + dc;
        if ((dr || dc) && nr >= 0 && nr < this.linhas && nc >= 0 && nc < this.colunas) {
          yield [nr, nc];
        }
      }
    }
  }

  _sortearMinas(seguro) {
    const livres = [];
    for (let r = 0; r < this.linhas; r++) {
      for (let c = 0; c < this.colunas; c++) {
        if (r !== seguro[0] || c !== seguro[1]) livres.push([r, c]);
      }
    }
    // sorteio sem repetição (Fisher-Yates parcial)
    for (let i = 0; i < this.minas; i++) {
      const j = i + Math.floor(this.rng() * (livres.length - i));
      [livres[i], livres[j]] = [livres[j], livres[i]];
      const [r, c] = livres[i];
      this.grade[r][c].mina = true;
    }
    for (let r = 0; r < this.linhas; r++) {
      for (let c = 0; c < this.colunas; c++) {
        let soma = 0;
        for (const [nr, nc] of this._vizinhos(r, c)) soma += this.grade[nr][nc].mina ? 1 : 0;
        this.grade[r][c].vizinhas = soma;
      }
    }
    this._minasSorteadas = true;
  }

  marcar(r, c) {
    const cel = this.grade[r][c];
    if (!cel.aberta) {
      cel.bandeira = !cel.bandeira;
    }
  }

  abrir(r, c) {
    if (this.status !== StatusJogo.EM_ANDAMENTO) {
      return this.status;
    }
    if (!this._minasSorteadas) {
      this._sortearMinas([r, c]);
    }
    const cel = this.grade[r][c];
    if (cel.bandeira || cel.aberta) {
      return this.status;
    }
    if (cel.mina) {
      cel.aberta = true;
      this.status = StatusJogo.DERROTA;
      return this.status;
    }
    this._abrirCascata(r, c);
    if (this._venceu()) {
      this.status = StatusJogo.VITORIA;
    }
    return this.status;
  }

  _abrirCascata(r, c) {
    const pilha = [[r, c]]; // iterativo: evita estouro de recursão em grades grandes
    while (pilha.length > 0) {
      const [cr, cc] = pilha.pop();
      const cel = this.grade[cr][cc];
      if (cel.aberta || cel.bandeira || cel.mina) {
        continue;
      }
      cel.aberta = true;
      if (cel.vizinhas === 0) {
        pilha.push(...this._vizinhos(cr, cc));
      }
    }
  }

  _venceu() {
    return this.grade.every((linha) => linha.every((c) => c.aberta || c.mina));
  }
}

// Gerador pseudoaleatório com semente (mulberry32), para o teste ser reproduzível.
function mulberry32(semente) {
  let a = semente;
  return () => {
    a = (a + 0x6d2b79f5) | 0;
    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

function verificar(condicao) {
  if (!condicao) throw new Error("verificação falhou");
}

const jogo = new Jogo(5, 5, 3, mulberry32(42));
verificar([StatusJogo.EM_ANDAMENTO, StatusJogo.VITORIA].includes(jogo.abrir(0, 0)));
verificar(!jogo.grade[0][0].mina); // primeiro clique sempre seguro
const minas = [];
for (let r = 0; r < 5; r++) {
  for (let c = 0; c < 5; c++) {
    if (jogo.grade[r][c].mina) minas.push([r, c]);
  }
}
verificar(minas.length === 3);
const [r, c] = minas[0];
verificar(jogo.abrir(r, c) === StatusJogo.DERROTA);
console.log("campo minado ok");
```

```java
import java.util.*;

enum StatusJogo { EM_ANDAMENTO, VITORIA, DERROTA }

class Celula {
    boolean mina = false;
    boolean aberta = false;
    boolean bandeira = false;
    int vizinhas = 0;
}

class Jogo {
    private final int linhas, colunas, minas;
    private final Random rng;
    final Celula[][] grade;
    StatusJogo status = StatusJogo.EM_ANDAMENTO;
    private boolean minasSorteadas = false;

    Jogo(int linhas, int colunas, int minas, Random rng) {
        if (minas >= linhas * colunas) {
            throw new IllegalArgumentException("minas demais");
        }
        this.linhas = linhas;
        this.colunas = colunas;
        this.minas = minas;
        this.rng = rng;
        this.grade = new Celula[linhas][colunas];
        for (int r = 0; r < linhas; r++) for (int c = 0; c < colunas; c++) grade[r][c] = new Celula();
    }

    private List<int[]> vizinhos(int r, int c) {
        List<int[]> resultado = new ArrayList<>();
        for (int dr = -1; dr <= 1; dr++) {
            for (int dc = -1; dc <= 1; dc++) {
                int nr = r + dr, nc = c + dc;
                if ((dr != 0 || dc != 0) && nr >= 0 && nr < linhas && nc >= 0 && nc < colunas) {
                    resultado.add(new int[] {nr, nc});
                }
            }
        }
        return resultado;
    }

    private void sortearMinas(int seguroR, int seguroC) {
        List<int[]> livres = new ArrayList<>();
        for (int r = 0; r < linhas; r++) {
            for (int c = 0; c < colunas; c++) {
                if (r != seguroR || c != seguroC) livres.add(new int[] {r, c});
            }
        }
        Collections.shuffle(livres, rng); // sorteio sem repetição
        for (int i = 0; i < minas; i++) {
            grade[livres.get(i)[0]][livres.get(i)[1]].mina = true;
        }
        for (int r = 0; r < linhas; r++) {
            for (int c = 0; c < colunas; c++) {
                int soma = 0;
                for (int[] v : vizinhos(r, c)) if (grade[v[0]][v[1]].mina) soma++;
                grade[r][c].vizinhas = soma;
            }
        }
        minasSorteadas = true;
    }

    void marcar(int r, int c) {
        Celula cel = grade[r][c];
        if (!cel.aberta) {
            cel.bandeira = !cel.bandeira;
        }
    }

    StatusJogo abrir(int r, int c) {
        if (status != StatusJogo.EM_ANDAMENTO) {
            return status;
        }
        if (!minasSorteadas) {
            sortearMinas(r, c);
        }
        Celula cel = grade[r][c];
        if (cel.bandeira || cel.aberta) {
            return status;
        }
        if (cel.mina) {
            cel.aberta = true;
            status = StatusJogo.DERROTA;
            return status;
        }
        abrirCascata(r, c);
        if (venceu()) {
            status = StatusJogo.VITORIA;
        }
        return status;
    }

    private void abrirCascata(int r, int c) {
        Deque<int[]> pilha = new ArrayDeque<>(); // iterativo: evita estouro de recursão em grades grandes
        pilha.push(new int[] {r, c});
        while (!pilha.isEmpty()) {
            int[] p = pilha.pop();
            Celula cel = grade[p[0]][p[1]];
            if (cel.aberta || cel.bandeira || cel.mina) {
                continue;
            }
            cel.aberta = true;
            if (cel.vizinhas == 0) {
                for (int[] v : vizinhos(p[0], p[1])) pilha.push(v);
            }
        }
    }

    private boolean venceu() {
        for (Celula[] linha : grade) for (Celula c : linha) if (!(c.aberta || c.mina)) return false;
        return true;
    }
}

static void verificar(boolean condicao) {
    if (!condicao) throw new AssertionError("verificação falhou");
}

public static void main(String[] args) {
    Jogo jogo = new Jogo(5, 5, 3, new Random(42));
    StatusJogo s = jogo.abrir(0, 0);
    verificar(s == StatusJogo.EM_ANDAMENTO || s == StatusJogo.VITORIA);
    verificar(!jogo.grade[0][0].mina); // primeiro clique sempre seguro
    List<int[]> minas = new ArrayList<>();
    for (int r = 0; r < 5; r++) for (int c = 0; c < 5; c++) if (jogo.grade[r][c].mina) minas.add(new int[] {r, c});
    verificar(minas.size() == 3);
    verificar(jogo.abrir(minas.get(0)[0], minas.get(0)[1]) == StatusJogo.DERROTA);
    System.out.println("campo minado ok");
}
```

```cpp
#include <algorithm>
#include <iostream>
#include <random>
#include <stdexcept>
#include <utility>
#include <vector>
using namespace std;

enum class StatusJogo { EM_ANDAMENTO, VITORIA, DERROTA };

struct Celula {
    bool mina = false;
    bool aberta = false;
    bool bandeira = false;
    int vizinhas = 0;
};

class Jogo {
    int linhas, colunas, minas;
    mt19937 rng;
    bool minasSorteadas = false;

    vector<pair<int, int>> vizinhos(int r, int c) const {
        vector<pair<int, int>> resultado;
        for (int dr = -1; dr <= 1; dr++) {
            for (int dc = -1; dc <= 1; dc++) {
                int nr = r + dr, nc = c + dc;
                if ((dr || dc) && nr >= 0 && nr < linhas && nc >= 0 && nc < colunas) resultado.push_back({nr, nc});
            }
        }
        return resultado;
    }

    void sortearMinas(int seguroR, int seguroC) {
        vector<pair<int, int>> livres;
        for (int r = 0; r < linhas; r++) {
            for (int c = 0; c < colunas; c++) {
                if (r != seguroR || c != seguroC) livres.push_back({r, c});
            }
        }
        shuffle(livres.begin(), livres.end(), rng); // sorteio sem repetição
        for (int i = 0; i < minas; i++) grade[livres[i].first][livres[i].second].mina = true;
        for (int r = 0; r < linhas; r++) {
            for (int c = 0; c < colunas; c++) {
                int soma = 0;
                for (auto [nr, nc] : vizinhos(r, c)) soma += grade[nr][nc].mina ? 1 : 0;
                grade[r][c].vizinhas = soma;
            }
        }
        minasSorteadas = true;
    }

    void abrirCascata(int r, int c) {
        vector<pair<int, int>> pilha = {{r, c}}; // iterativo: evita estouro de recursão em grades grandes
        while (!pilha.empty()) {
            auto [cr, cc] = pilha.back();
            pilha.pop_back();
            Celula& cel = grade[cr][cc];
            if (cel.aberta || cel.bandeira || cel.mina) continue;
            cel.aberta = true;
            if (cel.vizinhas == 0) {
                for (auto v : vizinhos(cr, cc)) pilha.push_back(v);
            }
        }
    }

    bool venceu() const {
        for (auto& linha : grade) for (auto& c : linha) if (!(c.aberta || c.mina)) return false;
        return true;
    }

public:
    vector<vector<Celula>> grade;
    StatusJogo status = StatusJogo::EM_ANDAMENTO;

    Jogo(int linhas, int colunas, int minas, unsigned semente)
        : linhas(linhas), colunas(colunas), minas(minas), rng(semente) {
        if (minas >= linhas * colunas) throw invalid_argument("minas demais");
        grade.assign(linhas, vector<Celula>(colunas));
    }

    void marcar(int r, int c) {
        Celula& cel = grade[r][c];
        if (!cel.aberta) cel.bandeira = !cel.bandeira;
    }

    StatusJogo abrir(int r, int c) {
        if (status != StatusJogo::EM_ANDAMENTO) return status;
        if (!minasSorteadas) sortearMinas(r, c);
        Celula& cel = grade[r][c];
        if (cel.bandeira || cel.aberta) return status;
        if (cel.mina) {
            cel.aberta = true;
            status = StatusJogo::DERROTA;
            return status;
        }
        abrirCascata(r, c);
        if (venceu()) status = StatusJogo::VITORIA;
        return status;
    }
};

void verificar(bool condicao) {
    if (!condicao) throw runtime_error("verificação falhou");
}

int main() {
```

```csharp
using System;
using System.Collections.Generic;
using System.Linq;

enum StatusJogo { EmAndamento, Vitoria, Derrota }

class Celula
{
    public bool Mina = false;
    public bool Aberta = false;
    public bool Bandeira = false;
    public int Vizinhas = 0;
}

class Jogo
{
    private readonly int linhas, colunas, minas;
    private readonly Random rng;
    public readonly Celula[,] Grade;
    public StatusJogo Status = StatusJogo.EmAndamento;
    private bool minasSorteadas = false;

    public Jogo(int linhas, int colunas, int minas, Random rng)
    {
        if (minas >= linhas * colunas)
        {
            throw new ArgumentException("minas demais");
        }
        this.linhas = linhas;
        this.colunas = colunas;
        this.minas = minas;
        this.rng = rng;
        Grade = new Celula[linhas, colunas];
        for (int r = 0; r < linhas; r++) for (int c = 0; c < colunas; c++) Grade[r, c] = new Celula();
    }

    private List<int[]> Vizinhos(int r, int c)
    {
        var resultado = new List<int[]>();
        for (int dr = -1; dr <= 1; dr++)
        {
            for (int dc = -1; dc <= 1; dc++)
            {
                int nr = r + dr, nc = c + dc;
                if ((dr != 0 || dc != 0) && nr >= 0 && nr < linhas && nc >= 0 && nc < colunas)
                {
                    resultado.Add(new[] { nr, nc });
                }
            }
        }
        return resultado;
    }

    private void SortearMinas(int seguroR, int seguroC)
    {
        var livres = new List<int[]>();
        for (int r = 0; r < linhas; r++)
        {
            for (int c = 0; c < colunas; c++)
            {
                if (r != seguroR || c != seguroC) livres.Add(new[] { r, c });
            }
        }
        // sorteio sem repetição
        foreach (var p in livres.OrderBy(x => rng.Next()).Take(minas)) Grade[p[0], p[1]].Mina = true;
        for (int r = 0; r < linhas; r++)
        {
            for (int c = 0; c < colunas; c++)
            {
                Grade[r, c].Vizinhas = Vizinhos(r, c).Count(v => Grade[v[0], v[1]].Mina);
            }
        }
        minasSorteadas = true;
    }

    public void Marcar(int r, int c)
    {
        var cel = Grade[r, c];
        if (!cel.Aberta)
        {
            cel.Bandeira = !cel.Bandeira;
        }
    }

    public StatusJogo Abrir(int r, int c)
    {
        if (Status != StatusJogo.EmAndamento)
        {
            return Status;
        }
        if (!minasSorteadas)
        {
            SortearMinas(r, c);
        }
        var cel = Grade[r, c];
        if (cel.Bandeira || cel.Aberta)
        {
            return Status;
        }
        if (cel.Mina)
        {
            cel.Aberta = true;
            Status = StatusJogo.Derrota;
            return Status;
        }
        AbrirCascata(r, c);
        if (Venceu())
        {
            Status = StatusJogo.Vitoria;
        }
        return Status;
    }

    private void AbrirCascata(int r, int c)
    {
        var pilha = new Stack<int[]>(); // iterativo: evita estouro de recursão em grades grandes
        pilha.Push(new[] { r, c });
        while (pilha.Count > 0)
        {
            var p = pilha.Pop();
            var cel = Grade[p[0], p[1]];
            if (cel.Aberta || cel.Bandeira || cel.Mina)
            {
                continue;
            }
            cel.Aberta = true;
            if (cel.Vizinhas == 0)
            {
                foreach (var v in Vizinhos(p[0], p[1])) pilha.Push(v);
            }
        }
    }

    private bool Venceu()
    {
        foreach (var cel in Grade) if (!(cel.Aberta || cel.Mina)) return false;
        return true;
    }
}

static void Verificar(bool condicao)
{
    if (!condicao) throw new Exception("verificação falhou");
}

static void Main()
{
    var jogo = new Jogo(5, 5, 3, new Random(42));
    var s = jogo.Abrir(0, 0);
    Verificar(s == StatusJogo.EmAndamento || s == StatusJogo.Vitoria);
    Verificar(!jogo.Grade[0, 0].Mina); // primeiro clique sempre seguro
    var minas = new List<int[]>();
    for (int r = 0; r < 5; r++) for (int c = 0; c < 5; c++) if (jogo.Grade[r, c].Mina) minas.Add(new[] { r, c });
    Verificar(minas.Count == 3);
    Verificar(jogo.Abrir(minas[0][0], minas[0][1]) == StatusJogo.Derrota);
    Console.WriteLine("campo minado ok");
}
```

```go
import (
	"fmt"
	"math/rand"
)

type StatusJogo string

const (
	EmAndamento StatusJogo = "em andamento"
	Vitoria     StatusJogo = "vitória"
	Derrota     StatusJogo = "derrota"
)

type Celula struct {
	mina, aberta, bandeira bool
	vizinhas               int
}

type Pos struct{ r, c int }

type Jogo struct {
	linhas, colunas, minas int
	rng                    *rand.Rand
	grade                  [][]*Celula
	status                 StatusJogo
	minasSorteadas         bool
}

func NovoJogo(linhas, colunas, minas int, rng *rand.Rand) (*Jogo, error) {
	if minas >= linhas*colunas {
		return nil, fmt.Errorf("minas demais")
	}
	j := &Jogo{linhas: linhas, colunas: colunas, minas: minas, rng: rng, status: EmAndamento}
	j.grade = make([][]*Celula, linhas)
	for r := range j.grade {
		j.grade[r] = make([]*Celula, colunas)
		for c := range j.grade[r] {
			j.grade[r][c] = &Celula{}
		}
	}
	return j, nil
}

func (j *Jogo) vizinhos(r, c int) []Pos {
	resultado := []Pos{}
	for dr := -1; dr <= 1; dr++ {
		for dc := -1; dc <= 1; dc++ {
			nr, nc := r+dr, c+dc
			if (dr != 0 || dc != 0) && nr >= 0 && nr < j.linhas && nc >= 0 && nc < j.colunas {
				resultado = append(resultado, Pos{nr, nc})
			}
		}
	}
	return resultado
}

func (j *Jogo) sortearMinas(seguro Pos) {
	livres := []Pos{}
	for r := 0; r < j.linhas; r++ {
		for c := 0; c < j.colunas; c++ {
			if (Pos{r, c}) != seguro {
				livres = append(livres, Pos{r, c})
			}
		}
	}
	j.rng.Shuffle(len(livres), func(a, b int) { livres[a], livres[b] = livres[b], livres[a] }) // sorteio sem repetição
	for _, p := range livres[:j.minas] {
		j.grade[p.r][p.c].mina = true
	}
	for r := 0; r < j.linhas; r++ {
		for c := 0; c < j.colunas; c++ {
			soma := 0
			for _, v := range j.vizinhos(r, c) {
				if j.grade[v.r][v.c].mina {
					soma++
				}
			}
			j.grade[r][c].vizinhas = soma
		}
	}
	j.minasSorteadas = true
}

func (j *Jogo) marcar(r, c int) {
	if cel := j.grade[r][c]; !cel.aberta {
		cel.bandeira = !cel.bandeira
	}
}

func (j *Jogo) abrir(r, c int) StatusJogo {
	if j.status != EmAndamento {
		return j.status
	}
	if !j.minasSorteadas {
		j.sortearMinas(Pos{r, c})
	}
	cel := j.grade[r][c]
	if cel.bandeira || cel.aberta {
		return j.status
	}
	if cel.mina {
		cel.aberta = true
		j.status = Derrota
		return j.status
	}
	j.abrirCascata(r, c)
	if j.venceu() {
		j.status = Vitoria
	}
	return j.status
}

func (j *Jogo) abrirCascata(r, c int) {
	pilha := []Pos{{r, c}} // iterativo: evita estouro de recursão em grades grandes
	for len(pilha) > 0 {
		p := pilha[len(pilha)-1]
		pilha = pilha[:len(pilha)-1]
		cel := j.grade[p.r][p.c]
		if cel.aberta || cel.bandeira || cel.mina {
			continue
		}
		cel.aberta = true
		if cel.vizinhas == 0 {
			pilha = append(pilha, j.vizinhos(p.r, p.c)...)
		}
	}
}

func (j *Jogo) venceu() bool {
	for _, linha := range j.grade {
		for _, c := range linha {
			if !(c.aberta || c.mina) {
				return false
			}
		}
	}
	return true
}

func verificar(condicao bool) {
	if !condicao {
		panic("verificação falhou")
	}
}

func main() {
	jogo, _ := NovoJogo(5, 5, 3, rand.New(rand.NewSource(42)))
	s := jogo.abrir(0, 0)
	verificar(s == EmAndamento || s == Vitoria)
	verificar(!jogo.grade[0][0].mina) // primeiro clique sempre seguro
	minas := []Pos{}
	for r := 0; r < 5; r++ {
		for c := 0; c < 5; c++ {
			if jogo.grade[r][c].mina {
				minas = append(minas, Pos{r, c})
			}
		}
	}
	verificar(len(minas) == 3)
	verificar(jogo.abrir(minas[0].r, minas[0].c) == Derrota)
	fmt.Println("campo minado ok")
}
```

## Decisões que merecem fala

- **Primeiro clique seguro**: o sorteio é adiado até o primeiro `abrir`, excluindo aquela célula.
- **Cascata iterativa**: uma pilha em vez de recursão evita estouro em tabuleiros grandes. As células com zero vizinhas minadas propagam a abertura.
- **RNG injetável**: com `random.Random(42)` os testes são determinísticos.
- **Bandeira protege**: célula marcada não abre, evitando um clique acidental.

## Padrões e extensões

O jogo é simples e quase não pede padrões. Quando houver variação, considere **Strategy** para o sorteio de minas (aleatório, posições fixas para teste) e **Observer** para a interface ser notificada de células abertas. Extensões comuns: níveis de dificuldade (Factory ou Builder de configuração), contador de bandeiras e tempo, desfazer com Memento.

<Callout type="tip" title="Teste a cascata">
Um bom teste: um tabuleiro com uma única mina num canto deve abrir quase tudo com um clique no canto oposto. Se a cascata falhar, esse teste mostra na hora.
</Callout>
