Um app de corridas faz o pareamento entre passageiros e motoristas próximos, calcula preço e acompanha o ciclo de vida da corrida. No LLD, concentre-se em duas peças: a estratégia de pareamento e a máquina de estados da corrida.

## Requisitos

- Motoristas informam posição e disponibilidade.
- Passageiro solicita corrida de A para B.
- O sistema escolhe o motorista disponível mais próximo.
- Corrida passa por: solicitada, aceita, em andamento, concluída (ou cancelada).
- Preço calculado por distância e tempo.

## Entidades

| Entidade | Responsabilidade |
| --- | --- |
| `Local` | Coordenadas e distância. |
| `Motorista` | Posição e disponibilidade. |
| `Corrida` | Passageiro, motorista, origem, destino e estado. |
| `Pareamento` | Strategy: escolhe o motorista. |
| `PrecoStrategy` | Strategy: calcula o valor. |
| `Servico` | Orquestra solicitação e conclusão. |

import math
from dataclasses import dataclass
from enum import Enum
from typing import List, Optional


@dataclass(frozen=True)
class Local:
    x: float
    y: float

    def distancia(self, outro: "Local") -> float:
        return math.hypot(self.x - outro.x, self.y - outro.y)


class Motorista:
    def __init__(self, nome: str, local: Local):
        self.nome, self.local, self.disponivel = nome, local, True


class Estado(Enum):
    SOLICITADA = 1
    ACEITA = 2
    EM_ANDAMENTO = 3
    CONCLUIDA = 4
    CANCELADA = 5


class Corrida:
    TRANSICOES = {
        Estado.SOLICITADA: {Estado.ACEITA, Estado.CANCELADA},
        Estado.ACEITA: {Estado.EM_ANDAMENTO, Estado.CANCELADA},
        Estado.EM_ANDAMENTO: {Estado.CONCLUIDA},
    }

    def __init__(self, passageiro: str, origem: Local, destino: Local):
        self.passageiro, self.origem, self.destino = passageiro, origem, destino
        self.estado = Estado.SOLICITADA
        self.motorista: Optional[Motorista] = None
        self.preco = 0.0

    def mudar(self, novo: Estado):
        if novo not in self.TRANSICOES.get(self.estado, set()):
            raise ValueError(f"{self.estado.name} -> {novo.name} inválido")
        self.estado = novo


class MaisProximo:
    def escolher(self, motoristas: List[Motorista], origem: Local) -> Optional[Motorista]:
        livres = [m for m in motoristas if m.disponivel]
        return min(livres, key=lambda m: m.local.distancia(origem), default=None)


class PrecoPorKm:
    def __init__(self, base=5.0, por_km=2.0):
        self.base, self.por_km = base, por_km

    def calcular(self, corrida: Corrida) -> float:
        return round(self.base + self.por_km * corrida.origem.distancia(corrida.destino), 2)


class Servico:
    def __init__(self, motoristas, pareamento, preco):
        self.motoristas, self.pareamento, self.preco = motoristas, pareamento, preco

    def solicitar(self, passageiro: str, origem: Local, destino: Local) -> Corrida:
        corrida = Corrida(passageiro, origem, destino)
        m = self.pareamento.escolher(self.motoristas, origem)
        if m is None:
            corrida.mudar(Estado.CANCELADA)
            return corrida
        m.disponivel = False           # reserva o motorista imediatamente
        corrida.motorista = m
        corrida.mudar(Estado.ACEITA)
        corrida.preco = self.preco.calcular(corrida)
        return corrida

    def iniciar(self, c: Corrida):
        c.mudar(Estado.EM_ANDAMENTO)

    def concluir(self, c: Corrida):
        c.mudar(Estado.CONCLUIDA)
        c.motorista.local = c.destino
        c.motorista.disponivel = True

    this.y = y;

    return Math.hypot(this.x - outro.x, this.y - outro.y);

    [Estado.SOLICITADA]: new Set([Estado.ACEITA, Estado.CANCELADA]),

import java.math.*;
import java.util.*;

class Local {
    final double x, y;

    Local(double x, double y) {
        this.x = x;
        this.y = y;
    }

    double distancia(Local outro) {
        return Math.hypot(x - outro.x, y - outro.y);
    }

    @Override
    public String toString() {
        return "Local(x=" + x + ", y=" + y + ")";
    }
}

class Motorista {
    final String nome;
    Local local;
    boolean disponivel = true;

    Motorista(String nome, Local local) {
        this.nome = nome;
        this.local = local;
    }
}

enum Estado {
    SOLICITADA, ACEITA, EM_ANDAMENTO, CONCLUIDA, CANCELADA;

    Set<Estado> permitidos() {
        switch (this) {
            case SOLICITADA: return EnumSet.of(ACEITA, CANCELADA);
            case ACEITA: return EnumSet.of(EM_ANDAMENTO, CANCELADA);
            case EM_ANDAMENTO: return EnumSet.of(CONCLUIDA);
            default: return EnumSet.noneOf(Estado.class);
        }
    }
}

class Corrida {
    final String passageiro;
    final Local origem, destino;
    Estado estado = Estado.SOLICITADA;
    Motorista motorista = null;
    double preco = 0.0;

    Corrida(String passageiro, Local origem, Local destino) {
        this.passageiro = passageiro;
        this.origem = origem;
        this.destino = destino;
    }

    void mudar(Estado novo) {
        if (!estado.permitidos().contains(novo)) {
            throw new IllegalStateException(estado.name() + " -> " + novo.name() + " inválido");
        }
        estado = novo;
    }
}

interface Pareamento {
    Motorista escolher(List<Motorista> motoristas, Local origem);
}

    public Motorista escolher(List<Motorista> motoristas, Local origem) {

        }
        return melhor;
    }
}

interface Preco {
    double calcular(Corrida corrida);
}

    return new BigDecimal(x).setScale(casas, RoundingMode.HALF_EVEN).doubleValue();
}

string nome(Estado e) {
    switch (e) {
        case Estado::SOLICITADA: return "SOLICITADA";
        case Estado::ACEITA: return "ACEITA";
        case Estado::EM_ANDAMENTO: return "EM_ANDAMENTO";
        case Estado::CONCLUIDA: return "CONCLUIDA";
        default: return "CANCELADA";
    }
}

set<Estado> permitidos(Estado e) {
    switch (e) {
        case Estado::SOLICITADA: return {Estado::ACEITA, Estado::CANCELADA};
        case Estado::ACEITA: return {Estado::EM_ANDAMENTO, Estado::CANCELADA};
        case Estado::EM_ANDAMENTO: return {Estado::CONCLUIDA};
        default: return {};
    }
}

struct Corrida {
    string passageiro;
    Local origem, destino;
    Estado estado = Estado::SOLICITADA;
    Motorista* motorista = nullptr;
    double preco = 0.0;

    void mudar(Estado novo) {
        if (!permitidos(estado).count(novo)) {
            throw logic_error(nome(estado) + " -> " + nome(novo) + " inválido");
        }
        estado = novo;
    }
};

public:
    virtual ~Pareamento() = default;
    virtual Motorista* escolher(vector<Motorista>& motoristas, const Local& origem) = 0;

public:

        for (auto& m : motoristas) {
            if (m.disponivel && (!melhor || m.local.distancia(origem) < melhor->local.distancia(origem))) melhor = &m;
        }
        return melhor;
    }
};

class Preco {
public:
    virtual ~Preco() = default;
    virtual double calcular(const Corrida& corrida) const = 0;
};

    char texto[64];
    snprintf(texto, sizeof texto, "%.*f", casas, x);
    return strtod(texto, nullptr);
}

        X = x;
        Y = y;
    }

    public double Distancia(Local outro)
    {
        return Math.Sqrt((X - outro.X) * (X - outro.X) + (Y - outro.Y) * (Y - outro.Y));
    }

    public override string ToString()
    {
        return "Local(x=" + X + ", y=" + Y + ")";
    }
}

class Motorista
{
    public readonly string Nome;
    public Local Local;
    public bool Disponivel = true;

    public Motorista(string nome, Local local)
    {
        Nome = nome;
        Local = local;
    }

enum Estado { SOLICITADA, ACEITA, EM_ANDAMENTO, CONCLUIDA, CANCELADA }

class Corrida
{
    static readonly Dictionary<Estado, HashSet<Estado>> TRANSICOES = new Dictionary<Estado, HashSet<Estado>>
    {
        { Estado.SOLICITADA, new HashSet<Estado> { Estado.ACEITA, Estado.CANCELADA } },
        { Estado.ACEITA, new HashSet<Estado> { Estado.EM_ANDAMENTO, Estado.CANCELADA } },
        { Estado.EM_ANDAMENTO, new HashSet<Estado> { Estado.CONCLUIDA } },
    };

    public readonly string Passageiro;
    public readonly Local Origem, Destino;
    public Estado Estado = Estado.SOLICITADA;
    public Motorista Motorista = null;
    public double Preco = 0.0;

    public Corrida(string passageiro, Local origem, Local destino)
    {
        Passageiro = passageiro;
        Origem = origem;
        Destino = destino;
    }

    public void Mudar(Estado novo)
    {
        HashSet<Estado> permitidos;
        if (!TRANSICOES.TryGetValue(Estado, out permitidos) || !permitidos.Contains(novo))
        {
            throw new InvalidOperationException(Estado + " -> " + novo + " inválido");
        }
        Estado = novo;
    }
}

interface IPareamento
{
    Motorista Escolher(List<Motorista> motoristas, Local origem);
}

class MaisProximo : IPareamento
{
    public Motorista Escolher(List<Motorista> motoristas, Local origem)
    {
        return motoristas.Where(m => m.Disponivel).OrderBy(m => m.Local.Distancia(origem)).FirstOrDefault();
    }
}

interface IPreco
{
    double Calcular(Corrida corrida);
}

// o F2 do C# arredonda a forma decimal curta, e o Math.Round(x, 2) multiplica por 100 antes (2.675 vira 267.5);
// o Python arredonda o valor binário exato do double, com empate para o par. Aqui o valor exato
// (mantissa * 2^expoente) é montado em decimal, com 28 dígitos, e arredondado para o par.
static string FixoPar(double x, int casas)
{
    long bits = BitConverter.DoubleToInt64Bits(Math.Abs(x));
    int campo = (int)(bits >> 52);
    decimal d = bits & 0xFFFFFFFFFFFFFL;
    if (campo > 0) d += 1L << 52;
    for (int e = Math.Max(campo, 1) - 1075; e > 0; e--) d *= 2;
    for (int e = Math.Max(campo, 1) - 1075; e < 0; e++) d /= 2;
    if (x < 0) d = -d;
    return Math.Round(d, casas, MidpointRounding.ToEven).ToString("F" + casas, CultureInfo.InvariantCulture);
}

// round() do Python: o mesmo arredondamento, de volta para double
static double Arred(double x, int casas)
{
    return double.Parse(FixoPar(x, casas), CultureInfo.InvariantCulture);
}

class PrecoPorKm : IPreco
{
    private readonly double baseValor, porKm;

    public PrecoPorKm(double baseValor = 5.0, double porKm = 2.0)
    {
        this.baseValor = baseValor;
        this.porKm = porKm;
    }

    public double Calcular(Corrida corrida)
    {
        return Arred(baseValor + porKm * corrida.Origem.Distancia(corrida.Destino), 2);
    }
}

class Servico
{
    private readonly List<Motorista> motoristas;
    private readonly IPareamento pareamento;
    private readonly IPreco preco;

    public Servico(List<Motorista> motoristas, IPareamento pareamento, IPreco preco)
    {
        this.motoristas = motoristas;
        this.pareamento = pareamento;
        this.preco = preco;
    }

    public Corrida Solicitar(string passageiro, Local origem, Local destino)
    {
        var corrida = new Corrida(passageiro, origem, destino);
        var m = pareamento.Escolher(motoristas, origem);
        if (m == null)
        {
            corrida.Mudar(Estado.CANCELADA);
            return corrida;
        }
        m.Disponivel = false; // reserva o motorista imediatamente
        corrida.Motorista = m;
        corrida.Mudar(Estado.ACEITA);
        corrida.Preco = preco.Calcular(corrida);
        return corrida;
    }

    public void Iniciar(Corrida c)
    {
        c.Mudar(Estado.EM_ANDAMENTO);
    }

    public void Concluir(Corrida c)
    {
        c.Mudar(Estado.CONCLUIDA);
        c.Motorista.Local = c.Destino;
        c.Motorista.Disponivel = true;
    }
}

static void Main()
{
    var ms = new List<Motorista> { new Motorista("Rui", new Local(0, 1)), new Motorista("Léo", new Local(5, 5)) };
    var svc = new Servico(ms, new MaisProximo(), new PrecoPorKm());
    var c1 = svc.Solicitar("Ana", new Local(0, 0), new Local(3, 4));
    Console.WriteLine(c1.Motorista.Nome + " " + c1.Preco); // Rui 15.0
    var c2 = svc.Solicitar("Bia", new Local(0, 0), new Local(1, 1));
    Console.WriteLine(c2.Motorista.Nome); // Léo (Rui já ocupado)
    svc.Iniciar(c1);
    svc.Concluir(c1);
    Console.WriteLine(ms[0].Disponivel.ToString().ToLower() + " " + ms[0].Local);
}

		}
	}
	return fmt.Errorf("%v -> %v inválido", c.estado, novo)

type Pareamento interface {

type MaisProximo struct{}

func (MaisProximo) escolher(motoristas []*Motorista, origem Local) *Motorista {
	var melhor *Motorista
	for _, m := range motoristas {
		if m.disponivel && (melhor == nil || m.local.distancia(origem) < melhor.local.distancia(origem)) {
			melhor = m
		}
	}
	return melhor
}

	calcular(c *Corrida) float64
}

// round() do Python: o strconv arredonda o valor exato do double, com empate para o par
func arred(x float64, casas int) float64 {
	r, _ := strconv.ParseFloat(strconv.FormatFloat(x, 'f', casas, 64), 64)
	return r
}

type PrecoPorKm struct {
	base, porKm float64
}

func (p PrecoPorKm) calcular(c *Corrida) float64 {
	return arred(p.base+p.porKm*c.origem.distancia(c.destino), 2)
}

type Servico struct {
	motoristas []*Motorista
	pareamento Pareamento
	preco      Preco
}

func (s *Servico) solicitar(passageiro string, origem, destino Local) *Corrida {
	corrida := &Corrida{passageiro: passageiro, origem: origem, destino: destino, estado: Solicitada}
	m := s.pareamento.escolher(s.motoristas, origem)
	if m == nil {
		corrida.mudar(Cancelada)
		return corrida
	}
	m.disponivel = false // reserva o motorista imediatamente
	corrida.motorista = m
	corrida.mudar(Aceita)
	corrida.preco = s.preco.calcular(corrida)
	return corrida
}

func (s *Servico) iniciar(c *Corrida) error { return c.mudar(EmAndamento) }

func (s *Servico) concluir(c *Corrida) error {
	if err := c.mudar(Concluida); err != nil {
		return errors.New("não foi possível concluir: " + err.Error())
	}
	c.motorista.local = c.destino
	c.motorista.disponivel = true
	return nil
}

Marcar o motorista como indisponível no mesmo passo em que ele é escolhido evita que dois passageiros peguem o mesmo carro. Em um sistema real isso é uma operação atômica.

## Extensões

Oferecer a corrida ao motorista e esperar aceite (com timeout, passando ao próximo candidato), preço dinâmico por demanda (nova `PrecoStrategy`), avaliações, tipos de veículo e, em escala, índice espacial (grade ou quadtree) para achar motoristas próximos sem varrer todos. Esse índice é o assunto de busca por proximidade em System Design.

## Concorrência

Muitas solicitações simultâneas disputam poucos motoristas. Proteja a transição de disponível para ocupado com lock por motorista ou comparação-e-troca; quem perde tenta o próximo candidato.

## Na entrevista

- **"E se o motorista recusar?"** A corrida volta a SOLICITADA e o pareamento exclui quem recusou.
- **"Como achar o mais próximo com 100 mil motoristas?"** Índice geoespacial; varrer todos não escala.
