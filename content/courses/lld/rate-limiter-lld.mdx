Um rate limiter decide se uma requisição passa ou é recusada. O LLD pede a interface limpa, algoritmos intercambiáveis e cuidado com concorrência e consumo de memória. Aqui implementamos token bucket e janela deslizante atrás da mesma interface.

## Requisitos

- Limitar requisições por chave (usuário, IP ou API key).
- Algoritmos trocáveis.
- Tempo injetável para testes.
- Seguro para várias threads.

## Algoritmos

| Algoritmo | Ideia | Pontos fortes e fracos |
| --- | --- | --- |
| Janela fixa | Contador por intervalo | Simples; permite rajada dupla na virada |
| Janela deslizante | Guarda os instantes recentes | Preciso; usa mais memória |
| Token bucket | Fichas repostas a taxa constante | Permite rajadas controladas; memória mínima |

```python
import threading
from collections import deque
from typing import Callable, Dict


class Limiter:
    def permitir(self, chave: str) -> bool:
        raise NotImplementedError


class TokenBucket(Limiter):
    def __init__(self, capacidade: int, repor_por_seg: float, relogio: Callable[[], float]):
        self.capacidade, self.taxa, self.relogio = capacidade, repor_por_seg, relogio
        self.baldes: Dict[str, list] = {}       # chave -> [fichas, ultimo instante]
        self._lock = threading.Lock()

    def permitir(self, chave: str) -> bool:
        with self._lock:
            agora = self.relogio()
            fichas, ultimo = self.baldes.get(chave, [self.capacidade, agora])
            fichas = min(self.capacidade, fichas + (agora - ultimo) * self.taxa)
            ok = fichas >= 1
            if ok:
                fichas -= 1
            self.baldes[chave] = [fichas, agora]
            return ok


class JanelaDeslizante(Limiter):
    def __init__(self, limite: int, janela: float, relogio: Callable[[], float]):
        self.limite, self.janela, self.relogio = limite, janela, relogio
        self.registros: Dict[str, deque] = {}
        self._lock = threading.Lock()

    def permitir(self, chave: str) -> bool:
        with self._lock:
            agora = self.relogio()
            q = self.registros.setdefault(chave, deque())
            while q and q[0] <= agora - self.janela:
                q.popleft()
            if len(q) < self.limite:
                q.append(agora)
                return True
            return False
```

```java
import java.util.*;
import java.util.function.DoubleSupplier;

interface Limiter {
    boolean permitir(String chave);
}

class TokenBucket implements Limiter {
    private final int capacidade;
    private final double taxa;
    private final DoubleSupplier relogio;
    private final Map<String, double[]> baldes = new HashMap<>(); // chave -> {fichas, último instante}

    TokenBucket(int capacidade, double reporPorSeg, DoubleSupplier relogio) {
        this.capacidade = capacidade;
        this.taxa = reporPorSeg;
        this.relogio = relogio;
    }

    public synchronized boolean permitir(String chave) {
        double agora = relogio.getAsDouble();
        double[] balde = baldes.getOrDefault(chave, new double[] {capacidade, agora});
        double fichas = Math.min(capacidade, balde[0] + (agora - balde[1]) * taxa);
        boolean ok = fichas >= 1;
        if (ok) {
            fichas -= 1;
        }
        baldes.put(chave, new double[] {fichas, agora});
        return ok;
    }
}

class JanelaDeslizante implements Limiter {
    private final int limite;
    private final double janela;
    private final DoubleSupplier relogio;
    private final Map<String, Deque<Double>> registros = new HashMap<>();

    JanelaDeslizante(int limite, double janela, DoubleSupplier relogio) {
        this.limite = limite;
        this.janela = janela;
        this.relogio = relogio;
    }

    public synchronized boolean permitir(String chave) {
        double agora = relogio.getAsDouble();
        Deque<Double> q = registros.computeIfAbsent(chave, k -> new ArrayDeque<>());
        while (!q.isEmpty() && q.peekFirst() <= agora - janela) {
            q.pollFirst();
        }
        if (q.size() < limite) {
            q.addLast(agora);
            return true;
        }
        return false;
    }
}

static String mostrar(List<Boolean> lista) {
    List<String> partes = new ArrayList<>();
    for (boolean b : lista) partes.add(b ? "True" : "False");
    return "[" + String.join(", ", partes) + "]";
}

    TokenBucket tb = new TokenBucket(3, 1, () -> t[0]);
    List<Boolean> primeira = new ArrayList<>();
    for (int i = 0; i < 4; i++) primeira.add(tb.permitir("u1"));
    System.out.println(mostrar(primeira)); // [True, True, True, False]
    t[0] = 2.0;

    for (int i = 0; i < 3; i++) segunda.add(tb.permitir("u1"));
```

```cpp
#include <deque>
#include <functional>
#include <iostream>
#include <map>
#include <mutex>
#include <string>
#include <utility>
#include <vector>
using namespace std;

class Limiter {
public:
    virtual ~Limiter() = default;
    virtual bool permitir(const string& chave) = 0;
};

class TokenBucket : public Limiter {
    int capacidade;
    double taxa;
    function<double()> relogio;
    map<string, pair<double, double>> baldes; // chave -> {fichas, último instante}
    mutex trava;

public:
    TokenBucket(int capacidade, double reporPorSeg, function<double()> relogio)
        : capacidade(capacidade), taxa(reporPorSeg), relogio(relogio) {}

    bool permitir(const string& chave) override {
        lock_guard<mutex> g(trava);
        double agora = relogio();
        auto it = baldes.find(chave);
        pair<double, double> balde = it != baldes.end() ? it->second : make_pair((double)capacidade, agora);
        double fichas = min((double)capacidade, balde.first + (agora - balde.second) * taxa);
        bool ok = fichas >= 1;
        if (ok) {
            fichas -= 1;
        }
        baldes[chave] = {fichas, agora};
        return ok;
    }
};

class JanelaDeslizante : public Limiter {
    int limite;
    double janela;
    function<double()> relogio;
    map<string, deque<double>> registros;
    mutex trava;

public:
    JanelaDeslizante(int limite, double janela, function<double()> relogio)
        : limite(limite), janela(janela), relogio(relogio) {}

    bool permitir(const string& chave) override {
        lock_guard<mutex> g(trava);
        double agora = relogio();
        deque<double>& q = registros[chave];
        while (!q.empty() && q.front() <= agora - janela) {
            q.pop_front();
        }
        if ((int)q.size() < limite) {
            q.push_back(agora);
            return true;
        }
        return false;
    }
};

    for (size_t i = 0; i < lista.size(); i++) r += string(i ? ", " : "") + (lista[i] ? "True" : "False");
    return r + "]";
}
```

```csharp
using System;
using System.Collections.Generic;
using System.Linq;

interface ILimiter
{
    bool Permitir(string chave);
}

class TokenBucket : ILimiter
{
    private readonly int capacidade;
    private readonly double taxa;
    private readonly Func<double> relogio;
    private readonly Dictionary<string, double[]> baldes = new Dictionary<string, double[]>(); // chave -> {fichas, último instante}
    private readonly object trava = new object();

    public TokenBucket(int capacidade, double reporPorSeg, Func<double> relogio)
    {
        this.capacidade = capacidade;
        taxa = reporPorSeg;
        this.relogio = relogio;
    }

    public bool Permitir(string chave)
    {
        lock (trava)
        {
            double agora = relogio();
            double[] balde;
            if (!baldes.TryGetValue(chave, out balde)) balde = new double[] { capacidade, agora };
            double fichas = Math.Min(capacidade, balde[0] + (agora - balde[1]) * taxa);
            bool ok = fichas >= 1;
            if (ok)
            {
                fichas -= 1;
            }
            baldes[chave] = new[] { fichas, agora };
            return ok;
        }
    }
}

class JanelaDeslizante : ILimiter
{
    private readonly int limite;
    private readonly double janela;
    private readonly Func<double> relogio;
    private readonly Dictionary<string, LinkedList<double>> registros = new Dictionary<string, LinkedList<double>>();
    private readonly object trava = new object();

    public JanelaDeslizante(int limite, double janela, Func<double> relogio)
    {
        this.limite = limite;
        this.janela = janela;
        this.relogio = relogio;
    }

    public bool Permitir(string chave)
    {
        lock (trava)
        {
            double agora = relogio();
            if (!registros.ContainsKey(chave)) registros[chave] = new LinkedList<double>();
            var q = registros[chave];
            while (q.Count > 0 && q.First.Value <= agora - janela)
            {
                q.RemoveFirst();
            }
            if (q.Count < limite)
            {
                q.AddLast(agora);
                return true;
            }
            return false;
        }
    }
}
```

```go
import (
	"fmt"
	"strings"
	"sync"
)

type Limiter interface {
	permitir(chave string) bool
}

type balde struct {
	fichas, ultimo float64
}

type TokenBucket struct {
	capacidade float64
	taxa       float64
	relogio    func() float64
	baldes     map[string]balde
	trava      sync.Mutex
}

func NovoTokenBucket(capacidade int, reporPorSeg float64, relogio func() float64) *TokenBucket {
	return &TokenBucket{capacidade: float64(capacidade), taxa: reporPorSeg, relogio: relogio, baldes: map[string]balde{}}
}

func (tb *TokenBucket) permitir(chave string) bool {
	tb.trava.Lock()
	defer tb.trava.Unlock()
	agora := tb.relogio()
	b, existe := tb.baldes[chave]
	if !existe {
		b = balde{tb.capacidade, agora}
	}
	fichas := b.fichas + (agora-b.ultimo)*tb.taxa
	if fichas > tb.capacidade {
		fichas = tb.capacidade
	}
	ok := fichas >= 1
	if ok {
		fichas--
	}
	tb.baldes[chave] = balde{fichas, agora}
	return ok
}

type JanelaDeslizante struct {
	limite    int
	janela    float64
	relogio   func() float64
	registros map[string][]float64
	trava     sync.Mutex
}

func (jd *JanelaDeslizante) permitir(chave string) bool {
	jd.trava.Lock()
	defer jd.trava.Unlock()
	agora := jd.relogio()
	q := jd.registros[chave]
	for len(q) > 0 && q[0] <= agora-jd.janela {
		q = q[1:]
	}
	if len(q) < jd.limite {
		jd.registros[chave] = append(q, agora)
		return true
	}
	jd.registros[chave] = q
	return false
}

func mostrar(lista []bool) string {
	partes := []string{}
	for _, b := range lista {
		if b {
			partes = append(partes, "True")
		} else {
			partes = append(partes, "False")
		}
	}
	return "[" + strings.Join(partes, ", ") + "]"
}

func main() {
	t := 0.0
	tb := NovoTokenBucket(3, 1, func() float64 { return t })
	primeira := []bool{}
	for i := 0; i < 4; i++ {
		primeira = append(primeira, tb.permitir("u1"))
	}
	fmt.Println(mostrar(primeira)) // [True, True, True, False]
	t = 2.0
	segunda := []bool{}
	for i := 0; i < 3; i++ {
		segunda = append(segunda, tb.permitir("u1"))
	}
	fmt.Println(mostrar(segunda)) // [True, True, False]
	fmt.Println(tb.permitir("u2")) // chaves independentes: true

	jd := &JanelaDeslizante{limite: 2, janela: 10, relogio: func() float64 { return t }, registros: map[string][]float64{}}
	t = 0
	a, b, c := jd.permitir("a"), jd.permitir("a"), jd.permitir("a")
	fmt.Println(a, b, c) // true true false
	t = 10
	fmt.Println(jd.permitir("a")) // true
}
```

Token bucket não precisa de um processo reabastecendo: as fichas são calculadas na leitura pelo tempo decorrido. É o truque que mantém a memória em dois números por chave.

## Concorrência e memória

O lock global é correto e simples, mas serializa todas as chaves. Alternativas: um lock por chave (ou fatiado por hash) e operações atômicas num armazenamento compartilhado. Em sistema distribuído, a contagem mora num Redis com script atômico, assunto de System Design. Também é preciso expirar chaves inativas, senão o dicionário cresce sem fim.

## Extensões

Limites por plano do cliente (política por chave), resposta com tempo restante até poder tentar de novo, regras compostas (por segundo e por dia) combinadas, e métricas de requisições recusadas.

## Na entrevista

- **"Qual algoritmo escolher?"** Token bucket quando rajadas curtas são aceitáveis; janela deslizante quando a precisão importa.
- **"Por que injetar o relógio?"** Para testar sem dormir.
