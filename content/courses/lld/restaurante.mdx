Um sistema de restaurante gira em torno do ciclo de vida de um pedido: o garçom registra itens, a cozinha prepara, o pedido é servido e a conta é fechada. O núcleo do problema é uma máquina de estados bem definida e a divisão de conta.

## Requisitos

- Cardápio com itens e preços.
- Mesas com pedidos abertos; itens podem ser adicionados enquanto o pedido está aberto.
- A cozinha avança o estado de cada item: pedido, preparando, pronto, servido.
- Fechar a conta com taxa de serviço e divisão entre pessoas.

## Entidades

| Entidade | Responsabilidade |
| --- | --- |
| `ItemCardapio` | Nome e preço. |
| `ItemPedido` | Item do cardápio com quantidade e estado. |
| `Pedido` | Itens de uma mesa; calcula o total. |
| `Mesa` | Número e pedido atual. |
| `Restaurante` | Cardápio, mesas e fila da cozinha. |

Estados válidos formam uma sequência. Centralizar a transição num método impede saltos como servir algo que nem foi preparado.

```python
from dataclasses import dataclass, field
from enum import Enum
from collections import deque
from typing import List, Optional


class Estado(Enum):
    PEDIDO = 1
    PREPARANDO = 2
    PRONTO = 3
    SERVIDO = 4


@dataclass(frozen=True)
class ItemCardapio:
    nome: str
    preco: float


@dataclass
class ItemPedido:
    prato: ItemCardapio
    qtd: int = 1
    estado: Estado = Estado.PEDIDO

    def avancar(self):
        if self.estado is Estado.SERVIDO:
            raise ValueError("item já servido")
        self.estado = Estado(self.estado.value + 1)

    @property
    def subtotal(self):
        return self.prato.preco * self.qtd


class Pedido:
    def __init__(self):
        self.itens: List[ItemPedido] = []
        self.fechado = False

    def adicionar(self, prato: ItemCardapio, qtd: int = 1) -> ItemPedido:
        if self.fechado:
            raise ValueError("pedido fechado")
        item = ItemPedido(prato, qtd)
        self.itens.append(item)
        return item

    def total(self, taxa_servico: float = 0.10) -> float:
        base = sum(i.subtotal for i in self.itens)
        return round(base * (1 + taxa_servico), 2)

    def dividir(self, pessoas: int) -> List[float]:
        total_centavos = round(self.total() * 100)
        base, resto = divmod(total_centavos, pessoas)
        return [(base + (1 if i < resto else 0)) / 100 for i in range(pessoas)]


class Cozinha:
    def __init__(self):
        self.fila = deque()

    def enviar(self, item: ItemPedido):
        item.avancar()  # PREPARANDO
        self.fila.append(item)

    def concluir_proximo(self) -> Optional[ItemPedido]:
        if not self.fila:
            return None
        item = self.fila.popleft()
        item.avancar()  # PRONTO
        return item


pizza, suco = ItemCardapio("Pizza", 40.0), ItemCardapio("Suco", 9.0)
pedido, cozinha = Pedido(), Cozinha()
for it in (pedido.adicionar(pizza, 2), pedido.adicionar(suco, 3)):
    cozinha.enviar(it)
pronto = cozinha.concluir_proximo()
pronto.avancar()  # SERVIDO
print(pronto.estado.name, pedido.total())
print(pedido.dividir(3))
```

```javascript
const Estado = Object.freeze({ PEDIDO: 1, PREPARANDO: 2, PRONTO: 3, SERVIDO: 4 });
const nomeEstado = (v) => Object.keys(Estado).find((k) => Estado[k] === v);

    this.preco = preco;
    Object.freeze(this);
  }

  constructor(prato, qtd = 1) {
    this.prato = prato;
    this.qtd = qtd;
    this.estado = Estado.PEDIDO;
  }

  avancar() {
    if (this.estado === Estado.SERVIDO) {
      throw new Error("item já servido");
    }
    this.estado += 1;
  }

  get subtotal() {
    return this.prato.preco * this.qtd;
  }

    this.itens = [];
    this.fechado = false;
  }

    if (this.fechado) {
      throw new Error("pedido fechado");
    }
    const item = new ItemPedido(prato, qtd);
    this.itens.push(item);
    return item;
  }

  total(taxaServico = 0.10) {
    const base = this.itens.reduce((soma, i) => soma + i.subtotal, 0);
    return Math.round(base * (1 + taxaServico) * 100) / 100;

    const totalCentavos = Math.round(this.total() * 100);
    const base = Math.floor(totalCentavos / pessoas);

      return null;
    }
    const item = this.fila.shift();
    item.avancar(); // PRONTO
```

```java
import java.util.*;

enum Estado {
    PEDIDO, PREPARANDO, PRONTO, SERVIDO;

    Estado proximo() {
        return values()[ordinal() + 1];
    }
}

class ItemCardapio {
    final String nome;
    final double preco;

    ItemCardapio(String nome, double preco) {
        this.nome = nome;
        this.preco = preco;
    }
}

class ItemPedido {
    final ItemCardapio prato;
    final int qtd;
    Estado estado = Estado.PEDIDO;

    ItemPedido(ItemCardapio prato, int qtd) {
        this.prato = prato;
        this.qtd = qtd;
    }

    void avancar() {
        if (estado == Estado.SERVIDO) {
            throw new IllegalStateException("item já servido");
        }
        estado = estado.proximo();
    }

    double subtotal() {
        return prato.preco * qtd;
    }
}

class Pedido {
    final List<ItemPedido> itens = new ArrayList<>();
    boolean fechado = false;

    ItemPedido adicionar(ItemCardapio prato, int qtd) {
        if (fechado) {
            throw new IllegalStateException("pedido fechado");
        }
        ItemPedido item = new ItemPedido(prato, qtd);
        itens.add(item);
        return item;
    }

    double total(double taxaServico) {
        double base = 0;
        for (ItemPedido i : itens) base += i.subtotal();
        return Math.round(base * (1 + taxaServico) * 100) / 100.0;
    }

    double total() {
        return total(0.10);
    }

    List<Double> dividir(int pessoas) {
        long totalCentavos = Math.round(total() * 100);
        long base = totalCentavos / pessoas;
        long resto = totalCentavos % pessoas;
        List<Double> partes = new ArrayList<>();
        for (int i = 0; i < pessoas; i++) partes.add((base + (i < resto ? 1 : 0)) / 100.0);
        return partes;
    }
}

class Cozinha {
    private final Deque<ItemPedido> fila = new ArrayDeque<>();

    void enviar(ItemPedido item) {
        item.avancar(); // PREPARANDO
        fila.addLast(item);
    }

    ItemPedido concluirProximo() {
        if (fila.isEmpty()) {
            return null;
        }
        ItemPedido item = fila.pollFirst();
        item.avancar(); // PRONTO
        return item;
    }
}

public static void main(String[] args) {
    ItemCardapio pizza = new ItemCardapio("Pizza", 40.0), suco = new ItemCardapio("Suco", 9.0);
    Pedido pedido = new Pedido();
    Cozinha cozinha = new Cozinha();
    for (ItemPedido it : List.of(pedido.adicionar(pizza, 2), pedido.adicionar(suco, 3))) {
        cozinha.enviar(it);
    }
    ItemPedido pronto = cozinha.concluirProximo();
    pronto.avancar(); // SERVIDO
    System.out.println(pronto.estado + " " + pedido.total());
    System.out.println(pedido.dividir(3));
}
```

```cpp
enum class Estado { PEDIDO = 1, PREPARANDO = 2, PRONTO = 3, SERVIDO = 4 };

string nomeEstado(Estado e) {
    switch (e) {
        case Estado::PEDIDO: return "PEDIDO";
        case Estado::PREPARANDO: return "PREPARANDO";
        case Estado::PRONTO: return "PRONTO";
        default: return "SERVIDO";
    }
}

struct ItemCardapio {
    string nome;
    double preco;
};

struct ItemPedido {
    ItemCardapio prato;
    int qtd = 1;
    Estado estado = Estado::PEDIDO;

    void avancar() {
        if (estado == Estado::SERVIDO) {
            throw logic_error("item já servido");
        }
        estado = static_cast<Estado>((int)estado + 1);
    }

    double subtotal() const { return prato.preco * qtd; }
};

class Pedido {
    vector<unique_ptr<ItemPedido>> itens;
    bool fechado = false;

public:
    ItemPedido* adicionar(const ItemCardapio& prato, int qtd = 1) {
        if (fechado) {
            throw logic_error("pedido fechado");
        }
        itens.push_back(make_unique<ItemPedido>(ItemPedido{prato, qtd}));
        return itens.back().get();
    }

    double total(double taxaServico = 0.10) const {
        double base = 0;
        for (auto& i : itens) base += i->subtotal();
        return round(base * (1 + taxaServico) * 100) / 100;
    }

    vector<double> dividir(int pessoas) const {
        long totalCentavos = lround(total() * 100);
        long base = totalCentavos / pessoas;
        long resto = totalCentavos % pessoas;
        vector<double> partes;
        for (int i = 0; i < pessoas; i++) partes.push_back((base + (i < resto ? 1 : 0)) / 100.0);
        return partes;
    }
};

class Cozinha {
    deque<ItemPedido*> fila;

public:
    void enviar(ItemPedido* item) {
        item->avancar(); // PREPARANDO
        fila.push_back(item);
    }

    ItemPedido* concluirProximo() {
        if (fila.empty()) {
            return nullptr;
        }
        ItemPedido* item = fila.front();
        fila.pop_front();
        item->avancar(); // PRONTO
        return item;
    }
```

```csharp
using System;
using System.Collections.Generic;
using System.Linq;

enum Estado { Pedido = 1, Preparando = 2, Pronto = 3, Servido = 4 }

class ItemCardapio
{
    public readonly string Nome;
    public readonly double Preco;

    public ItemCardapio(string nome, double preco)
    {
        Nome = nome;
        Preco = preco;
    }
}

class ItemPedido
{
    public readonly ItemCardapio Prato;
    public readonly int Qtd;
    public Estado Estado = Estado.Pedido;

    public ItemPedido(ItemCardapio prato, int qtd = 1)
    {
        Prato = prato;
        Qtd = qtd;
    }

    public void Avancar()
    {
        if (Estado == Estado.Servido)
        {
            throw new InvalidOperationException("item já servido");
        }
        Estado = (Estado)((int)Estado + 1);
    }

    public double Subtotal
    {
        get { return Prato.Preco * Qtd; }
    }
}

class PedidoRestaurante
{
    public readonly List<ItemPedido> Itens = new List<ItemPedido>();
    public bool Fechado = false;

    public ItemPedido Adicionar(ItemCardapio prato, int qtd = 1)
    {
        if (Fechado)
        {
            throw new InvalidOperationException("pedido fechado");
        }
        var item = new ItemPedido(prato, qtd);
        Itens.Add(item);
        return item;
    }

    public double Total(double taxaServico = 0.10)
    {
        double baseValor = Itens.Sum(i => i.Subtotal);
        return Math.Round(baseValor * (1 + taxaServico), 2);
    }

    public List<double> Dividir(int pessoas)
    {
        long totalCentavos = (long)Math.Round(Total() * 100);
        long baseCentavos = totalCentavos / pessoas;
        long resto = totalCentavos % pessoas;
        var partes = new List<double>();
        for (int i = 0; i < pessoas; i++) partes.Add((baseCentavos + (i < resto ? 1 : 0)) / 100.0);
        return partes;
    }
}

class Cozinha
{
    private readonly Queue<ItemPedido> fila = new Queue<ItemPedido>();

    public void Enviar(ItemPedido item)
    {
        item.Avancar(); // PREPARANDO
        fila.Enqueue(item);
    }

    public ItemPedido ConcluirProximo()
    {
        if (fila.Count == 0)
        {
            return null;
        }
        var item = fila.Dequeue();
        item.Avancar(); // PRONTO
        return item;
    }
}

static void Main()
{
    ItemCardapio pizza = new ItemCardapio("Pizza", 40.0), suco = new ItemCardapio("Suco", 9.0);
    var pedido = new PedidoRestaurante();

    foreach (var it in new[] { pedido.Adicionar(pizza, 2), pedido.Adicionar(suco, 3) })
    {
        cozinha.Enviar(it);
    }
    var pronto = cozinha.ConcluirProximo();
    pronto.Avancar(); // SERVIDO
    Console.WriteLine(pronto.Estado.ToString().ToUpper() + " " + pedido.Total());
    Console.WriteLine("[" + string.Join(", ", pedido.Dividir(3)) + "]");
}
```

```go
import (
	"errors"
	"fmt"
	"math"
	"strings"
)

type Estado int

const (
	PEDIDO     Estado = 1
	PREPARANDO Estado = 2
	PRONTO     Estado = 3
	SERVIDO    Estado = 4
)

func (e Estado) String() string {
	return [...]string{"", "PEDIDO", "PREPARANDO", "PRONTO", "SERVIDO"}[e]
}

type ItemCardapio struct {
	nome  string
	preco float64
}

type ItemPedido struct {
	prato  ItemCardapio
	qtd    int
	estado Estado
}

func (i *ItemPedido) avancar() error {
	if i.estado == SERVIDO {
		return errors.New("item já servido")
	}
	i.estado++
	return nil
}

func (i *ItemPedido) subtotal() float64 { return i.prato.preco * float64(i.qtd) }

type Pedido struct {
	itens   []*ItemPedido
	fechado bool
}

func (p *Pedido) adicionar(prato ItemCardapio, qtd int) (*ItemPedido, error) {
	if p.fechado {
		return nil, errors.New("pedido fechado")
	}
	item := &ItemPedido{prato, qtd, PEDIDO}
	p.itens = append(p.itens, item)
	return item, nil
}

func (p *Pedido) total(taxaServico float64) float64 {
	base := 0.0
	for _, i := range p.itens {
		base += i.subtotal()
	}
	return math.Round(base*(1+taxaServico)*100) / 100
}

func (p *Pedido) dividir(pessoas int) []float64 {
	totalCentavos := int(math.Round(p.total(0.10) * 100))
	base, resto := totalCentavos/pessoas, totalCentavos%pessoas
	partes := make([]float64, pessoas)
	for i := range partes {
		extra := 0
		if i < resto {
			extra = 1
		}
		partes[i] = float64(base+extra) / 100
	}
	return partes
}

type Cozinha struct {
	fila []*ItemPedido
}

func (c *Cozinha) enviar(item *ItemPedido) {
	item.avancar() // PREPARANDO
	c.fila = append(c.fila, item)
}

func (c *Cozinha) concluirProximo() *ItemPedido {
	if len(c.fila) == 0 {
		return nil
	}
	item := c.fila[0]
	c.fila = c.fila[1:]
	item.avancar() // PRONTO
	return item
}

func main() {
	pizza, suco := ItemCardapio{"Pizza", 40.0}, ItemCardapio{"Suco", 9.0}
	pedido, cozinha := &Pedido{}, &Cozinha{}
	i1, _ := pedido.adicionar(pizza, 2)
	i2, _ := pedido.adicionar(suco, 3)
	for _, it := range []*ItemPedido{i1, i2} {
		cozinha.enviar(it)
	}
	pronto := cozinha.concluirProximo()
	pronto.avancar() // SERVIDO
	fmt.Println(pronto.estado, pedido.total(0.10))
	partes := pedido.dividir(3)
	textos := []string{}
	for _, v := range partes {
		textos = append(textos, fmt.Sprint(v))
	}
	fmt.Println("[" + strings.Join(textos, ", ") + "]")
}
```

A divisão trabalha em centavos inteiros e distribui o resto, assim a soma das partes sempre bate com o total.

## Extensões

Várias estações na cozinha (bar, grill) com uma fila por estação, cancelamento de item (permitido só antes de PREPARANDO), descontos (Decorator ou Strategy sobre o total) e Observer para avisar o garçom quando o item fica pronto.

## Concorrência

Vários garçons adicionam itens ao mesmo pedido. Proteja a lista de itens com lock por pedido, e a fila da cozinha com uma estrutura thread-safe como `queue.Queue`.

## Na entrevista

- **"Como validar transições?"** Método no próprio item, ou um mapa de transições permitidas se houver desvios (cancelado).
- **"Por que não usar float para dinheiro?"** Erros de arredondamento; use centavos inteiros ou `Decimal`.
