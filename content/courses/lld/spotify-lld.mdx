Um player de música é um bom exercício de State, Strategy e Iterator: o estado da reprodução muda o que cada botão faz, e o modo de reprodução (sequencial, aleatório, repetir) muda qual faixa vem a seguir.

## Requisitos

- Biblioteca de músicas e playlists.
- Tocar, pausar, retomar, próxima e anterior.
- Modos: sequencial, aleatório e repetir uma.
- A fila pode receber músicas extras.

## Entidades

| Entidade | Responsabilidade |
| --- | --- |
| `Musica` | Título, artista e duração. |
| `Playlist` | Lista ordenada de músicas. |
| `ModoReproducao` | Strategy: decide o próximo índice. |
| `Player` | Estado atual, fila e comandos. |

```python
import random
from dataclasses import dataclass
from enum import Enum
from typing import List, Optional


@dataclass(frozen=True)
class Musica:
    titulo: str
    artista: str
    duracao: int


class Modo:
    def proximo(self, atual: int, total: int) -> Optional[int]:
        raise NotImplementedError

    def anterior(self, atual: int, total: int) -> Optional[int]:
        return max(0, atual - 1)


class Sequencial(Modo):
    def proximo(self, atual, total):
        return atual + 1 if atual + 1 < total else None


class RepetirUma(Modo):
    def proximo(self, atual, total):
        return atual


class Aleatorio(Modo):
    def __init__(self, semente=None):
        self.rng = random.Random(semente)

    def proximo(self, atual, total):
        return self.rng.randrange(total)


class Estado(Enum):
    PARADO = "parado"
    TOCANDO = "tocando"
    PAUSADO = "pausado"


class Player:
    def __init__(self, fila: List[Musica], modo: Modo):
        self.fila, self.modo = list(fila), modo
        self.indice = 0
        self.estado = Estado.PARADO

    @property
    def atual(self) -> Optional[Musica]:
        return self.fila[self.indice] if self.fila else None

    def tocar(self):
        if not self.fila:
            raise ValueError("fila vazia")
        self.estado = Estado.TOCANDO

    def pausar(self):
        if self.estado is Estado.TOCANDO:
            self.estado = Estado.PAUSADO

    def retomar(self):
        if self.estado is Estado.PAUSADO:
            self.estado = Estado.TOCANDO

    def proxima(self):
        novo = self.modo.proximo(self.indice, len(self.fila))
        if novo is None:
            self.estado = Estado.PARADO
            self.indice = 0
        else:
            self.indice = novo

    def anterior(self):
        self.indice = self.modo.anterior(self.indice, len(self.fila))

    def enfileirar(self, m: Musica):
        self.fila.append(m)


ms = [Musica("A", "X", 180), Musica("B", "X", 200), Musica("C", "Y", 150)]
p = Player(ms, Sequencial())
p.tocar()
p.proxima()
p.pausar()
print(p.atual.titulo, p.estado.value)   # B pausado
p.retomar()
p.proxima()
p.proxima()                             # fim da fila
print(p.estado.value)                   # parado
p.modo = Aleatorio(semente=1)
p.tocar()
p.proxima()
print(p.atual.titulo in {"A", "B", "C"})
```

```javascript
class Musica {
```

```java
abstract class Modo {
    abstract Integer proximo(int atual, int total); // null = fim da fila

    int anterior(int atual, int total) {
        return Math.max(0, atual - 1);
    }
}

class Sequencial extends Modo {
    Integer proximo(int atual, int total) {
        return atual + 1 < total ? atual + 1 : null;
    }
}

class RepetirUma extends Modo {
    Integer proximo(int atual, int total) {
        return atual;
    }
}

class Aleatorio extends Modo {
    private final Random rng;

    Aleatorio(long semente) {
        this.rng = new Random(semente);
    }

    Integer proximo(int atual, int total) {
        return rng.nextInt(total);
    }
}

enum Estado {
    PARADO("parado"), TOCANDO("tocando"), PAUSADO("pausado");

    final String valor;

    Estado(String valor) {
        this.valor = valor;
    }
}

    Modo modo;
    int indice = 0;

    Player(List<Musica> fila, Modo modo) {
        this.fila = new ArrayList<>(fila);
        this.modo = modo;
    }

    Musica atual() {
        return fila.isEmpty() ? null : fila.get(indice);
    }

    void tocar() {
        if (fila.isEmpty()) {
            throw new IllegalStateException("fila vazia");
        }
        estado = Estado.TOCANDO;
    }

    void pausar() {
        if (estado == Estado.TOCANDO) {
            estado = Estado.PAUSADO;
        }
    }

    void retomar() {
        if (estado == Estado.PAUSADO) {
            estado = Estado.TOCANDO;
        }
    }

        if (novo == null) {
            estado = Estado.PARADO;
            indice = 0;
        } else {
            indice = novo;
        }
    }

    void anterior() {
        indice = modo.anterior(indice, fila.size());
    }

    void enfileirar(Musica m) {
        fila.add(m);
    }
}
```

```cpp
class Modo {
public:
    virtual ~Modo() = default;
    virtual optional<int> proximo(int atual, int total) = 0; // nullopt = fim da fila
    virtual int anterior(int atual, int total) { return max(0, atual - 1); }
};

class Sequencial : public Modo {
public:
    optional<int> proximo(int atual, int total) override {
        if (atual + 1 < total) return atual + 1;
        return nullopt;
    }
};

class RepetirUma : public Modo {
public:
    optional<int> proximo(int atual, int total) override { return atual; }
};

class Aleatorio : public Modo {
    mt19937 rng;

public:
    Aleatorio(unsigned semente) : rng(semente) {}

    optional<int> proximo(int atual, int total) override { return uniform_int_distribution<int>(0, total - 1)(rng); }
};

enum class Estado { PARADO, TOCANDO, PAUSADO };

string valor(Estado e) {
    switch (e) {
        case Estado::PARADO: return "parado";
        case Estado::TOCANDO: return "tocando";
        default: return "pausado";
    }
}

class Player {
public:
    vector<Musica> fila;
    unique_ptr<Modo> modo;
    int indice = 0;
    Estado estado = Estado::PARADO;

    Player(vector<Musica> fila, unique_ptr<Modo> modo) : fila(fila), modo(move(modo)) {}

    const Musica* atual() const { return fila.empty() ? nullptr : &fila[indice]; }

    void tocar() {
        if (fila.empty()) {
            throw logic_error("fila vazia");
        }
        estado = Estado::TOCANDO;
    }

    void pausar() {
        if (estado == Estado::TOCANDO) estado = Estado::PAUSADO;
    }

    void retomar() {
        if (estado == Estado::PAUSADO) estado = Estado::TOCANDO;
    }

    void proxima() {
        auto novo = modo->proximo(indice, (int)fila.size());
        if (!novo) {
            estado = Estado::PARADO;
            indice = 0;
        } else {
            indice = *novo;
        }
    }

    void anterior() { indice = modo->anterior(indice, (int)fila.size()); }

    void enfileirar(const Musica& m) { fila.push_back(m); }
};
```

```go
import (
	"errors"
	"fmt"
	"math/rand"
)

type Musica struct {
	titulo, artista string
	duracao         int
}

type Modo interface {
	proximo(atual, total int) (int, bool) // false = fim da fila
	anterior(atual, total int) int
}

func anteriorPadrao(atual, total int) int {
	if atual-1 > 0 {
		return atual - 1
	}
	return 0
}

type Sequencial struct{}

func (Sequencial) proximo(atual, total int) (int, bool) {
	if atual+1 < total {
		return atual + 1, true
	}
	return 0, false
}
func (Sequencial) anterior(atual, total int) int { return anteriorPadrao(atual, total) }

type RepetirUma struct{}

func (RepetirUma) proximo(atual, total int) (int, bool) { return atual, true }
func (RepetirUma) anterior(atual, total int) int        { return anteriorPadrao(atual, total) }

type Aleatorio struct {
	rng *rand.Rand
}

func (a *Aleatorio) proximo(atual, total int) (int, bool) { return a.rng.Intn(total), true }
func (a *Aleatorio) anterior(atual, total int) int        { return anteriorPadrao(atual, total) }

type Estado string

const (
	Parado  Estado = "parado"
	Tocando Estado = "tocando"
	Pausado Estado = "pausado"
)

type Player struct {
	fila   []Musica
	modo   Modo
	indice int
	estado Estado
}

func NovoPlayer(fila []Musica, modo Modo) *Player {
	return &Player{fila: append([]Musica{}, fila...), modo: modo, estado: Parado}
}

func (p *Player) atual() *Musica {
	if len(p.fila) == 0 {
		return nil
	}
	return &p.fila[p.indice]
}

func (p *Player) tocar() error {
	if len(p.fila) == 0 {
		return errors.New("fila vazia")
	}
	p.estado = Tocando
	return nil
}

func (p *Player) pausar() {
	if p.estado == Tocando {
		p.estado = Pausado
	}
}

func (p *Player) retomar() {
	if p.estado == Pausado {
		p.estado = Tocando
	}
}

func (p *Player) proxima() {
	novo, ok := p.modo.proximo(p.indice, len(p.fila))
	if !ok {
		p.estado = Parado
		p.indice = 0
	} else {
		p.indice = novo
	}
}

func (p *Player) anterior() { p.indice = p.modo.anterior(p.indice, len(p.fila)) }

func (p *Player) enfileirar(m Musica) { p.fila = append(p.fila, m) }

func main() {
	ms := []Musica{{"A", "X", 180}, {"B", "X", 200}, {"C", "Y", 150}}
	p := NovoPlayer(ms, Sequencial{})
	p.tocar()
	p.proxima()
	p.pausar()
	fmt.Println(p.atual().titulo, p.estado) // B pausado
	p.retomar()
	p.proxima()
	p.proxima() // fim da fila
	fmt.Println(p.estado) // parado
	p.modo = &Aleatorio{rand.New(rand.NewSource(1))}
	p.tocar()
	p.proxima()
	t := p.atual().titulo
	fmt.Println(t == "A" || t == "B" || t == "C")
}
```

Trocar `p.modo` em tempo de execução é a vantagem do Strategy: o player não conhece as regras de ordem.

## Extensões

Embaralhar sem repetição (gerar uma permutação e percorrer, em vez de sortear sempre), histórico de reprodução para voltar à faixa realmente anterior, volume e equalizador como Decorator sobre o fluxo de áudio, e Observer para a interface atualizar a faixa atual.

## Estado explícito

O enum `Estado` evita dois booleanos que podem entrar em combinação inválida (tocando e pausado ao mesmo tempo). Se cada estado tiver comportamento rico, evolua para o padrão State com uma classe por estado.

## Na entrevista

- **"Como tocar sem repetir no aleatório?"** Embaralhe os índices e percorra; reembaralhe ao fim.
- **"Onde fica o áudio em si?"** Atrás de uma interface `Saida`; o design não depende de hardware.
