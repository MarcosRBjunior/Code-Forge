Controle de estoque parece um mapa de produto para quantidade, mas as perguntas reais são: como evitar estoque negativo, como registrar o histórico das movimentações e como avisar quando o nível fica baixo.

## Requisitos

- Cadastro de produtos com SKU.
- Entrada e saída de quantidade, por depósito.
- Nunca permitir saldo negativo.
- Alerta quando o saldo cai abaixo de um mínimo.
- Histórico de movimentações para auditoria.

## Entidades

| Entidade | Responsabilidade |
| --- | --- |
| `Produto` | SKU, nome e estoque mínimo. |
| `Movimentacao` | Registro imutável: SKU, delta, motivo. |
| `Deposito` | Saldos por SKU, histórico e ouvintes de alerta. |
| `transferir` | Move quantidade entre dois depósitos. |

A decisão central: guardamos o saldo atual para consulta rápida, mas toda alteração passa por um único método que valida e registra a movimentação.

```python
from dataclasses import dataclass
from typing import Callable, Dict, List


@dataclass(frozen=True)
class Produto:
    sku: str
    nome: str
    minimo: int = 0


@dataclass(frozen=True)
class Movimentacao:
    sku: str
    delta: int
    motivo: str


class Deposito:
    def __init__(self, nome: str):
        self.nome = nome
        self.saldos: Dict[str, int] = {}
        self.historico: List[Movimentacao] = []
        self.produtos: Dict[str, Produto] = {}
        self.ouvintes: List[Callable[[Produto, int], None]] = []

    def cadastrar(self, p: Produto):
        self.produtos[p.sku] = p
        self.saldos.setdefault(p.sku, 0)

    def _aplicar(self, sku: str, delta: int, motivo: str):
        novo = self.saldos[sku] + delta
        if novo < 0:
            raise ValueError(f"estoque insuficiente de {sku}")
        self.saldos[sku] = novo
        self.historico.append(Movimentacao(sku, delta, motivo))
        p = self.produtos[sku]
        if novo < p.minimo:
            for f in self.ouvintes:
                f(p, novo)

    def entrada(self, sku, qtd, motivo="compra"):
        self._aplicar(sku, qtd, motivo)

    def saida(self, sku, qtd, motivo="venda"):
        self._aplicar(sku, -qtd, motivo)


def transferir(origem: Deposito, destino: Deposito, sku: str, qtd: int):
    origem.saida(sku, qtd, f"transferência para {destino.nome}")
    destino.entrada(sku, qtd, f"transferência de {origem.nome}")

try:
    d1.saida("CAN-1", 10)
```

```java
import java.util.*;
import java.util.function.BiConsumer;

class Produto {
    final String sku, nome;
    final int minimo;

    Produto(String sku, String nome, int minimo) {
        this.sku = sku;
        this.nome = nome;
        this.minimo = minimo;
    }
}

class Movimentacao {
    final String sku;
    final int delta;
    final String motivo;

    Movimentacao(String sku, int delta, String motivo) {
        this.sku = sku;
        this.delta = delta;
        this.motivo = motivo;
    }
}

class Deposito {
    final String nome;
    final Map<String, Integer> saldos = new LinkedHashMap<>();
    final List<Movimentacao> historico = new ArrayList<>();
    final Map<String, Produto> produtos = new HashMap<>();
    final List<BiConsumer<Produto, Integer>> ouvintes = new ArrayList<>();

    Deposito(String nome) {
        this.nome = nome;
    }

    void cadastrar(Produto p) {
        produtos.put(p.sku, p);
        saldos.putIfAbsent(p.sku, 0);
    }

    private void aplicar(String sku, int delta, String motivo) {
        int novo = saldos.get(sku) + delta;
        if (novo < 0) {
            throw new IllegalArgumentException("estoque insuficiente de " + sku);
        }
        saldos.put(sku, novo);
        historico.add(new Movimentacao(sku, delta, motivo));
        Produto p = produtos.get(sku);
        if (novo < p.minimo) {
            for (BiConsumer<Produto, Integer> f : ouvintes) f.accept(p, novo);
        }
    }

    void entrada(String sku, int qtd, String motivo) {
        aplicar(sku, qtd, motivo);
    }

    void entrada(String sku, int qtd) {
        entrada(sku, qtd, "compra");
    }

    void saida(String sku, int qtd, String motivo) {
        aplicar(sku, -qtd, motivo);
    }

    void saida(String sku, int qtd) {
        saida(sku, qtd, "venda");
    }
}

static void transferir(Deposito origem, Deposito destino, String sku, int qtd) {
    origem.saida(sku, qtd, "transferência para " + destino.nome);
    destino.entrada(sku, qtd, "transferência de " + origem.nome);
}

static String mostrarSaldos(Map<String, Integer> m) {
    List<String> pares = new ArrayList<>();
    for (Map.Entry<String, Integer> e : m.entrySet()) pares.add("'" + e.getKey() + "': " + e.getValue());
    return "{" + String.join(", ", pares) + "}";
}

public static void main(String[] args) {
    Deposito d1 = new Deposito("SP"), d2 = new Deposito("RJ");
    Produto caneta = new Produto("CAN-1", "Caneta", 5);
    for (Deposito d : List.of(d1, d2)) {
        d.cadastrar(caneta);
    }
    d1.ouvintes.add((p, s) -> System.out.println("ALERTA: " + p.nome + " com " + s));

    d1.saida("CAN-1", 5);            // saldo SP = 3 -> alerta

        d1.saida("CAN-1", 10);
    } catch (IllegalArgumentException e) {
        System.out.println("erro: " + e.getMessage());
    }
    System.out.println(mostrarSaldos(d1.saldos) + " " + mostrarSaldos(d2.saldos) + " " + d1.historico.size());
```

```go
import (
	"fmt"
	"strings"
)

type Produto struct {
	sku, nome string
	minimo    int
}

type Movimentacao struct {
	sku    string
	delta  int
	motivo string
}

type Deposito struct {
	nome      string
	saldos    map[string]int
	ordem     []string // mantém a ordem de cadastro para imprimir os saldos
	historico []Movimentacao
	produtos  map[string]Produto
	ouvintes  []func(Produto, int)
}

func NovoDeposito(nome string) *Deposito {
	return &Deposito{nome: nome, saldos: map[string]int{}, produtos: map[string]Produto{}}
}

func (d *Deposito) cadastrar(p Produto) {
	d.produtos[p.sku] = p
	if _, existe := d.saldos[p.sku]; !existe {
		d.saldos[p.sku] = 0
		d.ordem = append(d.ordem, p.sku)
	}
}

func (d *Deposito) aplicar(sku string, delta int, motivo string) error {
	novo := d.saldos[sku] + delta
	if novo < 0 {
		return fmt.Errorf("estoque insuficiente de %s", sku)
	}
	d.saldos[sku] = novo
	d.historico = append(d.historico, Movimentacao{sku, delta, motivo})
	p := d.produtos[sku]
	if novo < p.minimo {
		for _, f := range d.ouvintes {
			f(p, novo)
		}
	}
	return nil
}

func (d *Deposito) entrada(sku string, qtd int, motivo string) error { return d.aplicar(sku, qtd, motivo) }

func (d *Deposito) saida(sku string, qtd int, motivo string) error { return d.aplicar(sku, -qtd, motivo) }

func (d *Deposito) mostrarSaldos() string {
	pares := []string{}
	for _, sku := range d.ordem {
		pares = append(pares, fmt.Sprintf("'%s': %d", sku, d.saldos[sku]))
	}
	return "{" + strings.Join(pares, ", ") + "}"
}

func transferir(origem, destino *Deposito, sku string, qtd int) error {
	if err := origem.saida(sku, qtd, "transferência para "+destino.nome); err != nil {
		return err
	}
	return destino.entrada(sku, qtd, "transferência de "+origem.nome)
}

func main() {
	d1, d2 := NovoDeposito("SP"), NovoDeposito("RJ")
	caneta := Produto{"CAN-1", "Caneta", 5}
	for _, d := range []*Deposito{d1, d2} {
		d.cadastrar(caneta)
	}
	d1.ouvintes = append(d1.ouvintes, func(p Produto, s int) { fmt.Printf("ALERTA: %s com %d\n", p.nome, s) })
	d1.entrada("CAN-1", 20, "compra")
	transferir(d1, d2, "CAN-1", 12) // saldo SP = 8
	d1.saida("CAN-1", 5, "venda")    // saldo SP = 3 -> alerta
	if err := d1.saida("CAN-1", 10, "venda"); err != nil {
		fmt.Println("erro:", err)
	}
	fmt.Println(d1.mostrarSaldos(), d2.mostrarSaldos(), len(d1.historico))
}
```

Nota sobre a transferência: se a entrada no destino falhar depois da saída na origem, o estoque some. Em produção isso exige transação ou compensação (desfazer a saída).

## Concorrência

Duas vendas simultâneas do último item são o caso clássico. Faça a verificação e a subtração de forma atômica: lock por SKU na memória, ou no banco um `UPDATE` condicional (só subtrai se a quantidade for suficiente) conferindo as linhas afetadas.

## Extensões

Reservas de estoque durante o checkout (saldo disponível é o saldo menos o reservado), lotes com validade (FEFO: o que vence primeiro sai primeiro) e inventário cíclico com ajuste de contagem.

## Na entrevista

- **"Como auditar?"** Histórico imutável de movimentações; o saldo pode ser recalculado.
- **"Por que Observer no alerta?"** Para que e-mail, painel e pedido automático de compra reajam sem acoplar ao depósito.
