Um framework de logging é pequeno, mas combina vários padrões: níveis de severidade, múltiplos destinos e formatação intercambiável. Frequentemente o entrevistador usa o problema para testar Chain of Responsibility ou Observer, além de thread safety.

## Requisitos

- Níveis: DEBUG, INFO, WARN, ERROR.
- O logger descarta mensagens abaixo do nível configurado.
- Vários destinos ao mesmo tempo (console, arquivo, memória).
- Formatação configurável.
- Seguro para uso em várias threads.

## Entidades

| Entidade | Responsabilidade |
| --- | --- |
| `Nivel` | Enum ordenado. |
| `Registro` | Nível, mensagem e instante. |
| `Formatador` | Transforma registro em texto. |
| `Destino` | Interface `escrever(texto)`: console, arquivo, memória. |
| `Logger` | Filtra por nível e despacha a todos os destinos. |

Cada destino pode ter seu próprio nível mínimo (erros vão para o arquivo, tudo vai para o console). Isso aproxima o design do Chain of Responsibility: cada destino decide se trata o registro.

```python
import sys
import threading
from dataclasses import dataclass
from enum import IntEnum
from typing import List


class Nivel(IntEnum):
    DEBUG = 10
    INFO = 20
    WARN = 30
    ERROR = 40


@dataclass(frozen=True)
class Registro:
    nivel: Nivel
    mensagem: str
    instante: int


class Formatador:
    def formatar(self, r: Registro) -> str:
        return f"[{r.instante}] {r.nivel.name}: {r.mensagem}"


class Destino:
    def __init__(self, minimo: Nivel = Nivel.DEBUG, formatador: Formatador = None):
        self.minimo = minimo
        self.formatador = formatador or Formatador()

    def tratar(self, r: Registro):
        if r.nivel >= self.minimo:
            self.escrever(self.formatador.formatar(r))

    def escrever(self, texto: str):
        raise NotImplementedError


class Console(Destino):
    def escrever(self, texto):
        print(texto, file=sys.stdout)


class Memoria(Destino):
    def __init__(self, *a, **k):
        super().__init__(*a, **k)
        self.linhas: List[str] = []

    def escrever(self, texto):
        self.linhas.append(texto)


class Logger:
    def __init__(self, nivel: Nivel = Nivel.INFO):
        self.nivel = nivel
        self.destinos: List[Destino] = []
        self._lock = threading.Lock()
        self._tempo = 0

    def adicionar(self, d: Destino):
        self.destinos.append(d)

    def log(self, nivel: Nivel, mensagem: str):
        if nivel < self.nivel:
            return
        with self._lock:                # preserva a ordem e evita linhas misturadas
            self._tempo += 1
            r = Registro(nivel, mensagem, self._tempo)
            for d in self.destinos:
                d.tratar(r)

    def debug(self, m): self.log(Nivel.DEBUG, m)
    def info(self, m): self.log(Nivel.INFO, m)
    def warn(self, m): self.log(Nivel.WARN, m)
    def error(self, m): self.log(Nivel.ERROR, m)


mem = Memoria(minimo=Nivel.WARN)
log = Logger(Nivel.INFO)
log.adicionar(Console())
log.adicionar(mem)
log.debug("ignorado")
log.info("servidor iniciado")
log.error("falha no pagamento")
print(mem.linhas)
```

```java
import java.util.*;

enum Nivel {
    DEBUG(10), INFO(20), WARN(30), ERROR(40);

    final int valor;

    Nivel(int valor) {
        this.valor = valor;
    }
}

class Registro {
    final Nivel nivel;
    final String mensagem;
    final int instante;

    Registro(Nivel nivel, String mensagem, int instante) {
        this.nivel = nivel;
        this.mensagem = mensagem;
        this.instante = instante;
    }
}

class Formatador {
    String formatar(Registro r) {
        return "[" + r.instante + "] " + r.nivel.name() + ": " + r.mensagem;
    }
}

abstract class Destino {
    protected final Nivel minimo;
    protected final Formatador formatador;

    Destino(Nivel minimo, Formatador formatador) {
        this.minimo = minimo;
        this.formatador = formatador != null ? formatador : new Formatador();
    }

    Destino() {
        this(Nivel.DEBUG, null);
    }

    void tratar(Registro r) {
        if (r.nivel.valor >= minimo.valor) {
            escrever(formatador.formatar(r));
        }
    }

    abstract void escrever(String texto);
}

class Console extends Destino {
    void escrever(String texto) {
        System.out.println(texto);
    }
}

class Memoria extends Destino {
    final List<String> linhas = new ArrayList<>();

    Memoria(Nivel minimo) {
        super(minimo, null);
    }

    void escrever(String texto) {
        linhas.add(texto);
    }
}

class Logger {
    private final Nivel nivel;
    private final List<Destino> destinos = new ArrayList<>();
    private final Object trava = new Object();
    private int tempo = 0;

    Logger(Nivel nivel) {
        this.nivel = nivel;
    }

    void adicionar(Destino d) {
        destinos.add(d);
    }

    void log(Nivel n, String mensagem) {
        if (n.valor < nivel.valor) {
            return;
        }
        synchronized (trava) { // preserva a ordem e evita linhas misturadas
            tempo += 1;
            Registro r = new Registro(n, mensagem, tempo);
            for (Destino d : destinos) {
                d.tratar(r);
            }
        }
    }

    void debug(String m) { log(Nivel.DEBUG, m); }
    void info(String m) { log(Nivel.INFO, m); }
    void warn(String m) { log(Nivel.WARN, m); }
    void error(String m) { log(Nivel.ERROR, m); }
}

public static void main(String[] args) {
    Memoria mem = new Memoria(Nivel.WARN);
    Logger log = new Logger(Nivel.INFO);
    log.adicionar(new Console());
    log.adicionar(mem);
    log.debug("ignorado");
    log.info("servidor iniciado");
    log.error("falha no pagamento");
    List<String> citadas = new ArrayList<>();
    for (String l : mem.linhas) citadas.add("'" + l + "'");
    System.out.println("[" + String.join(", ", citadas) + "]");
}
```

```cpp
#include <iostream>
#include <memory>
#include <mutex>
#include <string>
#include <vector>
using namespace std;

enum class Nivel { DEBUG = 10, INFO = 20, WARN = 30, ERROR = 40 };

string nomeDoNivel(Nivel n) {
    switch (n) {
        case Nivel::DEBUG: return "DEBUG";
        case Nivel::INFO: return "INFO";
        case Nivel::WARN: return "WARN";
        default: return "ERROR";
    }
}

struct Registro {
    Nivel nivel;
    string mensagem;
    int instante;
};

class Formatador {
public:
    string formatar(const Registro& r) const {
        return "[" + to_string(r.instante) + "] " + nomeDoNivel(r.nivel) + ": " + r.mensagem;
    }
};

class Destino {
protected:
    Nivel minimo;
    Formatador formatador;

public:
    Destino(Nivel minimo = Nivel::DEBUG) : minimo(minimo) {}
    virtual ~Destino() = default;

    void tratar(const Registro& r) {
        if ((int)r.nivel >= (int)minimo) {
            escrever(formatador.formatar(r));
        }
    }

    virtual void escrever(const string& texto) = 0;
};

class Console : public Destino {
public:
    using Destino::Destino;
    void escrever(const string& texto) override { cout << texto << endl; }
};

class Memoria : public Destino {
public:
    vector<string> linhas;

    using Destino::Destino;
    void escrever(const string& texto) override { linhas.push_back(texto); }
};

class Logger {
    Nivel nivel;
    vector<Destino*> destinos;
    mutex trava;
    int tempo = 0;

public:
    Logger(Nivel nivel = Nivel::INFO) : nivel(nivel) {}

    void adicionar(Destino* d) { destinos.push_back(d); }

    void log(Nivel n, const string& mensagem) {
        if ((int)n < (int)nivel) {
            return;
        }
        lock_guard<mutex> g(trava); // preserva a ordem e evita linhas misturadas
        tempo += 1;
        Registro r{n, mensagem, tempo};
        for (auto* d : destinos) d->tratar(r);
    }

    void debug(const string& m) { log(Nivel::DEBUG, m); }
    void info(const string& m) { log(Nivel::INFO, m); }
    void warn(const string& m) { log(Nivel::WARN, m); }
    void error(const string& m) { log(Nivel::ERROR, m); }
};

int main() {
    Memoria mem(Nivel::WARN);
    Console console;
    Logger logger(Nivel::INFO);
    logger.adicionar(&console);
    logger.adicionar(&mem);
    logger.debug("ignorado");
    logger.info("servidor iniciado");
    logger.error("falha no pagamento");
    cout << "[";
    for (size_t i = 0; i < mem.linhas.size(); i++) cout << (i ? ", '" : "'") << mem.linhas[i] << "'";
    cout << "]" << endl;
}
```

```go
import (
	"fmt"
	"strings"
	"sync"
)

type Nivel int

const (
	Debug Nivel = 10
	Info  Nivel = 20
	Warn  Nivel = 30
	Error Nivel = 40
)

func (n Nivel) String() string {
	switch n {
	case Debug:
		return "DEBUG"
	case Info:
		return "INFO"
	case Warn:
		return "WARN"
	}
	return "ERROR"
}

type Registro struct {
	nivel    Nivel
	mensagem string
	instante int
}

type Formatador struct{}

func (Formatador) formatar(r Registro) string {
	return fmt.Sprintf("[%d] %s: %s", r.instante, r.nivel, r.mensagem)
}

type Destino interface {
	tratar(r Registro)
}

type Console struct {
	minimo     Nivel
	formatador Formatador
}

func (c Console) tratar(r Registro) {
	if r.nivel >= c.minimo {
		fmt.Println(c.formatador.formatar(r))
	}
}

type Memoria struct {
	minimo     Nivel
	formatador Formatador
	linhas     []string
}

func (m *Memoria) tratar(r Registro) {
	if r.nivel >= m.minimo {
		m.linhas = append(m.linhas, m.formatador.formatar(r))
	}
}

type Logger struct {
	nivel    Nivel
	destinos []Destino
	trava    sync.Mutex
	tempo    int
}

func (l *Logger) adicionar(d Destino) { l.destinos = append(l.destinos, d) }

func (l *Logger) log(n Nivel, mensagem string) {
	if n < l.nivel {
		return
	}
	l.trava.Lock() // preserva a ordem e evita linhas misturadas
	defer l.trava.Unlock()
	l.tempo++
	r := Registro{n, mensagem, l.tempo}
	for _, d := range l.destinos {
		d.tratar(r)
	}
}

func (l *Logger) debug(m string) { l.log(Debug, m) }
func (l *Logger) info(m string)  { l.log(Info, m) }
func (l *Logger) warn(m string)  { l.log(Warn, m) }
func (l *Logger) error(m string) { l.log(Error, m) }

func main() {
	mem := &Memoria{minimo: Warn}
	logger := &Logger{nivel: Info}
	logger.adicionar(Console{minimo: Debug})
	logger.adicionar(mem)
	logger.debug("ignorado")
	logger.info("servidor iniciado")
	logger.error("falha no pagamento")
	citadas := []string{}
	for _, l := range mem.linhas {
		citadas = append(citadas, "'"+l+"'")
	}
	fmt.Println("[" + strings.Join(citadas, ", ") + "]")
}
```

A checagem de nível acontece antes do lock e antes de montar o texto, então logs descartados custam quase nada. Na prática, evite também formatar a mensagem do chamador quando o nível estiver desligado (formatação preguiçosa).

## Trade-offs

| Decisão | Opção | Custo |
| --- | --- | --- |
| Escrita síncrona | Simples e ordenada | Bloqueia a thread de negócio |
| Escrita assíncrona | Fila e uma thread escritora | Pode perder logs num travamento |
| Singleton global | Fácil de usar | Dificulta testes; prefira injetar |

## Extensões

Rotação de arquivo por tamanho ou data, contexto estruturado (JSON com campos), loggers hierárquicos por nome com herança de configuração (como no módulo `logging` do Python) e amostragem de logs de debug.

## Na entrevista

- **"Como não bloquear a aplicação?"** Fila em memória e thread de escrita, com política de descarte se encher.
- **"Singleton?"** Um logger por nome obtido de um registro central é melhor que um global rígido.
