Um serviço de notificações envia mensagens por vários canais (e-mail, SMS, push) respeitando preferências do usuário. O LLD gira em torno de duas ideias: canais intercambiáveis (Strategy) e um fluxo de decisão antes do envio (preferências, templates, tentativas).

## Requisitos

- Canais: e-mail, SMS e push, extensíveis.
- Cada usuário escolhe quais canais aceita por tipo de notificação.
- Mensagens vêm de templates com variáveis.
- Falha em um canal não impede os demais; há novas tentativas limitadas.

## Entidades

| Entidade | Responsabilidade |
| --- | --- |
| `Canal` | Interface `enviar(destino, texto)`. |
| `Template` | Texto com variáveis. |
| `Preferencias` | Canais habilitados por tipo. |
| `ServicoNotificacao` | Resolve canais, renderiza e envia com retentativas. |

from typing import Dict, List, Set


class Canal:
    nome = "base"

    def enviar(self, destino: str, texto: str) -> None:
        raise NotImplementedError

    nome = "email"

    def __init__(self, falhas_iniciais=0):
        self.falhas = falhas_iniciais
        self.enviados = []

    def enviar(self, destino, texto):
        if self.falhas > 0:
            self.falhas -= 1
            raise ConnectionError("operadora indisponível")
        self.enviados.append((destino, texto))


class Usuario:
    def __init__(self, nome: str, contatos: Dict[str, str], preferencias: Dict[str, Set[str]]):
        self.nome, self.contatos, self.preferencias = nome, contatos, preferencias


class ServicoNotificacao:
    def __init__(self, canais: List[Canal], tentativas: int = 3):
        self.canais = {c.nome: c for c in canais}
        self.templates: Dict[str, Modelo] = {}
        self.tentativas = tentativas

    def registrar(self, tipo: str, texto: str):
        self.templates[tipo] = Modelo(texto)

    def notificar(self, usuario: Usuario, tipo: str, **vars) -> Dict[str, bool]:
        texto = self.templates[tipo].substitute(vars)
        resultado = {}
        for nome in usuario.preferencias.get(tipo, set()):
            canal = self.canais.get(nome)
            destino = usuario.contatos.get(nome)
            if canal is None or destino is None:
                continue
            resultado[nome] = self._com_retentativa(canal, destino, texto)
        return resultado

    def _com_retentativa(self, canal: Canal, destino: str, texto: str) -> bool:
        for _ in range(self.tentativas):
            try:
                canal.enviar(destino, texto)
                return True
            except ConnectionError:
                continue
        return False

import java.util.*;
import java.util.regex.*;

class ConexaoIndisponivel extends RuntimeException {
    ConexaoIndisponivel(String mensagem) {
        super(mensagem);
    }
}

interface Canal {
    String nome();
    void enviar(String destino, String texto);
}

class Email implements Canal {
    final List<String[]> enviados = new ArrayList<>();

    public String nome() {
        return "email";
    }

    public void enviar(String destino, String texto) {
        enviados.add(new String[] {destino, texto});
    }
}

class Sms implements Canal {
    private int falhas;
    final List<String[]> enviados = new ArrayList<>();

    Sms(int falhasIniciais) {
        this.falhas = falhasIniciais;
    }

    public String nome() {
        return "sms";
    }

    public void enviar(String destino, String texto) {
        if (falhas > 0) {
            falhas -= 1;
            throw new ConexaoIndisponivel("operadora indisponível");
        }
        enviados.add(new String[] {destino, texto});
    }
}

class Usuario {
    final String nome;
    final Map<String, String> contatos;
    final Map<String, List<String>> preferencias;

    Usuario(String nome, Map<String, String> contatos, Map<String, List<String>> preferencias) {
        this.nome = nome;
        this.contatos = contatos;
        this.preferencias = preferencias;
    }
}

class ServicoNotificacao {
    private final Map<String, Canal> canais = new HashMap<>();
    private final Map<String, String> templates = new HashMap<>();
    private final int tentativas;

    ServicoNotificacao(List<Canal> lista, int tentativas) {
        for (Canal c : lista) canais.put(c.nome(), c);
        this.tentativas = tentativas;
    }

    ServicoNotificacao(List<Canal> lista) {
        this(lista, 3);
    }

    void registrar(String tipo, String texto) {
        templates.put(tipo, texto);
    }

    // Troca cada $variavel pelo valor recebido (o equivalente do string.Template do Python).
    private String substituir(String modelo, Map<String, String> vars) {
        Matcher m = Pattern.compile("\\$(\\w+)").matcher(modelo);
        StringBuffer saida = new StringBuffer();
        while (m.find()) {
            String valor = vars.get(m.group(1));
            if (valor == null) throw new IllegalArgumentException("variável ausente: " + m.group(1));
            m.appendReplacement(saida, Matcher.quoteReplacement(valor));
        }
        m.appendTail(saida);
        return saida.toString();
    }

    Map<String, Boolean> notificar(Usuario usuario, String tipo, Map<String, String> vars) {
        String texto = substituir(templates.get(tipo), vars);
        Map<String, Boolean> resultado = new LinkedHashMap<>();
        for (String nome : usuario.preferencias.getOrDefault(tipo, new ArrayList<>())) {
            Canal canal = canais.get(nome);
            String destino = usuario.contatos.get(nome);
            if (canal == null || destino == null) {
                continue;
            }
            resultado.put(nome, comRetentativa(canal, destino, texto));
        }
        return resultado;
    }

    private boolean comRetentativa(Canal canal, String destino, String texto) {
        for (int i = 0; i < tentativas; i++) {
            try {
                canal.enviar(destino, texto);
                return true;
            } catch (ConexaoIndisponivel e) {
                continue;
            }
        }
        return false;
    }
}

public static void main(String[] args) {
    Email email = new Email();
    Sms sms = new Sms(2);
    ServicoNotificacao svc = new ServicoNotificacao(List.of(email, sms));

}

    // Troca cada $variavel pelo valor recebido (o equivalente do string.Template do Python).
    string substituir(const string& modelo, const map<string, string>& vars) const {
        string saida;
        size_t i = 0;
        while (i < modelo.size()) {
            if (modelo[i] == '$') {
                size_t j = i + 1;
                while (j < modelo.size() && (isalnum((unsigned char)modelo[j]) || modelo[j] == '_')) j++;
                string nome = modelo.substr(i + 1, j - i - 1);
                auto it = vars.find(nome);
                if (it == vars.end()) throw invalid_argument("variável ausente: " + nome);
                saida += it->second;
                i = j;
            } else {
                saida += modelo[i++];
            }
        }
        return saida;
    }

    bool comRetentativa(Canal& canal, const string& destino, const string& texto) {
        for (int i = 0; i < tentativas; i++) {
            try {
                canal.enviar(destino, texto);
                return true;
            } catch (const ConexaoIndisponivel&) {
                continue;
            }
        }
        return false;
    }

public:

        for (auto* c : lista) canais[c->nome()] = c;
    }

            auto destino = usuario.contatos.find(nome);

                continue;
            }
            resultado.push_back({nome, comRetentativa(*canal->second, destino->second, texto)});
        }
        return resultado;
    }
};

class ConexaoIndisponivel : Exception
{
    public ConexaoIndisponivel(string mensagem) : base(mensagem) { }

interface ICanal
{
    string Nome { get; }

class Email : ICanal

    }

}

    public Sms(int falhasIniciais = 0)
    {

        get { return "sms"; }

        }
        Enviados.Add(Tuple.Create(destino, texto));
    }
}

{
    public readonly string Nome;
    public readonly Dictionary<string, string> Contatos;
    public readonly Dictionary<string, List<string>> Preferencias;

    public Usuario(string nome, Dictionary<string, string> contatos, Dictionary<string, List<string>> preferencias)
    {
        Nome = nome;
        Contatos = contatos;
        Preferencias = preferencias;
    }

{
    private readonly Dictionary<string, ICanal> canais;
    private readonly Dictionary<string, string> templates = new Dictionary<string, string>();
    private readonly int tentativas;

    public ServicoNotificacao(List<ICanal> lista, int tentativas = 3)
    {
        canais = lista.ToDictionary(c => c.Nome);
        this.tentativas = tentativas;
    }

    public void Registrar(string tipo, string texto)
    {
        templates[tipo] = texto;

    {
        return Regex.Replace(modelo, @"\$(\w+)", m =>
        {
            string valor;
            if (!vars.TryGetValue(m.Groups[1].Value, out valor)) throw new ArgumentException("variável ausente: " + m.Groups[1].Value);
            return valor;
        });

            if (!canais.TryGetValue(nome, out canal) || !usuario.Contatos.TryGetValue(nome, out destino))
            {
                continue;

            resultado.Add(new KeyValuePair<string, bool>(nome, ComRetentativa(canal, destino, texto)));

                return true;

            catch (ConexaoIndisponivel)

                continue;
            }
        }
        return false;
    }

	if s.falhas > 0 {
		s.falhas--
		return errConexao
	}
	s.enviados = append(s.enviados, [2]string{destino, texto})
	return nil
}

type Usuario struct {
	nome         string
	contatos     map[string]string
	preferencias map[string][]string
}

type ServicoNotificacao struct {
	canais     map[string]Canal
	templates  map[string]string
	tentativas int
}

func NovoServico(canais []Canal, tentativas int) *ServicoNotificacao {
	s := &ServicoNotificacao{canais: map[string]Canal{}, templates: map[string]string{}, tentativas: tentativas}
	for _, c := range canais {
		s.canais[c.nome()] = c
	}
	return s
}

func (s *ServicoNotificacao) registrar(tipo, texto string) { s.templates[tipo] = texto }

var variavel = regexp.MustCompile(`\$(\w+)`)

// Troca cada $variavel pelo valor recebido (o equivalente do string.Template do Python).
func substituir(modelo string, vars map[string]string) (string, error) {
	var ausente string
	saida := variavel.ReplaceAllStringFunc(modelo, func(m string) string {
		nome := m[1:]
		valor, ok := vars[nome]
		if !ok {
			ausente = nome
		}
		return valor
	})
	if ausente != "" {
		return "", fmt.Errorf("variável ausente: %s", ausente)
	}
	return saida, nil
}

type Resultado struct {
	canal string
	ok    bool
}

func (s *ServicoNotificacao) notificar(u Usuario, tipo string, vars map[string]string) ([]Resultado, error) {
	texto, err := substituir(s.templates[tipo], vars)
	if err != nil {
		return nil, err
	}
	resultado := []Resultado{}
	for _, nome := range u.preferencias[tipo] {
		canal, temCanal := s.canais[nome]
		destino, temDestino := u.contatos[nome]
		if !temCanal || !temDestino {
			continue
		}
		resultado = append(resultado, Resultado{nome, s.comRetentativa(canal, destino, texto)})
	}
	return resultado, nil
}

func (s *ServicoNotificacao) comRetentativa(canal Canal, destino, texto string) bool {
	for i := 0; i < s.tentativas; i++ {
		if err := canal.enviar(destino, texto); err == nil {
			return true
		} else if errors.Is(err, errConexao) {
			continue
		}
	}
	return false
}

## Extensões

Fila assíncrona para desacoplar o produto que dispara do envio, prioridade (urgentes passam na frente), limite de frequência por usuário (evitar spam), agendamento e relatório de entrega. Novos canais exigem só uma nova classe, sem mexer no serviço (Open/Closed).

## Concorrência

Os envios são independentes e lentos (rede), então vale usar um pool de workers consumindo uma fila. Garanta idempotência: se o worker cair depois de enviar e antes de confirmar, o reenvio não pode duplicar a notificação; um id de notificação resolve.
