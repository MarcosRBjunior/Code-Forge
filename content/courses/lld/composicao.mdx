**Composição** é a forma mais forte de "todo e parte": a parte **não existe sem o todo** e pertence a um único todo. Um pedido é composto de itens de pedido; apagar o pedido apaga os itens, porque um item solto, sem pedido, não significa nada. Uma casa é composta de cômodos; o cômodo não muda de casa.

## Como reconhecer

- O todo **cria** as partes (ou recebe só os dados para criá-las).
- A parte não é compartilhada com outros todos.
- O ciclo de vida da parte acompanha o do todo.
- De fora, normalmente ninguém manipula a parte diretamente; tudo passa pelo todo.

Em UML, composição é desenhada com um **losango cheio** do lado do todo:

```text
Pedido ◆────────── ItemPedido
        1      1..*
```

```python
from dataclasses import dataclass


@dataclass(frozen=True)
class ItemPedido:
    sku: str
    quantidade: int
    preco_unitario: float

    @property
    def subtotal(self) -> float:
        return self.quantidade * self.preco_unitario


class Pedido:
    def __init__(self, numero: int) -> None:
        self.numero = numero
        self._itens: list[ItemPedido] = []
        self._fechado = False

    def adicionar_item(self, sku: str, quantidade: int, preco: float) -> None:
        if self._fechado:
            raise RuntimeError("pedido já fechado")
        if quantidade <= 0:
            raise ValueError("quantidade inválida")
        self._itens.append(ItemPedido(sku, quantidade, preco))  # o todo cria a parte

    def fechar(self) -> float:
        if not self._itens:
            raise RuntimeError("pedido vazio")
        self._fechado = True
        return self.total()

    def total(self) -> float:
        return round(sum(i.subtotal for i in self._itens), 2)

    def resumo(self) -> list[str]:
        return [f"{i.quantidade}x {i.sku}" for i in self._itens]


pedido = Pedido(42)
pedido.adicionar_item("CAFE-500G", 2, 18.90)
pedido.adicionar_item("FILTRO-102", 1, 7.50)
print(pedido.resumo())
print(pedido.fechar())  # 45.3
try:
    pedido.adicionar_item("ACUCAR", 1, 5.0)
except RuntimeError as e:
    print(e)
```

```javascript
class ItemPedido {
  constructor(sku, quantidade, precoUnitario) {
    this.sku = sku;
    this.quantidade = quantidade;
    this.precoUnitario = precoUnitario;
    Object.freeze(this);
  }

  get subtotal() {
    return this.quantidade * this.precoUnitario;
  }
}

class Pedido {
  #itens = [];
  #fechado = false;

  constructor(numero) {
    this.numero = numero;
  }

  adicionarItem(sku, quantidade, preco) {
    if (this.#fechado) throw new Error("pedido já fechado");
    if (quantidade <= 0) throw new RangeError("quantidade inválida");
    this.#itens.push(new ItemPedido(sku, quantidade, preco)); // o todo cria a parte
  }

  fechar() {
    if (this.#itens.length === 0) throw new Error("pedido vazio");
    this.#fechado = true;
    return this.total();
  }

  total() {
    const soma = this.#itens.reduce((s, i) => s + i.subtotal, 0);
    return Math.round(soma * 100) / 100;
  }

  resumo() {
    return this.#itens.map((i) => `${i.quantidade}x ${i.sku}`);
  }
}
```

```java
import java.util.*;

class ItemPedido {
    final String sku;
    final int quantidade;
    final double precoUnitario;

    ItemPedido(String sku, int quantidade, double precoUnitario) {
        this.sku = sku;
        this.quantidade = quantidade;
        this.precoUnitario = precoUnitario;
    }

    double subtotal() {
        return quantidade * precoUnitario;
    }
}

class Pedido {
    final int numero;
    private final List<ItemPedido> itens = new ArrayList<>();
    private boolean fechado = false;

    Pedido(int numero) {
        this.numero = numero;
    }

    void adicionarItem(String sku, int quantidade, double preco) {
        if (fechado) throw new IllegalStateException("pedido já fechado");
        if (quantidade <= 0) throw new IllegalArgumentException("quantidade inválida");
        itens.add(new ItemPedido(sku, quantidade, preco)); // o todo cria a parte
    }

    double fechar() {
        if (itens.isEmpty()) throw new IllegalStateException("pedido vazio");
        fechado = true;
        return total();
    }

    double total() {
        double soma = 0;
        for (ItemPedido i : itens) soma += i.subtotal();
        return Math.round(soma * 100) / 100.0;
    }

    List<String> resumo() {
        List<String> r = new ArrayList<>();
        for (ItemPedido i : itens) r.add(i.quantidade + "x " + i.sku);
        return r;
    }
}

public static void main(String[] args) {
    Pedido pedido = new Pedido(42);
    pedido.adicionarItem("CAFE-500G", 2, 18.90);
    pedido.adicionarItem("FILTRO-102", 1, 7.50);
    System.out.println(pedido.resumo());
    System.out.println(pedido.fechar()); // 45.3
    try {
        pedido.adicionarItem("ACUCAR", 1, 5.0);
    } catch (IllegalStateException e) {
        System.out.println(e.getMessage());
    }
}
```

```go
import (
	"errors"
	"fmt"
	"math"
)

type ItemPedido struct {
	sku           string
	quantidade    int
	precoUnitario float64
}

func (i ItemPedido) subtotal() float64 { return float64(i.quantidade) * i.precoUnitario }

type Pedido struct {
	numero  int
	itens   []ItemPedido
	fechado bool
}

func (p *Pedido) adicionarItem(sku string, quantidade int, preco float64) error {
	if p.fechado {
		return errors.New("pedido já fechado")
	}
	if quantidade <= 0 {
		return errors.New("quantidade inválida")
	}
	p.itens = append(p.itens, ItemPedido{sku, quantidade, preco}) // o todo cria a parte
	return nil
}

func (p *Pedido) fechar() (float64, error) {
	if len(p.itens) == 0 {
		return 0, errors.New("pedido vazio")
	}
	p.fechado = true
	return p.total(), nil
}

func (p *Pedido) total() float64 {
	soma := 0.0
	for _, i := range p.itens {
		soma += i.subtotal()
	}
	return math.Round(soma*100) / 100
}

func (p *Pedido) resumo() []string {
	r := []string{}
	for _, i := range p.itens {
		r = append(r, fmt.Sprintf("%dx %s", i.quantidade, i.sku))
	}
	return r
}

func main() {
	pedido := &Pedido{numero: 42}
	pedido.adicionarItem("CAFE-500G", 2, 18.90)
	pedido.adicionarItem("FILTRO-102", 1, 7.50)
	fmt.Println(pedido.resumo())
	total, _ := pedido.fechar()
	fmt.Println(total) // 45.3
	if err := pedido.adicionarItem("ACUCAR", 1, 5.0); err != nil {
		fmt.Println(err)
	}
}
```

Repare que o chamador nunca instancia `ItemPedido`. Ele passa dados ao `Pedido`, que decide como criar e guardar os itens. Isso dá ao todo **controle total** sobre as regras: não dá para incluir item num pedido fechado porque não existe outro caminho para chegar à lista.

## Composição para montar comportamento

Composição também é a base de um dos princípios mais citados em design: **prefira composição a herança**. Em vez de herdar de uma classe para ganhar comportamento, o objeto **é composto** de outros objetos que fazem partes do trabalho:

<CodeTabs>

```python
class Motor:
    def __init__(self, potencia_cv: int) -> None:
        self.potencia_cv = potencia_cv
        self.ligado = False

    def ligar(self) -> None:
        self.ligado = True


class Cambio:
    def __init__(self, marchas: int) -> None:
        self.marchas = marchas
        self.atual = 0

    def subir(self) -> None:
        self.atual = min(self.atual + 1, self.marchas)


class Carro:
    def __init__(self, modelo: str, potencia_cv: int, marchas: int) -> None:
        self.modelo = modelo
        self._motor = Motor(potencia_cv)   # criados e possuídos pelo carro
        self._cambio = Cambio(marchas)

    def partir(self) -> str:
        self._motor.ligar()
        self._cambio.subir()
        return f"{self.modelo}: motor ligado, marcha {self._cambio.atual}"


print(Carro("Sedan", 120, 6).partir())
```

Para criar um carro com câmbio automático, você trocaria o componente, e não criaria uma nova subclasse para cada combinação de motor e câmbio.

## Trade-offs

| Vantagem | Custo |
| --- | --- |
| O todo garante as invariantes das partes | o todo pode crescer e virar uma classe grande |
| Ciclo de vida simples: parte morre com o todo | menos flexível se a parte precisar ser compartilhada depois |
| Facilita trocar componentes internos | exige métodos de delegação no todo |

<Callout type="warning" title="Não vaze as partes">
Se `Pedido` tiver um método que retorna a lista interna de itens, qualquer um pode alterá-la e a composição vira só nominal. Exponha cópias, tuplas ou visões de leitura.
</Callout>

## Armadilhas

- **Partes compartilhadas por engano.** Passar o mesmo objeto `Motor` para dois carros quebra a semântica de composição.
- **Classe "todo" inchada.** Se o todo tem dezenas de métodos de delegação, talvez as partes devam ser expostas por interfaces específicas ou o todo deva ser dividido.
- **Confundir com agregação.** Pergunte sempre: a parte faz sentido sozinha?
