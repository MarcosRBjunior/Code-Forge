O geohash divide o mapa numa grade de tamanho fixo: cada célula de 6 caracteres tem o mesmo tamanho no centro de uma metrópole e no meio do oceano. Se um bairro tem 20 mil restaurantes e uma cidade do interior tem 30, uma grade única fica ou grossa demais para o bairro ou fina demais para o interior. A **quadtree** resolve isso com uma divisão **adaptativa**: só subdivide onde há muitos pontos.

## Como funciona

Uma quadtree é uma árvore em que cada nó representa um retângulo do espaço.

- Um nó folha guarda até `capacidade` pontos (por exemplo, 100).
- Quando uma folha passa desse limite, ela se divide em **quatro filhos** iguais: noroeste, nordeste, sudoeste e sudeste. Os pontos são redistribuídos entre eles.
- Regiões vazias continuam como folhas grandes; regiões densas viram árvores profundas.

```text
+---------------+---------------+
|               |       |   |   |
|               |       +---+---+
|   poucos      |       |   |   |
|   pontos      +-------+---+---+
|               |       |       |
|               |       |       |
+---------------+---------------+
|               |               |
|   (vazio)     |   poucos      |
|               |   pontos      |
+---------------+---------------+
o quadrante nordeste, mais denso, foi subdividido duas vezes
```

class Retangulo:
    def __init__(self, x, y, larg, alt):
        self.x, self.y, self.larg, self.alt = x, y, larg, alt

    def contem(self, px, py):
        return self.x <= px < self.x + self.larg and self.y <= py < self.y + self.alt

    def intersecta(self, outro):
        return not (outro.x >= self.x + self.larg or outro.x + outro.larg <= self.x or
                    outro.y >= self.y + self.alt or outro.y + outro.alt <= self.y)


class QuadTree:
    def __init__(self, limite, capacidade=4):
        self.limite = limite
        self.capacidade = capacidade
        self.pontos = []
        self.filhos = None

    def inserir(self, px, py):
        if not self.limite.contem(px, py):
            return False
        if self.filhos is None:
            if len(self.pontos) < self.capacidade:
                self.pontos.append((px, py))
                return True
            self._dividir()
        return any(f.inserir(px, py) for f in self.filhos)

    def _dividir(self):
        x, y = self.limite.x, self.limite.y
        m_l, m_a = self.limite.larg / 2, self.limite.alt / 2
        self.filhos = [QuadTree(Retangulo(x + dx, y + dy, m_l, m_a), self.capacidade)
                       for dx in (0, m_l) for dy in (0, m_a)]
        for p in self.pontos:
            any(f.inserir(*p) for f in self.filhos)
        self.pontos = []

    def buscar(self, area, achados=None):
        achados = [] if achados is None else achados
        if not self.limite.intersecta(area):
            return achados          # poda: subárvore inteira descartada
        achados.extend(p for p in self.pontos if area.contem(*p))
        for f in self.filhos or []:
            f.buscar(area, achados)
        return achados

class Retangulo {
  constructor(x, y, larg, alt) {
    this.x = x;
    this.y = y;
    this.larg = larg;
    this.alt = alt;
  }

  contem(px, py) {
    return this.x <= px && px < this.x + this.larg && this.y <= py && py < this.y + this.alt;
  }

  intersecta(outro) {
    return !(
      outro.x >= this.x + this.larg ||
      outro.x + outro.larg <= this.x ||
      outro.y >= this.y + this.alt ||
      outro.y + outro.alt <= this.y
    );
  }
}

class QuadTree {
  constructor(limite, capacidade = 4) {
    this.limite = limite;
    this.capacidade = capacidade;
    this.pontos = [];
    this.filhos = null;
  }

  inserir(px, py) {
    if (!this.limite.contem(px, py)) {
      return false;
    }
    if (this.filhos === null) {
      if (this.pontos.length < this.capacidade) {
        this.pontos.push([px, py]);
        return true;
      }
      this._dividir();
    }
    return this.filhos.some((f) => f.inserir(px, py));
  }

  _dividir() {
    const { x, y } = this.limite;
    const mL = this.limite.larg / 2;
    const mA = this.limite.alt / 2;
    this.filhos = [];
    for (const dx of [0, mL]) {
      for (const dy of [0, mA]) {
        this.filhos.push(new QuadTree(new Retangulo(x + dx, y + dy, mL, mA), this.capacidade));
      }
    }
    for (const [px, py] of this.pontos) {
      this.filhos.some((f) => f.inserir(px, py));
    }
    this.pontos = [];
  }

  buscar(area, achados = []) {
    if (!this.limite.intersecta(area)) {
      return achados; // poda: subárvore inteira descartada
    }
    for (const p of this.pontos) {
      if (area.contem(p[0], p[1])) achados.push(p);
    }
    for (const f of this.filhos ?? []) {
      f.buscar(area, achados);
    }
    return achados;
  }
}

  let a = semente;

    a = (a + 0x6d2b79f5) | 0;

    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

type Ponto = [number, number];

class QuadTree {
  pontos: Ponto[] = [];
  filhos: QuadTree[] | null = null;

  constructor(readonly limite: Retangulo, readonly capacidade: number = 4) {}

  inserir(px: number, py: number): boolean {
    if (!this.limite.contem(px, py)) {
      return false;
    }
    if (this.filhos === null) {
      if (this.pontos.length < this.capacidade) {
        this.pontos.push([px, py]);
        return true;
      }
      this.dividir();
    }
    return this.filhos!.some((f) => f.inserir(px, py));
  }

  private dividir(): void {
    const { x, y } = this.limite;
    const mL = this.limite.larg / 2;
    const mA = this.limite.alt / 2;
    const filhos: QuadTree[] = [];

      for (const dy of [0, mA]) {
        filhos.push(new QuadTree(new Retangulo(x + dx, y + dy, mL, mA), this.capacidade));
      }
    }
    this.filhos = filhos;
    for (const [px, py] of this.pontos) {
      filhos.some((f) => f.inserir(px, py));
    }
    this.pontos = [];
  }

  buscar(area: Retangulo, achados: Ponto[] = []): Ponto[] {
    if (!this.limite.intersecta(area)) {
      return achados; // poda: subárvore inteira descartada

      if (area.contem(p[0], p[1])) achados.push(p);
    }
    for (const f of this.filhos ?? []) {
      f.buscar(area, achados);
    }
    return achados;
  }
}

import java.util.*;

class Retangulo {
    final double x, y, larg, alt;

    Retangulo(double x, double y, double larg, double alt) {
        this.x = x;
        this.y = y;
        this.larg = larg;
        this.alt = alt;
    }

    boolean contem(double px, double py) {
        return x <= px && px < x + larg && y <= py && py < y + alt;
    }

    boolean intersecta(Retangulo o) {
        return !(o.x >= x + larg || o.x + o.larg <= x || o.y >= y + alt || o.y + o.alt <= y);
    }
}

class QuadTree {
    final Retangulo limite;
    final int capacidade;
    List<double[]> pontos = new ArrayList<>();
    List<QuadTree> filhos = null;

    QuadTree(Retangulo limite, int capacidade) {
        this.limite = limite;
        this.capacidade = capacidade;
    }

    QuadTree(Retangulo limite) {
        this(limite, 4);
    }

    boolean inserir(double px, double py) {
        if (!limite.contem(px, py)) {
            return false;
        }
        if (filhos == null) {
            if (pontos.size() < capacidade) {
                pontos.add(new double[] {px, py});
                return true;
            }
            dividir();
        }
        for (QuadTree f : filhos) if (f.inserir(px, py)) return true;
        return false;
    }

    private void dividir() {
        double mL = limite.larg / 2, mA = limite.alt / 2;
        filhos = new ArrayList<>();
        for (double dx : new double[] {0, mL}) {
            for (double dy : new double[] {0, mA}) {
                filhos.add(new QuadTree(new Retangulo(limite.x + dx, limite.y + dy, mL, mA), capacidade));
            }
        }
        for (double[] p : pontos) {
            for (QuadTree f : filhos) if (f.inserir(p[0], p[1])) break;
        }
        pontos = new ArrayList<>();
    }

    List<double[]> buscar(Retangulo area, List<double[]> achados) {
        if (!limite.intersecta(area)) {
            return achados; // poda: subárvore inteira descartada
        }
        for (double[] p : pontos) if (area.contem(p[0], p[1])) achados.add(p);
        if (filhos != null) for (QuadTree f : filhos) f.buscar(area, achados);
        return achados;
    }

    List<double[]> buscar(Retangulo area) {
        return buscar(area, new ArrayList<>());
    }
}

public static void main(String[] args) {
    Random random = new Random(7); // a sequência difere da do Python, então o número final também
    QuadTree arvore = new QuadTree(new Retangulo(0, 0, 100, 100));
    for (int i = 0; i < 500; i++) {
        arvore.inserir(random.nextDouble() * 100, random.nextDouble() * 100);
    }
    System.out.println(arvore.buscar(new Retangulo(40, 40, 10, 10)).size() + " pontos na janela");
}

class QuadTree {
    Retangulo limite;
    int capacidade;
    vector<pair<double, double>> pontos;
    vector<unique_ptr<QuadTree>> filhos;

    void dividir() {
        double mL = limite.larg / 2, mA = limite.alt / 2;
        for (double dx : {0.0, mL}) {
            for (double dy : {0.0, mA}) {
                filhos.push_back(make_unique<QuadTree>(Retangulo{limite.x + dx, limite.y + dy, mL, mA}, capacidade));
            }
        }
        for (auto& p : pontos) {
            for (auto& f : filhos) if (f->inserir(p.first, p.second)) break;
        }
        pontos.clear();
    }

public:
    QuadTree(Retangulo limite, int capacidade = 4) : limite(limite), capacidade(capacidade) {}

    bool inserir(double px, double py) {
        if (!limite.contem(px, py)) {
            return false;
        }
        if (filhos.empty()) {
            if ((int)pontos.size() < capacidade) {
                pontos.push_back({px, py});
                return true;
            }
            dividir();
        }
        for (auto& f : filhos) if (f->inserir(px, py)) return true;
        return false;
    }

    void buscar(const Retangulo& area, vector<pair<double, double>>& achados) const {
        if (!limite.intersecta(area)) {
            return; // poda: subárvore inteira descartada
        }
        for (auto& p : pontos) if (area.contem(p.first, p.second)) achados.push_back(p);
        for (auto& f : filhos) f->buscar(area, achados);
    }
};

import (
	"fmt"
	"math/rand"

type Retangulo struct {
	x, y, larg, alt float64
}

func (r Retangulo) contem(px, py float64) bool {
	return r.x <= px && px < r.x+r.larg && r.y <= py && py < r.y+r.alt
}

func (r Retangulo) intersecta(o Retangulo) bool {
	return !(o.x >= r.x+r.larg || o.x+o.larg <= r.x || o.y >= r.y+r.alt || o.y+o.alt <= r.y)
}

type Ponto struct{ x, y float64 }

type QuadTree struct {
	limite     Retangulo
	capacidade int
	pontos     []Ponto
	filhos     []*QuadTree
}

func NovaQuadTree(limite Retangulo, capacidade int) *QuadTree {
	return &QuadTree{limite: limite, capacidade: capacidade}
}

func (q *QuadTree) inserir(px, py float64) bool {
	if !q.limite.contem(px, py) {
		return false
	}
	if q.filhos == nil {
		if len(q.pontos) < q.capacidade {
			q.pontos = append(q.pontos, Ponto{px, py})
			return true
		}
		q.dividir()
	}
	for _, f := range q.filhos {
		if f.inserir(px, py) {
			return true
		}
	}
	return false
}

func (q *QuadTree) dividir() {
	mL, mA := q.limite.larg/2, q.limite.alt/2
	for _, dx := range []float64{0, mL} {
		for _, dy := range []float64{0, mA} {
			q.filhos = append(q.filhos, NovaQuadTree(Retangulo{q.limite.x + dx, q.limite.y + dy, mL, mA}, q.capacidade))
		}
	}
	for _, p := range q.pontos {
		for _, f := range q.filhos {
			if f.inserir(p.x, p.y) {
				break
			}
		}
	}
	q.pontos = nil
}

func (q *QuadTree) buscar(area Retangulo, achados []Ponto) []Ponto {
	if !q.limite.intersecta(area) {
		return achados // poda: subárvore inteira descartada
	}
	for _, p := range q.pontos {
		if area.contem(p.x, p.y) {
			achados = append(achados, p)
		}
	}
	for _, f := range q.filhos {
		achados = f.buscar(area, achados)
	}
	return achados
}

A busca por janela é eficiente porque descarta subárvores inteiras cujo retângulo não toca a área procurada. Para "os K mais próximos", explora-se primeiro o nó que contém o ponto e depois os vizinhos, em ordem de distância, até ter K candidatos e nenhum nó restante poder conter algo mais perto.

## Custos

| Operação | Custo típico |
| --- | --- |
| Inserção | Proporcional à profundidade, O(log n) em dados bem espalhados |
| Busca por janela | Profundidade mais o número de nós visitados na região |
| Pior caso | Pontos muito aglomerados criam árvores profundas; limite a profundidade máxima |

## Quadtrees em sistemas reais

Num serviço de "lugares próximos", os lugares mudam pouco. Uma estratégia comum:

- Construir a quadtree **em memória** a partir do banco ao iniciar o servidor. Com centenas de milhões de lugares, isso cabe em algumas dezenas de GB e leva minutos.
- Ter várias réplicas do serviço de busca, cada uma com a árvore completa, atrás de um load balancer.
- Aplicar inserções novas em lote (por exemplo, à noite) ou reconstruir a árvore periodicamente, em vez de manter escrita concorrente.

Para pontos que se movem o tempo todo, como motoristas, atualizar a árvore a cada poucos segundos gera muito rebalanceamento. Nesse caso, uma grade fixa (geohash ou H3) em memória costuma ser mais simples.

<Callout type="tip" title="Primo em 3D">
A mesma ideia com oito filhos em vez de quatro se chama **octree** e é usada em jogos, renderização e nuvens de pontos 3D.
</Callout>

## Quadtree versus geohash

| Critério | Geohash | Quadtree |
| --- | --- | --- |
| Granularidade | Fixa por precisão | Adapta-se à densidade |
| Onde vive | Em qualquer índice de string no banco | Normalmente em memória |
| Atualizações frequentes | Baratas | Podem causar divisões e fusões |
| Busca de K vizinhos | Consultar células e expandir | Natural na árvore |
