Nenhum algoritmo que só compara elementos consegue ordenar em menos de O(n log n) no pior caso. **Counting sort** e **bucket sort** fogem desse limite usando informação extra sobre os valores: eles são inteiros num intervalo pequeno, ou estão bem espalhados num intervalo conhecido.

## Counting sort

Se os valores estão entre 0 e k, crie um array de contagens com k + 1 posições, conte quantas vezes cada valor aparece e reescreva a saída na ordem dos índices.

<CodeTabs>

```python
def counting_sort(nums, k):
    contagem = [0] * (k + 1)
    for x in nums:
        contagem[x] += 1
    saida = []
    for valor, vezes in enumerate(contagem):
        saida.extend([valor] * vezes)
    return saida

print(counting_sort([4, 1, 3, 4, 0, 2, 1], 4))  # [0, 1, 1, 2, 3, 4, 4]
```

### Versão estável, para ordenar registros

Quando você ordena objetos por uma chave inteira (e não números soltos), precisa preservar a ordem dos empatados. A técnica: transforme as contagens em **somas de prefixo**, que dizem onde cada valor termina na saída, e percorra a entrada de trás para frente.

<CodeTabs>

```python
def counting_sort_estavel(itens, chave, k):
    contagem = [0] * (k + 1)
    for it in itens:
        contagem[chave(it)] += 1
    for v in range(1, k + 1):
        contagem[v] += contagem[v - 1]   # posição final (exclusiva) de cada valor
    saida = [None] * len(itens)
    for it in reversed(itens):
        contagem[chave(it)] -= 1
        saida[contagem[chave(it)]] = it
    return saida

alunos = [("lia", 2), ("rui", 1), ("eva", 2), ("max", 0)]
print(counting_sort_estavel(alunos, lambda a: a[1], 2))
# [('max', 0), ('rui', 1), ('lia', 2), ('eva', 2)]
```

```javascript
function countingSortEstavel(itens, chave, k) {
  const contagem = new Array(k + 1).fill(0);
  for (const it of itens) {
    contagem[chave(it)]++;
  }
  for (let v = 1; v <= k; v++) {
    contagem[v] += contagem[v - 1];   // posição final (exclusiva) de cada valor
  }
  const saida = new Array(itens.length).fill(null);
  for (let i = itens.length - 1; i >= 0; i--) {
    const it = itens[i];
    contagem[chave(it)]--;
    saida[contagem[chave(it)]] = it;
  }
  return saida;
}
```

```java
import java.util.*;
import java.util.function.ToIntFunction;

static <T> List<T> countingSortEstavel(List<T> itens, ToIntFunction<T> chave, int k) {
    int[] contagem = new int[k + 1];
    for (T it : itens) {
        contagem[chave.applyAsInt(it)]++;
    }
    for (int v = 1; v <= k; v++) {
        contagem[v] += contagem[v - 1];   // posição final (exclusiva) de cada valor
    }
    List<T> saida = new ArrayList<>(Collections.nCopies(itens.size(), (T) null));
    for (int i = itens.size() - 1; i >= 0; i--) {
        T it = itens.get(i);
        contagem[chave.applyAsInt(it)]--;
        saida.set(contagem[chave.applyAsInt(it)], it);
    }
    return saida;
}
```

## Bucket sort

Se os valores estão espalhados de forma razoavelmente uniforme num intervalo, divida o intervalo em n baldes, jogue cada valor no seu balde, ordene cada balde (que terá poucos elementos) e concatene.

<CodeTabs>

```python
def bucket_sort(nums):
    if not nums:
        return []
    lo, hi = min(nums), max(nums)
    n = len(nums)
    largura = (hi - lo) / n or 1
    baldes = [[] for _ in range(n)]
    for x in nums:
        i = min(int((x - lo) / largura), n - 1)
        baldes[i].append(x)
    saida = []
    for b in baldes:
        saida.extend(sorted(b))
    return saida

print(bucket_sort([0.42, 0.32, 0.23, 0.52, 0.25, 0.47, 0.51]))
# [0.23, 0.25, 0.32, 0.42, 0.47, 0.51, 0.52]
```

```javascript
function bucketSort(nums) {
  if (nums.length === 0) {
    return [];
  }
  const lo = Math.min(...nums), hi = Math.max(...nums);
  const n = nums.length;
  const largura = (hi - lo) / n || 1;
  const baldes = Array.from({ length: n }, () => []);
  for (const x of nums) {
    const i = Math.min(Math.floor((x - lo) / largura), n - 1);
    baldes[i].push(x);
  }
  const saida = [];
  for (const b of baldes) {
    saida.push(...b.sort((a, c) => a - c));
  }
  return saida;
}
```

```go
import (
	"fmt"
	"sort"
)

func bucketSort(nums []float64) []float64 {
	if len(nums) == 0 {
		return []float64{}
	}
	lo, hi := nums[0], nums[0]
	for _, x := range nums {
		if x < lo {
			lo = x
		}
		if x > hi {
			hi = x
		}
	}
	n := len(nums)
	largura := (hi - lo) / float64(n)
	if largura == 0 {
		largura = 1
	}
	baldes := make([][]float64, n)
	for _, x := range nums {
		i := int((x - lo) / largura)
		if i > n-1 {
			i = n - 1
		}
		baldes[i] = append(baldes[i], x)
	}
	saida := []float64{}
	for _, b := range baldes {
		sort.Float64s(b)
		saida = append(saida, b...)
	}
	return saida
}

func main() {
	fmt.Println(bucketSort([]float64{0.42, 0.32, 0.23, 0.52, 0.25, 0.47, 0.51}))
	// [0.23 0.25 0.32 0.42 0.47 0.51 0.52]
}
```

## Baldes por frequência: o uso mais comum em entrevista

Em entrevistas, a ideia de baldes aparece mais como truque do que como sort completo. Exemplo: os k elementos mais frequentes. A frequência de qualquer elemento está entre 1 e n, então dá para usar a frequência como índice do balde:

<CodeTabs>

```python
from collections import Counter

def top_k_frequentes(nums, k):
    freq = Counter(nums)
    baldes = [[] for _ in range(len(nums) + 1)]
    for valor, f in freq.items():
        baldes[f].append(valor)
    resp = []
    for f in range(len(baldes) - 1, 0, -1):
        for valor in baldes[f]:
            resp.append(valor)
            if len(resp) == k:
                return resp
    return resp

print(top_k_frequentes([5, 5, 5, 2, 2, 9], 2))  # [5, 2]
```

```javascript
function topKFrequentes(nums, k) {
  const freq = new Map();
  for (const x of nums) {
    freq.set(x, (freq.get(x) ?? 0) + 1);

      resp.push(valor);
      if (resp.length === k) {
        return resp;
      }
    }
  }
  return resp;
}
```

```cpp
    vector<vector<int>> baldes(nums.size() + 1);
    for (const auto& [valor, f] : freq) {
        baldes[f].push_back(valor);
    }
    vector<int> resp;
    for (int f = (int)baldes.size() - 1; f > 0; f--) {
        for (int valor : baldes[f]) {
            resp.push_back(valor);
            if (resp.size() == k) {
                return resp;
            }
        }
    }
    return resp;
}

int main() {
    vector<int> r = topKFrequentes({5, 5, 5, 2, 2, 9}, 2);
    cout << "[";
    for (size_t i = 0; i < r.size(); i++) cout << (i ? ", " : "") << r[i];
    cout << "]" << endl;  // [5, 2]
}
```

```csharp
static List<int> TopKFrequentes(int[] nums, int k)

        freq[x] = freq.TryGetValue(x, out int c) ? c + 1 : 1;

}
```

```go
import "fmt"

func topKFrequentes(nums []int, k int) []int {
	freq := map[int]int{}
	for _, x := range nums {
		freq[x]++
	}
	baldes := make([][]int, len(nums)+1)
	for valor, f := range freq {
		baldes[f] = append(baldes[f], valor)
	}
	resp := []int{}
	for f := len(baldes) - 1; f > 0; f-- {
		for _, valor := range baldes[f] {
			resp = append(resp, valor)
			if len(resp) == k {
				return resp
			}
		}
	}
	return resp
}

	fmt.Println(topKFrequentes([]int{5, 5, 5, 2, 2, 9}, 2)) // [5 2]
```

## Quando reconhecer

Sinais no enunciado: valores inteiros com **limite pequeno** ("notas de 0 a 100", "idades", "letras minúsculas"), **frequências** que você quer agrupar, ou pedido explícito de tempo linear para ordenar. Se os valores não têm limite conhecido ou são arbitrários, volte aos sorts por comparação.

## Na entrevista

Quando o entrevistador pergunta "dá para fazer melhor que O(n log n)?", verifique o intervalo dos valores. Responda com a condição: "se os valores estão entre 0 e k, counting sort faz em O(n + k); isso só compensa se k não for muito maior que n". Esse cuidado em declarar a hipótese conta muito.
