Um **grafo** é um conjunto de vértices ligados por arestas. Ele modela redes sociais, mapas, dependências entre tarefas e qualquer situação em que coisas se relacionam. Árvores e listas ligadas são casos particulares de grafos.

## Representação

A forma mais usada em entrevistas é a **lista de adjacência**: para cada vértice, a lista dos vizinhos.

<CodeTabs>

```python
    grafo[b].append(a)   # omita esta linha se o grafo for dirigido
```

| Representação | Espaço | Checar se existe aresta | Quando usar |
| --- | --- | --- | --- |
| Lista de adjacência | O(V + E) | O(grau) | Quase sempre |
| Matriz de adjacência | O(V²) | O(1) | Grafos densos e pequenos |
| Grade implícita | — | — | Vizinhos são as células acima, abaixo e aos lados |

Em problemas de grade, você nem monta o grafo: os vizinhos de `(l, c)` são calculados na hora.

## BFS e DFS

Os dois percursos visitam cada vértice e cada aresta uma vez, O(V + E). A diferença é a ordem.

<CodeTabs>

```python
from collections import deque

def bfs_distancias(grafo, origem):
    dist = {origem: 0}
    fila = deque([origem])
    while fila:
        v = fila.popleft()
        for w in grafo[v]:
            if w not in dist:
                dist[w] = dist[v] + 1
                fila.append(w)
    return dist

def dfs(grafo, v, visitado):
    visitado.add(v)
    for w in grafo[v]:
        if w not in visitado:
            dfs(grafo, w, visitado)
    return visitado

print(bfs_distancias(grafo, 0))   # distância em arestas a partir de 0
print(sorted(dfs(grafo, 0, set())))
```

```javascript
function bfsDistancias(grafo, origem) {
  const dist = new Map([[origem, 0]]);
  const fila = [origem];
  for (let ini = 0; ini < fila.length; ini++) {   // índice de leitura no lugar de shift(), que é O(n)
    const v = fila[ini];
    for (const w of grafo.get(v)) {
      if (!dist.has(w)) {
        dist.set(w, dist.get(v) + 1);
        fila.push(w);
      }
    }
  }
  return dist;
}

function dfs(grafo, v, visitado) {
  visitado.add(v);
  for (const w of grafo.get(v)) {
    if (!visitado.has(w)) {
      dfs(grafo, w, visitado);
    }
  }
  return visitado;
}

console.log(Object.fromEntries(bfsDistancias(grafo, 0)));   // distância em arestas a partir de 0
console.log([...dfs(grafo, 0, new Set())].sort((a, b) => a - b));
```

```csharp
static SortedDictionary<int, int> BfsDistancias(SortedDictionary<int, List<int>> grafo, int origem)
{
    var dist = new SortedDictionary<int, int> { [origem] = 0 };
    var fila = new Queue<int>();
    fila.Enqueue(origem);
    while (fila.Count > 0)
    {
        int v = fila.Dequeue();
        foreach (int w in grafo[v])
        {
            if (!dist.ContainsKey(w))
            {
                dist[w] = dist[v] + 1;
                fila.Enqueue(w);
            }
        }
    }
    return dist;
}
```

```go
import "sort"

func bfsDistancias(grafo map[int][]int, origem int) map[int]int {
	dist := map[int]int{origem: 0}
	fila := []int{origem}
	for len(fila) > 0 {
		v := fila[0]
		fila = fila[1:]
		for _, w := range grafo[v] {
			if _, visto := dist[w]; !visto {
				dist[w] = dist[v] + 1
				fila = append(fila, w)
			}
		}
	}
	return dist
}

func dfs(grafo map[int][]int, v int, visitado map[int]bool) map[int]bool {
	visitado[v] = true
	for _, w := range grafo[v] {
		if !visitado[w] {
			dfs(grafo, w, visitado)
		}
	}
	return visitado
}

func main() {
	fmt.Println(bfsDistancias(grafo, 0)) // distância em arestas a partir de 0
	vis := dfs(grafo, 0, map[int]bool{})
	ordenados := []int{}
	for v := range vis {
		ordenados = append(ordenados, v)
	}
	sort.Ints(ordenados)
	fmt.Println(ordenados)
}
```

</CodeTabs>

**BFS** encontra o **menor caminho em número de arestas** em grafos sem peso. **DFS** é mais simples para explorar componentes, detectar ciclos e contar ilhas.

<Callout type="warning" title="Marque ao enfileirar">
Na BFS, marque o vértice como visitado no momento em que ele entra na fila, não quando sai. Caso contrário, o mesmo vértice pode ser enfileirado várias vezes e a complexidade explode.
</Callout>

## Ordenação topológica

Em um grafo **dirigido e sem ciclos**, a ordenação topológica lista os vértices de modo que toda aresta vá de um vértice anterior para um posterior. É o problema de "em que ordem fazer as matérias, respeitando os pré-requisitos". O algoritmo de Kahn processa primeiro quem não tem dependências pendentes:

<CodeTabs>

```python
from collections import deque

def ordem_topologica(n, dependencias):
    grafo = [[] for _ in range(n)]
    grau_entrada = [0] * n
    for antes, depois in dependencias:
        grafo[antes].append(depois)
        grau_entrada[depois] += 1
    fila = deque(v for v in range(n) if grau_entrada[v] == 0)
    ordem = []
    while fila:
        v = fila.popleft()
        ordem.append(v)
        for w in grafo[v]:
            grau_entrada[w] -= 1
            if grau_entrada[w] == 0:
                fila.append(w)
    return ordem if len(ordem) == n else []  # vazio: existe ciclo

print(ordem_topologica(4, [(0, 1), (0, 2), (1, 3), (2, 3)]))  # [0, 1, 2, 3]
```

```javascript
function ordemTopologica(n, dependencias) {
  const grafo = Array.from({ length: n }, () => []);
  const grauEntrada = new Array(n).fill(0);
  for (const [antes, depois] of dependencias) {
    grafo[antes].push(depois);
    grauEntrada[depois]++;
  }
  const fila = [];
  for (let v = 0; v < n; v++) {
    if (grauEntrada[v] === 0) fila.push(v);
  }
  const ordem = [];
  for (let ini = 0; ini < fila.length; ini++) {
    const v = fila[ini];
    ordem.push(v);
    for (const w of grafo[v]) {
      grauEntrada[w]--;
      if (grauEntrada[w] === 0) {
        fila.push(w);
      }
    }
  }
  return ordem.length === n ? ordem : [];   // vazio: existe ciclo
}

console.log(ordemTopologica(4, [[0, 1], [0, 2], [1, 3], [2, 3]]));  // [0, 1, 2, 3]
```

```java
import java.util.*;

static List<Integer> ordemTopologica(int n, int[][] dependencias) {
    List<List<Integer>> grafo = new ArrayList<>();
    for (int i = 0; i < n; i++) grafo.add(new ArrayList<>());
    int[] grauEntrada = new int[n];
    for (int[] d : dependencias) {
        grafo.get(d[0]).add(d[1]);
        grauEntrada[d[1]]++;
    }
    Deque<Integer> fila = new ArrayDeque<>();
    for (int v = 0; v < n; v++) {
        if (grauEntrada[v] == 0) fila.add(v);
    }
    List<Integer> ordem = new ArrayList<>();
    while (!fila.isEmpty()) {
        int v = fila.poll();
        ordem.add(v);
        for (int w : grafo.get(v)) {
            grauEntrada[w]--;
            if (grauEntrada[w] == 0) {
                fila.add(w);
            }
        }
    }
    return ordem.size() == n ? ordem : new ArrayList<>();   // vazio: existe ciclo
}
```

```csharp
static List<int> OrdemTopologica(int n, int[][] dependencias)
{
    var grafo = new List<int>[n];
    for (int i = 0; i < n; i++) grafo[i] = new List<int>();
    var grauEntrada = new int[n];
    foreach (int[] d in dependencias)
    {
        grafo[d[0]].Add(d[1]);

                fila.Enqueue(w);
```

Se nem todos os vértices entram na ordem, sobrou um ciclo, e isso já responde perguntas como "é possível concluir todas as matérias?".

## Na entrevista

Comece esclarecendo: o grafo é dirigido? Tem pesos? Pode ser desconexo? Pode ter ciclos? Cada resposta muda o algoritmo. Grafos desconexos pedem um laço externo que inicie uma busca a partir de cada vértice ainda não visitado; esquecer isso é o erro mais comum.
