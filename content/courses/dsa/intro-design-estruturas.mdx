Alguns problemas não pedem uma função, e sim uma **classe**: "implemente um cache LRU", "projete uma pilha que devolve o mínimo em O(1)", "crie um contador de acessos nos últimos 5 minutos". São os problemas de **design de estruturas de dados**. O desafio não é um algoritmo novo, mas **combinar** estruturas que você já conhece para que cada operação atinja a complexidade pedida.

## O roteiro

1. **Liste as operações e a complexidade exigida de cada uma.** Escreva isso antes de tudo. É a especificação.
2. **Para cada operação, pergunte qual estrutura a faz no custo pedido.** Busca por chave em O(1): hash map. Ordem de inserção ou de uso: lista ligada ou deque. Mínimo/máximo: heap ou pilha auxiliar. Ordem com busca: árvore balanceada.
3. **Se nenhuma estrutura sozinha atende tudo, combine duas** e mantenha-as sincronizadas: toda operação que altera uma altera a outra.
4. **Defina os invariantes** (o que é sempre verdade sobre o estado) e verifique cada método contra eles.

## Exemplo: pilha com mínimo em O(1)

Operações: `empilhar`, `desempilhar`, `topo`, `minimo`, todas O(1). Uma pilha comum faz as três primeiras; o mínimo exigiria varrer tudo. A solução é guardar, junto de cada elemento, o mínimo **até aquele ponto**:

<CodeTabs>

```python
class PilhaMin:
    def __init__(self):
        self.dados = []  # pares (valor, minimo_ate_aqui)

    def empilhar(self, x):
        menor = min(x, self.dados[-1][1]) if self.dados else x
        self.dados.append((x, menor))

    def desempilhar(self):
        return self.dados.pop()[0]

    def topo(self):
        return self.dados[-1][0]

    def minimo(self):
        return self.dados[-1][1]

p = PilhaMin()
for v in [5, 3, 7, 2]:
    p.empilhar(v)
print(p.minimo())  # 2
p.desempilhar()
print(p.minimo())  # 3
```

```go
import "fmt"

type par struct{ valor, minimo int }

type PilhaMin struct {
	dados []par // pares (valor, minimoAteAqui)
}

func (p *PilhaMin) empilhar(x int) {
	menor := x
	if len(p.dados) > 0 && p.dados[len(p.dados)-1].minimo < x {
		menor = p.dados[len(p.dados)-1].minimo
	}
	p.dados = append(p.dados, par{x, menor})
}

func (p *PilhaMin) desempilhar() int {
	topo := p.dados[len(p.dados)-1]
	p.dados = p.dados[:len(p.dados)-1]
	return topo.valor
}

func (p *PilhaMin) topo() int {
	return p.dados[len(p.dados)-1].valor
}

func (p *PilhaMin) minimo() int {
	return p.dados[len(p.dados)-1].minimo
}

func main() {
	p := &PilhaMin{}
	for _, v := range []int{5, 3, 7, 2} {
		p.empilhar(v)
	}
	fmt.Println(p.minimo()) // 2
	p.desempilhar()
	fmt.Println(p.minimo()) // 3
}
```

Invariante: o segundo campo do topo é sempre o mínimo de toda a pilha. Como desempilhar remove o par inteiro, o mínimo anterior "reaparece" sozinho. Espaço O(n).

## Exemplo: cache LRU

Operações: `get(chave)` e `put(chave, valor)` em O(1), descartando o item **menos recentemente usado** quando a capacidade estoura. Precisamos de busca por chave (hash map) e de ordem de uso com remoção e movimentação em O(1) (lista duplamente ligada). Em Python, `OrderedDict` já é essa combinação pronta:

<CodeTabs>

```python
from collections import OrderedDict

class CacheLRU:
    def __init__(self, capacidade):
        self.cap = capacidade
        self.itens = OrderedDict()

    def get(self, chave):
        if chave not in self.itens:
            return -1
        self.itens.move_to_end(chave)      # passa a ser o mais recente
        return self.itens[chave]

    def put(self, chave, valor):
        if chave in self.itens:
            self.itens.move_to_end(chave)
        self.itens[chave] = valor
        if len(self.itens) > self.cap:
            self.itens.popitem(last=False)  # remove o mais antigo

c = CacheLRU(2)
c.put("a", 1); c.put("b", 2)
c.get("a")          # "a" vira o mais recente
c.put("c", 3)       # estoura: remove "b"
print(c.get("b"), c.get("a"), c.get("c"))  # -1 1 3
```

```javascript
class CacheLRU {
  constructor(capacidade) {
    this.cap = capacidade;
    this.itens = new Map();             // o Map guarda a ordem de inserção: o primeiro é o mais antigo
  }

  get(chave) {
    if (!this.itens.has(chave)) {
      return -1;
    }
    const valor = this.itens.get(chave);
    this.itens.delete(chave);           // reinserir passa a chave a ser a mais recente
    this.itens.set(chave, valor);
    return valor;
  }

  put(chave, valor) {
    if (this.itens.has(chave)) {
      this.itens.delete(chave);
    }
    this.itens.set(chave, valor);
    if (this.itens.size > this.cap) {
      const maisAntigo = this.itens.keys().next().value;
      this.itens.delete(maisAntigo);    // remove o mais antigo
    }
  }
}

const c = new CacheLRU(2);
c.put("a", 1); c.put("b", 2);
c.get("a");          // "a" vira o mais recente
c.put("c", 3);       // estoura: remove "b"
console.log(c.get("b"), c.get("a"), c.get("c"));  // -1 1 3
```

```java
import java.util.*;

class CacheLRU {
    private final int cap;
    // accessOrder = true: cada get/put move a chave para o fim (a mais recente)
    private final LinkedHashMap<String, Integer> itens = new LinkedHashMap<>(16, 0.75f, true);

    CacheLRU(int capacidade) {
        this.cap = capacidade;
    }

    int get(String chave) {
        return itens.getOrDefault(chave, -1);
    }

    void put(String chave, int valor) {
        itens.put(chave, valor);
        if (itens.size() > cap) {
            String maisAntigo = itens.keySet().iterator().next();
            itens.remove(maisAntigo);       // remove o mais antigo
        }
    }
}

public static void main(String[] args) {
    CacheLRU c = new CacheLRU(2);
    c.put("a", 1); c.put("b", 2);
    c.get("a");          // "a" vira o mais recente
    c.put("c", 3);       // estoura: remove "b"
    System.out.println(c.get("b") + " " + c.get("a") + " " + c.get("c"));  // -1 1 3
}
```

```cpp
#include <iostream>
#include <list>
#include <string>
#include <unordered_map>
#include <utility>
using namespace std;

class CacheLRU {
    int cap;
    list<pair<string, int>> ordem;   // frente = mais recente, fundo = mais antigo
    unordered_map<string, list<pair<string, int>>::iterator> indice;

public:
    CacheLRU(int capacidade) : cap(capacidade) {}

    int get(const string& chave) {
        auto it = indice.find(chave);
        if (it == indice.end()) {
            return -1;
        }
        ordem.splice(ordem.begin(), ordem, it->second);   // passa a ser o mais recente
        return it->second->second;
    }

    void put(const string& chave, int valor) {
        auto it = indice.find(chave);
        if (it != indice.end()) {
            it->second->second = valor;
            ordem.splice(ordem.begin(), ordem, it->second);
            return;
        }
        ordem.push_front({chave, valor});
        indice[chave] = ordem.begin();
        if ((int)ordem.size() > cap) {
            indice.erase(ordem.back().first);             // remove o mais antigo
            ordem.pop_back();
        }
    }
};

int main() {
    CacheLRU c(2);
    c.put("a", 1); c.put("b", 2);
    c.get("a");          // "a" vira o mais recente
    c.put("c", 3);       // estoura: remove "b"
    cout << c.get("b") << " " << c.get("a") << " " << c.get("c") << endl;  // -1 1 3
}
```

```csharp
using System;
using System.Collections.Generic;

class CacheLRU
{
    private readonly int cap;
    private readonly LinkedList<KeyValuePair<string, int>> ordem = new LinkedList<KeyValuePair<string, int>>();   // início = mais recente
    private readonly Dictionary<string, LinkedListNode<KeyValuePair<string, int>>> indice =
        new Dictionary<string, LinkedListNode<KeyValuePair<string, int>>>();

    public CacheLRU(int capacidade)
    {
        cap = capacidade;
    }

    public int Get(string chave)
    {
        if (!indice.TryGetValue(chave, out var no))
        {
            return -1;
        }
        ordem.Remove(no);
        ordem.AddFirst(no);                // passa a ser o mais recente
        return no.Value.Value;
    }

    public void Put(string chave, int valor)
    {
        if (indice.TryGetValue(chave, out var existente))
        {
            ordem.Remove(existente);
            indice.Remove(chave);
        }
        indice[chave] = ordem.AddFirst(new KeyValuePair<string, int>(chave, valor));
        if (ordem.Count > cap)
        {
            var maisAntigo = ordem.Last;
            ordem.RemoveLast();            // remove o mais antigo
            indice.Remove(maisAntigo.Value.Key);
        }
    }
}

static void Main()
{
    var c = new CacheLRU(2);
    c.Put("a", 1); c.Put("b", 2);
    c.Get("a");          // "a" vira o mais recente
    c.Put("c", 3);       // estoura: remove "b"
    Console.WriteLine(c.Get("b") + " " + c.Get("a") + " " + c.Get("c"));  // -1 1 3
}
```

```go
import (
	"container/list"
	"fmt"
)

type entrada struct {
	chave string
	valor int
}

type CacheLRU struct {
	cap    int
	ordem  *list.List // frente = mais recente, fundo = mais antigo
	indice map[string]*list.Element
}

func novoCache(capacidade int) *CacheLRU {
	return &CacheLRU{cap: capacidade, ordem: list.New(), indice: map[string]*list.Element{}}
}

func (c *CacheLRU) get(chave string) int {
	el, ok := c.indice[chave]
	if !ok {
		return -1
	}
	c.ordem.MoveToFront(el) // passa a ser o mais recente
	return el.Value.(entrada).valor
}

func (c *CacheLRU) put(chave string, valor int) {
	if el, ok := c.indice[chave]; ok {
		el.Value = entrada{chave, valor}
		c.ordem.MoveToFront(el)
		return
	}
	c.indice[chave] = c.ordem.PushFront(entrada{chave, valor})
	if c.ordem.Len() > c.cap {
		maisAntigo := c.ordem.Back() // remove o mais antigo
		c.ordem.Remove(maisAntigo)
		delete(c.indice, maisAntigo.Value.(entrada).chave)
	}
}

func main() {
	c := novoCache(2)
	c.put("a", 1)
	c.put("b", 2)
	c.get("a")    // "a" vira o mais recente
	c.put("c", 3) // estoura: remove "b"
	fmt.Println(c.get("b"), c.get("a"), c.get("c")) // -1 1 3
}
```

## Combinações recorrentes

| Requisito | Combinação típica |
| --- | --- |
| Busca O(1) + ordem de uso | hash map + lista duplamente ligada |
| Inserir, remover e sortear aleatório em O(1) | lista + hash map de valor para índice (troca com o último ao remover) |
| Mínimo/máximo em O(1) com pilha | pilha de pares ou pilha auxiliar |
| Mediana de um fluxo | dois heaps (max-heap da metade baixa, min-heap da alta) |
| Eventos numa janela de tempo | deque, descartando da frente o que expirou |

## Quando reconhecer

O enunciado descreve uma **classe** com vários métodos e dá a complexidade de cada um, ou diz "projete", "implemente uma estrutura que suporte...". A pista mais forte é uma exigência de O(1) para operações que nenhuma estrutura isolada oferece juntas.

## Na entrevista

Escreva a tabela operação → complexidade exigida antes de codificar e mostre qual estrutura cobre cada linha. Fale dos invariantes. Depois implemente método a método, testando um cenário curto ao final, como no exemplo do cache. Follow-ups frequentes: segurança com múltiplas threads, capacidade zero e chaves repetidas.
