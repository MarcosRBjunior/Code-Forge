A BFS encontra o caminho com menos arestas, mas no mundo real as arestas têm custo: quilômetros, milissegundos de latência, reais de pedágio. Quando os pesos variam, o caminho com menos arestas pode ser o mais caro. O algoritmo de **Dijkstra** resolve o caminho mínimo a partir de uma origem em grafos com pesos **não negativos**.

## A ideia

Mantenha a melhor distância conhecida até cada vértice. A cada passo, pegue o vértice **ainda não finalizado com a menor distância** e finalize-o: como não há pesos negativos, nenhum caminho que passe por vértices mais distantes poderia chegar a ele por menos. Em seguida, **relaxe** suas arestas, ou seja, verifique se passar por ele melhora a distância dos vizinhos.

Para pegar o menor rapidamente, use um **min-heap** de pares `(distância, vértice)`.

```text
       4
   A ----- B
   |       |
 1 |       | 1
   |       |
   C ----- D
       2
Origem A: dist(C)=1, dist(D)=3 (via C), dist(B)=4.
Note que B pela rota A-C-D-B também custa 4: empate.
```

```python
import heapq

def dijkstra(n, arestas, origem):
    g = [[] for _ in range(n)]
    for u, v, w in arestas:
        g[u].append((v, w))
        g[v].append((u, w))          # remova esta linha se o grafo for dirigido
    dist = [float("inf")] * n
    dist[origem] = 0
    heap = [(0, origem)]
    while heap:
        d, u = heapq.heappop(heap)
        if d > dist[u]:
            continue                 # entrada velha: já achamos algo melhor
        for v, w in g[u]:
            nd = d + w
            if nd < dist[v]:
                dist[v] = nd
                heapq.heappush(heap, (nd, v))
    return dist

# A=0, B=1, C=2, D=3
print(dijkstra(4, [(0, 1, 4), (0, 2, 1), (2, 3, 2), (3, 1, 1)], 0))  # [0, 4, 1, 3]
```

```javascript
// JavaScript não tem heap na biblioteca padrão: esta classe mínima guarda pares [prioridade, vértice]
// e compara pela prioridade (é reaproveitada no exemplo seguinte)
class MinHeap {
  constructor() { this.d = []; }
  get tamanho() { return this.d.length; }
  push(x) {
    const d = this.d;
    d.push(x);
    let i = d.length - 1;
    while (i > 0) {
      const p = (i - 1) >> 1;
      if (d[p][0] <= d[i][0]) break;
      [d[p], d[i]] = [d[i], d[p]];
      i = p;
    }
  }
  pop() {
    const d = this.d;
    const topo = d[0];
    const ultimo = d.pop();
    if (d.length > 0) {
      d[0] = ultimo;
      let i = 0;
      while (true) {
        let m = i;
        const l = 2 * i + 1, r = 2 * i + 2;
        if (l < d.length && d[l][0] < d[m][0]) m = l;
        if (r < d.length && d[r][0] < d[m][0]) m = r;
        if (m === i) break;
        [d[m], d[i]] = [d[i], d[m]];
        i = m;
      }
    }
    return topo;
  }
}

function dijkstra(n, arestas, origem) {
  const g = Array.from({ length: n }, () => []);
  for (const [u, v, w] of arestas) {
    g[u].push([v, w]);
    g[v].push([u, w]);               // remova esta linha se o grafo for dirigido
  }
  const dist = new Array(n).fill(Infinity);
  dist[origem] = 0;
  const heap = new MinHeap();
  heap.push([0, origem]);
  while (heap.tamanho > 0) {
    const [d, u] = heap.pop();
    if (d > dist[u]) {
      continue;                      // entrada velha: já achamos algo melhor
    }
    for (const [v, w] of g[u]) {
      const nd = d + w;
      if (nd < dist[v]) {
        dist[v] = nd;
        heap.push([nd, v]);
      }
    }
  }
  return dist;
}
```

```java
                heap.add(new int[]{nd, v});

        }
    }
    return dist;
}
```

```csharp
using System;
using System.Collections.Generic;

// PriorityQueue só existe a partir do .NET 6; aqui um SortedSet de (distância, id, vértice) faz o papel do heap
// (o id desempata entradas iguais, porque um SortedSet não aceita duplicatas)
static int[] Dijkstra(int n, int[][] arestas, int origem)
{
    var g = new List<int[]>[n];
    for (int i = 0; i < n; i++) g[i] = new List<int[]>();
    foreach (int[] a in arestas)
    {
        g[a[0]].Add(new[] { a[1], a[2] });
        g[a[1]].Add(new[] { a[0], a[2] });     // remova esta linha se o grafo for dirigido
    }
    var dist = new int[n];
    for (int i = 0; i < n; i++) dist[i] = int.MaxValue;
    dist[origem] = 0;
    var heap = new SortedSet<Tuple<int, int, int>>();
    int id = 0;
    heap.Add(Tuple.Create(0, id++, origem));
    while (heap.Count > 0)
    {
        var topo = heap.Min;
        heap.Remove(topo);
        int d = topo.Item1, u = topo.Item3;
        if (d > dist[u])
        {
            continue;                           // entrada velha: já achamos algo melhor
        }
        foreach (int[] aresta in g[u])
        {
            int v = aresta[0], w = aresta[1];
            int nd = d + w;
            if (nd < dist[v])
            {
                dist[v] = nd;
                heap.Add(Tuple.Create(nd, id++, v));
            }
        }
    }
    return dist;
}

{
```

```go
import (
	"container/heap"
	"fmt"
	"math"
)

type item struct{ dist, vertice int }
type filaMin []item

func (h filaMin) Len() int           { return len(h) }
func (h filaMin) Less(i, j int) bool { return h[i].dist < h[j].dist }
func (h filaMin) Swap(i, j int)      { h[i], h[j] = h[j], h[i] }
func (h *filaMin) Push(x any)        { *h = append(*h, x.(item)) }
func (h *filaMin) Pop() any {
	antigo := *h
	n := len(antigo)
	x := antigo[n-1]
	*h = antigo[:n-1]
	return x
}

type aresta struct{ v, w int }

func dijkstra(n int, arestas [][3]int, origem int) []int {
	g := make([][]aresta, n)
	for _, a := range arestas {
		u, v, w := a[0], a[1], a[2]
		g[u] = append(g[u], aresta{v, w})
		g[v] = append(g[v], aresta{u, w}) // remova esta linha se o grafo for dirigido
	}
	dist := make([]int, n)
	for i := range dist {
		dist[i] = math.MaxInt64
	}
	dist[origem] = 0
	h := &filaMin{{0, origem}}
	for h.Len() > 0 {
		topo := heap.Pop(h).(item)
		d, u := topo.dist, topo.vertice
		if d > dist[u] {
			continue // entrada velha: já achamos algo melhor
		}
		for _, e := range g[u] {
			nd := d + e.w
			if nd < dist[e.v] {
				dist[e.v] = nd
				heap.Push(h, item{nd, e.v})
			}
		}
	}
	return dist
}

	fmt.Println(dijkstra(4, [][3]int{{0, 1, 4}, {0, 2, 1}, {2, 3, 2}, {3, 1, 1}}, 0)) // [0 4 1 3]
```

</CodeTabs>

Em vez de atualizar prioridades dentro do heap, empurramos uma entrada nova e ignoramos as velhas ao retirá-las (a linha `if d > dist[u]`). É a forma mais simples e a esperada em entrevistas.

**Complexidade:** O((V + E) log V) de tempo, já que cada relaxamento bem-sucedido pode empurrar uma entrada no heap. Espaço O(V + E) para o grafo e o heap.

## Reconstruindo o caminho

Para devolver o caminho, e não só o custo, guarde de onde cada vértice foi alcançado:

```python
import heapq

def caminho(g, origem, destino):
    dist, anterior = {origem: 0}, {origem: None}
    heap = [(0, origem)]
    while heap:
        d, u = heapq.heappop(heap)
        if u == destino:
            break                    # parada antecipada: destino finalizado
        if d > dist[u]:
            continue
        for v, w in g.get(u, []):
            if d + w < dist.get(v, float("inf")):
                dist[v], anterior[v] = d + w, u
                heapq.heappush(heap, (d + w, v))
    if destino not in dist:
        return None
    rota, x = [], destino
    while x is not None:
        rota.append(x)
        x = anterior[x]
    return rota[::-1], dist[destino]

g = {"A": [("B", 4), ("C", 1)], "C": [("D", 2)], "D": [("B", 1)]}
print(caminho(g, "A", "D"))  # (['A', 'C', 'D'], 3)
```

```javascript
function caminho(g, origem, destino) {
  const dist = new Map([[origem, 0]]);
  const anterior = new Map([[origem, null]]);
  const heap = new MinHeap();
  heap.push([0, origem]);
  while (heap.tamanho > 0) {
    const [d, u] = heap.pop();
    if (u === destino) {
      break;                         // parada antecipada: destino finalizado
    }
    if (d > dist.get(u)) {
      continue;
    }
    for (const [v, w] of g[u] ?? []) {
      if (d + w < (dist.get(v) ?? Infinity)) {
        dist.set(v, d + w);
        anterior.set(v, u);
        heap.push([d + w, v]);
      }
    }
  }
  if (!dist.has(destino)) {
    return null;
  }
  const rota = [];
  let x = destino;
  while (x !== null) {
    rota.push(x);
    x = anterior.get(x);
  }
  return [rota.reverse(), dist.get(destino)];
}
```

```java
import java.util.*;

static String caminho(Map<String, List<Object[]>> g, String origem, String destino) {
    Map<String, Integer> dist = new HashMap<>();
    Map<String, String> anterior = new HashMap<>();
    dist.put(origem, 0);
    anterior.put(origem, null);
    PriorityQueue<Object[]> heap = new PriorityQueue<>(Comparator.comparingInt(x -> (Integer) x[0]));
    heap.add(new Object[]{0, origem});
    while (!heap.isEmpty()) {
        Object[] topo = heap.poll();
        int d = (Integer) topo[0];
        String u = (String) topo[1];
        if (u.equals(destino)) {
            break;                       // parada antecipada: destino finalizado
        }
        if (d > dist.get(u)) {
            continue;
        }
        for (Object[] aresta : g.getOrDefault(u, List.of())) {
            String v = (String) aresta[0];
            int w = (Integer) aresta[1];
            if (d + w < dist.getOrDefault(v, Integer.MAX_VALUE)) {
                dist.put(v, d + w);
                anterior.put(v, u);
                heap.add(new Object[]{d + w, v});
            }
        }
    }
    if (!dist.containsKey(destino)) {
        return null;
    }
    LinkedList<String> rota = new LinkedList<>();
    for (String x = destino; x != null; x = anterior.get(x)) {
        rota.addFirst(x);
    }
    return "(" + rota + ", " + dist.get(destino) + ")";
}

public static void main(String[] args) {
    Map<String, List<Object[]>> g = new HashMap<>();
    g.put("A", List.of(new Object[]{"B", 4}, new Object[]{"C", 1}));
    g.put("C", List.<Object[]>of(new Object[]{"D", 2}));
    g.put("D", List.<Object[]>of(new Object[]{"B", 1}));

}
```

```cpp
#include <algorithm>
#include <climits>
#include <iostream>
#include <map>
#include <queue>
#include <string>
#include <utility>
#include <vector>
using namespace std;

using Grafo = map<string, vector<pair<string, int>>>;

// devolve a rota; o custo vai em custo (-1 se não há caminho)
vector<string> caminho(const Grafo& g, const string& origem, const string& destino, int& custo) {
    map<string, int> dist = {{origem, 0}};
    map<string, string> anterior;
    priority_queue<pair<int, string>, vector<pair<int, string>>, greater<pair<int, string>>> heap;
    heap.push({0, origem});
    while (!heap.empty()) {
        auto [d, u] = heap.top();
        heap.pop();
        if (u == destino) {
            break;                       // parada antecipada: destino finalizado
        }
        if (d > dist[u]) {
            continue;
        }
        auto it = g.find(u);
        if (it == g.end()) {
            continue;
        }
        for (const auto& [v, w] : it->second) {
            auto atual = dist.find(v);
            if (d + w < (atual == dist.end() ? INT_MAX : atual->second)) {
                dist[v] = d + w;
                anterior[v] = u;
                heap.push({d + w, v});
            }
        }
    }
    custo = -1;
    if (!dist.count(destino)) {
        return {};
    }
    custo = dist[destino];
    vector<string> rota;
    for (string x = destino;; x = anterior[x]) {
        rota.push_back(x);
        if (x == origem) break;
    }
    reverse(rota.begin(), rota.end());
    return rota;
}

int main() {
    Grafo g = {{"A", {{"B", 4}, {"C", 1}}}, {"C", {{"D", 2}}}, {"D", {{"B", 1}}}};
    int custo;
    vector<string> rota = caminho(g, "A", "D", custo);
    cout << "([";
    for (size_t i = 0; i < rota.size(); i++) cout << (i ? ", " : "") << rota[i];
    cout << "], " << custo << ")" << endl;  // ([A, C, D], 3)
}
```

```csharp
using System;
using System.Collections.Generic;
using System.Linq;

// SortedSet de (distância, id, vértice) faz o papel do heap; o id desempata entradas iguais
static string Caminho(Dictionary<string, List<Tuple<string, int>>> g, string origem, string destino)
{
    var dist = new Dictionary<string, int> { [origem] = 0 };
    var anterior = new Dictionary<string, string> { [origem] = null };
    var heap = new SortedSet<Tuple<int, int, string>>();
    int id = 0;
    heap.Add(Tuple.Create(0, id++, origem));
    while (heap.Count > 0)
    {
        var topo = heap.Min;
        heap.Remove(topo);
        int d = topo.Item1;
        string u = topo.Item3;
        if (u == destino)
        {
            break;                       // parada antecipada: destino finalizado
        }
        if (d > dist[u])
        {
            continue;
        }
        if (!g.ContainsKey(u))
        {
            continue;
        }
        foreach (var aresta in g[u])
        {
            string v = aresta.Item1;
            int w = aresta.Item2;
            int atual = dist.ContainsKey(v) ? dist[v] : int.MaxValue;
            if (d + w < atual)
            {
                dist[v] = d + w;
                anterior[v] = u;
                heap.Add(Tuple.Create(d + w, id++, v));
            }
        }
    }
    if (!dist.ContainsKey(destino))
    {
        return null;
    }
    var rota = new List<string>();
    for (string x = destino; x != null; x = anterior[x])
    {
        rota.Add(x);
    }
    rota.Reverse();
    return "([" + string.Join(", ", rota) + "], " + dist[destino] + ")";
}
```

```go
type itemStr struct {
	dist int
	no   string
}
type filaStr []itemStr

func (h filaStr) Len() int           { return len(h) }
func (h filaStr) Less(i, j int) bool { return h[i].dist < h[j].dist }
func (h filaStr) Swap(i, j int)      { h[i], h[j] = h[j], h[i] }
func (h *filaStr) Push(x any)        { *h = append(*h, x.(itemStr)) }
func (h *filaStr) Pop() any {
	antigo := *h
	n := len(antigo)
	x := antigo[n-1]
	*h = antigo[:n-1]
	return x
}

type arestaStr struct {
	para string
	w    int
}

// devolve a rota e o custo; ok é false quando não há caminho
func caminho(g map[string][]arestaStr, origem, destino string) ([]string, int, bool) {
	dist := map[string]int{origem: 0}
	anterior := map[string]string{}
	h := &filaStr{{0, origem}}
	for h.Len() > 0 {
		topo := heap.Pop(h).(itemStr)
		d, u := topo.dist, topo.no
		if u == destino {
			break // parada antecipada: destino finalizado
		}
		if d > dist[u] {
			continue
		}
		for _, e := range g[u] {
			atual, existe := dist[e.para]
			if !existe || d+e.w < atual {
				dist[e.para] = d + e.w
				anterior[e.para] = u
				heap.Push(h, itemStr{d + e.w, e.para})
			}
		}
	}
	if _, ok := dist[destino]; !ok {
		return nil, 0, false
	}
	rota := []string{}
	for x := destino; ; x = anterior[x] {
		rota = append([]string{x}, rota...)
		if x == origem {
			break
		}
	}
	return rota, dist[destino], true
}

func main() {
	g := map[string][]arestaStr{
		"A": {{"B", 4}, {"C", 1}},
		"C": {{"D", 2}},
		"D": {{"B", 1}},
	}
	rota, custo, _ := caminho(g, "A", "D")
	fmt.Println(rota, custo) // [A C D] 3
}
```

<Callout type="warning" title="Pesos negativos quebram Dijkstra">
Se uma aresta tem peso negativo, um vértice já finalizado pode ganhar um caminho mais barato depois, e o algoritmo devolve respostas erradas. Use Bellman-Ford, O(V·E), que também detecta ciclos negativos.
</Callout>

## Quando reconhecer

- "Menor custo", "menor tempo", "menor esforço" com **pesos diferentes** nas conexões.
- Tempo de propagação de um sinal pela rede a partir de um nó: é o maior valor de `dist`.
- Grades em que cada célula tem um custo de entrada ou o esforço é a diferença de altura.
- Se todos os pesos são iguais, BFS basta. Se os pesos são só 0 ou 1, uma deque (BFS 0-1) resolve em O(V + E).

| Situação | Algoritmo |
| --- | --- |
| Sem pesos | BFS |
| Pesos 0 ou 1 | BFS 0-1 com deque |
| Pesos não negativos | Dijkstra |
| Pesos negativos | Bellman-Ford |
| Limite de k arestas | Bellman-Ford com k rodadas ou BFS por níveis |
