Uma **fila** (queue) é a estrutura da fila do banco: quem chega primeiro é atendido primeiro (FIFO, first in, first out). Um **deque** (double-ended queue) generaliza a ideia: permite inserir e remover nas **duas pontas** em O(1). Em entrevistas, filas aparecem principalmente em BFS e simulações, e deques aparecem em janelas deslizantes com máximo ou mínimo.

## Não use list.pop(0)

O erro mais comum em Python é usar uma lista como fila:


Remover do começo de uma lista obriga a mover todos os outros elementos. Em um BFS com n nós, isso transforma O(n) em O(n²). Use `collections.deque`:

<CodeTabs>

```python
from collections import deque

fila = deque()
fila.append("a")       # entra no fim: O(1)
fila.append("b")
print(fila.popleft())  # sai do começo: O(1) -> a
fila.appendleft("z")   # entra no começo: O(1)
print(fila.pop())      # sai do fim: O(1) -> b
print(list(fila))      # ['z']
```

```javascript
// JavaScript não tem deque na biblioteca padrão. Esta versão mínima usa um objeto com dois índices,
// o que torna as quatro operações O(1) (um array com shift()/unshift() seria O(n)).
class Deque {
  constructor() {
    this.itens = {};
    this.ini = 0;    // posição do primeiro elemento
    this.fim = 0;    // posição seguinte à do último
  }
  get tamanho() { return this.fim - this.ini; }
  append(x) { this.itens[this.fim++] = x; }
  appendleft(x) { this.itens[--this.ini] = x; }
  popleft() {
    const x = this.itens[this.ini];
    delete this.itens[this.ini++];
    return x;
  }
  pop() {
    const x = this.itens[--this.fim];
    delete this.itens[this.fim];
    return x;
  }
  front() { return this.itens[this.ini]; }
  back() { return this.itens[this.fim - 1]; }
  toArray() {
    const saida = [];
    for (let i = this.ini; i < this.fim; i++) saida.push(this.itens[i]);
    return saida;
  }
}

const fila2 = new Deque();
fila2.append("a");              // entra no fim: O(1)
fila2.append("b");
console.log(fila2.popleft());   // sai do começo: O(1) -> a
fila2.appendleft("z");          // entra no começo: O(1)
console.log(fila2.pop());       // sai do fim: O(1) -> b
console.log(fila2.toArray());   // [z]
```

| Operação | `list` | `deque` |
| --- | --- | --- |
| Inserir/remover no fim | O(1) amortizado | O(1) |
| Inserir/remover no começo | O(n) | O(1) |
| Acesso por índice no meio | O(1) | O(n) |

O deque troca o acesso aleatório rápido por operações rápidas nas pontas. Se você precisa de `fila[i]` com frequência, ele não é a escolha certa.

## Fila em BFS

O uso mais clássico: explorar um grafo ou uma grade em **camadas**, garantindo que você visita primeiro os nós mais próximos da origem. Por isso BFS encontra o caminho mínimo em número de arestas quando elas não têm peso.

<CodeTabs>

```python
from collections import deque

def distancias(grafo, origem):
    dist = {origem: 0}
    fila = deque([origem])
    while fila:
        no = fila.popleft()
        for viz in grafo[no]:
            if viz not in dist:
                dist[viz] = dist[no] + 1
                fila.append(viz)
    return dist

g = {"A": ["B", "C"], "B": ["D"], "C": ["D"], "D": []}
print(distancias(g, "A"))  # {'A': 0, 'B': 1, 'C': 1, 'D': 2}
```

```javascript
function distancias(grafo, origem) {
  const dist = new Map([[origem, 0]]);
  const fila = [origem];
  for (let ini = 0; ini < fila.length; ini++) {   // índice de leitura no lugar de shift(), que é O(n)
    const no = fila[ini];
    for (const viz of grafo[no]) {
      if (!dist.has(viz)) {
        dist.set(viz, dist.get(no) + 1);
        fila.push(viz);
      }
    }
  }
  return dist;
}

console.log(Object.fromEntries(distancias(g, "A")));  // { A: 0, B: 1, C: 1, D: 2 }
```

```java
import java.util.*;

static Map<String, Integer> distancias(Map<String, List<String>> grafo, String origem) {
    Map<String, Integer> dist = new TreeMap<>();
    dist.put(origem, 0);
    Deque<String> fila = new ArrayDeque<>();
    fila.add(origem);
    while (!fila.isEmpty()) {
        String no = fila.poll();
        for (String viz : grafo.get(no)) {
            if (!dist.containsKey(viz)) {
                dist.put(viz, dist.get(no) + 1);
                fila.add(viz);
            }
        }
    }
    return dist;
}
```

```cpp
map<string, int> distancias(const map<string, vector<string>>& grafo, const string& origem) {
```

```csharp
        foreach (string viz in grafo[no])
```

## Deque monotônico: máximo da janela

Problema: para cada janela de tamanho k, qual é o maior valor? Recalcular o máximo a cada janela custa O(n·k). Com um deque de índices em que os valores ficam **decrescentes**, o máximo está sempre na frente:

<CodeTabs>

```python
from collections import deque

def maximo_janelas(nums, k):
    dq = deque()   # índices; nums[dq] decrescente
    resultado = []
    for i, x in enumerate(nums):
        if dq and dq[0] <= i - k:       # frente saiu da janela
            dq.popleft()
        while dq and nums[dq[-1]] <= x:  # não serão mais o máximo
            dq.pop()
        dq.append(i)
        if i >= k - 1:
            resultado.append(nums[dq[0]])
    return resultado

print(maximo_janelas([1, 3, -1, -3, 5, 3, 6, 7], 3))  # [3, 3, 5, 5, 6, 7]
```

```javascript
function maximoJanelas(nums, k) {
  // reaproveita a classe Deque do exemplo anterior
  const dq = new Deque();   // índices; nums[dq] decrescente
  const resultado = [];
  nums.forEach((x, i) => {
    if (dq.tamanho > 0 && dq.front() <= i - k) {         // frente saiu da janela
      dq.popleft();
    }
    while (dq.tamanho > 0 && nums[dq.back()] <= x) { // não serão mais o máximo
      dq.pop();
    }
    dq.append(i);
    if (i >= k - 1) {
      resultado.push(nums[dq.front()]);
    }
  });
  return resultado;
}

console.log(maximoJanelas([1, 3, -1, -3, 5, 3, 6, 7], 3));  // [3, 3, 5, 5, 6, 7]
```

```typescript
function maximoJanelas(nums: number[], k: number): number[] {
  // aqui basta um array com índice de início: a frente nunca volta atrás
  const dq: number[] = [];   // índices; nums[dq] decrescente
  let frente = 0;
  const resultado: number[] = [];
  nums.forEach((x, i) => {
    if (dq.length > frente && dq[frente] <= i - k) {            // frente saiu da janela
      frente++;
    }
    while (dq.length > frente && nums[dq[dq.length - 1]] <= x) { // não serão mais o máximo
      dq.pop();
    }
    dq.push(i);
    if (i >= k - 1) {
      resultado.push(nums[dq[frente]]);
    }
  });
  return resultado;
}

console.log(maximoJanelas([1, 3, -1, -3, 5, 3, 6, 7], 3));  // [3, 3, 5, 5, 6, 7]
```

```java
import java.util.*;

static List<Integer> maximoJanelas(int[] nums, int k) {
    Deque<Integer> dq = new ArrayDeque<>();   // índices; nums[dq] decrescente
    List<Integer> resultado = new ArrayList<>();
    for (int i = 0; i < nums.length; i++) {
        if (!dq.isEmpty() && dq.peekFirst() <= i - k) {          // frente saiu da janela
            dq.pollFirst();
        }
        while (!dq.isEmpty() && nums[dq.peekLast()] <= nums[i]) { // não serão mais o máximo
            dq.pollLast();
        }
        dq.addLast(i);
        if (i >= k - 1) {
            resultado.add(nums[dq.peekFirst()]);
        }
    }
    return resultado;
}
```

```csharp
using System;
using System.Collections.Generic;

static List<int> MaximoJanelas(int[] nums, int k)
{
    var dq = new LinkedList<int>();   // índices; nums[dq] decrescente
    var resultado = new List<int>();
    for (int i = 0; i < nums.Length; i++)
    {
        if (dq.Count > 0 && dq.First.Value <= i - k)           // frente saiu da janela
        {
            dq.RemoveFirst();
        }
        while (dq.Count > 0 && nums[dq.Last.Value] <= nums[i]) // não serão mais o máximo
        {
            dq.RemoveLast();
        }
        dq.AddLast(i);
        if (i >= k - 1)
        {
            resultado.Add(nums[dq.First.Value]);
        }
    }
    return resultado;
}

{
    Console.WriteLine("[" + string.Join(", ", MaximoJanelas(new[] { 1, 3, -1, -3, 5, 3, 6, 7 }, 3)) + "]");  // [3, 3, 5, 5, 6, 7]
}
```

```go
import "fmt"

func maximoJanelas(nums []int, k int) []int {
	dq := []int{} // índices; nums[dq] decrescente (a frente é dq[0])
	resultado := []int{}
	for i, x := range nums {
		if len(dq) > 0 && dq[0] <= i-k { // frente saiu da janela
			dq = dq[1:]
		}
		for len(dq) > 0 && nums[dq[len(dq)-1]] <= x { // não serão mais o máximo
			dq = dq[:len(dq)-1]
		}
		dq = append(dq, i)
		if i >= k-1 {
			resultado = append(resultado, nums[dq[0]])
		}
	}
	return resultado
}

func main() {
	fmt.Println(maximoJanelas([]int{1, 3, -1, -3, 5, 3, 6, 7}, 3)) // [3 3 5 5 6 7]
}
```

</CodeTabs>

Cada índice entra e sai do deque no máximo uma vez: O(n) de tempo e O(k) de espaço. Um elemento menor que o recém-chegado nunca mais será o máximo, porque o novo é maior e vai durar mais tempo na janela.

<Callout type="tip" title="Quando reconhecer">
"Menor número de passos", "níveis", "camada por camada", "ordem de chegada" pedem fila (BFS ou simulação). "Máximo ou mínimo de cada janela" pede deque monotônico. Precisar mexer nas duas pontas pede deque.
</Callout>

## Outras aparições

- **Processamento por níveis de árvore:** guarde `len(fila)` no início de cada nível e processe exatamente essa quantidade.
- **Simulação de tarefas:** um sistema que atende pedidos em ordem de chegada, como um contador de requisições nos últimos 5 minutos, em que você descarta da frente o que expirou.
- **Implementar fila com duas pilhas:** pergunta clássica de design; cada elemento muda de pilha uma vez, O(1) amortizado.
