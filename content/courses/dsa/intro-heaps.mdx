Um **heap** é uma estrutura que mantém o menor (ou o maior) elemento sempre acessível no topo, e permite inserir e remover em O(log n). É a implementação usual de uma **fila de prioridade**.

## Como funciona

Um heap mínimo é uma árvore binária quase completa em que todo pai é menor ou igual aos seus filhos. Ele é guardado em um array: o filho esquerdo do índice `i` fica em `2i + 1` e o direito em `2i + 2`. Inserir coloca o elemento no fim e o faz "subir" trocando com o pai; remover o topo coloca o último elemento na raiz e o faz "descer".

| Operação | Custo |
| --- | --- |
| Ver o topo | O(1) |
| Inserir | O(log n) |
| Remover o topo | O(log n) |
| Montar a partir de uma lista | O(n) |

## heapq em Python

O módulo `heapq` transforma uma `list` comum em heap mínimo:

<CodeTabs>

```python
import heapq

nums = [5, 1, 8, 3, 2]
heapq.heapify(nums)          # O(n), in-place
heapq.heappush(nums, 0)
menor = heapq.heappop(nums)  # 0
print(menor, nums[0])        # 0 1
```

```javascript
// JavaScript não tem heap na biblioteca padrão: esta classe mínima é reaproveitada no exemplo seguinte
class MinHeap {
  constructor(itens = []) {
    this.d = [];
    for (const x of itens) this.push(x);    // aqui é O(n log n); o heapify de verdade é O(n)
  }

  get tamanho() { return this.d.length; }

  peek() { return this.d[0]; }

  push(x) {
    const d = this.d;
    d.push(x);
    let i = d.length - 1;
    while (i > 0) {
      const p = (i - 1) >> 1;
      if (d[p] <= d[i]) break;
      [d[p], d[i]] = [d[i], d[p]];
      i = p;
    }
  }

  pop() {
    const d = this.d;
    const topo = d[0];
    const ultimo = d.pop();
    if (d.length > 0) {
      d[0] = ultimo;
      let i = 0;
      while (true) {
        let m = i;
        const l = 2 * i + 1, r = 2 * i + 2;
        if (l < d.length && d[l] < d[m]) m = l;
        if (r < d.length && d[r] < d[m]) m = r;
        if (m === i) break;
        [d[m], d[i]] = [d[i], d[m]];
        i = m;
      }
    }
    return topo;
  }
}

const nums = new MinHeap([5, 1, 8, 3, 2]);
nums.push(0);
const menor = nums.pop();            // 0
console.log(menor, nums.peek());     // 0 1
```

```csharp
using System;
using System.Collections.Generic;

// PriorityQueue só existe a partir do .NET 6; esta classe mínima funciona em qualquer versão
class MinHeap
{
    private List<int> d = new List<int>();

    public int Count { get { return d.Count; } }
    public int Peek() { return d[0]; }

    public void Push(int x)
    {
        d.Add(x);
        int i = d.Count - 1;
        while (i > 0)
        {
            int p = (i - 1) / 2;
            if (d[p] <= d[i]) break;
            int t = d[p]; d[p] = d[i]; d[i] = t;
            i = p;
        }
    }

    public int Pop()
    {
        int topo = d[0];
        int ultimo = d[d.Count - 1];
        d.RemoveAt(d.Count - 1);
        if (d.Count > 0)
        {
            d[0] = ultimo;
            int i = 0;
            while (true)
            {
                int m = i, l = 2 * i + 1, r = 2 * i + 2;
                if (l < d.Count && d[l] < d[m]) m = l;
                if (r < d.Count && d[r] < d[m]) m = r;
                if (m == i) break;
                int t = d[m]; d[m] = d[i]; d[i] = t;
                i = m;
            }
        }
        return topo;
    }
}

static void Main()
{
    var nums = new MinHeap();
    foreach (int x in new[] { 5, 1, 8, 3, 2 }) nums.Push(x);
    nums.Push(0);
    int menor = nums.Pop();                  // 0
    Console.WriteLine(menor + " " + nums.Peek());  // 0 1
}
```

```go
import (
	"container/heap"
	"fmt"
)

// container/heap trabalha sobre qualquer tipo que implemente esta interface
type IntHeap []int

func (h IntHeap) Len() int           { return len(h) }
func (h IntHeap) Less(i, j int) bool { return h[i] < h[j] }
func (h IntHeap) Swap(i, j int)      { h[i], h[j] = h[j], h[i] }
func (h *IntHeap) Push(x any)        { *h = append(*h, x.(int)) }
func (h *IntHeap) Pop() any {
	antigo := *h
	n := len(antigo)
	x := antigo[n-1]
	*h = antigo[:n-1]
	return x
}

func main() {
	nums := &IntHeap{5, 1, 8, 3, 2}
	heap.Init(nums) // O(n), in-place
	heap.Push(nums, 0)
	menor := heap.Pop(nums).(int) // 0
	fmt.Println(menor, (*nums)[0]) // 0 1
}
```

</CodeTabs>

Não existe heap máximo pronto. O truque é inserir os valores com o sinal invertido e desinverter ao retirar. Para ordenar por uma chave, insira tuplas: `(prioridade, desempate, item)`. O desempate evita que o Python tente comparar itens que não são comparáveis.

<Callout type="tip" title="Tuplas com desempate">
Ao guardar objetos, use `(prioridade, contador, objeto)` com um contador crescente. Se duas prioridades empatarem, a comparação para no contador e nunca chega ao objeto.
</Callout>

## O padrão top-k

Para achar os k maiores elementos, não ordene tudo. Mantenha um heap **mínimo de tamanho k**: o topo é o menor dos k maiores vistos até agora, e qualquer elemento novo maior que ele entra no lugar.

<CodeTabs>

```python
import heapq

def k_maiores(nums, k):
    heap = []
    for x in nums:
        if len(heap) < k:
            heapq.heappush(heap, x)
        elif x > heap[0]:
            heapq.heapreplace(heap, x)  # remove o topo e insere x
    return sorted(heap, reverse=True)

print(k_maiores([3, 1, 5, 12, 2, 11, 9], 3))  # [12, 11, 9]
```

```javascript
function kMaiores(nums, k) {
  const heap = new MinHeap();
  for (const x of nums) {
    if (heap.tamanho < k) {
      heap.push(x);
    } else if (x > heap.peek()) {
      heap.pop();       // remove o topo...
      heap.push(x);     // ...e insere x
    }
  }
  const saida = [];
  while (heap.tamanho > 0) saida.push(heap.pop());
  return saida.reverse();   // do maior para o menor
}

console.log(kMaiores([3, 1, 5, 12, 2, 11, 9], 3));  // [12, 11, 9]
```

import java.util.*;

static List<Integer> kMaiores(int[] nums, int k) {
    PriorityQueue<Integer> heap = new PriorityQueue<>();
    for (int x : nums) {
        if (heap.size() < k) {
            heap.add(x);

            heap.poll();     // remove o topo...
            heap.add(x);     // ...e insere x

    return saida;
}

```cpp
#include <iostream>
#include <queue>
#include <vector>
using namespace std;

vector<int> kMaiores(const vector<int>& nums, size_t k) {
    priority_queue<int, vector<int>, greater<int>> heap;
    for (int x : nums) {
        if (heap.size() < k) {
            heap.push(x);
        } else if (x > heap.top()) {
            heap.pop();      // remove o topo...
            heap.push(x);    // ...e insere x
        }
    }
    vector<int> saida;
    while (!heap.empty()) {
        saida.insert(saida.begin(), heap.top());   // o menor sai primeiro: vai para a frente
        heap.pop();
    }
    return saida;
}

int main() {
    vector<int> r = kMaiores({3, 1, 5, 12, 2, 11, 9}, 3);
    cout << "[";
    for (size_t i = 0; i < r.size(); i++) cout << (i ? ", " : "") << r[i];
    cout << "]" << endl;  // [12, 11, 9]
}
```

```csharp
using System;
using System.Collections.Generic;

static List<int> KMaiores(int[] nums, int k)
{
    var heap = new MinHeap();
    foreach (int x in nums)
    {
        if (heap.Count < k)
        {
            heap.Push(x);
        }
        else if (x > heap.Peek())
        {
            heap.Pop();      // remove o topo...
            heap.Push(x);    // ...e insere x
        }
    }
    var saida = new List<int>();
    while (heap.Count > 0)
    {
        saida.Add(heap.Pop());
    }
    saida.Reverse();         // do maior para o menor
    return saida;
}

static void Main()
{
    Console.WriteLine("[" + string.Join(", ", KMaiores(new[] { 3, 1, 5, 12, 2, 11, 9 }, 3)) + "]");  // [12, 11, 9]
}
```

```go
func kMaiores(nums []int, k int) []int {
	h := &IntHeap{}
	for _, x := range nums {
		if h.Len() < k {
			heap.Push(h, x)
		} else if x > (*h)[0] {
			(*h)[0] = x
			heap.Fix(h, 0) // equivale ao heapreplace: troca o topo por x
		}
	}
	saida := make([]int, h.Len())
	for i := len(saida) - 1; i >= 0; i-- {
		saida[i] = heap.Pop(h).(int) // o menor sai primeiro: preenche de trás para a frente
	}
	return saida
}

func main() {
	fmt.Println(kMaiores([]int{3, 1, 5, 12, 2, 11, 9}, 3)) // [12 11 9]
}
```

</CodeTabs>

Isso custa O(n log k) e O(k) de espaço, melhor que O(n log n) quando k é pequeno. A mesma ideia resolve "k pontos mais próximos", "k palavras mais frequentes" e o k-ésimo maior elemento.

Outros usos clássicos: juntar k listas ordenadas (o heap guarda a cabeça de cada lista), a mediana de um fluxo (dois heaps, um máximo e um mínimo) e o algoritmo de Dijkstra.
