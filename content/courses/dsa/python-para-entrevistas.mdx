## Listas, tuplas e fatiamento

<CodeTabs>

```python
nums = [5, 2, 8, 1]
nums.append(3)          # O(1) amortizado
ultimo = nums.pop()     # O(1); pop(0) é O(n)!
nums.sort()             # O(n log n), in-place
copia = nums[::-1]      # fatia invertida, O(n), nova lista
grade = [[0] * 4 for _ in range(3)]   # matriz 3x4 sem referência compartilhada
```

```go
	nums = append(nums, 3)               // O(1) amortizado

	for i := range grade {
		grade[i] = make([]int, 4)
	}
```

```javascript
// JavaScript não tem heap nem deque na biblioteca padrão: o heap mínimo abaixo é a versão mais enxuta possível
class MinHeap {
  constructor(itens = []) {
    this.d = [];
    for (const x of itens) this.push(x);
  }
  get tamanho() { return this.d.length; }
  push(x) {
    const d = this.d;
    d.push(x);
    for (let i = d.length - 1; i > 0 && d[(i - 1) >> 1] > d[i]; i = (i - 1) >> 1) {
      [d[(i - 1) >> 1], d[i]] = [d[i], d[(i - 1) >> 1]];
    }
  }
  pop() {
    const d = this.d;
    const topo = d[0];
    const ultimo = d.pop();
    if (d.length > 0) {
      d[0] = ultimo;
      for (let i = 0; ;) {
        let m = i;
        for (const c of [2 * i + 1, 2 * i + 2]) if (c < d.length && d[c] < d[m]) m = c;
        if (m === i) break;
        [d[m], d[i]] = [d[i], d[m]];
        i = m;
      }
    }
    return topo;
  }
  peek() { return this.d[0]; }
}

const pilha = [];
pilha.push(1); pilha.push(2);
console.log(pilha.pop());          // 2 (LIFO)

const fila = [1, 2];               // para filas pequenas, shift() basta; em volume use dois índices
fila.push(3);
console.log(fila.shift());         // 1 (FIFO)

const heap = new MinHeap([5, 1, 4]);
heap.push(0);                      // O(log n)
console.log(heap.pop());           // 0: min-heap

// max-heap: guarde o valor negado
```

```csharp
using System;
using System.Collections.Generic;
using System.Linq;

// PriorityQueue só existe a partir do .NET 6: aqui um SortedSet de (valor, id) faz o papel do heap
// (o id desempata valores iguais, porque um SortedSet não aceita duplicatas)
class MinHeap
{
    private SortedSet<Tuple<int, int>> d = new SortedSet<Tuple<int, int>>();
    private int id = 0;
    public int Count { get { return d.Count; } }
    public void Push(int x) { d.Add(Tuple.Create(x, id++)); }
    public int Peek() { return d.Min.Item1; }
    public int Pop()
    {
        var topo = d.Min;
        d.Remove(topo);
        return topo.Item1;
    }
}

static void Main()
{
    var pilha = new Stack<int>();
    pilha.Push(1); pilha.Push(2);
    Console.WriteLine(pilha.Pop());            // 2 (LIFO)

    var fila = new Queue<int>(new[] { 1, 2 });
    fila.Enqueue(3);
    Console.WriteLine(fila.Dequeue());         // 1 (FIFO), O(1)

    var heap = new MinHeap();
    foreach (int x in new[] { 5, 1, 4 }) heap.Push(x);
    heap.Push(0);                              // O(log n)
    Console.WriteLine(heap.Pop());             // 0: min-heap

    // max-heap: guarde o valor negado
    var mx = new MinHeap();
    foreach (int x in new[] { 3, 7, 2 }) mx.Push(-x);
    Console.WriteLine(-mx.Peek());             // 7

    // k maiores sem ordenar tudo (aqui, por simplicidade, ordenando uma cópia)
    var k = new[] { 4, 9, 1, 7 }.OrderByDescending(x => x).Take(2);
    Console.WriteLine("[" + string.Join(", ", k) + "]");   // [9, 7]
}
```

```go
func (h IntHeap) Len() int           { return len(h) }
func (h IntHeap) Less(i, j int) bool { return h[i] < h[j] }
func (h IntHeap) Swap(i, j int)      { h[i], h[j] = h[j], h[i] }
func (h *IntHeap) Push(x any)        { *h = append(*h, x.(int)) }
func (h *IntHeap) Pop() any {
	antigo := *h
	n := len(antigo)
	x := antigo[n-1]
	*h = antigo[:n-1]
	return x
}

func main() {
	pilha := []int{}
	pilha = append(pilha, 1, 2)
	fmt.Println(pilha[len(pilha)-1]) // 2 (LIFO)
	pilha = pilha[:len(pilha)-1]

	fila := []int{1, 2}
	fila = append(fila, 3)
	fmt.Println(fila[0]) // 1 (FIFO); fila = fila[1:] é O(1) em Go
	fila = fila[1:]

	h := &IntHeap{5, 1, 4}
	heap.Init(h)                          // O(n)
	heap.Push(h, 0)                       // O(log n)
	fmt.Println(heap.Pop(h))              // 0: min-heap

	// max-heap: guarde o valor negado
	mx := &IntHeap{}
	for _, x := range []int{3, 7, 2} {
		heap.Push(mx, -x)
	}
	fmt.Println(-(*mx)[0]) // 7

	// k maiores sem ordenar tudo (aqui, por simplicidade, ordenando uma cópia)
	v := []int{4, 9, 1, 7}
	sort.Sort(sort.Reverse(sort.IntSlice(v)))
	fmt.Println(v[:2]) // [9 7]
}
```

## Qual estrutura o enunciado está pedindo

Muitas vezes a escolha da estrutura já é metade da solução. Alguns atalhos de reconhecimento:

- "Já vi esse valor?" ou "quantas vezes aparece?": `set` ou `Counter`.
- "Agrupe por alguma característica" (anagramas, mesmo domínio): `defaultdict(list)` com a característica como chave, por exemplo a tupla da palavra ordenada.
- "Os k maiores", "o próximo mais urgente": `heapq`.
- "Processe na ordem de chegada" ou "camada por camada": `deque`.
- "O último aberto precisa fechar primeiro" (parênteses, desfazer): lista usada como pilha.
- "Primeira posição maior ou igual a X" num array ordenado: `bisect`.
