Comparar duas strings de tamanho m custa O(m). Comparar dois **números** custa O(1). O Rabin-Karp transforma cada janela do texto num número (um hash) e só compara caracteres quando os números batem. O truque que torna isso eficiente é o **rolling hash**: o hash da próxima janela sai do hash da anterior em O(1), sem reler os m caracteres.

## O hash polinomial

Trate a string como um número escrito numa base B (por exemplo, 256), reduzido por um primo grande M:

```text
hash("abc") = (a·B² + b·B¹ + c·B⁰) mod M
```

Para deslizar a janela de `"abc"` para `"bcd"`:

1. Remova a contribuição do caractere que sai: subtraia `a·B²`.
2. Desloque o resto uma casa: multiplique por B.
3. Some o caractere que entra: some `d`.

O valor `B^(m-1) mod M` é calculado uma vez no início, então cada passo custa O(1).

```python
def rabin_karp(texto, padrao):
    m, n = len(padrao), len(texto)
    if m == 0:
        return [0]
    if m > n:
        return []
    BASE, MOD = 256, 1_000_000_007
    potencia = pow(BASE, m - 1, MOD)        # peso do caractere que sai
    hp = hj = 0
    for i in range(m):
        hp = (hp * BASE + ord(padrao[i])) % MOD
        hj = (hj * BASE + ord(texto[i])) % MOD
    ocorrencias = []
    for i in range(n - m + 1):
        if hp == hj and texto[i:i + m] == padrao:   # confirma: evita falso positivo
            ocorrencias.append(i)
        if i + m < n:
            hj = (hj - ord(texto[i]) * potencia) % MOD   # remove o da esquerda
            hj = (hj * BASE + ord(texto[i + m])) % MOD   # adiciona o da direita
    return ocorrencias

print(rabin_karp("abracadabra", "abra"))   # [0, 7]
print(rabin_karp("aaaaa", "aa"))           # [0, 1, 2, 3]
```

```javascript
function rabinKarp(texto, padrao) {
  const m = padrao.length, n = texto.length;
  if (m === 0) {
    return [0];
  }
  if (m > n) {
    return [];
  }
  const BASE = 256, MOD = 1_000_000_007;
  let potencia = 1;                            // peso do caractere que sai: BASE^(m-1) mod MOD
  for (let i = 0; i < m - 1; i++) {
    potencia = (potencia * BASE) % MOD;
  }
  let hp = 0, hj = 0;
  for (let i = 0; i < m; i++) {
    hp = (hp * BASE + padrao.charCodeAt(i)) % MOD;
    hj = (hj * BASE + texto.charCodeAt(i)) % MOD;
  }
  const ocorrencias = [];
  for (let i = 0; i <= n - m; i++) {
    if (hp === hj && texto.substring(i, i + m) === padrao) {   // confirma: evita falso positivo
      ocorrencias.push(i);
    }
    if (i + m < n) {
      // o resto de uma subtração pode ser negativo em JavaScript: somamos MOD antes de reduzir
      hj = ((hj - texto.charCodeAt(i) * potencia) % MOD + MOD) % MOD;   // remove o da esquerda
      hj = (hj * BASE + texto.charCodeAt(i + m)) % MOD;                  // adiciona o da direita
    }
  }
  return ocorrencias;
}
```

**Complexidade:** O(n + m) no caso esperado, porque colisões são raras com um primo grande. No pior caso, com muitas colisões, cada uma exige a verificação em O(m), e o total vai a O(n·m). Espaço O(1) extra.

<Callout type="warning" title="Colisões existem">
Dois textos diferentes podem ter o mesmo hash. Sempre confirme comparando os caracteres, ou use dois hashes independentes (bases e módulos diferentes) quando a verificação for cara. Em Python, o `%` de número negativo já devolve valor positivo; em Java e JavaScript, some M antes do módulo.
</Callout>

## Hash de qualquer substring em O(1)

Calculando os hashes de todos os prefixos e as potências da base, o hash de `s[i:i+t]` sai de uma conta só, parecida com soma de prefixos. Isso permite comparar substrings arbitrárias em O(1) e combina muito bem com busca binária. Exemplo: a maior substring que aparece pelo menos duas vezes. Se existe uma repetida de tamanho t, existe uma de tamanho t-1, então dá para buscar t binariamente.

<CodeTabs>

```python
def maior_repetida(s):
    MOD, BASE = (1 << 61) - 1, 131
    n = len(s)
    pref = [0] * (n + 1)
    pot = [1] * (n + 1)
    for i, c in enumerate(s):
        pref[i + 1] = (pref[i] * BASE + ord(c)) % MOD
        pot[i + 1] = pot[i] * BASE % MOD

    def hash_de(i, tam):                    # hash de s[i:i+tam] em O(1)
        return (pref[i + tam] - pref[i] * pot[tam]) % MOD

    def acha(tam):
        vistos = {}
        for i in range(n - tam + 1):
            h = hash_de(i, tam)
            if h in vistos and s[vistos[h]:vistos[h] + tam] == s[i:i + tam]:
                return i
            vistos.setdefault(h, i)
        return -1

    lo, hi, melhor = 1, n - 1, ""
    while lo <= hi:
        meio = (lo + hi) // 2
        i = acha(meio)
        if i != -1:
            melhor, lo = s[i:i + meio], meio + 1
        else:
            hi = meio - 1
    return melhor

print(maior_repetida("banana"))            # ana
```

```javascript
function maiorRepetida(s) {
  // BigInt: o produto de dois valores de 61 bits não cabe em um Number
  const MOD = (1n << 61n) - 1n, BASE = 131n;
  const n = s.length;
  const pref = new Array(n + 1).fill(0n);
  const pot = new Array(n + 1).fill(1n);
  for (let i = 0; i < n; i++) {
    pref[i + 1] = (pref[i] * BASE + BigInt(s.charCodeAt(i))) % MOD;
    pot[i + 1] = pot[i] * BASE % MOD;
  }

  const hashDe = (i, tam) => {              // hash de s[i:i+tam] em O(1)
    return (((pref[i + tam] - pref[i] * pot[tam]) % MOD) + MOD) % MOD;
  };

  const acha = (tam) => {
    const vistos = new Map();
    for (let i = 0; i <= n - tam; i++) {
      const h = hashDe(i, tam);
      if (vistos.has(h) && s.substr(vistos.get(h), tam) === s.substr(i, tam)) {
        return i;
      }
      if (!vistos.has(h)) {
        vistos.set(h, i);
      }
    }
    return -1;
  };

  let lo = 1, hi = n - 1, melhor = "";
  while (lo <= hi) {
    const meio = Math.floor((lo + hi) / 2);
    const i = acha(meio);
    if (i !== -1) {
      melhor = s.substr(i, meio);
      lo = meio + 1;
    } else {
      hi = meio - 1;
    }
  }
  return melhor;
}
```

```java
import java.util.*;

static String maiorRepetida(String s) {
    // módulo menor que o do Python: o produto de dois valores cabe em long (com 2^61 - 1 seriam necessários 128 bits)
    final long MOD = 1_000_000_007L, BASE = 131;
    int n = s.length();
    long[] pref = new long[n + 1];
    long[] pot = new long[n + 1];
    pot[0] = 1;
    for (int i = 0; i < n; i++) {
        pref[i + 1] = (pref[i] * BASE + s.charAt(i)) % MOD;
        pot[i + 1] = pot[i] * BASE % MOD;
    }

    java.util.function.BiFunction<Integer, Integer, Long> hashDe =   // hash de s[i:i+tam] em O(1)
        (i, tam) -> Math.floorMod(pref[i + tam] - pref[i] * pot[tam], MOD);

    java.util.function.IntUnaryOperator acha = tam -> {
        Map<Long, Integer> vistos = new HashMap<>();
        for (int i = 0; i <= n - tam; i++) {
            long h = hashDe.apply(i, tam);
            Integer antes = vistos.get(h);
            if (antes != null && s.substring(antes, antes + tam).equals(s.substring(i, i + tam))) {
                return i;
            }
            vistos.putIfAbsent(h, i);
        }
        return -1;
    };

    int lo = 1, hi = n - 1;
    String melhor = "";
    while (lo <= hi) {
        int meio = (lo + hi) / 2;
        int i = acha.applyAsInt(meio);
        if (i != -1) {
            melhor = s.substring(i, i + meio);
            lo = meio + 1;
        } else {
            hi = meio - 1;
        }
    }
    return melhor;
}

    System.out.println(maiorRepetida("banana"));            // ana
```

```cpp
#include <functional>
#include <iostream>
#include <string>
#include <unordered_map>
#include <vector>
using namespace std;

string maiorRepetida(const string& s) {
    typedef unsigned long long u64;
    typedef unsigned __int128 u128;                 // produto de dois valores de 61 bits cabe em 128 bits
    const u64 MOD = (1ULL << 61) - 1, BASE = 131;
    int n = s.size();
    vector<u64> pref(n + 1, 0), pot(n + 1, 1);
    for (int i = 0; i < n; i++) {
        pref[i + 1] = ((u128)pref[i] * BASE + (unsigned char)s[i]) % MOD;
        pot[i + 1] = (u128)pot[i] * BASE % MOD;
    }

    auto hashDe = [&](int i, int tam) -> u64 {      // hash de s[i:i+tam] em O(1)
        return (pref[i + tam] + MOD - (u128)pref[i] * pot[tam] % MOD) % MOD;
    };

    auto acha = [&](int tam) -> int {
        unordered_map<u64, int> vistos;
        for (int i = 0; i + tam <= n; i++) {
            u64 h = hashDe(i, tam);
            auto it = vistos.find(h);
            if (it != vistos.end() && s.compare(it->second, tam, s, i, tam) == 0) {
                return i;
            }
            vistos.emplace(h, i);                   // emplace não sobrescreve: equivale a setdefault
        }
        return -1;
    };

    int lo = 1, hi = n - 1;
    string melhor;
    while (lo <= hi) {
        int meio = (lo + hi) / 2;
        int i = acha(meio);
        if (i != -1) {
            melhor = s.substr(i, meio);
            lo = meio + 1;
        } else {
            hi = meio - 1;
        }
    }
    return melhor;
}
```

```go
		return ((pref[i+tam]-pref[i]*pot[tam])%MOD + MOD) % MOD
	}

	acha := func(tam int) int {
		vistos := map[int]int{}
		for i := 0; i+tam <= n; i++ {
			h := hashDe(i, tam)
			if antes, ok := vistos[h]; ok && s[antes:antes+tam] == s[i:i+tam] {
				return i
			}
			if _, ok := vistos[h]; !ok {
				vistos[h] = i
			}
		}
		return -1
	}

	lo, hi, melhor := 1, n-1, ""
	for lo <= hi {
		meio := (lo + hi) / 2
		if i := acha(meio); i != -1 {
			melhor, lo = s[i:i+meio], meio+1
		} else {
			hi = meio - 1
		}
	}
	return melhor
}

func main() {
	fmt.Println(maiorRepetida("banana")) // ana
}
```

</CodeTabs>

Cada chamada de `acha` é O(n) esperado, e a busca binária faz O(log n) chamadas, totalizando O(n log n).

## KMP ou Rabin-Karp?

| Situação | Melhor escolha |
| --- | --- |
| Um padrão, garantia de pior caso | KMP |
| Vários padrões do mesmo tamanho | Rabin-Karp (guarde os hashes num set) |
| Comparar substrings de tamanhos variados | Rolling hash com prefixos |
| Padrão em 2D ou detectar janelas repetidas | Rolling hash |
