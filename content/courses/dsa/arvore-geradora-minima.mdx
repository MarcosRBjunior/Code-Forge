Uma operadora quer ligar cinco cidades com fibra ótica. Cada par de cidades tem um custo de instalação, e basta que todas fiquem conectadas, direta ou indiretamente. Qual o menor custo total? A resposta é uma **árvore geradora mínima** (AGM, ou *MST*, de *minimum spanning tree*): um subconjunto de arestas que conecta todos os vértices, sem ciclos, com a menor soma de pesos possível.

## Propriedades úteis

- Um grafo conexo com V vértices tem AGM com exatamente **V − 1 arestas**.
- **Propriedade do corte:** se você divide os vértices em dois grupos, a aresta mais barata que cruza a divisão pertence a alguma AGM. Os dois algoritmos clássicos são aplicações gulosas dessa ideia.
- Se o grafo for desconexo, não existe árvore geradora; o máximo é uma **floresta** geradora mínima.

## Kruskal: arestas em ordem de custo

Ordene todas as arestas pelo peso e percorra da mais barata para a mais cara. Aceite a aresta se ela liga dois componentes diferentes; descarte se fecharia um ciclo. Quem responde "estão no mesmo componente?" é o Union-Find.

<CodeTabs>

```python
def kruskal(n, arestas):
    pai = list(range(n))
    def find(x):
        while pai[x] != x:
            pai[x] = pai[pai[x]]
            x = pai[x]
        return x

    custo, usadas = 0, []
    for w, u, v in sorted(arestas):
        ru, rv = find(u), find(v)
        if ru != rv:
            pai[ru] = rv
            custo += w
            usadas.append((u, v, w))
            if len(usadas) == n - 1:
                break
    return custo if len(usadas) == n - 1 else None, usadas

arestas = [(4, 0, 1), (1, 0, 2), (3, 1, 2), (2, 1, 3), (5, 2, 3), (7, 3, 4)]
print(kruskal(5, arestas))  # (13, [(0, 2, 1), (1, 3, 2), (1, 2, 3), (3, 4, 7)])
```

```go
func kruskal(n int, arestas [][3]int) (int, [][3]int) {
	pai = make([]int, n)
	for i := range pai {
		pai[i] = i
	}

	custo := 0
	usadas := [][3]int{}
	ordenadas := append([][3]int{}, arestas...)
	sort.Slice(ordenadas, func(i, j int) bool {
		a, b := ordenadas[i], ordenadas[j]
		if a[0] != b[0] {
			return a[0] < b[0]
		}
		if a[1] != b[1] {
			return a[1] < b[1]
		}
		return a[2] < b[2]
	})

		w, u, v := a[0], a[1], a[2]
		ru, rv := find(u), find(v)
		if ru != rv {
			pai[ru] = rv
			custo += w
			usadas = append(usadas, [3]int{u, v, w})
			if len(usadas) == n-1 {
				break
			}
		}
	}
	if len(usadas) != n-1 {
		return -1, usadas
	}
	return custo, usadas
}
```

## Prim: crescendo a partir de um vértice

Comece com um vértice qualquer. Repetidamente, adicione a aresta mais barata que liga a árvore atual a um vértice de fora. Um min-heap mantém as candidatas, de forma parecida com Dijkstra; a diferença é que a prioridade é o peso da aresta, não a distância acumulada.

<CodeTabs>

```python
import heapq

def prim(n, arestas):
    g = [[] for _ in range(n)]
    for w, u, v in arestas:
        g[u].append((w, v))
        g[v].append((w, u))
    na_arvore = [False] * n
    heap = [(0, 0)]                  # (peso da aresta, vértice)
    custo = adicionados = 0
    while heap and adicionados < n:
        w, u = heapq.heappop(heap)
        if na_arvore[u]:
            continue
        na_arvore[u] = True
        custo += w
        adicionados += 1
        for peso, v in g[u]:
            if not na_arvore[v]:
                heapq.heappush(heap, (peso, v))
    return custo if adicionados == n else None

arestas = [(4, 0, 1), (1, 0, 2), (3, 1, 2), (2, 1, 3), (5, 2, 3), (7, 3, 4)]
print(prim(5, arestas))  # 13
```

```javascript
// Aqui ela guarda pares [peso, vértice] comparados pelo peso.
class MinHeap {
  constructor() { this.d = []; }
  get tamanho() { return this.d.length; }
  push(x) {
    const d = this.d;
    d.push(x);
    let i = d.length - 1;
    while (i > 0) {
      const p = (i - 1) >> 1;
      if (d[p][0] <= d[i][0]) break;
      [d[p], d[i]] = [d[i], d[p]];
      i = p;
    }
  }
  pop() {
    const d = this.d;
    const topo = d[0];
    const ultimo = d.pop();
    if (d.length > 0) {
      d[0] = ultimo;
      let i = 0;
      while (true) {
        let m = i;
        const l = 2 * i + 1, r = 2 * i + 2;
        if (l < d.length && d[l][0] < d[m][0]) m = l;
        if (r < d.length && d[r][0] < d[m][0]) m = r;
        if (m === i) break;
        [d[m], d[i]] = [d[i], d[m]];
        i = m;
      }
    }
    return topo;
  }
}

function prim(n, arestas) {
  const g = Array.from({ length: n }, () => []);
  for (const [w, u, v] of arestas) {
    g[u].push([w, v]);
    g[v].push([w, u]);
  }
  const naArvore = new Array(n).fill(false);
  const heap = new MinHeap();
  heap.push([0, 0]);                  // [peso da aresta, vértice]
  let custo = 0, adicionados = 0;
  while (heap.tamanho > 0 && adicionados < n) {
    const [w, u] = heap.pop();
    if (naArvore[u]) {
      continue;
    }
    naArvore[u] = true;
    custo += w;
    adicionados++;
    for (const [peso, v] of g[u]) {
      if (!naArvore[v]) {
        heap.push([peso, v]);
      }
    }
  }
  return adicionados === n ? custo : null;
}
```

```java
import java.util.*;

// arestas no formato {peso, u, v}; devolve -1 se o grafo não for conexo
static int prim(int n, int[][] arestas) {
    List<List<int[]>> g = new ArrayList<>();
    for (int i = 0; i < n; i++) g.add(new ArrayList<>());
    for (int[] a : arestas) {
        g.get(a[1]).add(new int[]{a[0], a[2]});
        g.get(a[2]).add(new int[]{a[0], a[1]});
    }
    boolean[] naArvore = new boolean[n];
    PriorityQueue<int[]> heap = new PriorityQueue<>(Comparator.comparingInt(x -> x[0]));
    heap.add(new int[]{0, 0});                  // {peso da aresta, vértice}
    int custo = 0, adicionados = 0;
    while (!heap.isEmpty() && adicionados < n) {
        int[] topo = heap.poll();
        int w = topo[0], u = topo[1];
        if (naArvore[u]) {
            continue;
        }
        naArvore[u] = true;
        custo += w;
        adicionados++;
        for (int[] viz : g.get(u)) {
            if (!naArvore[viz[1]]) {
                heap.add(new int[]{viz[0], viz[1]});
            }
        }
    }
    return adicionados == n ? custo : -1;
}
```

```csharp
using System;
using System.Collections.Generic;

// PriorityQueue só existe a partir do .NET 6; aqui um SortedSet de (peso, vértice, id) faz o papel do heap
// (o id desempata entradas iguais, porque um SortedSet não aceita duplicatas)
// arestas no formato {peso, u, v}; devolve -1 se o grafo não for conexo
static int Prim(int n, int[][] arestas)
{
    var g = new List<int[]>[n];
    for (int i = 0; i < n; i++) g[i] = new List<int[]>();
    foreach (int[] a in arestas)
    {
        g[a[1]].Add(new[] { a[0], a[2] });
        g[a[2]].Add(new[] { a[0], a[1] });
    }
    var naArvore = new bool[n];
    var heap = new SortedSet<Tuple<int, int, int>>();
    int id = 0;
    heap.Add(Tuple.Create(0, 0, id++));        // (peso da aresta, vértice, id)
    int custo = 0, adicionados = 0;
    while (heap.Count > 0 && adicionados < n)
    {
        var topo = heap.Min;
        heap.Remove(topo);
        int w = topo.Item1, u = topo.Item2;
        if (naArvore[u])
        {
            continue;
        }
        naArvore[u] = true;
        custo += w;
        adicionados++;
        foreach (int[] viz in g[u])
        {
            if (!naArvore[viz[1]])
            {
                heap.Add(Tuple.Create(viz[0], viz[1], id++));
            }
        }
    }
    return adicionados == n ? custo : -1;
}
```

```go
func (h filaMin) Swap(i, j int)      { h[i], h[j] = h[j], h[i] }
func (h *filaMin) Push(x any)        { *h = append(*h, x.(item)) }
func (h *filaMin) Pop() any {
	antigo := *h
	n := len(antigo)
	x := antigo[n-1]
	*h = antigo[:n-1]
	return x
}

// arestas no formato {peso, u, v}; devolve -1 se o grafo não for conexo
func prim(n int, arestas [][3]int) int {
	g := make([][]item, n)
	for _, a := range arestas {
		w, u, v := a[0], a[1], a[2]
		g[u] = append(g[u], item{w, v})
		g[v] = append(g[v], item{w, u})
	}
	naArvore := make([]bool, n)
	h := &filaMin{{0, 0}} // (peso da aresta, vértice)
	custo, adicionados := 0, 0
	for h.Len() > 0 && adicionados < n {
		topo := heap.Pop(h).(item)
		if naArvore[topo.vertice] {
			continue
		}
		naArvore[topo.vertice] = true
		custo += topo.peso
		adicionados++
		for _, viz := range g[topo.vertice] {
			if !naArvore[viz.vertice] {
				heap.Push(h, viz)
			}
		}
	}
	if adicionados != n {
		return -1
	}
	return custo
}

func main() {
	arestas := [][3]int{{4, 0, 1}, {1, 0, 2}, {3, 1, 2}, {2, 1, 3}, {5, 2, 3}, {7, 3, 4}}
	fmt.Println(prim(5, arestas)) // 13
}
```

</CodeTabs>

**Complexidade:** O(E log V) de tempo com heap binário e O(V + E) de espaço.

## Kruskal ou Prim?

| Critério | Kruskal | Prim |
| --- | --- | --- |
| Entrada natural | Lista de arestas | Lista de adjacência |
| Melhor em | Grafos esparsos | Grafos densos (com matriz, O(V²) sem heap) |
| Estrutura auxiliar | Union-Find | Min-heap |
| Código em entrevista | Geralmente mais curto | Parecido com Dijkstra |

<Callout type="note" title="Grafo implícito completo">
Em problemas como "conectar todos os pontos do plano com custo igual à distância de Manhattan", toda dupla de pontos é uma aresta: E é cerca de V²/2. Aqui Prim com um vetor de menor custo, sem heap, roda em O(V²) e evita gerar e ordenar todas as arestas.
</Callout>

## Quando reconhecer

- "Conectar todos" (cidades, computadores, pontos) com **custo total mínimo**.
- Não importa a distância entre dois pontos específicos, só a soma de todas as conexões. Se importasse, seria caminho mínimo.
- Variações: custo mínimo para fornecer água a casas (crie um vértice virtual ligado a cada casa com o custo de um poço), ou identificar arestas críticas da AGM.
