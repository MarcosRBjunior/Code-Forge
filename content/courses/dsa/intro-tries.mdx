Pense no autocompletar de um campo de busca: você digita "pro" e o sistema sugere "programa", "projeto", "prova". Guardar as palavras num conjunto responde bem "esta palavra existe?", mas é ruim para "quais palavras **começam com** este prefixo?". A **trie** (árvore de prefixos) foi feita exatamente para isso.

## Como funciona

Cada nó representa um prefixo. As arestas são caracteres. A raiz é o prefixo vazio, e descer pela aresta `p`, depois `r`, depois `o` leva ao nó do prefixo "pro". Palavras que compartilham prefixo compartilham o caminho. Cada nó também tem uma marca dizendo se uma palavra **termina** ali, porque "pro" pode ser apenas prefixo de "prova" sem ser uma palavra.

```text
(raiz)
  └ p
    └ r
      └ o
        ├ v
        │ └ a*        * = fim de palavra
        └ j
          └ e
            └ t
              └ o*
```

## Implementação

Em Python, o jeito mais simples é usar um dicionário de filhos por nó:

```python
class NoTrie:
    def __init__(self):
        self.filhos = {}
        self.fim = False

class Trie:
    def __init__(self):
        self.raiz = NoTrie()

    def inserir(self, palavra):
        no = self.raiz
        for ch in palavra:
            no = no.filhos.setdefault(ch, NoTrie())
        no.fim = True

    def _descer(self, texto):
        no = self.raiz
        for ch in texto:
            if ch not in no.filhos:
                return None
            no = no.filhos[ch]
        return no

    def contem(self, palavra):
        no = self._descer(palavra)
        return no is not None and no.fim

    def tem_prefixo(self, prefixo):
        return self._descer(prefixo) is not None

    def com_prefixo(self, prefixo):
        no = self._descer(prefixo)
        saida = []
        def coletar(n, atual):
            if n.fim:
                saida.append(atual)
            for ch in sorted(n.filhos):
                coletar(n.filhos[ch], atual + ch)
        if no:
            coletar(no, prefixo)
        return saida

t = Trie()
for p in ["prova", "projeto", "pro", "pato"]:
    t.inserir(p)
print(t.contem("pro"), t.contem("proj"))   # True False
print(t.tem_prefixo("proj"))               # True
print(t.com_prefixo("pro"))                # ['pro', 'projeto', 'prova']
```

```java
import java.util.*;

class NoTrie {
    Map<Character, NoTrie> filhos = new TreeMap<>();   // TreeMap já itera em ordem alfabética
    boolean fim = false;
}

class Trie {
    NoTrie raiz = new NoTrie();

    void inserir(String palavra) {
        NoTrie no = raiz;
        for (char ch : palavra.toCharArray()) {
            no = no.filhos.computeIfAbsent(ch, k -> new NoTrie());
        }
        no.fim = true;
    }

    private NoTrie descer(String texto) {
        NoTrie no = raiz;
        for (char ch : texto.toCharArray()) {
            no = no.filhos.get(ch);
            if (no == null) {
                return null;
            }
        }
        return no;
    }

    boolean contem(String palavra) {
        NoTrie no = descer(palavra);
        return no != null && no.fim;
    }

    boolean temPrefixo(String prefixo) {
        return descer(prefixo) != null;
    }

    List<String> comPrefixo(String prefixo) {
        NoTrie no = descer(prefixo);
        List<String> saida = new ArrayList<>();
        if (no != null) {
            coletar(no, new StringBuilder(prefixo), saida);
        }
        return saida;
    }

    private void coletar(NoTrie n, StringBuilder atual, List<String> saida) {
        if (n.fim) {
            saida.add(atual.toString());
        }
        for (Map.Entry<Character, NoTrie> e : n.filhos.entrySet()) {
            atual.append(e.getKey());
            coletar(e.getValue(), atual, saida);
            atual.deleteCharAt(atual.length() - 1);
        }
    }
}

public static void main(String[] args) {
    Trie t = new Trie();
    for (String p : new String[]{"prova", "projeto", "pro", "pato"}) {
        t.inserir(p);
    }
    System.out.println(t.contem("pro") + " " + t.contem("proj"));   // true false
    System.out.println(t.temPrefixo("proj"));                       // true
    System.out.println(t.comPrefixo("pro"));                        // [pro, projeto, prova]
}
```

```cpp
#include <iostream>
#include <map>
#include <string>
#include <vector>
using namespace std;

struct NoTrie {
    map<char, NoTrie*> filhos;   // map já itera em ordem alfabética
    bool fim = false;
};

class Trie {
public:
    void inserir(const string& palavra) {
        NoTrie* no = &raiz;
        for (char ch : palavra) {
            if (!no->filhos.count(ch)) {
                no->filhos[ch] = new NoTrie();
            }
            no = no->filhos[ch];
        }
        no->fim = true;
    }

    bool contem(const string& palavra) {
        NoTrie* no = descer(palavra);
        return no != nullptr && no->fim;
    }

    bool temPrefixo(const string& prefixo) {
        return descer(prefixo) != nullptr;
    }

    vector<string> comPrefixo(const string& prefixo) {
        NoTrie* no = descer(prefixo);
        vector<string> saida;
        if (no) {
            coletar(no, prefixo, saida);
        }
        return saida;
    }

private:
    NoTrie raiz;

    NoTrie* descer(const string& texto) {
        NoTrie* no = &raiz;
        for (char ch : texto) {
            auto it = no->filhos.find(ch);
            if (it == no->filhos.end()) {
                return nullptr;
            }
            no = it->second;
        }
        return no;
    }

    void coletar(NoTrie* n, const string& atual, vector<string>& saida) {
        if (n->fim) {
            saida.push_back(atual);
        }
        for (auto& [ch, filho] : n->filhos) {
            coletar(filho, atual + ch, saida);
        }
    }
};

int main() {
    Trie t;
    for (string p : {"prova", "projeto", "pro", "pato"}) {
        t.inserir(p);
    }
    cout << boolalpha << t.contem("pro") << " " << t.contem("proj") << endl;   // true false
    cout << t.temPrefixo("proj") << endl;                                      // true
    vector<string> r = t.comPrefixo("pro");
    cout << "[";
    for (size_t i = 0; i < r.size(); i++) cout << (i ? ", " : "") << r[i];
    cout << "]" << endl;                                                       // [pro, projeto, prova]
}
```

```csharp
using System;
using System.Collections.Generic;

class NoTrie
{
    public SortedDictionary<char, NoTrie> Filhos = new SortedDictionary<char, NoTrie>();   // itera em ordem alfabética
    public bool Fim = false;
}

class Trie
{
    private NoTrie raiz = new NoTrie();

    public void Inserir(string palavra)
    {
        NoTrie no = raiz;
        foreach (char ch in palavra)
        {
            if (!no.Filhos.ContainsKey(ch))
            {
                no.Filhos[ch] = new NoTrie();
            }
            no = no.Filhos[ch];
        }
        no.Fim = true;
    }

    private NoTrie Descer(string texto)
    {
        NoTrie no = raiz;
        foreach (char ch in texto)
        {
            if (!no.Filhos.TryGetValue(ch, out no))
            {
                return null;
            }
        }
        return no;
    }

    public bool Contem(string palavra)
    {
        NoTrie no = Descer(palavra);
        return no != null && no.Fim;
    }

    public bool TemPrefixo(string prefixo)
    {
        return Descer(prefixo) != null;
    }

    public List<string> ComPrefixo(string prefixo)
    {
        NoTrie no = Descer(prefixo);
        var saida = new List<string>();
        if (no != null)
        {
            Coletar(no, prefixo, saida);
        }
        return saida;
    }

    private void Coletar(NoTrie n, string atual, List<string> saida)
    {
        if (n.Fim)
        {
            saida.Add(atual);
        }
        foreach (var par in n.Filhos)
        {
            Coletar(par.Value, atual + par.Key, saida);
        }
    }
}

static void Main()
{
    var t = new Trie();
    foreach (string p in new[] { "prova", "projeto", "pro", "pato" })
    {
        t.Inserir(p);
    }
    Console.WriteLine(t.Contem("pro") + " " + t.Contem("proj"));   // True False
    Console.WriteLine(t.TemPrefixo("proj"));                       // True
    Console.WriteLine("[" + string.Join(", ", t.ComPrefixo("pro")) + "]");   // [pro, projeto, prova]
}
```

```go
import (
	"fmt"
	"sort"
)

type NoTrie struct {
	filhos map[rune]*NoTrie
	fim    bool
}

func novoNo() *NoTrie {
	return &NoTrie{filhos: map[rune]*NoTrie{}}
}

type Trie struct {
	raiz *NoTrie
}

func (t *Trie) inserir(palavra string) {
	no := t.raiz
	for _, ch := range palavra {
		if _, ok := no.filhos[ch]; !ok {
			no.filhos[ch] = novoNo()
		}
		no = no.filhos[ch]
	}
	no.fim = true
}

func (t *Trie) descer(texto string) *NoTrie {
	no := t.raiz
	for _, ch := range texto {
		prox, ok := no.filhos[ch]
		if !ok {
			return nil
		}
		no = prox
	}
	return no
}

func (t *Trie) contem(palavra string) bool {
	no := t.descer(palavra)
	return no != nil && no.fim
}

func (t *Trie) temPrefixo(prefixo string) bool {
	return t.descer(prefixo) != nil
}

func (t *Trie) comPrefixo(prefixo string) []string {
	no := t.descer(prefixo)
	saida := []string{}
	var coletar func(n *NoTrie, atual string)
	coletar = func(n *NoTrie, atual string) {
		if n.fim {
			saida = append(saida, atual)
		}
		chaves := []rune{}
		for ch := range n.filhos {
			chaves = append(chaves, ch)
		}
		sort.Slice(chaves, func(i, j int) bool { return chaves[i] < chaves[j] })
		for _, ch := range chaves {
			coletar(n.filhos[ch], atual+string(ch))
		}
	}
	if no != nil {
		coletar(no, prefixo)
	}
	return saida
}

func main() {
	t := &Trie{raiz: novoNo()}
	for _, p := range []string{"prova", "projeto", "pro", "pato"} {
		t.inserir(p)
	}
	fmt.Println(t.contem("pro"), t.contem("proj")) // true false
	fmt.Println(t.temPrefixo("proj"))              // true
	fmt.Println(t.comPrefixo("pro"))               // [pro projeto prova]
}
```

## Complexidade

Seja L o tamanho da palavra ou prefixo consultado:

| Operação | Trie | Conjunto (hash) |
| --- | --- | --- |
| Inserir | O(L) | O(L) para calcular o hash |
| Buscar palavra exata | O(L) | O(L) |
| Existe alguma palavra com prefixo? | O(L) | O(n·L), testando todas |
| Listar palavras com prefixo | O(L + tamanho da saída) | O(n·L) |

O custo não depende de **quantas** palavras há na trie, só do tamanho da consulta. O espaço é O(total de caracteres inseridos) no pior caso, menos quando há muitos prefixos compartilhados. Dicionários por nó têm overhead; com alfabeto fixo de 26 letras, um array de 26 posições por nó é mais compacto em linguagens como Java.

<Callout type="note" title="Trie com DFS">
Problemas como "encontre todas as palavras de um dicionário numa grade de letras" combinam DFS na grade com uma trie: você só continua explorando um caminho enquanto ele ainda é prefixo de alguma palavra. Isso poda a busca de forma drástica.
</Callout>

## Variações úteis

- **Contador de prefixo:** guarde em cada nó quantas palavras passam por ali. Responde "quantas palavras começam com X" em O(L).
- **Curinga:** busca com `.` significando qualquer letra; no `.` você desce por todos os filhos.
- **Trie de bits:** inserir números bit a bit (do mais significativo ao menos) permite achar o par com maior XOR em O(n · bits).

## Quando reconhecer

- Palavras-chave: prefixo, autocompletar, dicionário de palavras, "começa com", corretor ortográfico.
- Várias consultas contra o mesmo conjunto de palavras, em que um hash set testaria tudo a cada consulta.
- Busca de muitas palavras ao mesmo tempo numa grade ou texto.

## Na entrevista

Comece justificando por que um conjunto não basta (consultas de prefixo). Deixe clara a diferença entre "o caminho existe" e "uma palavra termina aqui"; esquecer a marca `fim` é o bug mais comum. Ao falar de espaço, mencione o trade-off entre dicionário por nó (flexível) e array fixo (rápido, mas desperdiça memória em alfabetos grandes).
